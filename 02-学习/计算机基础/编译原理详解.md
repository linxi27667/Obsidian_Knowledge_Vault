# 编译原理详解

## 核心概念

- **词法分析** - 将源代码转换为token流
- **语法分析** - 构建抽象语法树(AST)
- **语义分析** - 类型检查、作用域分析
- **代码生成** - 生成目标代码

---

## 一、编译流程

### 1.1 编译阶段

```
源代码 → 词法分析 → 语法分析 → 语义分析 → 中间代码 → 优化 → 目标代码
  │         │         │         │         │        │        │
  └─────────┴─────────┴─────────┴─────────┴────────┴────────┘
                        编译器
```

## 二、词法分析

### 2.1 词法分析器

```c
// 词法分析器
typedef struct {
    const char *source;
    int pos;
    int line;
    int column;
} lexer_t;

// 初始化词法分析器
void lexer_init(lexer_t *lexer, const char *source) {
    lexer->source = source;
    lexer->pos = 0;
    lexer->line = 1;
    lexer->column = 1;
}

// 跳过空白
void skip_whitespace(lexer_t *lexer) {
    while (lexer->source[lexer->pos] == ' ' ||
           lexer->source[lexer->pos] == '\t' ||
           lexer->source[lexer->pos] == '\n') {
        if (lexer->source[lexer->pos] == '\n') {
            lexer->line++;
            lexer->column = 1;
        } else {
            lexer->column++;
        }
        lexer->pos++;
    }
}

// 跳过注释
void skip_comment(lexer_t *lexer) {
    if (lexer->source[lexer->pos] == '/' && lexer->source[lexer->pos + 1] == '/') {
        // 单行注释
        while (lexer->source[lexer->pos] != '\n' && lexer->source[lexer->pos] != '\0') {
            lexer->pos++;
        }
    } else if (lexer->source[lexer->pos] == '/' && lexer->source[lexer->pos + 1] == '*') {
        // 多行注释
        lexer->pos += 2;
        while (!(lexer->source[lexer->pos] == '*' && lexer->source[lexer->pos + 1] == '/')) {
            if (lexer->source[lexer->pos] == '\n') {
                lexer->line++;
                lexer->column = 1;
            }
            lexer->pos++;
        }
        lexer->pos += 2;
    }
}

// 读取数字
token_t read_number(lexer_t *lexer) {
    token_t token;
    token.type = TOKEN_NUMBER;
    token.line = lexer->line;
    token.column = lexer->column;

    int i = 0;
    while (isdigit(lexer->source[lexer->pos]) || lexer->source[lexer->pos] == '.') {
        token.value[i++] = lexer->source[lexer->pos++];
        lexer->column++;
    }
    token.value[i] = '\0';

    return token;
}

// 读取标识符
token_t read_identifier(lexer_t *lexer) {
    token_t token;
    token.type = TOKEN_IDENTIFIER;
    token.line = lexer->line;
    token.column = lexer->column;

    int i = 0;
    while (isalnum(lexer->source[lexer->pos]) || lexer->source[lexer->pos] == '_') {
        token.value[i++] = lexer->source[lexer->pos++];
        lexer->column++;
    }
    token.value[i] = '\0';

    // 检查是否为关键字
    const char *keywords[] = {"if", "else", "while", "for", "return", "int", "float", "void"};
    for (int j = 0; j < sizeof(keywords) / sizeof(keywords[0]); j++) {
        if (strcmp(token.value, keywords[j]) == 0) {
            token.type = TOKEN_KEYWORD;
            break;
        }
    }

    return token;
}

// 获取下一个token
token_t lexer_next(lexer_t *lexer) {
    skip_whitespace(lexer);
    skip_comment(lexer);

    token_t token;
    token.line = lexer->line;
    token.column = lexer->column;

    char c = lexer->source[lexer->pos];

    if (c == '\0') {
        token.type = TOKEN_EOF;
        token.value[0] = '\0';
        return token;
    }

    if (isdigit(c)) {
        return read_number(lexer);
    }

    if (isalpha(c) || c == '_') {
        return read_identifier(lexer);
    }

    // 运算符
    token.type = TOKEN_OPERATOR;
    token.value[0] = c;
    token.value[1] = '\0';
    lexer->pos++;
    lexer->column++;

    return token;
}
```

### 2.2 正则表达式引擎

```c
// 简单正则表达式匹配
typedef enum {
    RE_CHAR,       // 字面字符
    RE_ANY,        // .
    RE_STAR,       // *
    RE_PLUS,       // +
    RE_QUESTION,   // ?
    RE_CLASS,      // [abc]
    RE_NOT_CLASS,  // [^abc]
    RE_START,      // ^
    RE_END,        // $
} re_type_t;

typedef struct re_node {
    re_type_t type;
    char ch;
    char *class;
    struct re_node *next;
} re_node_t;

// 正则表达式编译
re_node_t *re_compile(const char *pattern) {
    re_node_t *head = NULL;
    re_node_t *tail = NULL;

    while (*pattern) {
        re_node_t *node = malloc(sizeof(re_node_t));

        switch (*pattern) {
            case '.':
                node->type = RE_ANY;
                break;
            case '*':
                node->type = RE_STAR;
                break;
            case '+':
                node->type = RE_PLUS;
                break;
            case '?':
                node->type = RE_QUESTION;
                break;
            case '^':
                node->type = RE_START;
                break;
            case '$':
                node->type = RE_END;
                break;
            case '[':
                // 解析字符类
                pattern++;
                if (*pattern == '^') {
                    node->type = RE_NOT_CLASS;
                    pattern++;
                } else {
                    node->type = RE_CLASS;
                }
                // 读取字符类内容
                break;
            default:
                node->type = RE_CHAR;
                node->ch = *pattern;
                break;
        }

        node->next = NULL;

        if (head == NULL) {
            head = node;
            tail = node;
        } else {
            tail->next = node;
            tail = node;
        }

        pattern++;
    }

    return head;
}

// 正则表达式匹配
int re_match(re_node_t *pattern, const char *text) {
    if (pattern == NULL) return 1;

    switch (pattern->type) {
        case RE_CHAR:
            if (*text == pattern->ch) {
                return re_match(pattern->next, text + 1);
            }
            return 0;

        case RE_ANY:
            if (*text != '\0') {
                return re_match(pattern->next, text + 1);
            }
            return 0;

        case RE_STAR:
            // 匹配0次或多次
            while (*text != '\0' && re_match(pattern->next, text)) {
                text++;
            }
            return re_match(pattern->next, text);

        case RE_START:
            return re_match(pattern->next, text);

        case RE_END:
            return *text == '\0';
    }

    return 0;
}
```

## 三、语法分析

### 3.1 递归下降分析

```c
// 语法分析器
typedef struct {
    lexer_t lexer;
    token_t current;
} parser_t;

// 初始化分析器
void parser_init(parser_t *parser, const char *source) {
    lexer_init(&parser->lexer, source);
    parser->current = lexer_next(&parser->lexer);
}

// 消费token
void parser_consume(parser_t *parser, token_type_t expected) {
    if (parser->current.type != expected) {
        printf("Syntax error at line %d: expected %d, got %d\n",
               parser->current.line, expected, parser->current.type);
        exit(1);
    }
    parser->current = lexer_next(&parser->lexer);
}

// 解析表达式
ast_node_t *parse_expression(parser_t *parser);

// 解析项
ast_node_t *parse_term(parser_t *parser) {
    if (parser->current.type == TOKEN_NUMBER) {
        ast_node_t *node = malloc(sizeof(ast_node_t));
        node->type = NODE_LITERAL;
        node->literal.int_val = atoi(parser->current.value);
        parser_consume(parser, TOKEN_NUMBER);
        return node;
    }

    if (parser->current.type == TOKEN_IDENTIFIER) {
        ast_node_t *node = malloc(sizeof(ast_node_t));
        node->type = NODE_IDENTIFIER;
        strcpy(node->identifier.name, parser->current.value);
        parser_consume(parser, TOKEN_IDENTIFIER);
        return node;
    }

    if (strcmp(parser->current.value, "(") == 0) {
        parser_consume(parser, TOKEN_PUNCTUATION);
        ast_node_t *expr = parse_expression(parser);
        parser_consume(parser, TOKEN_PUNCTUATION);
        return expr;
    }

    return NULL;
}

// 解析因子
ast_node_t *parse_factor(parser_t *parser) {
    ast_node_t *left = parse_term(parser);

    while (parser->current.type == TOKEN_OPERATOR &&
           (strcmp(parser->current.value, "*") == 0 ||
            strcmp(parser->current.value, "/") == 0)) {
        char op = parser->current.value[0];
        parser_consume(parser, TOKEN_OPERATOR);

        ast_node_t *right = parse_term(parser);

        ast_node_t *node = malloc(sizeof(ast_node_t));
        node->type = NODE_BINARY_OP;
        node->binary_op.op = op;
        node->binary_op.left = left;
        node->binary_op.right = right;
        left = node;
    }

    return left;
}

// 解析表达式
ast_node_t *parse_expression(parser_t *parser) {
    ast_node_t *left = parse_factor(parser);

    while (parser->current.type == TOKEN_OPERATOR &&
           (strcmp(parser->current.value, "+") == 0 ||
            strcmp(parser->current.value, "-") == 0)) {
        char op = parser->current.value[0];
        parser_consume(parser, TOKEN_OPERATOR);

        ast_node_t *right = parse_factor(parser);

        ast_node_t *node = malloc(sizeof(ast_node_t));
        node->type = NODE_BINARY_OP;
        node->binary_op.op = op;
        node->binary_op.left = left;
        node->binary_op.right = right;
        left = node;
    }

    return left;
}

// 解析语句
ast_node_t *parse_statement(parser_t *parser) {
    if (parser->current.type == TOKEN_KEYWORD &&
        strcmp(parser->current.value, "return") == 0) {
        parser_consume(parser, TOKEN_KEYWORD);
        ast_node_t *expr = parse_expression(parser);
        // 创建return节点
        ast_node_t *node = malloc(sizeof(ast_node_t));
        node->type = NODE_RETURN;
        return node;
    }

    // 表达式语句
    return parse_expression(parser);
}

// 解析函数定义
ast_node_t *parse_function(parser_t *parser) {
    // 返回类型
    parser_consume(parser, TOKEN_KEYWORD);

    // 函数名
    char name[64];
    strcpy(name, parser->current.value);
    parser_consume(parser, TOKEN_IDENTIFIER);

    // 参数列表
    parser_consume(parser, TOKEN_PUNCTUATION);  // (
    parser_consume(parser, TOKEN_PUNCTUATION);  // )

    // 函数体
    parser_consume(parser, TOKEN_PUNCTUATION);  // {
    ast_node_t *body = parse_statement(parser);
    parser_consume(parser, TOKEN_PUNCTUATION);  // }

    ast_node_t *node = malloc(sizeof(ast_node_t));
    node->type = NODE_FUNCTION;
    strcpy(node->function.name, name);
    node->function.body = body;

    return node;
}
```

### 3.2 表达式解析

```c
// 运算符优先级
int get_precedence(const char *op) {
    if (strcmp(op, "*") == 0 || strcmp(op, "/") == 0) return 3;
    if (strcmp(op, "+") == 0 || strcmp(op, "-") == 0) return 2;
    if (strcmp(op, "<") == 0 || strcmp(op, ">") == 0) return 1;
    if (strcmp(op, "=") == 0) return 0;
    return -1;
}

// Pratt解析器
ast_node_t *pratt_parse_expression(parser_t *parser, int min_prec) {
    ast_node_t *left = parse_term(parser);

    while (parser->current.type == TOKEN_OPERATOR &&
           get_precedence(parser->current.value) >= min_prec) {
        char op[8];
        strcpy(op, parser->current.value);
        int prec = get_precedence(op);
        parser_consume(parser, TOKEN_OPERATOR);

        ast_node_t *right = pratt_parse_expression(parser, prec + 1);

        ast_node_t *node = malloc(sizeof(ast_node_t));
        node->type = NODE_BINARY_OP;
        node->binary_op.op = op[0];
        node->binary_op.left = left;
        node->binary_op.right = right;
        left = node;
    }

    return left;
}
```

### 4.1 符号表

```c
// 符号类型
typedef enum {
    SYM_VARIABLE,
    SYM_FUNCTION,
    SYM_CONSTANT,
} symbol_type_t;

// 符号
typedef struct {
    char name[64];
    symbol_type_t type;
    char data_type[32];      // int, float, void等
    int scope_level;
    int offset;              // 栈偏移
    bool initialized;
} symbol_t;

// 符号表
typedef struct {
    symbol_t symbols[1000];
    int count;
    int scope_level;
} symbol_table_t;

// 查找符号
symbol_t *symbol_lookup(symbol_table_t *table, const char *name) {
    for (int i = table->count - 1; i >= 0; i--) {
        if (strcmp(table->symbols[i].name, name) == 0) {
            return &table->symbols[i];
        }
    }
    return NULL;
}

// 添加符号
int symbol_add(symbol_table_t *table, const char *name, symbol_type_t type,
               const char *data_type) {
    if (symbol_lookup(table, name) != NULL) {
        return -1;  // 重复定义
    }

    symbol_t *sym = &table->symbols[table->count++];
    strcpy(sym->name, name);
    sym->type = type;
    strcpy(sym->data_type, data_type);
    sym->scope_level = table->scope_level;
    sym->initialized = false;

    return 0;
}

// 进入作用域
void scope_enter(symbol_table_t *table) {
    table->scope_level++;
}

// 离开作用域
void scope_leave(symbol_table_t *table) {
    while (table->count > 0 &&
           table->symbols[table->count - 1].scope_level == table->scope_level) {
        table->count--;
    }
    table->scope_level--;
}
```

### 4.2 类型检查

```c
// 类型系统
typedef enum {
    TYPE_VOID,
    TYPE_INT,
    TYPE_FLOAT,
    TYPE_CHAR,
    TYPE_POINTER,
    TYPE_ARRAY,
    TYPE_STRUCT,
} type_kind_t;

typedef struct type_info {
    type_kind_t kind;
    int size;
    struct type_info *base;      // 指针/数组的基类型
    int array_size;
} type_info_t;

// 类型兼容性检查
bool types_compatible(type_info_t *a, type_info_t *b) {
    if (a->kind == b->kind) {
        if (a->kind == TYPE_POINTER || a->kind == TYPE_ARRAY) {
            return types_compatible(a->base, b->base);
        }
        return true;
    }

    // 隐式转换
    if (a->kind == TYPE_FLOAT && b->kind == TYPE_INT) return true;
    if (a->kind == TYPE_INT && b->kind == TYPE_FLOAT) return true;

    return false;
}

// 类型推导
type_info_t *infer_type(ast_node_t *node, symbol_table_t *table) {
    switch (node->type) {
        case NODE_LITERAL:
            // 整数字面量
            type_info_t *int_type = malloc(sizeof(type_info_t));
            int_type->kind = TYPE_INT;
            int_type->size = 4;
            return int_type;

        case NODE_IDENTIFIER: {
            symbol_t *sym = symbol_lookup(table, node->identifier.name);
            if (sym == NULL) {
                printf("Undefined variable: %s\n", node->identifier.name);
                return NULL;
            }
            // 返回变量类型
            break;
        }

        case NODE_BINARY_OP: {
            type_info_t *left = infer_type(node->binary_op.left, table);
            type_info_t *right = infer_type(node->binary_op.right, table);

            if (!types_compatible(left, right)) {
                printf("Type mismatch in binary operation\n");
                return NULL;
            }

            // 返回较宽的类型
            if (left->kind == TYPE_FLOAT || right->kind == TYPE_FLOAT) {
                type_info_t *float_type = malloc(sizeof(type_info_t));
                float_type->kind = TYPE_FLOAT;
                float_type->size = 4;
                return float_type;
            }

            return left;
        }
    }

    return NULL;
}
```

### 5.1 三地址码

```c
// 三地址码操作
typedef enum {
    TAC_ADD,       // x = y + z
    TAC_SUB,       // x = y - z
    TAC_MUL,       // x = y * z
    TAC_DIV,       // x = y / z
    TAC_ASSIGN,    // x = y
    TAC_LABEL,     // L:
    TAC_GOTO,      // goto L
    TAC_IF,        // if x goto L
    TAC_IF_FALSE,  // if !x goto L
    TAC_CALL,      // x = call f, n
    TAC_PARAM,     // param x
    TAC_RETURN,    // return x
} tac_op_t;

// 三地址码
typedef struct {
    tac_op_t op;
    char result[64];
    char arg1[64];
    char arg2[64];
    int label;
} tac_t;

// 三地址码生成器
typedef struct {
    tac_t code[10000];
    int count;
    int temp_count;
    int label_count;
} tac_generator_t;

// 生成临时变量
char *new_temp(tac_generator_t *gen) {
    char *temp = malloc(32);
    sprintf(temp, "t%d", gen->temp_count++);
    return temp;
}

// 生成标签
int new_label(tac_generator_t *gen) {
    return gen->label_count++;
}

// 添加三地址码
void emit_tac(tac_generator_t *gen, tac_t *tac) {
    gen->code[gen->count++] = *tac;
}

// 生成表达式代码
char *gen_expression(tac_generator_t *gen, ast_node_t *node) {
    if (node->type == NODE_LITERAL) {
        char *temp = new_temp(gen);
        tac_t tac = {TAC_ASSIGN, temp, {node->literal.int_val}, ""};
        emit_tac(gen, &tac);
        return temp;
    }

    if (node->type == NODE_BINARY_OP) {
        char *left = gen_expression(gen, node->binary_op.left);
        char *right = gen_expression(gen, node->binary_op.right);
        char *result = new_temp(gen);

        tac_op_t op;
        switch (node->binary_op.op) {
            case '+': op = TAC_ADD; break;
            case '-': op = TAC_SUB; break;
            case '*': op = TAC_MUL; break;
            case '/': op = TAC_DIV; break;
        }

        tac_t tac = {op, result, left, right};
        emit_tac(gen, &tac);
        return result;
    }

    return NULL;
}
```

## 六、代码优化

### 6.1 常量传播

```c
// 常量传播
typedef struct {
    char var[64];
    int value;
    bool is_constant;
} const_value_t;

const_value_t const_table[1000];
int const_count = 0;

void constant_propagation(tac_t *code, int count) {
    for (int i = 0; i < count; i++) {
        // 检查是否为常量赋值
        if (code[i].op == TAC_ASSIGN && is_number(code[i].arg1)) {
            const_value_t *cv = &const_table[const_count++];
            strcpy(cv->var, code[i].result);
            cv->value = atoi(code[i].arg1);
            cv->is_constant = true;
        }

        // 替换为常量
        const_value_t *arg1_const = find_const(code[i].arg1);
        if (arg1_const) {
            sprintf(code[i].arg1, "%d", arg1_const->value);
        }

        const_value_t *arg2_const = find_const(code[i].arg2);
        if (arg2_const) {
            sprintf(code[i].arg2, "%d", arg2_const->value);
        }
    }
```

```c
void dead_code_elimination(tac_t *code, int count) {
    bool *used = calloc(count, sizeof(bool));

    // 标记使用的变量
    for (int i = 0; i < count; i++) {
        if (strlen(code[i].arg1) > 0) {
            mark_used(code, count, code[i].arg1, used);
        }
        if (strlen(code[i].arg2) > 0) {
            mark_used(code, count, code[i].arg2, used);
        }
    }

    // 删除未使用的赋值
    for (int i = 0; i < count; i++) {
        if (!used[i] && code[i].op == TAC_ASSIGN) {
            code[i].op = TAC_ADD;  // 标记为删除
            code[i].result[0] = '\0';
        }
    }

    free(used);
}
```

```c
void loop_invariant_motion(tac_t *code, int loop_start, int loop_end) {
    for (int i = loop_start; i <= loop_end; i++) {
        // 检查是否为循环不变量

void loop_unroll(tac_t *code, int loop_start, int loop_end, int unroll_factor) {
    // 复制循环体
    int body_size = loop_end - loop_start;
    for (int i = 0; i < unroll_factor - 1; i++) {
        // 复制并修改循环变量
        for (int j = loop_start; j <= loop_end; j++) {
            tac_t copy = code[j];
            // 修改循环变量
            update_loop_var(&copy, i + 1);
            insert_after(code, &copy, loop_end + i * body_size);
        }
    }
}
```

```c
// 寄存器分配
typedef struct {
    char var[64];
    int register_num;
    bool is_live;
} reg_alloc_t;

// 图着色寄存器分配
typedef struct {
    int interference[100][100];  // 干涉图
    int node_count;
    int color[100];              // 分配的颜色(寄存器)
} reg_allocator_t;

// 构建干涉图
void build_interference_graph(reg_allocator_t *alloc, tac_t *code, int count) {
    // 分析活跃变量
    for (int i = count - 1; i >= 0; i--) {
        // 同时活跃的变量之间有边
        for (int j = 0; j < alloc->node_count; j++) {
            if (is_live(j, i) && j != get_var_index(code[i].result)) {
                alloc->interference[get_var_index(code[i].result)][j] = 1;
                alloc->interference[j][get_var_index(code[i].result)] = 1;
            }
        }
    }
}

// 图着色
int graph_color(reg_allocator_t *alloc) {
    int colors = 0;
    int *degree = calloc(alloc->node_count, sizeof(int));

    // 计算度数
    for (int i = 0; i < alloc->node_count; i++) {
        for (int j = 0; j < alloc->node_count; j++) {
            if (alloc->interference[i][j]) {
                degree[i]++;
            }
        }
    }

    // 简化
    int *stack = malloc(alloc->node_count * sizeof(int));
    int top = -1;

    for (int i = 0; i < alloc->node_count; i++) {
        // 找到度数最小的节点
        int min_node = -1;
        int min_degree = alloc->node_count + 1;
        for (int j = 0; j < alloc->node_count; j++) {
            if (alloc->color[j] == 0 && degree[j] < min_degree) {
                min_degree = degree[j];
                min_node = j;
            }
        }

        stack[++top] = min_node;

        // 移除节点
        for (int j = 0; j < alloc->node_count; j++) {
            if (alloc->interference[min_node][j]) {
                degree[j]--;
            }
        }
    }

    // 选择
    while (top >= 0) {
        int node = stack[top--];
        int *used_colors = calloc(colors + 1, sizeof(int));

        // 找到已使用的颜色
        for (int j = 0; j < alloc->node_count; j++) {
            if (alloc->interference[node][j] && alloc->color[j] > 0) {
                used_colors[alloc->color[j]] = 1;
            }
        }

        // 分配最小可用颜色
        for (int c = 1; c <= colors + 1; c++) {
            if (!used_colors[c]) {
                alloc->color[node] = c;
                if (c > colors) colors = c;
                break;
            }
        }

        free(used_colors);
    }

    free(stack);
    free(degree);
    return colors;
}
```

```c
// 指令选择
typedef struct {
    const char *pattern;
    void (*emit)(tac_t *tac);
} pattern_t;

// 模式匹配
void emit_x86_add(tac_t *tac) {
    printf("    mov eax, [%s]\n", tac->arg1);
    printf("    add eax, [%s]\n", tac->arg2);
    printf("    mov [%s], eax\n", tac->result);
}

void emit_x86_mul(tac_t *tac) {
    printf("    mov eax, [%s]\n", tac->arg1);
    printf("    imul eax, [%s]\n", tac->arg2);
    printf("    mov [%s], eax\n", tac->result);
}

// 指令选择
void select_instructions(tac_t *code, int count) {
    pattern_t patterns[] = {
        {TAC_ADD, emit_x86_add},
        {TAC_SUB, emit_x86_sub},
        {TAC_MUL, emit_x86_mul},
        {TAC_DIV, emit_x86_div},
    };

    for (int i = 0; i < count; i++) {
        for (int j = 0; j < sizeof(patterns) / sizeof(patterns[0]); j++) {
            if (code[i].op == patterns[j].op) {
                patterns[j].emit(&code[i]);
                break;
            }
        }
    }
}
```
