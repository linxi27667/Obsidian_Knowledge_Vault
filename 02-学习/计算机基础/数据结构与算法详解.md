## 一、复杂度分析

### 1.1 时间复杂度

```c
// 常见时间复杂度
// O(1) - 常数时间
int get_first(int arr[], int n) {
    return arr[0];
}

// O(log n) - 对数时间
int binary_search(int arr[], int n, int target) {
    int left = 0, right = n - 1;
    while (left <= right) {
        int mid = left + (right - left) / 2;
        if (arr[mid] == target) return mid;
        if (arr[mid] < target) left = mid + 1;
        else right = mid - 1;
    }
    return -1;
}

// O(n) - 线性时间
int linear_search(int arr[], int n, int target) {
    for (int i = 0; i < n; i++) {
        if (arr[i] == target) return i;
    }
    return -1;
}

// O(n log n) - 线性对数时间
void merge_sort(int arr[], int left, int right) {
    if (left < right) {
        int mid = left + (right - left) / 2;
        merge_sort(arr, left, mid);
        merge_sort(arr, mid + 1, right);
        merge(arr, left, mid, right);
    }
}

// O(n^2) - 平方时间
void bubble_sort(int arr[], int n) {
    for (int i = 0; i < n - 1; i++) {
        for (int j = 0; j < n - i - 1; j++) {
            if (arr[j] > arr[j + 1]) {
                swap(&arr[j], &arr[j + 1]);
            }
        }
    }
}

// O(2^n) - 指数时间
int fibonacci(int n) {
    if (n <= 1) return n;
    return fibonacci(n - 1) + fibonacci(n - 2);
}
```

```c
    int size;

vector_t *vector_create(int capacity) {
    vector_t *v = malloc(sizeof(vector_t));
    v->data = malloc(capacity * sizeof(int));
    v->size = 0;
    v->capacity = capacity;
    return v;
}

// 扩容
void vector_resize(vector_t *v) {
    v->capacity *= 2;
    v->data = realloc(v->data, v->capacity * sizeof(int));
}

// 添加元素
void vector_push(vector_t *v, int value) {
    if (v->size >= v->capacity) {
        vector_resize(v);
    }
    v->data[v->size++] = value;
}

// 获取元素
int vector_get(vector_t *v, int index) {
    if (index < 0 || index >= v->size) {
        return -1;  // 越界
    }
    return v->data[index];
}

// 删除元素
void vector_remove(vector_t *v, int index) {
    if (index < 0 || index >= v->size) return;
    for (int i = index; i < v->size - 1; i++) {
        v->data[i] = v->data[i + 1];
    }
    v->size--;
}

// 查找
int vector_find(vector_t *v, int value) {
    for (int i = 0; i < v->size; i++) {
        if (v->data[i] == value) return i;
    }
    return -1;
}
```

```c
// 尾插法
node_t *insert_tail(node_t *head, int data) {
    node_t *node = create_node(data);
    if (head == NULL) return node;

    node_t *current = head;
    while (current->next != NULL) {
        current = current->next;
    }
    current->next = node;
    return head;
}

// 删除节点
node_t *delete_node(node_t *head, int data) {
    if (head == NULL) return NULL;

    if (head->data == data) {
        node_t *temp = head;
        head = head->next;
        free(temp);
        return head;
    }

    node_t *current = head;
    while (current->next != NULL && current->next->data != data) {
        current = current->next;
    }

    if (current->next != NULL) {
        node_t *temp = current->next;
        current->next = temp->next;
        free(temp);
    }

    return head;
}

// 反转链表
node_t *reverse_list(node_t *head) {
    node_t *prev = NULL;
    node_t *current = head;
    node_t *next = NULL;

    while (current != NULL) {
        next = current->next;
        current->next = prev;
        prev = current;
        current = next;
    }

    return prev;
}

// 检测环
bool has_cycle(node_t *head) {
    node_t *slow = head;
    node_t *fast = head;

    while (fast != NULL && fast->next != NULL) {
        slow = slow->next;
        fast = fast->next->next;
        if (slow == fast) return true;
    }

    return false;
}
```

### 2.3 栈

```c
// 栈
typedef struct {
    int *data;
    int top;
    int capacity;
} stack_t;

// 创建栈
stack_t *stack_create(int capacity) {
    stack_t *s = malloc(sizeof(stack_t));
    s->data = malloc(capacity * sizeof(int));
    s->top = -1;
    s->capacity = capacity;
    return s;
}

// 入栈
void stack_push(stack_t *s, int value) {
    if (s->top >= s->capacity - 1) {
        return;  // 栈满
    }
    s->data[++s->top] = value;
}

// 出栈
int stack_pop(stack_t *s) {
    if (s->top < 0) {
        return -1;  // 栈空
    }
    return s->data[s->top--];
}

// 栈顶
int stack_peek(stack_t *s) {
    if (s->top < 0) {
        return -1;
    }
    return s->data[s->top];
}

// 判断空
bool stack_empty(stack_t *s) {
    return s->top < 0;
}

// 括号匹配
bool is_valid_parentheses(const char *s) {
    stack_t *stack = stack_create(strlen(s));

    for (int i = 0; s[i]; i++) {
        if (s[i] == '(' || s[i] == '[' || s[i] == '{') {
            stack_push(stack, s[i]);
        } else {
            if (stack_empty(stack)) {
                return false;
            }
            char top = stack_pop(stack);
            if ((s[i] == ')' && top != '(') ||
                (s[i] == ']' && top != '[') ||
                (s[i] == '}' && top != '{')) {
                return false;
            }
        }
    }

    return stack_empty(stack);
}
```

### 2.4 队列

```c
// 循环队列
typedef struct {
    int *data;
    int front;
    int rear;
    int size;
    int capacity;
} queue_t;

// 创建队列
queue_t *queue_create(int capacity) {
    queue_t *q = malloc(sizeof(queue_t));
    q->data = malloc(capacity * sizeof(int));
    q->front = 0;
    q->rear = -1;
    q->size = 0;
    q->capacity = capacity;
    return q;
}

// 入队
void queue_enqueue(queue_t *q, int value) {
    if (q->size >= q->capacity) {
        return;  // 队满
    }
    q->rear = (q->rear + 1) % q->capacity;
    q->data[q->rear] = value;
    q->size++;
}

// 出队
int queue_dequeue(queue_t *q) {
    if (q->size <= 0) {
        return -1;  // 队空
    }
    int value = q->data[q->front];
    q->front = (q->front + 1) % q->capacity;
    q->size--;
    return value;
}

// 队首
int queue_front(queue_t *q) {
    if (q->size <= 0) return -1;
    return q->data[q->front];
}

// 判断空
bool queue_empty(queue_t *q) {
    return q->size <= 0;
}
```

```c
    if (root == NULL) return NULL;

    if (data < root->data) {
        root->left = bst_delete(root->left, data);
    } else if (data > root->data) {
        root->right = bst_delete(root->right, data);
    } else {
        // 找到节点
        if (root->left == NULL) {
            tree_node_t *temp = root->right;
            free(root);
            return temp;
        } else if (root->right == NULL) {
            tree_node_t *temp = root->left;
            free(root);
            return temp;

        tree_node_t *successor = root->right;
        while (successor->left != NULL) {
            successor = successor->left;
        }
        root->data = successor->data;
        root->right = bst_delete(root->right, successor->data);
    }

    return root;
}

    if (root->data <= min || root->data >= max) return false;
```

### 3.3 平衡树

```c
// AVL树
typedef struct avl_node {
    int data;
    struct avl_node *left;
    struct avl_node *right;
    int height;
} avl_node_t;

// 获取高度
int avl_height(avl_node_t *node) {
    if (node == NULL) return 0;
    return node->height;
}

// 更新高度
void avl_update_height(avl_node_t *node) {
    int left_h = avl_height(node->left);
    int right_h = avl_height(node->right);
    node->height = 1 + (left_h > right_h ? left_h : right_h);
}

// 获取平衡因子
int avl_balance_factor(avl_node_t *node) {
    if (node == NULL) return 0;
    return avl_height(node->left) - avl_height(node->right);
}

// 右旋
avl_node_t *avl_rotate_right(avl_node_t *y) {
    avl_node_t *x = y->left;
    avl_node_t *T2 = x->right;

    x->right = y;
    y->left = T2;

    avl_update_height(y);
    avl_update_height(x);

    return x;
}

// 左旋
avl_node_t *avl_rotate_left(avl_node_t *x) {
    avl_node_t *y = x->right;
    avl_node_t *T2 = y->left;

    y->left = x;
    x->right = T2;

    avl_update_height(x);
    avl_update_height(y);

    return y;
}

// AVL插入
avl_node_t *avl_insert(avl_node_t *node, int data) {
    if (node == NULL) {
        avl_node_t *new_node = malloc(sizeof(avl_node_t));
        new_node->data = data;
        new_node->left = NULL;
        new_node->right = NULL;
        new_node->height = 1;
        return new_node;
    }

    if (data < node->data) {
        node->left = avl_insert(node->left, data);
    } else if (data > node->data) {
        node->right = avl_insert(node->right, data);
    } else {
        return node;  // 重复
    }

    avl_update_height(node);

    int balance = avl_balance_factor(node);

    // LL
    if (balance > 1 && data < node->left->data) {
        return avl_rotate_right(node);
    }
    // RR
    if (balance < -1 && data > node->right->data) {
        return avl_rotate_left(node);
    }
    // LR
    if (balance > 1 && data > node->left->data) {
        node->left = avl_rotate_left(node->left);
        return avl_rotate_right(node);
    }
    // RL
    if (balance < -1 && data < node->right->data) {
        node->right = avl_rotate_right(node->right);
        return avl_rotate_left(node);
    }

    return node;
}
```

### 4.3 最短路径

```c
// Dijkstra算法
typedef struct {
    int vertex;
    int distance;
} heap_node_t;

void dijkstra(adj_list_t *g, int start, int *dist) {
    bool *visited = calloc(g->vertices, sizeof(bool));
    for (int i = 0; i < g->vertices; i++) {
        dist[i] = INT_MAX;
    }
    dist[start] = 0;

    // 最小堆
    heap_t *heap = heap_create(g->vertices);
    heap_insert(heap, start, 0);

    while (!heap_empty(heap)) {
        heap_node_t min = heap_extract_min(heap);
        int u = min.vertex;

        if (visited[u]) continue;
        visited[u] = true;

        adj_node_t *current = g->lists[u];
        while (current != NULL) {
            int v = current->vertex;
            int weight = current->weight;

            if (!visited[v] && dist[u] + weight < dist[v]) {
                dist[v] = dist[u] + weight;
                heap_insert(heap, v, dist[v]);
            }

            current = current->next;
        }
    }

    free(visited);
    heap_destroy(heap);
}
```

### 5.1 常见排序

```c
// 快速排序
int partition(int arr[], int low, int high) {
    int pivot = arr[high];
    int i = low - 1;

    for (int j = low; j < high; j++) {
        if (arr[j] <= pivot) {
            i++;
            swap(&arr[i], &arr[j]);
        }
    }
    swap(&arr[i + 1], &arr[high]);
    return i + 1;
}

void quick_sort(int arr[], int low, int high) {
    if (low < high) {
        int pi = partition(arr, low, high);
        quick_sort(arr, low, pi - 1);
        quick_sort(arr, pi + 1, high);
    }
}

// 归并排序
void merge(int arr[], int left, int mid, int right) {
    int n1 = mid - left + 1;
    int n2 = right - mid;

    int *L = malloc(n1 * sizeof(int));
    int *R = malloc(n2 * sizeof(int));

    for (int i = 0; i < n1; i++) L[i] = arr[left + i];
    for (int j = 0; j < n2; j++) R[j] = arr[mid + 1 + j];

    int i = 0, j = 0, k = left;
    while (i < n1 && j < n2) {
        if (L[i] <= R[j]) {
            arr[k++] = L[i++];
        } else {
            arr[k++] = R[j++];
        }
    }

    while (i < n1) arr[k++] = L[i++];
    while (j < n2) arr[k++] = R[j++];

    free(L);
    free(R);
}

// 堆排序
void heapify(int arr[], int n, int i) {
    int largest = i;
    int left = 2 * i + 1;
    int right = 2 * i + 2;

    if (left < n && arr[left] > arr[largest]) {
        largest = left;
    }
    if (right < n && arr[right] > arr[largest]) {
        largest = right;
    }

    if (largest != i) {
        swap(&arr[i], &arr[largest]);
        heapify(arr, n, largest);
    }
}

void heap_sort(int arr[], int n) {
    for (int i = n / 2 - 1; i >= 0; i--) {
        heapify(arr, n, i);
    }

    for (int i = n - 1; i > 0; i--) {
        swap(&arr[0], &arr[i]);
        heapify(arr, i, 0);
    }
}
```

### 6.1 哈希表

```c
// 哈希表
typedef struct hash_node {
    char key[64];
    int value;
    struct hash_node *next;
} hash_node_t;

typedef struct {
    hash_node_t **buckets;
    int size;
} hash_table_t;

// 哈希函数
unsigned int hash_function(const char *key, int size) {
    unsigned int hash = 0;
    for (int i = 0; key[i]; i++) {
        hash = hash * 31 + key[i];
    }
    return hash % size;
}

// 创建哈希表
hash_table_t *hash_create(int size) {
    hash_table_t *table = malloc(sizeof(hash_table_t));
    table->size = size;
    table->buckets = calloc(size, sizeof(hash_node_t *));
    return table;
}

// 插入
void hash_insert(hash_table_t *table, const char *key, int value) {
    unsigned int index = hash_function(key, table->size);

    hash_node_t *node = malloc(sizeof(hash_node_t));
    strcpy(node->key, key);
    node->value = value;
    node->next = table->buckets[index];
    table->buckets[index] = node;
}

// 查找
int hash_search(hash_table_t *table, const char *key) {
    unsigned int index = hash_function(key, table->size);

    hash_node_t *current = table->buckets[index];
    while (current != NULL) {
        if (strcmp(current->key, key) == 0) {
            return current->value;
        }
        current = current->next;
    }

    return -1;  // 未找到
}

// 删除
void hash_delete(hash_table_t *table, const char *key) {
    unsigned int index = hash_function(key, table->size);

    hash_node_t *current = table->buckets[index];
    hash_node_t *prev = NULL;

    while (current != NULL) {
        if (strcmp(current->key, key) == 0) {
            if (prev == NULL) {
                table->buckets[index] = current->next;
            } else {
                prev->next = current->next;
            }
            free(current);
            return;
        }
        prev = current;
        current = current->next;
    }
}
```
