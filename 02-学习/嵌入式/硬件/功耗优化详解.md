# 功耗优化详解

## 核心概念

- **动态功耗** - 开关功耗
- **静态功耗** - 漏电流功耗
- **占空比** - 工作时间比例
- **低功耗模式** - 睡眠/停止/待机

---

## 一、功耗分析

### 1.1 功耗模型

```c
// 功耗计算模型
typedef struct {
    float vdd;              // 供电电压
    float frequency;        // 工作频率
    float dynamic_current;  // 动态电流
    float static_current;   // 静态电流
    float temperature;      // 温度
} power_params_t;

// 动态功耗: P_dynamic = C * V² * f
float calc_dynamic_power(float capacitance, float voltage, float frequency) {
    return capacitance * voltage * voltage * frequency;
}

// 静态功耗: P_static = V * I_leak
float calc_static_power(float voltage, float leakage_current) {
    return voltage * leakage_current;
}

// 总功耗
float calc_total_power(power_params_t *params) {
    float p_dynamic = calc_dynamic_power(1e-9, params->vdd, params->frequency);
    float p_static = calc_static_power(params->vdd, params->static_current);
    return p_dynamic + p_static;
}

// 能量消耗
float calc_energy(float power, float duration_seconds) {
    return power * duration_seconds;  // 焦耳
}

// 电池寿命估算
float estimate_battery_life(float battery_capacity_mah,
                             float average_current_ma) {
    return battery_capacity_mah / average_current_ma;  // 小时
}
```

### 1.2 功耗测量

```c
// 电流测量
typedef struct {
    float shunt_resistor;   // 采样电阻
    float amplifier_gain;   // 放大倍数
    uint16_t adc_resolution;
    float adc_reference;
} current_sensor_t;

float measure_current(current_sensor_t *sensor, uint16_t adc_value) {
    float voltage = (float)adc_value / sensor->adc_resolution * sensor->adc_reference;
    float current = voltage / (sensor->shunt_resistor * sensor->amplifier_gain);
    return current;
}

// 功耗统计
typedef struct {
    uint32_t start_time;
    float total_energy;
    float peak_current;
    float average_current;
    uint32_t sample_count;
} power_stats_t;

void power_stats_update(power_stats_t *stats, float current) {
    stats->total_energy += current * get_vdd();
    stats->sample_count++;

    if (current > stats->peak_current) {
        stats->peak_current = current;
    }

    stats->average_current = stats->total_energy / (get_vdd() * stats->sample_count);
}
```

## 二、MCU低功耗

### 2.1 低功耗模式

```c
// STM32低功耗模式
typedef enum {
    LOW_POWER_RUN,      // 低功耗运行
    LOW_POWER_SLEEP,    // 低功耗睡眠
    STOP,               // 停止模式
    STANDBY,            // 待机模式
} low_power_mode_t;

// 进入低功耗模式
void enter_low_power_mode(low_power_mode_t mode) {
    switch (mode) {
        case LOW_POWER_RUN:
            // 降低时钟频率
            RCC->CFGR |= RCC_CFGR_HPRE_DIV16;  // 分频
            // 关闭不必要外设
            RCC->APB1ENR &= ~(RCC_APB1ENR_TIM2EN | RCC_APB1ENR_TIM3EN);
            break;

        case LOW_POWER_SLEEP:
            // 配置唤醒源
            configure_wakeup_source();
            // 进入睡眠
            __WFI();
            break;

        case STOP:
            // 配置唤醒源
            configure_wakeup_source();
            // 设置电压调节器
            PWR->CR |= PWR_CR_LPDS;
            // 进入停止模式
            SCB->SCR |= SCB_SCR_SLEEPDEEP_Msk;
            __WFI();
            break;

        case STANDBY:
            // 配置唤醒引脚
            PWR->CSR |= PWR_CSR_EWUP;
            // 进入待机模式
            PWR->CR |= PWR_CR_CWUF | PWR_CR_CSBF | PWR_CR_PDDS;
            SCB->SCR |= SCB_SCR_SLEEPDEEP_Msk;
            __WFI();
            break;
    }
}

// 唤醒后恢复
void exit_low_power_mode(void) {
    // 恢复时钟
    SystemClock_Config();

    // 恢复外设
    RCC->APB1ENR |= RCC_APB1ENR_TIM2EN | RCC_APB1ENR_TIM3EN;

    // 清除唤醒标志
    PWR->CR |= PWR_CR_CWUF;
}
```

### 2.2 时钟管理

```c
// 动态时钟调整
typedef struct {
    uint32_t required_frequency;
    uint32_t current_frequency;
    bool frequency_changed;
} clock_manager_t;

void optimize_clock(clock_manager_t *cm, uint32_t required_freq) {
    if (required_freq != cm->current_frequency) {
        if (required_freq < cm->current_frequency) {
            // 降频
            set_system_clock(required_freq);
            cm->current_frequency = required_freq;
            cm->frequency_changed = true;
        }
    }
}

// 外设时钟门控
void enable_peripheral_clock(peripheral_t periph) {
    switch (periph) {
        case PERIPH_GPIOA:
            RCC->AHB1ENR |= RCC_AHB1ENR_GPIOAEN;
            break;
        case PERIPH_USART1:
            RCC->APB2ENR |= RCC_APB2ENR_USART1EN;
            break;
        case PERIPH_TIM2:
            RCC->APB1ENR |= RCC_APB1ENR_TIM2EN;
            break;
    }
}

void disable_peripheral_clock(peripheral_t periph) {
    switch (periph) {
        case PERIPH_GPIOA:
            RCC->AHB1ENR &= ~RCC_AHB1ENR_GPIOAEN;
            break;
        case PERIPH_USART1:
            RCC->APB2ENR &= ~RCC_APB2ENR_USART1EN;
            break;
        case PERIPH_TIM2:
            RCC->APB1ENR &= ~RCC_APB1ENR_TIM2EN;
            break;
    }
}
```

## 三、外设功耗优化

### 3.1 GPIO优化

```c
// GPIO功耗优化
typedef struct {
    gpio_num_t pin;
    bool is_output;
    bool level;
    bool use_pullup;
    bool use_pulldown;
} gpio_config_t;

// 配置未使用GPIO
void configure_unused_gpio(void) {
    // 未使用的GPIO配置为输出低电平
    for (int i = 0; i < 16; i++) {
        if (!is_gpio_used(i)) {
            gpio_set_direction(i, GPIO_MODE_OUTPUT);
            gpio_set_level(i, 0);
        }
    }
}

// GPIO泄漏电流优化
void minimize_gpio_leakage(void) {
    // 输入引脚不要悬空
    // 配置上拉或下拉
    gpio_set_pull_mode(PIN_UNUSED, GPIO_PULLDOWN_ONLY);

    // 模拟输入引脚配置为高阻
    gpio_set_direction(PIN_ADC, GPIO_MODE_DISABLE);
}
```

### 3.2 通信接口优化

```c
// UART低功耗
typedef struct {
    uint32_t baudrate;
    bool use_dma;
    bool auto_baud;
    uint32_t idle_timeout;
} uart_low_power_config_t;

void uart_optimize(uart_low_power_config_t *config) {
    // 降低波特率
    if (config->baudrate > 115200) {
        config->baudrate = 115200;
    }

    // 使用DMA减少CPU唤醒
    if (config->use_dma) {
        uart_enable_dma();
    }

    // 空闲超时自动休眠
    if (config->idle_timeout > 0) {
        uart_set_idle_timeout(config->idle_timeout);
    }
}

// SPI低功耗
void spi_optimize(void) {
    // 降低SPI时钟
    SPI_SetClock(SPI_CLOCK_DIV256);

    // 使用后关闭SPI
    SPI_Disable();
}

// I2C低功耗
void i2c_optimize(void) {
    // 降低I2C速度
    I2C_SetSpeed(I2C_SPEED_STANDARD);  // 100kHz

    // 使用后关闭I2C
    I2C_Disable();
}
```

## 四、传感器低功耗

### 4.1 传感器占空比

```c
// 传感器电源管理
typedef struct {
    gpio_num_t power_pin;
    uint32_t warmup_time_ms;
    uint32_t sample_time_ms;
    uint32_t sleep_time_ms;
} sensor_power_config_t;

// 带占空比的传感器读取
float read_sensor_with_duty_cycle(sensor_power_config_t *config) {
    // 开启传感器电源
    gpio_set_level(config->power_pin, 1);

    // 等待稳定
    delay_ms(config->warmup_time_ms);

    // 读取数据
    float value = read_sensor();

    // 关闭传感器电源
    gpio_set_level(config->power_pin, 0);

    return value;
}

// 定时采样任务
void sensor_sampling_task(void *param) {
    sensor_power_config_t *config = (sensor_power_config_t *)param;

    while (1) {
        // 读取传感器
        float value = read_sensor_with_duty_cycle(config);

        // 处理数据
        process_sensor_data(value);

        // 休眠
        vTaskDelay(pdMS_TO_TICKS(config->sleep_time_ms));
    }
}
```

### 4.2 自适应采样

```c
// 自适应采样率
typedef struct {
    float last_value;
    float threshold;
    uint32_t min_interval_ms;
    uint32_t max_interval_ms;
    uint32_t current_interval_ms;
} adaptive_sampler_t;

uint32_t calculate_sampling_interval(adaptive_sampler_t *sampler, float current_value) {
    float delta = fabs(current_value - sampler->last_value);

    if (delta > sampler->threshold) {
        // 变化大，增加采样率
        sampler->current_interval_ms = sampler->min_interval_ms;
    } else {
        // 变化小，降低采样率
        sampler->current_interval_ms *= 2;
        if (sampler->current_interval_ms > sampler->max_interval_ms) {
            sampler->current_interval_ms = sampler->max_interval_ms;
        }
    }

    sampler->last_value = current_value;
    return sampler->current_interval_ms;
}
```

## 五、通信低功耗

### 5.1 WiFi低功耗

```c
// WiFi省电模式
typedef struct {
    enum {
        WIFI_PS_NONE,       // 不省电
        WIFI_PS_MIN_MODEM,  // 最小省电
        WIFI_PS_MAX_MODEM,  // 最大省电
    } power_save_mode;

    uint32_t listen_interval;  // 监听间隔
    uint32_t dtim_period;      // DTIM周期
} wifi_power_config_t;

void wifi_set_power_save(wifi_power_config_t *config) {
    // 设置省电模式
    esp_wifi_set_ps(config->power_save_mode);

    // 设置监听间隔
    esp_wifi_set_listen_interval(config->listen_interval);
}

// WiFi连接管理
void wifi_power_management(void) {
    // 不需要WiFi时断开连接
    if (!need_wifi_connection()) {
        esp_wifi_disconnect();
        esp_wifi_stop();
    }

    // 需要时重新连接
    if (need_wifi_connection()) {
        esp_wifi_start();
        esp_wifi_connect();
    }
}
```

// BLE省电配置
typedef struct {
    uint16_t min_interval;    // 最小连接间隔
    uint16_t max_interval;    // 最大连接间隔
    uint16_t slave_latency;   // 从机延迟

    // 增加从机延迟

void ble_advertising_optimize(void) {
    // 增加广播间隔

## 六、系统级优化

### 6.1 任务调度优化

```c
// 低功耗任务调度
typedef struct {
    task_t *tasks;
    int task_count;
    uint32_t idle_time_ms;
} low_power_scheduler_t;

void schedule_tasks(low_power_scheduler_t *scheduler) {
    while (1) {
        bool any_task_ready = false;

        // 检查就绪任务
        for (int i = 0; i < scheduler->task_count; i++) {
            if (is_task_ready(&scheduler->tasks[i])) {
                execute_task(&scheduler->tasks[i]);
                any_task_ready = true;
            }
        }

        // 没有任务就绪时休眠
        if (!any_task_ready) {
            enter_idle_mode(scheduler->idle_time_ms);
        }
    }
}

// 空闲模式
void enter_idle_mode(uint32_t duration_ms) {
    // 配置唤醒定时器
    configure_wakeup_timer(duration_ms);

    // 进入低功耗模式
    enter_low_power_mode(LOW_POWER_SLEEP);
}
```

### 6.2 电源域管理

```c
// 电源域控制
typedef struct {
    const char *name;
    gpio_num_t enable_pin;
    bool is_enabled;
    float voltage;
    float current_limit;
} power_domain_t;

// 电源域使能
void power_domain_enable(power_domain_t *domain) {
    if (!domain->is_enabled) {
        gpio_set_level(domain->enable_pin, 1);
        delay_ms(10);  // 等待稳定
        domain->is_enabled = true;
    }
}

// 电源域禁用
void power_domain_disable(power_domain_t *domain) {
    if (domain->is_enabled) {
        gpio_set_level(domain->enable_pin, 0);
        domain->is_enabled = false;
    }
}

// 按需电源管理
typedef struct {
    power_domain_t *domains;
    int domain_count;
    uint32_t *usage_counts;
} power_manager_t;

void request_power_domain(power_manager_t *pm, int domain_id) {
    pm->usage_counts[domain_id]++;
    if (pm->usage_counts[domain_id] == 1) {
        power_domain_enable(&pm->domains[domain_id]);
    }
}

void release_power_domain(power_manager_t *pm, int domain_id) {
    if (pm->usage_counts[domain_id] > 0) {
        pm->usage_counts[domain_id]--;
        if (pm->usage_counts[domain_id] == 0) {
            power_domain_disable(&pm->domains[domain_id]);
        }
    }
}
```
