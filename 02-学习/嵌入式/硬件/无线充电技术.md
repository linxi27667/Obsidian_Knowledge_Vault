# 无线充电技术

## 核心概念

- **Qi标准** - 无线充电联盟标准
- **电磁感应** - 近场能量传输
- **谐振耦合** - 中距离传输
- **整流电路** - 交流转直流

## 一、无线充电原理

### 1.1 电磁感应

```
发射端                    接收端
┌─────────┐              ┌─────────┐
│  逆变器  │              │  整流器  │
│  ┌───┐  │   电磁场     │  ┌───┐  │
│  │Tx │  │ ~~~~~~~~~~~~ │  │Rx │  │
│  │线圈│  │   耦合系数k  │  │线圈│  │
│  └───┘  │              │  └───┘  │
│  控制器  │              │  控制器  │
└─────────┘              └─────────┘

互感: M = k × √(L1 × L2)
传输效率: η ∝ k² × Q1 × Q2
```

### 1.2 Qi标准

```c
// Qi协议通信(调制在功率信号上)
// 接收端 → 发射端: 负载调制(ASK)
// 发射端 → 接收端: 频率调制(FSK)

// Qi功率等级
typedef enum {
    QI_BPP = 0,    // Baseline Power Profile: 5W
    QI_EPP = 1,    // Extended Power Profile: 15W
    QI_DELL = 2,   // Dell: 30W+

typedef struct {

    uint8_t reserved;

// 接收端功率控制
typedef struct {
    float voltage;        // 接收电压
    float current;        // 接收电流
    float temperature;    // 温度
```

## 二、发射端设计

### 2.1 全桥逆变器

```c
// 全桥逆变器驱动
typedef struct {
    gpio_num_t q1_pin;   // 高侧1
    gpio_num_t q2_pin;   // 低侧1
    gpio_num_t q3_pin;   // 高侧2
    gpio_num_t q4_pin;   // 低侧2
    uint32_t frequency;  // 工作频率
    uint32_t dead_time;  // 死区时间(ns)
} inverter_t;

void inverter_init(inverter_t *inv) {
    // 配置PWM定时器
    mcpwm_config_t pwm_config = {
        .frequency = inv->frequency,
        .cmpr_a = 50.0,
        .cmpr_b = 50.0,
        .duty_mode = MCPWM_DUTY_MODE_0,
        .counter_mode = MCPWM_UP_DOWN_COUNTER,
    };

    mcpwm_init(MCPWM_UNIT_0, MCPWM_TIMER_0, &pwm_config);

    // 设置死区
    mcpwm_deadtime_enable(MCPWM_UNIT_0, MCPWM_TIMER_0,
                          MCPWM_ACTIVE_HIGH_COMPLIMENT_MODE,
                          inv->dead_time, inv->dead_time);
}

// 频率调节(功率控制)
void inverter_set_frequency(inverter_t *inv, uint32_t freq) {
    inv->frequency = freq;
    mcpwm_set_frequency(MCPWM_UNIT_0, MCPWM_TIMER_0, freq);
}

// 占空比调节
void inverter_set_duty(inverter_t *inv, float duty) {
    mcpwm_set_duty(MCPWM_UNIT_0, MCPWM_TIMER_0, MCPWM_OPR_A, duty);
    mcpwm_set_duty(MCPWM_UNIT_0, MCPWM_TIMER_0, MCPWM_OPR_B, duty);
}
```

### 2.2 发射线圈设计

```c
// 发射线圈参数
typedef struct {
    float inductance_uh;    // 电感
    float resistance;       // 电阻
    float capacitance_nf;   // 谐振电容
    int turns;              // 匝数
    float diameter_mm;      // 直径
} tx_coil_t;

// 谐振频率计算
float calc_resonant_freq(float L_uH, float C_nF) {
    // f = 1 / (2π√(LC))
    float L = L_uH * 1e-6;
    float C = C_nF * 1e-9;
    return 1.0f / (2 * M_PI * sqrtf(L * C));
}

// Qi标准频率: 110-205kHz
void tx_coil_design(tx_coil_t *coil) {
    coil->inductance_uh = 10.0f;    // 典型值
    coil->capacitance_nf = 100.0f;   // 谐振电容
    coil->turns = 12;
    coil->diameter_mm = 40.0f;

    float f = calc_resonant_freq(coil->inductance_uh, coil->capacitance_nf);
    printf("Resonant frequency: %.1f kHz\n", f / 1000);
}
```

## 三、接收端设计

### 3.1 整流电路

```c
// 同步整流(效率更高)
typedef struct {
    gpio_num_t sr1_pin;
    gpio_num_t sr2_pin;
    gpio_num_t sr3_pin;
    gpio_num_t sr4_pin;
} sync_rectifier_t;

void sync_rect_init(sync_rectifier_t *sr) {
    // 配置为输入(检测电流方向)
    gpio_set_direction(sr->sr1_pin, GPIO_MODE_INPUT);
    gpio_set_direction(sr->sr2_pin, GPIO_MODE_INPUT);
    gpio_set_direction(sr->sr3_pin, GPIO_MODE_INPUT);
    gpio_set_direction(sr->sr4_pin, GPIO_MODE_INPUT);
}

// 电流检测控制整流
void sync_rect_control(sync_rectifier_t *sr, float *current) {
    // 检测电流方向
    if (*current > 0) {
        gpio_set_direction(sr->sr1_pin, GPIO_MODE_OUTPUT);
        gpio_set_level(sr->sr1_pin, 1);
        gpio_set_direction(sr->sr4_pin, GPIO_MODE_OUTPUT);
        gpio_set_level(sr->sr4_pin, 1);
    } else {
        gpio_set_direction(sr->sr2_pin, GPIO_MODE_OUTPUT);
        gpio_set_level(sr->sr2_pin, 1);
        gpio_set_direction(sr->sr3_pin, GPIO_MODE_OUTPUT);
        gpio_set_level(sr->sr3_pin, 1);
    }
}
```

### 3.2 功率控制

```c
// 接收端功率控制
typedef struct {
    float target_voltage;
    float target_current;
    float actual_voltage;
    float actual_current;
    uint8_t control_error;
} rx_power_control_t;

// 错误码
#define RX_ERR_NONE          0
#define RX_ERR_OVERVOLTAGE   1
#define RX_ERR_OVERCURRENT   2
#define RX_ERR_OVERTEMP      3
#define RX_ERR_NO_LOAD       4

void rx_power_control(rx_power_control_t *ctrl) {
    // 读取实际值
    ctrl->actual_voltage = adc_read_voltage() * VOLTAGE_DIVIDER_RATIO;
    ctrl->actual_current = adc_read_current();

    // 过压保护
    if (ctrl->actual_voltage > ctrl->target_voltage * 1.1f) {
        ctrl->control_error = RX_ERR_OVERVOLTAGE;
        qi_send_error_packet(ctrl->control_error);
        return;
    }

    // 过流保护
    if (ctrl->actual_current > ctrl->target_current * 1.1f) {
        ctrl->control_error = RX_ERR_OVERCURRENT;
        qi_send_error_packet(ctrl->control_error);
        return;
    }

    // 功率调节请求
    if (ctrl->actual_voltage < ctrl->target_voltage * 0.95f) {
        qi_send_power_request(QI_POWER_INCREASE);
    } else if (ctrl->actual_voltage > ctrl->target_voltage * 1.05f) {
        qi_send_power_request(QI_POWER_DECREASE);
    }
}
```

## 四、效率优化

### 4.1 线圈对齐检测

```c
// 对齐质量评估
typedef struct {
    float coupling_coefficient;  // 耦合系数
    float efficiency;            // 传输效率
    bool aligned;                // 是否对齐
    float offset_x;              // X偏移
    float offset_y;              // Y偏移
} alignment_status_t;

alignment_status_t check_alignment(float tx_voltage, float tx_current,
                                   float rx_voltage, float rx_current) {
    alignment_status_t status;

    // 计算功率
    float tx_power = tx_voltage * tx_current;
    float rx_power = rx_voltage * rx_current;

    // 效率
    status.efficiency = rx_power / tx_power * 100.0f;

    // 耦合系数估算
    status.coupling_coefficient = sqrtf(rx_power / tx_power);

    // 对齐判断(效率>70%认为对齐)
    status.aligned = (status.efficiency > 70.0f);

    return status;
}
```

### 4.2 异物检测(FOD)

```c
// 异物检测
typedef struct {
    float quality_factor;      // 品质因数
    float power_loss;          // 功率损耗
    bool foreign_object;       // 是否有异物
} fod_status_t;

fod_status_t detect_fod(float tx_power, float rx_power,
                        float rx_coil_q) {
    fod_status_t status;
    status.quality_factor = rx_coil_q;
    status.power_loss = tx_power - rx_power;

    // 异物判断
    // 1. 功率损耗过大
    if (status.power_loss > tx_power * 0.3f) {
        status.foreign_object = true;
        return status;
    }

    // 2. 品质因数异常
    if (rx_coil_q < 50.0f) {  // 正常Q>100
        status.foreign_object = true;
        return status;
    }

    status.foreign_object = false;
    return status;
}
```

## 五、磁共振无线充电

### 5.1 磁共振原理

```c
// 磁共振特性
/*
 * 与电磁感应的区别:
 * - 工作频率: 6.78MHz (vs 110-205kHz)
 * - 传输距离: 30-50cm (vs <1cm)
 * - 多设备充电: 支持
 * - 效率: 80-90%
 * - 标准: AirFuel Alliance
 */

// 谐振器设计
typedef struct {
    float frequency;        // 谐振频率
    float L;                // 电感
    float C;                // 电容
    float R;                // 电阻
    float Q;                // 品质因数
} resonator_t;

void design_resonator(resonator_t *res, float freq_hz) {
    res->frequency = freq_hz;

    // 选择电感(固定值)
    res->L = 10e-6;  // 10μH

    // 计算谐振电容
    res->C = 1.0f / (4 * M_PI * M_PI * freq_hz * freq_hz * res->L);

    // 品质因数
    res->R = 0.1f;  // 典型电阻
    res->Q = 2 * M_PI * freq_hz * res->L / res->R;
}
```

## 六、安全保护

### 6.1 温度监控

```c
// 温度保护
typedef struct {
    float warning_threshold;   // 警告温度
    float shutdown_threshold;  // 关断温度
    float current_temp;        // 当前温度
} thermal_protection_t;

bool thermal_check(thermal_protection_t *tp) {
    tp->current_temp = read_ntc_temperature();

    if (tp->current_temp >= tp->shutdown_threshold) {
        // 关断充电
        power_off_transmitter();
        return false;
    }

    if (tp->current_temp >= tp->warning_threshold) {
        // 降低功率
        reduce_power(50);
    }

    return true;
}
```
