# 嵌入式设计模式

## 核心概念

- **设计模式** - 可复用的软件设计方案
- **架构模式** - 系统级的结构方案
- **嵌入式约束** - 资源受限、实时性、可靠性

---

## 一、创建型模式

### 1.1 单例模式(Singleton)

**用途：** 确保全局唯一实例(如硬件管理器)

```c
typedef struct {
    int initialized;
    int value;
} singleton_t;

static singleton_t instance = {0};

singleton_t *singleton_get_instance(void) {
    if (!instance.initialized) {
        instance.value = 0;
        instance.initialized = 1;
    }
    return &instance;
}
```

---

### 1.2 工厂模式(Factory)

**用途：** 根据类型创建不同驱动

```c
typedef enum { DEV_UART, DEV_SPI, DEV_I2C } dev_type_t;

typedef struct {
    int (*init)(void);
    int (*read)(uint8_t *buf, int len);
    int (*write)(uint8_t *buf, int len);
} device_ops_t;

static int uart_init(void) { return 0; }
static int spi_init(void) { return 0; }

device_ops_t *device_create(dev_type_t type) {
    static device_ops_t uart_ops = { .init = uart_init };
    static device_ops_t spi_ops = { .init = spi_init };

    switch (type) {
        case DEV_UART: return &uart_ops;
        case DEV_SPI:  return &spi_ops;
        default:       return NULL;
    }
}
```

## 二、结构型模式

### 2.1 适配器模式(Adapter)

**用途：** 统一不同硬件接口

```c
// 不同传感器的统一接口
typedef struct {
    float (*read_temperature)(void);
    float (*read_humidity)(void);
} sensor_adapter_t;

// DS18B20适配器
static float ds18b20_read_temp(void) {
    return ds18b20_get_temperature();
}

// DHT22适配器
static float dht22_read_temp(void) {
    return dht22.temperature;
}

sensor_adapter_t ds18b20_adapter = {
    .read_temperature = ds18b20_read_temp,
    .read_humidity = NULL,
};

sensor_adapter_t dht22_adapter = {
    .read_temperature = dht22_read_temp,
    .read_humidity = dht22_read_humidity,
};
```

### 2.2 外观模式(Facade)

**用途：** 简化复杂子系统

```c
// 复杂的WiFi子系统
typedef struct {
    void (*init)(void);
    void (*connect)(const char *ssid, const char *pass);
    void (*disconnect)(void);
    int (*is_connected)(void);
} wifi_facade_t;

void wifi_system_init(void) {
    // 内部：初始化射频、协议栈、DHCP等
    rf_init();
    tcpip_init();
    dhcp_start();
}

void wifi_connect(const char *ssid, const char *pass) {
    // 内部：扫描、认证、关联、获取IP
    wifi_scan();
    wifi_auth(ssid, pass);
    wifi_associate();
    dhcp_request();
}
```

### 2.3 代理模式(Proxy)

**用途：** 控制对硬件的访问

```c
typedef struct {
    int (*read)(uint32_t addr, uint8_t *data, int len);
    int (*write)(uint32_t addr, uint8_t *data, int len);
} flash_proxy_t;

static int flash_locked = 1;

int flash_proxy_write(uint32_t addr, uint8_t *data, int len) {
    if (flash_locked) {
        return -1;  // 拒绝写入
    }
    return flash_write(addr, data, len);
}

void flash_unlock(void) { flash_locked = 0; }
void flash_lock(void) { flash_locked = 1; }
```

## 三、行为型模式

### 3.1 观察者模式(Observer)

**用途：** 事件通知系统

```c
#define MAX_OBSERVERS 10

typedef void (*observer_cb)(void *data);

typedef struct {
    observer_cb callbacks[MAX_OBSERVERS];
    int count;
} subject_t;

void subject_init(subject_t *s) {
    s->count = 0;
}

void subject_attach(subject_t *s, observer_cb cb) {
    if (s->count < MAX_OBSERVERS) {
        s->callbacks[s->count++] = cb;
    }
}

void subject_notify(subject_t *s, void *data) {
    for (int i = 0; i < s->count; i++) {
        s->callbacks[i](data);
    }
}

// 使用
subject_t button_event;

void led_handler(void *data) {
    led_toggle();
}

void buzzer_handler(void *data) {
    buzzer_beep();
}

void app_init(void) {
    subject_init(&button_event);
    subject_attach(&button_event, led_handler);
    subject_attach(&button_event, buzzer_handler);
}

void button_isr(void) {
    subject_notify(&button_event, NULL);
}
```

### 3.2 命令模式(Command)

**用途：** 命令队列、撤销/重做

```c
typedef void (*command_execute)(void *ctx);
typedef void (*command_undo)(void *ctx);

typedef struct {
    command_execute execute;
    command_undo undo;
    void *context;
} command_t;

#define CMD_QUEUE_SIZE 16
static command_t cmd_queue[CMD_QUEUE_SIZE];
static int cmd_head = 0, cmd_tail = 0;

void cmd_enqueue(command_t cmd) {
    cmd_queue[cmd_tail] = cmd;
    cmd_tail = (cmd_tail + 1) % CMD_QUEUE_SIZE;
}

void cmd_execute_next(void) {
    if (cmd_head != cmd_tail) {
        command_t *cmd = &cmd_queue[cmd_head];
        cmd->execute(cmd->context);
        cmd_head = (cmd_head + 1) % CMD_QUEUE_SIZE;
    }
}
```

### 3.3 策略模式(Strategy)

**用途：** 运行时切换算法

```c
typedef int (*sort_strategy)(int *arr, int len);

int bubble_sort(int *arr, int len) { /* ... */ return 0; }
int quick_sort(int *arr, int len) { /* ... */ return 0; }

typedef struct {
    sort_strategy sort;
} sorter_t;

void sorter_init(sorter_t *s, sort_strategy strategy) {
    s->sort = strategy;
}

void sorter_execute(sorter_t *s, int *arr, int len) {
    s->sort(arr, len);
}

// 使用
sorter_t sorter;
sorter_init(&sorter, quick_sort);
sorter_execute(&sorter, data, data_len);
```

**用途：** 定义算法骨架，子类实现细节

## 四、并发模式

### 4.1 生产者-消费者

```c
#include "freertos/FreeRTOS.h"
#include "freertos/queue.h"

QueueHandle_t data_queue;

void producer_task(void *arg) {
    sensor_data_t data;
    while (1) {
        data = read_sensor();
        xQueueSend(data_queue, &data, portMAX_DELAY);
        vTaskDelay(pdMS_TO_TICKS(1000));
    }
}

void consumer_task(void *arg) {
    sensor_data_t data;
    while (1) {
        if (xQueueReceive(data_queue, &data, portMAX_DELAY)) {
            process_data(&data);
        }
    }
}

void app_init(void) {
    data_queue = xQueueCreate(10, sizeof(sensor_data_t));
    xTaskCreate(producer_task, "producer", 2048, NULL, 5, NULL);
    xTaskCreate(consumer_task, "consumer", 2048, NULL, 5, NULL);
}
```

### 4.2 读者-写者

```c
#include "freertos/semphr.h"

SemaphoreHandle_t rw_mutex;
SemaphoreHandle_t read_count_mutex;
int read_count = 0;

void reader_enter(void) {
    xSemaphoreTake(read_count_mutex, portMAX_DELAY);
    read_count++;
    if (read_count == 1) {
        xSemaphoreTake(rw_mutex, portMAX_DELAY);
    }
    xSemaphoreGive(read_count_mutex);
}

void reader_exit(void) {
    xSemaphoreTake(read_count_mutex, portMAX_DELAY);
    read_count--;
    if (read_count == 0) {
        xSemaphoreGive(rw_mutex);
    }
    xSemaphoreGive(read_count_mutex);
}

void writer_enter(void) {
    xSemaphoreTake(rw_mutex, portMAX_DELAY);
}

void writer_exit(void) {
    xSemaphoreGive(rw_mutex);
}
```

### 4.3 任务链

```c
typedef struct task_node {
    void (*run)(void *data);
    void *data;
    struct task_node *next;
} task_node_t;

static task_node_t *task_chain = NULL;

void chain_add(task_node_t *node) {
    node->next = task_chain;
    task_chain = node;
}

void chain_execute(void) {
    task_node_t *current = task_chain;
    while (current) {
        current->run(current->data);
        current = current->next;
    }
}
```

## 五、状态管理模式

### 5.1 状态机(前文已述)

### 5.2 层次状态机(HSM)

```c
typedef enum {
    STATE_IDLE,
    STATE_RUNNING,
    STATE_RUNNING_FAST,
    STATE_RUNNING_SLOW,
    STATE_ERROR,
} state_t;

typedef struct {
    state_t parent;
    state_t current;
    void (*entry)(void);
    void (*exit)(void);
} hsm_state_t;

static hsm_state_t states[] = {
    [STATE_IDLE] = { .parent = STATE_IDLE, .entry = idle_entry, .exit = idle_exit },
    [STATE_RUNNING] = { .parent = STATE_RUNNING, .entry = running_entry, .exit = running_exit },
    [STATE_RUNNING_FAST] = { .parent = STATE_RUNNING, .entry = fast_entry, .exit = fast_exit },
    [STATE_RUNNING_SLOW] = { .parent = STATE_RUNNING, .entry = slow_entry, .exit = slow_exit },
};

static state_t current_state = STATE_IDLE;

void hsm_transition(state_t new_state) {
    // 退出当前状态
    states[current_state].exit();

    // 如果父状态不同，也退出
    if (states[new_state].parent != states[current_state].parent) {
        states[states[current_state].parent].exit();
    }

    // 进入新状态
    if (states[new_state].parent != states[current_state].parent) {
        states[states[new_state].parent].entry();
    }
    states[new_state].entry();

    current_state = new_state;
}
```

## 六、资源管理模式

### 6.1 资源池

```c
#define POOL_SIZE 8

typedef struct {
    uint8_t *buffer;
    int in_use;
} pool_item_t;

static uint8_t pool_memory[POOL_SIZE][1024];
static pool_item_t pool[POOL_SIZE];

void *pool_alloc(void) {
    for (int i = 0; i < POOL_SIZE; i++) {
        if (!pool[i].in_use) {
            pool[i].in_use = 1;
            return pool[i].buffer;
        }
    }
    return NULL;
}

void pool_free(void *ptr) {
    for (int i = 0; i < POOL_SIZE; i++) {
        if (pool[i].buffer == ptr) {
            pool[i].in_use = 0;
            return;
        }
    }
}
```

### 6.2 对象池(环形缓冲)

```c
typedef struct {
    uint8_t *buffer;
    int size;
    int head;
    int tail;
    int count;
} ring_buffer_t;

void ring_init(ring_buffer_t *rb, uint8_t *buf, int size) {
    rb->buffer = buf;
    rb->size = size;
    rb->head = 0;
    rb->tail = 0;
    rb->count = 0;
}

int ring_put(ring_buffer_t *rb, uint8_t data) {
    if (rb->count >= rb->size) return -1;
    rb->buffer[rb->head] = data;
    rb->head = (rb->head + 1) % rb->size;
    rb->count++;
    return 0;
}

int ring_get(ring_buffer_t *rb, uint8_t *data) {
    if (rb->count <= 0) return -1;
    *data = rb->buffer[rb->tail];
    rb->tail = (rb->tail + 1) % rb->size;
    rb->count--;
    return 0;
}
```

## 七、通信模式

### 7.1 发布-订阅

```c
#define MAX_TOPICS 8
#define MAX_SUBSCRIBERS 4

typedef void (*subscriber_cb)(const char *topic, void *data, int len);

typedef struct {
    const char *topic;
    subscriber_cb subscribers[MAX_SUBSCRIBERS];
    int count;
} pubsub_topic_t;

static pubsub_topic_t topics[MAX_TOPICS];
static int topic_count = 0;

void pubsub_subscribe(const char *topic, subscriber_cb cb) {
    for (int i = 0; i < topic_count; i++) {
        if (strcmp(topics[i].topic, topic) == 0) {
            topics[i].subscribers[topics[i].count++] = cb;
            return;
        }
    }
    // 新topic
    topics[topic_count].topic = topic;
    topics[topic_count].subscribers[0] = cb;
    topics[topic_count].count = 1;
    topic_count++;
}

void pubsub_publish(const char *topic, void *data, int len) {
    for (int i = 0; i < topic_count; i++) {
        if (strcmp(topics[i].topic, topic) == 0) {
            for (int j = 0; j < topics[i].count; j++) {
                topics[i].subscribers[j](topic, data, len);
            }
            return;
        }
    }
}
```
