## 一、代码优化

### 1.1 算法优化

```c
// 快速排序（优化版）
void quicksort(int *arr, int left, int right) {
    if (left >= right) return;

    // 三数取中法选择pivot
    int mid = left + (right - left) / 2;
    if (arr[left] > arr[mid]) swap(&arr[left], &arr[mid]);
    if (arr[left] > arr[right]) swap(&arr[left], &arr[right]);
    if (arr[mid] > arr[right]) swap(&arr[mid], &arr[right]);

    int pivot = arr[mid];
    swap(&arr[mid], &arr[right - 1]);

    int i = left, j = right - 1;
    while (1) {
        while (arr[++i] < pivot);
        while (arr[--j] > pivot);
        if (i >= j) break;
        swap(&arr[i], &arr[j]);
    }
    swap(&arr[i], &arr[right - 1]);

    quicksort(arr, left, i - 1);
    quicksort(arr, i + 1, right);
}

// 位运算优化
int count_bits_set(uint32_t x) {
    x = x - ((x >> 1) & 0x55555555);
    x = (x & 0x33333333) + ((x >> 2) & 0x33333333);
    x = (x + (x >> 4)) & 0x0F0F0F0F;
    x = x + (x >> 8);
    x = x + (x >> 16);
    return x & 0x0000003F;
}

// 查表法
static const uint8_t bit_count_table[256] = {
    0,1,1,2,1,2,2,3,1,2,2,3,2,3,3,4,
    // ... 256 entries
};

int count_bits_table(uint32_t x) {
    return bit_count_table[x & 0xFF] +
           bit_count_table[(x >> 8) & 0xFF] +
           bit_count_table[(x >> 16) & 0xFF] +
           bit_count_table[(x >> 24) & 0xFF];
}
```

```c
// 紧凑数据结构
typedef struct {
    uint32_t id : 12;      // 12位ID
    uint32_t type : 4;     // 4位类型
    uint32_t status : 4;   // 4位状态
    uint32_t value : 12;   // 12位值
} packed_record_t;  // 4字节

// 环形缓冲区（无锁）
typedef struct {
    uint8_t *buffer;
    size_t size;
    volatile size_t head;
    volatile size_t tail;
} lock_free_ring_t;

bool ring_push(lock_free_ring_t *ring, uint8_t data) {
    size_t next = (ring->head + 1) % ring->size;
    if (next == ring->tail) return false;  // 满

    ring->buffer[ring->head] = data;
    __DMB();  // 数据内存屏障
    ring->head = next;
    return true;
}

bool ring_pop(lock_free_ring_t *ring, uint8_t *data) {
    if (ring->head == ring->tail) return false;  // 空

    *data = ring->buffer[ring->tail];
    __DMB();
    ring->tail = (ring->tail + 1) % ring->size;
    return true;
}

// 内存池
typedef struct {
    uint8_t *pool;
    size_t block_size;
    size_t block_count;
    uint32_t *bitmap;
} memory_pool_t;

void *pool_alloc(memory_pool_t *pool) {
    for (size_t i = 0; i < pool->block_count / 32; i++) {
        if (pool->bitmap[i] != 0xFFFFFFFF) {
            int bit = __builtin_ffz(pool->bitmap[i]);
            pool->bitmap[i] |= (1 << bit);
            return pool->pool + (i * 32 + bit) * pool->block_size;
        }
    }
    return NULL;
}

void pool_free(memory_pool_t *pool, void *ptr) {
    size_t offset = (uint8_t *)ptr - pool->pool;
    size_t block = offset / pool->block_size;
    pool->bitmap[block / 32] &= ~(1 << (block % 32));
}
```

## 二、内存优化

### 2.1 内存分配优化

```c
// 内存分配跟踪
typedef struct {
    void *ptr;
    size_t size;
    const char *file;
    int line;
    uint32_t timestamp;
} alloc_record_t;

#define MAX_ALLOC_RECORDS 100
static alloc_record_t alloc_records[MAX_ALLOC_RECORDS];
static int alloc_count = 0;
static size_t total_allocated = 0;
static size_t peak_allocated = 0;

void *tracked_malloc(size_t size, const char *file, int line) {
    void *ptr = malloc(size);
    if (!ptr) return NULL;

    if (alloc_count < MAX_ALLOC_RECORDS) {
        alloc_records[alloc_count].ptr = ptr;
        alloc_records[alloc_count].size = size;
        alloc_records[alloc_count].file = file;
        alloc_records[alloc_count].line = line;
        alloc_records[alloc_count].timestamp = get_tick_count();
        alloc_count++;
    }

    total_allocated += size;
    if (total_allocated > peak_allocated) {
        peak_allocated = total_allocated;
    }

    return ptr;
}

void tracked_free(void *ptr, const char *file, int line) {
    for (int i = 0; i < alloc_count; i++) {
        if (alloc_records[i].ptr == ptr) {
            total_allocated -= alloc_records[i].size;
            alloc_records[i] = alloc_records[--alloc_count];
            break;
        }
    }
    free(ptr);
}

#define malloc(size) tracked_malloc(size, __FILE__, __LINE__)
#define free(ptr) tracked_free(ptr, __FILE__, __LINE__)

// 内存使用报告
void memory_report(void) {
    printf("\n=== Memory Report ===\n");
    printf("Total allocated: %zu bytes\n", total_allocated);
    printf("Peak allocated: %zu bytes\n", peak_allocated);
    printf("Active allocations: %d\n", alloc_count);

    printf("\nTop allocations:\n");
    // 按大小排序
    for (int i = 0; i < alloc_count - 1; i++) {
        for (int j = i + 1; j < alloc_count; j++) {
            if (alloc_records[j].size > alloc_records[i].size) {
                alloc_record_t temp = alloc_records[i];
                alloc_records[i] = alloc_records[j];
                alloc_records[j] = temp;
            }
        }
    }

    for (int i = 0; i < 10 && i < alloc_count; i++) {
        printf("  %zu bytes at %s:%d\n",
               alloc_records[i].size,
               alloc_records[i].file,
               alloc_records[i].line);
    }
}
```

### 2.2 内存碎片整理

```c
// 内存碎片检测
typedef struct {
    size_t total_free;
    size_t largest_block;
    int fragment_count;
    float fragmentation;  // 碎片率
} memory_fragmentation_t;

memory_fragmentation_t analyze_fragmentation(void) {
    memory_fragmentation_t frag = {0};

    // 遍历空闲链表
    heap_block_t *block = free_list;
    while (block) {
        frag.total_free += block->size;
        frag.fragment_count++;

        if (block->size > frag.largest_block) {
            frag.largest_block = block->size;
        }

        block = block->next;
    }

    // 计算碎片率
    if (frag.total_free > 0) {
        frag.fragmentation = 1.0f - (float)frag.largest_block / frag.total_free;
    }

    return frag;
}

// 内存整理
void memory_defragment(void) {
    // 合并相邻空闲块
    heap_block_t *block = free_list;
    while (block && block->next) {
        if ((uint8_t *)block + block->size + sizeof(heap_block_t) ==
            (uint8_t *)block->next) {
            // 合并
            block->size += block->next->size + sizeof(heap_block_t);
            block->next = block->next->next;
        } else {
            block = block->next;
        }
    }
}
```

## 三、实时性优化

### 3.1 中断优化

```c
// 中断延迟测量
typedef struct {
    uint32_t min_latency;
    uint32_t max_latency;
    uint32_t avg_latency;
    uint32_t sample_count;
} interrupt_latency_t;

static interrupt_latency_t latency_stats = {
    .min_latency = UINT32_MAX,
    .max_latency = 0,
};

void measure_interrupt_latency(uint32_t trigger_time) {
    uint32_t now = get_tick_us();
    uint32_t latency = now - trigger_time;

    if (latency < latency_stats.min_latency) {
        latency_stats.min_latency = latency;
    }
    if (latency > latency_stats.max_latency) {
        latency_stats.max_latency = latency;
    }

    latency_stats.avg_latency =
        (latency_stats.avg_latency * latency_stats.sample_count + latency) /
        (latency_stats.sample_count + 1);
    latency_stats.sample_count++;
}

// 中断优先级优化
void optimize_interrupt_priorities(void) {
    // 高优先级：时间关键
    NVIC_SetPriority(TIM2_IRQn, 0);  // 最高
    NVIC_SetPriority(USART1_IRQn, 1);

    // 中优先级：一般外设
    NVIC_SetPriority(SPI1_IRQn, 2);
    NVIC_SetPriority(I2C1_IRQn, 3);

    // 低优先级：非关键
    NVIC_SetPriority(DMA1_Channel1_IRQn, 4);
    NVIC_SetPriority(ADC1_IRQn, 5);
}
```

### 3.2 任务调度优化

```c
// 任务执行时间测量
typedef struct {
    const char *name;
    uint32_t min_time;
    uint32_t max_time;
    uint32_t avg_time;
    uint32_t exec_count;
    uint32_t deadline_miss;
} task_timing_t;

#define MAX_TASKS 16
static task_timing_t task_timings[MAX_TASKS];
static int task_count = 0;

void task_timing_start(int task_id) {
    task_timings[task_id].min_time = get_tick_us();
}

void task_timing_end(int task_id, uint32_t deadline_us) {
    uint32_t elapsed = get_tick_us() - task_timings[task_id].min_time;

    if (elapsed < task_timings[task_id].min_time) {
        task_timings[task_id].min_time = elapsed;
    }
    if (elapsed > task_timings[task_id].max_time) {
        task_timings[task_id].max_time = elapsed;
    }

    task_timings[task_id].avg_time =
        (task_timings[task_id].avg_time * task_timings[task_id].exec_count + elapsed) /
        (task_timings[task_id].exec_count + 1);
    task_timings[task_id].exec_count++;

    if (elapsed > deadline_us) {
        task_timings[task_id].deadline_miss++;
    }
}

// 任务调度分析
void print_task_timing_report(void) {
    printf("\n=== Task Timing Report ===\n");
    printf("%-20s %10s %10s %10s %10s %10s\n",
           "Task", "Min(us)", "Max(us)", "Avg(us)", "Count", "Misses");
    printf("--------------------------------------------------------------\n");

    for (int i = 0; i < task_count; i++) {
        printf("%-20s %10lu %10lu %10lu %10lu %10lu\n",
               task_timings[i].name,
               task_timings[i].min_time,
               task_timings[i].max_time,
               task_timings[i].avg_time,
               task_timings[i].exec_count,
               task_timings[i].deadline_miss);
    }
}
```

### 4.1 DMA优化

```c
// DMA传输配置
typedef struct {
    DMA_Channel_TypeDef *channel;
    uint32_t direction;
    uint32_t src_addr;
    uint32_t dst_addr;
    uint32_t size;
    bool circular;
    void (*callback)(void);
} dma_config_t;

// DMA传输
void dma_transfer(dma_config_t *config) {
    // 配置DMA
    config->channel->CCR = 0;
    config->channel->CPAR = config->dst_addr;
    config->channel->CMAR = config->src_addr;
    config->channel->CNDTR = config->size;

    uint32_t ccr = DMA_CCR_MINC | DMA_CCR_TCIE;
    if (config->circular) {
        ccr |= DMA_CCR_CIRC;
    }
    config->channel->CCR = ccr | DMA_CCR_EN;
}

// 零拷贝缓冲区
typedef struct {
    uint8_t *buffer;
    size_t size;
    volatile size_t write_pos;
    volatile size_t read_pos;
    bool is_full;
} zero_copy_buffer_t;

size_t zcb_write(zero_copy_buffer_t *buf, const uint8_t *data, size_t len) {
    size_t written = 0;
    while (written < len && !buf->is_full) {
        buf->buffer[buf->write_pos] = data[written];
        buf->write_pos = (buf->write_pos + 1) % buf->size;
        if (buf->write_pos == buf->read_pos) {
            buf->is_full = true;
        }
        written++;
    }
    return written;
}

size_t zcb_read(zero_copy_buffer_t *buf, uint8_t *data, size_t len) {
    size_t read = 0;
    while (read < len && (buf->read_pos != buf->write_pos || buf->is_full)) {
        data[read] = buf->buffer[buf->read_pos];
        buf->read_pos = (buf->read_pos + 1) % buf->size;
        buf->is_full = false;
        read++;
    }
    return read;
}
```

### 4.2 协议优化

```c
// 协议压缩
typedef struct {
    uint8_t *input;
    size_t input_len;
    uint8_t *output;
    size_t output_len;
} protocol_compress_t;

// 简单RLE压缩
size_t rle_compress(const uint8_t *input, size_t input_len,
                    uint8_t *output, size_t output_len) {
    size_t out_pos = 0;
    size_t in_pos = 0;

    while (in_pos < input_len && out_pos < output_len - 2) {
        uint8_t value = input[in_pos];
        uint8_t count = 1;

        while (in_pos + count < input_len &&
               input[in_pos + count] == value &&
               count < 255) {
            count++;
        }

        output[out_pos++] = count;
        output[out_pos++] = value;
        in_pos += count;
    }

    return out_pos;
}

// 协议批处理
typedef struct {
    uint8_t *buffer;
    size_t buffer_size;
    size_t current_size;
    uint32_t last_send;
    uint32_t batch_timeout;
} batch_processor_t;

void batch_add(batch_processor_t *bp, const uint8_t *data, size_t len) {
    if (bp->current_size + len > bp->buffer_size) {
        batch_flush(bp);
    }

    memcpy(bp->buffer + bp->current_size, data, len);
    bp->current_size += len;
}

void batch_flush(batch_processor_t *bp) {
    if (bp->current_size > 0) {
        send_data(bp->buffer, bp->current_size);
        bp->current_size = 0;
        bp->last_send = get_tick_count();
    }
}

void batch_check_timeout(batch_processor_t *bp) {
    if (bp->current_size > 0 &&
        get_tick_count() - bp->last_send > bp->batch_timeout) {
        batch_flush(bp);
    }
}
```

### 5.1 动态功耗管理

```c
// 功耗状态机
typedef enum {
    POWER_STATE_ACTIVE,
    POWER_STATE_IDLE,
    POWER_STATE_SLEEP,
    POWER_STATE_DEEP_SLEEP,
} power_state_t;

typedef struct {
    power_state_t current_state;
    uint32_t idle_timeout_ms;
    uint32_t sleep_timeout_ms;
    uint32_t last_activity;
} power_manager_t;

void power_manager_update(power_manager_t *pm) {
    uint32_t now = get_tick_count();
    uint32_t idle_time = now - pm->last_activity;

    switch (pm->current_state) {
        case POWER_STATE_ACTIVE:
            if (idle_time > pm->idle_timeout_ms) {
                pm->current_state = POWER_STATE_IDLE;
                enter_idle_mode();
            }
            break;

        case POWER_STATE_IDLE:
            if (idle_time > pm->sleep_timeout_ms) {
                pm->current_state = POWER_STATE_SLEEP;
                enter_sleep_mode();
            }
            break;

        case POWER_STATE_SLEEP:
            // 等待唤醒中断
            break;
    }
}

void power_manager_wakeup(power_manager_t *pm) {
    pm->last_activity = get_tick_count();
    if (pm->current_state != POWER_STATE_ACTIVE) {
        pm->current_state = POWER_STATE_ACTIVE;
        exit_low_power_mode();
    }
}
```
