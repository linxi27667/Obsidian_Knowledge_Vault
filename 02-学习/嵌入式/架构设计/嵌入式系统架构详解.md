# 嵌入式系统架构详解

## 核心概念

- **分层架构** - 层次化设计
- **模块化** - 高内聚低耦合
- **状态机** - 事件驱动设计
- **实时性** - 时间约束

---

## 一、架构模式

### 1.1 分层架构

```c
// 分层架构示例
// 应用层 - 业务逻辑
// 中间件层 - RTOS、协议栈
// 驱动层 - 外设驱动
// 硬件抽象层 - HAL

// 硬件抽象层
typedef struct {
    void (*gpio_set)(int pin, int level);
    int (*gpio_get)(int pin);
    void (*uart_send)(int port, const uint8_t *data, size_t len);
    int (*uart_recv)(int port, uint8_t *buf, size_t len);
    void (*spi_transfer)(int port, const uint8_t *tx, uint8_t *rx, size_t len);
    void (*i2c_write)(int port, uint8_t addr, const uint8_t *data, size_t len);
    int (*i2c_read)(int port, uint8_t addr, uint8_t *buf, size_t len);
} hal_interface_t;

// 驱动层
typedef struct {
    const char *name;
    int (*init)(void);
    int (*read)(void *buf, size_t len);
    int (*write)(const void *buf, size_t len);
    int (*ioctl)(int cmd, void *arg);
    void (*deinit)(void);
} driver_t;

// 注册驱动
#define MAX_DRIVERS 32
static driver_t *drivers[MAX_DRIVERS];
static int driver_count = 0;

int driver_register(driver_t *drv) {
    if (driver_count >= MAX_DRIVERS) return -1;
    drivers[driver_count++] = drv;
    return 0;
}

driver_t *driver_find(const char *name) {
    for (int i = 0; i < driver_count; i++) {
        if (strcmp(drivers[i]->name, name) == 0) {
            return drivers[i];
        }
    }
    return NULL;
}
```

### 1.2 事件驱动架构

```c
// 事件系统
typedef enum {
    EVENT_NONE = 0,
    EVENT_BUTTON_PRESS,
    EVENT_BUTTON_RELEASE,
    EVENT_SENSOR_DATA,
    EVENT_TIMER_EXPIRED,
    EVENT_UART_RECEIVED,
    EVENT_WIFI_CONNECTED,
    EVENT_WIFI_DISCONNECTED,
    EVENT_MQTT_MESSAGE,
    EVENT_MAX,
} event_type_t;

typedef struct {
    event_type_t type;
    uint32_t timestamp;
    void *data;
    size_t data_len;
} event_t;

// 事件队列
#define EVENT_QUEUE_SIZE 32
static event_t event_queue[EVENT_QUEUE_SIZE];
static int event_head = 0;
static int event_tail = 0;
static int event_count = 0;

// 发送事件
int event_send(event_type_t type, void *data, size_t len) {
    if (event_count >= EVENT_QUEUE_SIZE) return -1;

    event_t *evt = &event_queue[event_head];
    evt->type = type;
    evt->timestamp = get_tick_count();
    evt->data = data;
    evt->data_len = len;

    event_head = (event_head + 1) % EVENT_QUEUE_SIZE;
    event_count++;
    return 0;
}

// 接收事件
int event_receive(event_t *evt) {
    if (event_count <= 0) return -1;

    *evt = event_queue[event_tail];
    event_tail = (event_tail + 1) % EVENT_QUEUE_SIZE;
    event_count--;
    return 0;
}

// 事件处理器
typedef void (*event_handler_t)(event_t *evt);
static event_handler_t event_handlers[EVENT_MAX] = {0};

void event_register_handler(event_type_t type, event_handler_t handler) {
    event_handlers[type] = handler;
}

// 事件分发
void event_dispatch(void) {
    event_t evt;
    while (event_receive(&evt) == 0) {
        if (evt.type < EVENT_MAX && event_handlers[evt.type]) {
            event_handlers[evt.type](&evt);
        }
    }
}
```

### 1.3 状态机架构

```c
// 状态机
typedef enum {
    STATE_IDLE,
    STATE_INIT,
    STATE_RUNNING,
    STATE_ERROR,
    STATE_SLEEP,
    STATE_MAX,
} state_t;

typedef enum {
    ACTION_NONE,
    ACTION_ENTER,
    ACTION_EXIT,
    ACTION_UPDATE,
} action_t;

typedef state_t (*state_handler_t)(action_t action, void *data);

// 状态表
typedef struct {
    state_t state;
    state_handler_t handler;
} state_entry_t;

static state_entry_t state_table[STATE_MAX];
static state_t current_state = STATE_IDLE;

// 注册状态
void state_register(state_t state, state_handler_t handler) {
    state_table[state].state = state;
    state_table[state].handler = handler;
}

// 状态转换
void state_transition(state_t new_state, void *data) {
    // 退出当前状态
    if (state_table[current_state].handler) {
        state_table[current_state].handler(ACTION_EXIT, data);
    }

    // 进入新状态
    current_state = new_state;
    if (state_table[current_state].handler) {
        state_table[current_state].handler(ACTION_ENTER, data);
    }
}

// 状态更新
void state_update(void *data) {
    if (state_table[current_state].handler) {
        state_t next = state_table[current_state].handler(ACTION_UPDATE, data);
        if (next != current_state) {
            state_transition(next, data);
        }
    }
}

// 示例状态处理
state_t idle_handler(action_t action, void *data) {
    switch (action) {
        case ACTION_ENTER:
            printf("进入空闲状态\n");
            break;
        case ACTION_UPDATE:
            if (button_pressed()) {
                return STATE_RUNNING;
            }
            break;
        case ACTION_EXIT:
            printf("退出空闲状态\n");
            break;
    }
    return STATE_IDLE;
}

state_t running_handler(action_t action, void *data) {
    switch (action) {
        case ACTION_ENTER:
            printf("进入运行状态\n");
            break;
        case ACTION_UPDATE:
            // 执行任务
            do_work();
            if (error_occurred()) {
                return STATE_ERROR;
            }
            if (should_sleep()) {
                return STATE_SLEEP;
            }
            break;
        case ACTION_EXIT:
            printf("退出运行状态\n");
            break;
    }
    return STATE_RUNNING;
}
```

## 二、模块化设计

### 2.1 模块接口

```c
// 模块接口定义
typedef struct {
    const char *name;
    int version;
    int (*init)(void);
    int (*start)(void);
    int (*stop)(void);
    int (*deinit)(void);
    void (*update)(void);
    int (*ioctl)(int cmd, void *arg);
} module_interface_t;

// 模块注册
#define MAX_MODULES 16
static module_interface_t *modules[MAX_MODULES];
static int module_count = 0;

int module_register(module_interface_t *mod) {
    if (module_count >= MAX_MODULES) return -1;
    modules[module_count++] = mod;
    return 0;
}

// 初始化所有模块
int modules_init_all(void) {
    for (int i = 0; i < module_count; i++) {
        if (modules[i]->init) {
            int ret = modules[i]->init();
            if (ret != 0) {
                printf("Module %s init failed: %d\n", modules[i]->name, ret);
                return ret;
            }
        }
    }
    return 0;
}

// 启动所有模块
int modules_start_all(void) {
    for (int i = 0; i < module_count; i++) {
        if (modules[i]->start) {
            int ret = modules[i]->start();
            if (ret != 0) {
                printf("Module %s start failed: %d\n", modules[i]->name, ret);
                return ret;
            }
        }
    }
    return 0;
}

// 更新所有模块
void modules_update_all(void) {
    for (int i = 0; i < module_count; i++) {
        if (modules[i]->update) {
            modules[i]->update();
        }
    }
}
```

### 2.2 依赖管理

```c
// 模块依赖
typedef struct {
    const char *module_name;
    const char *dependencies[8];
    int dep_count;
} module_dependency_t;

// 依赖检查
int check_dependencies(module_dependency_t *deps, int count) {
    for (int i = 0; i < count; i++) {
        for (int j = 0; j < deps[i].dep_count; j++) {
            bool found = false;
            for (int k = 0; k < module_count; k++) {
                if (strcmp(modules[k]->name, deps[i].dependencies[j]) == 0) {
                    found = true;
                    break;
                }
            }
            if (!found) {
                printf("Missing dependency: %s for %s\n",
                       deps[i].dependencies[j], deps[i].module_name);
                return -1;
            }
        }
    }
    return 0;
}

// 拓扑排序（初始化顺序）
void topological_sort(module_dependency_t *deps, int count, int *order) {
    int in_degree[count];
    memset(in_degree, 0, sizeof(in_degree));

    // 计算入度
    for (int i = 0; i < count; i++) {
        for (int j = 0; j < deps[i].dep_count; j++) {
            for (int k = 0; k < count; k++) {
                if (strcmp(modules[k]->name, deps[i].dependencies[j]) == 0) {
                    in_degree[i]++;
                }
            }
        }
    }

    // 排序
    int sorted = 0;
    bool visited[count];
    memset(visited, 0, sizeof(visited));

    while (sorted < count) {
        bool found = false;
        for (int i = 0; i < count; i++) {
            if (!visited[i] && in_degree[i] == 0) {
                order[sorted++] = i;
                visited[i] = true;

                // 更新入度
                for (int j = 0; j < count; j++) {
                    for (int k = 0; k < deps[j].dep_count; k++) {
                        if (strcmp(modules[j].dependencies[k], modules[i]->name) == 0) {
                            in_degree[j]--;
                        }
                    }
                }

                found = true;
                break;
            }
        }
        if (!found) break;
    }
}
```

## 三、实时系统设计

### 3.1 实时任务

```c
// 实时任务
typedef struct {
    const char *name;
    uint32_t period_ms;       // 周期
    uint32_t deadline_ms;     // 截止时间
    uint32_t execution_ms;    // 执行时间
    uint32_t priority;        // 优先级
    void (*task_func)(void);  // 任务函数
    bool is_periodic;         // 是否周期性
} rt_task_t;

// 最早截止时间优先调度
typedef struct {
    rt_task_t *tasks;
    int task_count;
    uint32_t current_time;
} edf_scheduler_t;

// EDF调度
int edf_schedule(edf_scheduler_t *sched) {
    int earliest_idx = -1;
    uint32_t earliest_deadline = UINT32_MAX;

    for (int i = 0; i < sched->task_count; i++) {
        uint32_t absolute_deadline = sched->tasks[i].deadline_ms;
        if (absolute_deadline < earliest_deadline) {
            earliest_deadline = absolute_deadline;
            earliest_idx = i;
        }
    }

    return earliest_idx;
}

// 速率单调调度
int rms_compare(const void *a, const void *b) {
    const rt_task_t *ta = (const rt_task_t *)a;
    const rt_task_t *tb = (const rt_task_t *)b;
    return ta->period_ms - tb->period_ms;
}

void rms_schedule(rt_task_t *tasks, int count) {
    // 按周期排序（周期越短优先级越高）
    qsort(tasks, count, sizeof(rt_task_t), rms_compare);
}
```

### 3.2 时间管理

```c
// 软件定时器
typedef struct {
    uint32_t interval_ms;
    uint32_t elapsed_ms;
    bool is_active;
    bool is_periodic;
    void (*callback)(void);
} soft_timer_t;

#define MAX_TIMERS 32
static soft_timer_t timers[MAX_TIMERS];
static int timer_count = 0;

int timer_create(uint32_t interval_ms, bool periodic, void (*callback)(void)) {
    if (timer_count >= MAX_TIMERS) return -1;

    soft_timer_t *t = &timers[timer_count];
    t->interval_ms = interval_ms;
    t->elapsed_ms = 0;
    t->is_active = true;
    t->is_periodic = periodic;
    t->callback = callback;

    return timer_count++;
}

void timer_tick(uint32_t elapsed_ms) {
    for (int i = 0; i < timer_count; i++) {
        if (!timers[i].is_active) continue;

        timers[i].elapsed_ms += elapsed_ms;
        if (timers[i].elapsed_ms >= timers[i].interval_ms) {
            if (timers[i].callback) {
                timers[i].callback();
            }

            if (timers[i].is_periodic) {
                timers[i].elapsed_ms = 0;
            } else {
                timers[i].is_active = false;
            }
        }
    }
}

// 时间戳
typedef struct {
    uint32_t seconds;
    uint32_t microseconds;
} timestamp_t;

timestamp_t timestamp_now(void) {
    timestamp_t ts;
    uint32_t tick = get_tick_count();
    ts.seconds = tick / 1000;
    ts.microseconds = (tick % 1000) * 1000;
    return ts;
}

uint32_t timestamp_diff_ms(timestamp_t *a, timestamp_t *b) {
    return (a->seconds - b->seconds) * 1000 + (a->microseconds - b->microseconds) / 1000;
}
```

## 四、通信架构

### 4.1 发布-订阅模式

```c
// 主题
typedef struct {
    const char *name;
    void (*subscribers[8])(const void *data, size_t len);
    int subscriber_count;
} topic_t;

#define MAX_TOPICS 16
static topic_t topics[MAX_TOPICS];
static int topic_count = 0;

// 创建主题
int pubsub_create_topic(const char *name) {
    if (topic_count >= MAX_TOPICS) return -1;

    topics[topic_count].name = name;
    topics[topic_count].subscriber_count = 0;
    return topic_count++;
}

// 订阅主题
int pubsub_subscribe(const char *topic_name, void (*callback)(const void *data, size_t len)) {
    for (int i = 0; i < topic_count; i++) {
        if (strcmp(topics[i].name, topic_name) == 0) {
            if (topics[i].subscriber_count >= 8) return -1;
            topics[i].subscribers[topics[i].subscriber_count++] = callback;
            return 0;
        }
    }
    return -1;
}

// 发布消息
int pubsub_publish(const char *topic_name, const void *data, size_t len) {
    for (int i = 0; i < topic_count; i++) {
        if (strcmp(topics[i].name, topic_name) == 0) {
            for (int j = 0; j < topics[i].subscriber_count; j++) {
                topics[i].subscribers[j](data, len);
            }
            return 0;
        }
    }
    return -1;
}
```

### 4.2 命令模式

```c
// 命令接口
typedef struct {
    const char *name;
    const char *description;
    int (*execute)(int argc, char *argv[]);
} command_t;

#define MAX_COMMANDS 32
static command_t commands[MAX_COMMANDS];
static int command_count = 0;

// 注册命令
int command_register(const char *name, const char *desc, int (*execute)(int, char **)) {
    if (command_count >= MAX_COMMANDS) return -1;

    commands[command_count].name = name;
    commands[command_count].description = desc;
    commands[command_count].execute = execute;
    command_count++;
    return 0;
}

// 执行命令
int command_execute(const char *input) {
    char buf[256];
    strncpy(buf, input, sizeof(buf) - 1);

    // 解析参数
    char *argv[16];
    int argc = 0;
    char *token = strtok(buf, " ");
    while (token && argc < 16) {
        argv[argc++] = token;
        token = strtok(NULL, " ");
    }

    if (argc == 0) return -1;

    // 查找并执行命令
    for (int i = 0; i < command_count; i++) {
        if (strcmp(commands[i].name, argv[0]) == 0) {
            return commands[i].execute(argc, argv);
        }
    }

    printf("Unknown command: %s\n", argv[0]);
    return -1;
}

// 列出所有命令
void command_list(void) {
    for (int i = 0; i < command_count; i++) {
        printf("  %-16s %s\n", commands[i].name, commands[i].description);
    }
}
```

## 五、错误处理

### 5.1 错误管理

```c
// 错误码
typedef enum {
    ERR_NONE = 0,
    ERR_INVALID_PARAM,
    ERR_OUT_OF_MEMORY,
    ERR_TIMEOUT,
    ERR_NOT_FOUND,
    ERR_BUSY,
    ERR_IO,
    ERR_COMM,
    ERR_STATE,
    ERR_MAX,
} error_code_t;

// 错误信息
typedef struct {
    error_code_t code;
    const char *file;
    int line;
    uint32_t timestamp;
    char message[64];
} error_info_t;

#define MAX_ERROR_HISTORY 16
static error_info_t error_history[MAX_ERROR_HISTORY];
static int error_head = 0;
static int error_count = 0;

// 记录错误
void error_record(error_code_t code, const char *file, int line, const char *msg) {
    error_info_t *err = &error_history[error_head];
    err->code = code;
    err->file = file;
    err->line = line;
    err->timestamp = get_tick_count();
    strncpy(err->message, msg, sizeof(err->message) - 1);

    error_head = (error_head + 1) % MAX_ERROR_HISTORY;
    if (error_count < MAX_ERROR_HISTORY) error_count++;
}

#define ERROR_RECORD(code, msg) error_record(code, __FILE__, __LINE__, msg)

// 获取最近错误
error_info_t *error_get_last(void) {
    if (error_count == 0) return NULL;
    int idx = (error_head - 1 + MAX_ERROR_HISTORY) % MAX_ERROR_HISTORY;
    return &error_history[idx];
}

// 错误恢复
typedef void (*error_recovery_t)(error_code_t code);
static error_recovery_t recovery_handlers[ERR_MAX] = {0};

void error_register_recovery(error_code_t code, error_recovery_t handler) {
    recovery_handlers[code] = handler;
}

void error_try_recover(error_code_t code) {
    if (code < ERR_MAX && recovery_handlers[code]) {
        recovery_handlers[code](code);
    }
}
```

## 附录：架构对比

| 架构 | 优点 | 缺点 | 适用场景 |
|------|------|------|----------|
| 分层架构 | 清晰、易维护 | 性能损失 | 通用系统 |
| 事件驱动 | 响应快、解耦 | 调试难 | GUI、网络 |
| 状态机 | 逻辑清晰 | 复杂度高 | 协议、控制 |
| 微内核 | 可扩展 | IPC开销 | 大型系统 |
