# 数据结构与算法

## 核心概念

- **数据结构** - 数据的组织、管理和存储方式
- **算法** - 解决问题的步骤和方法
- **时间复杂度** - 算法执行时间随数据规模增长的趋势
- **空间复杂度** - 算法所需内存空间随数据规模增长的趋势

## 一、复杂度分析

```c
// 头插法
void insert_head(struct node **head, int data) {

}

// 尾插法
void insert_tail(struct node **head, int data) {
    struct node *new = create_node(data);
    if (*head == NULL) {
        *head = new;
        return;
    }
    struct node *curr = *head;
    while (curr->next != NULL) {
        curr = curr->next;
    }
    curr->next = new;
}

// 删除节点
void delete_node(struct node **head, int data) {
    struct node *curr = *head;
    struct node *prev = NULL;
    while (curr != NULL) {
        if (curr->data == data) {
            if (prev == NULL) {
                *head = curr->next;
            } else {
                prev->next = curr->next;
            }
            free(curr);
            return;
        }
        prev = curr;
        curr = curr->next;
    }
}
```

### 2.3 栈

**特点：** LIFO（后进先出）

**顺序栈：**
```c
struct stack {
    int data[MAX_SIZE];
    int top;
};

void push(struct stack *s, int val) {
    if (s->top < MAX_SIZE - 1) {
        s->data[++s->top] = val;
    }
}

int pop(struct stack *s) {
    if (s->top >= 0) {
        return s->data[s->top--];
    }
    return -1;
}
```

**应用：**
- 函数调用栈
- 表达式求值

### 2.4 队列

**特点：** FIFO（先进先出）

**循环队列：**
```c
struct queue {
    int data[MAX_SIZE];
    int front;
    int rear;
    int count;
};

void enqueue(struct queue *q, int val) {
    if (q->count < MAX_SIZE) {
        q->data[q->rear] = val;
        q->rear = (q->rear + 1) % MAX_SIZE;
        q->count++;
    }
}

int dequeue(struct queue *q) {
    if (q->count > 0) {
        int val = q->data[q->front];
        q->front = (q->front + 1) % MAX_SIZE;
        q->count--;
        return val;
    }
    return -1;
}
```

**应用：**
- 任务调度
- 缓冲区
- BFS

### 3.3 二叉搜索树(BST)

**性质：**

- 右子树所有节点 > 根节点

```c
        return root;
    }
    if (val < root->data) {
        return search(root->left, val);
    }
    return search(root->right, val);
}
```

**插入：**
```c
struct tree* insert(struct tree *root, int val) {
    if (root == NULL) {
        return create_node(val);
    }
    if (val < root->data) {
        root->left = insert(root->left, val);
    } else if (val > root->data) {
        root->right = insert(root->right, val);
    }
    return root;
}
```

**删除：**
```c
struct tree* delete(struct tree *root, int val) {
    if (root == NULL) return NULL;
    if (val < root->data) {
        root->left = delete(root->left, val);
    } else if (val > root->data) {
        root->right = delete(root->right, val);
    } else {
        if (root->left == NULL) {
            struct tree *temp = root->right;
            free(root);
            return temp;
        }
        if (root->right == NULL) {
            struct tree *temp = root->left;
            free(root);
            return temp;
        }
        struct tree *temp = find_min(root->right);
        root->data = temp->data;
        root->right = delete(root->right, temp->data);
    }
    return root;
}
```

```c
int visited[MAX];

    printf("%d ", v);
```

```c
void bfs(int start) {
    int visited[MAX] = {0};

    init_queue(&q);
    enqueue(&q, start);
    visited[start] = 1;
    while (!is_empty(&q)) {
        int v = dequeue(&q);
        printf("%d ", v);
        struct node *curr = adj[v];
        while (curr != NULL) {
            if (!visited[curr->vertex]) {
                enqueue(&q, curr->vertex);
                visited[curr->vertex] = 1;

            curr = curr->next;

    }
}
```

### 4.3 最短路径

**Dijkstra算法：**
```c
int dist[MAX];
int visited[MAX];

void dijkstra(int start, int n) {
    for (int i = 0; i < n; i++) {
        dist[i] = INF;
        visited[i] = 0;
    }
    dist[start] = 0;
    
    for (int i = 0; i < n; i++) {
        int u = -1;
        for (int j = 0; j < n; j++) {
            if (!visited[j] && (u == -1 || dist[j] < dist[u])) {
                u = j;
            }
        }
        if (dist[u] == INF) break;
        visited[u] = 1;
        
        struct node *curr = adj[u];
        while (curr != NULL) {
            int v = curr->vertex;
            int w = curr->weight;
            if (dist[u] + w < dist[v]) {
                dist[v] = dist[u] + w;
            }
            curr = curr->next;
        }
    }
}
```

**Floyd算法：**
```c
int dist[MAX][MAX];

void floyd(int n) {
    for (int k = 0; k < n; k++) {
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (dist[i][k] + dist[k][j] < dist[i][j]) {
                    dist[i][j] = dist[i][k] + dist[k][j];
                }
            }
        }
    }
}
```

### 4.4 最小生成树

**Prim算法：**
```c
int parent[MAX];
int key[MAX];
int visited[MAX];

void prim(int n) {
    for (int i = 0; i < n; i++) {
        key[i] = INF;
        visited[i] = 0;
    }
    key[0] = 0;
    parent[0] = -1;
    
    for (int i = 0; i < n-1; i++) {
        int u = -1;
        for (int j = 0; j < n; j++) {
            if (!visited[j] && (u == -1 || key[j] < key[u])) {
                u = j;
            }
        }
        visited[u] = 1;
        
        struct node *curr = adj[u];
        while (curr != NULL) {
            int v = curr->vertex;
            int w = curr->weight;
            if (!visited[v] && w < key[v]) {
                parent[v] = u;
                key[v] = w;
            }
            curr = curr->next;
        }
    }
}
```

### 5.3 插入排序

```c
void insertion_sort(int arr[], int n) {
    for (int i = 1; i < n; i++) {
        int key = arr[i];
        int j = i - 1;
        while (j >= 0 && arr[j] > key) {
            arr[j+1] = arr[j];

        }
        arr[j+1] = key;
```

### 5.4 快速排序

**时间复杂度：** O(n log n)
**空间复杂度：** O(log n)
**稳定性：** 不稳定

```c
int partition(int arr[], int low, int high) {
    int pivot = arr[high];
    int i = low - 1;
    for (int j = low; j < high; j++) {
        if (arr[j] <= pivot) {
            i++;
            swap(&arr[i], &arr[j]);
        }
    }
    swap(&arr[i+1], &arr[high]);
    return i + 1;
}

void quick_sort(int arr[], int low, int high) {
    if (low < high) {
        int pi = partition(arr, low, high);
        quick_sort(arr, low, pi - 1);
        quick_sort(arr, pi + 1, high);
    }
}
```

### 5.5 归并排序

**时间复杂度：** O(n log n)
**空间复杂度：** O(n)
**稳定性：** 稳定

```c
void merge(int arr[], int left, int mid, int right) {
    int n1 = mid - left + 1;
    int n2 = right - mid;
    int L[n1], R[n2];
    
    for (int i = 0; i < n1; i++) L[i] = arr[left + i];
    for (int i = 0; i < n2; i++) R[i] = arr[mid + 1 + i];
    
    int i = 0, j = 0, k = left;
    while (i < n1 && j < n2) {
        if (L[i] <= R[j]) {
            arr[k++] = L[i++];
        } else {
            arr[k++] = R[j++];
        }
    }
    while (i < n1) arr[k++] = L[i++];
    while (j < n2) arr[k++] = R[j++];
}

void merge_sort(int arr[], int left, int right) {
    if (left < right) {
        int mid = left + (right - left) / 2;
        merge_sort(arr, left, mid);
        merge_sort(arr, mid + 1, right);
        merge(arr, left, mid, right);
    }
}
```

### 5.6 堆排序

**时间复杂度：** O(n log n)
**空间复杂度：** O(1)
**稳定性：** 不稳定

```c
void heapify(int arr[], int n, int i) {
    int largest = i;
    int left = 2 * i + 1;
    int right = 2 * i + 2;
    
    if (left < n && arr[left] > arr[largest]) {
        largest = left;
    }
    if (right < n && arr[right] > arr[largest]) {
        largest = right;
    }
    if (largest != i) {
        swap(&arr[i], &arr[largest]);
        heapify(arr, n, largest);
    }
}

void heap_sort(int arr[], int n) {
    for (int i = n / 2 - 1; i >= 0; i--) {
        heapify(arr, n, i);
    }
    for (int i = n - 1; i > 0; i--) {
        swap(&arr[0], &arr[i]);
        heapify(arr, i, 0);
    }
}
```

```c
int binary_search(int arr[], int n, int target) {
    int left = 0, right = n - 1;
    while (left <= right) {
        int mid = left + (right - left) / 2;
        if (arr[mid] == target) {
            return mid;
        } else if (arr[mid] < target) {
            left = mid + 1;
        } else {
            right = mid - 1;
        }
    }

}
```

### 8.2 经典问题

**背包问题：**
```c
int dp[MAX_W + 1];

int knapsack(int weights[], int values[], int n, int W) {
    memset(dp, 0, sizeof(dp));
    for (int i = 0; i < n; i++) {
        for (int w = W; w >= weights[i]; w--) {
            dp[w] = max(dp[w], dp[w - weights[i]] + values[i]);
        }
    }
    return dp[W];
}
```

**最长公共子序列：**
```c
int dp[MAX][MAX];

int lcs(char *X, char *Y, int m, int n) {
    for (int i = 0; i <= m; i++) {
        for (int j = 0; j <= n; j++) {
            if (i == 0 || j == 0) {
                dp[i][j] = 0;
            } else if (X[i-1] == Y[j-1]) {
                dp[i][j] = dp[i-1][j-1] + 1;
            } else {
                dp[i][j] = max(dp[i-1][j], dp[i][j-1]);
            }
        }
    }
    return dp[m][n];
}
```

```c
    // 按结束时间排序
    sort_by_finish_time(start, finish, n);
    
    int last_finish = 0;
    for (int i = 0; i < n; i++) {
        if (start[i] >= last_finish) {
            printf("Activity %d\n", i);
            last_finish = finish[i];
        }
    }
}
```

## 十、嵌入式常用数据结构

### 10.1 环形缓冲区

```c
struct ring_buffer {
    uint8_t *buffer;
    int head;
    int tail;
    int size;
    int count;
};

void ring_buffer_put(struct ring_buffer *rb, uint8_t data) {
    if (rb->count < rb->size) {
        rb->buffer[rb->head] = data;
        rb->head = (rb->head + 1) % rb->size;
        rb->count++;
    }
}

uint8_t ring_buffer_get(struct ring_buffer *rb) {
    if (rb->count > 0) {
        uint8_t data = rb->buffer[rb->tail];
        rb->tail = (rb->tail + 1) % rb->size;
        rb->count--;
        return data;
    }
    return 0;
}
```

**应用：** 串口缓冲、音频流

### 10.2 位图

```c
#define BITMAP_SIZE 32

uint32_t bitmap[BITMAP_SIZE / 32];

void set_bit(int pos) {
    bitmap[pos / 32] |= (1 << (pos % 32));
}

void clear_bit(int pos) {
    bitmap[pos / 32] &= ~(1 << (pos % 32));
}

int get_bit(int pos) {
    return (bitmap[pos / 32] >> (pos % 32)) & 1;
}
```

## 附录：算法模板

### 二分查找

```c
int binary_search(int arr[], int n, int target) {
    int left = 0, right = n - 1;
    while (left <= right) {
        int mid = left + (right - left) / 2;
        if (arr[mid] == target) return mid;
        if (arr[mid] < target) left = mid + 1;
        else right = mid - 1;
    }
    return -1;
}
```

### 快速排序

```c
void quick_sort(int arr[], int low, int high) {
    if (low < high) {
        int pi = partition(arr, low, high);
        quick_sort(arr, low, pi - 1);
        quick_sort(arr, pi + 1, high);
    }
}
```

### BFS

```c
void bfs(int start) {
    int visited[MAX] = {0};
    struct queue q;
    init_queue(&q);
    enqueue(&q, start);
    visited[start] = 1;
    while (!is_empty(&q)) {
        int v = dequeue(&q);
        // 处理v
        for (每个邻接点u) {
            if (!visited[u]) {
                enqueue(&q, u);
                visited[u] = 1;
            }
        }
    }
}
```

### DFS

```c
void dfs(int v) {
    visited[v] = 1;
    // 处理v
    for (每个邻接点u) {
        if (!visited[u]) {
            dfs(u);
        }
    }
}
```
