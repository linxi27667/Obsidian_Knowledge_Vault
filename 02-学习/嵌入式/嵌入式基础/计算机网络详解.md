# 计算机网络详解

## 核心概念

- **OSI模型** - 七层网络模型
- **路由** - 数据包转发
- **NAT** - 网络地址转换
- **网络安全** - 防火墙、加密

## 一、OSI模型

### 1.1 七层架构

```
┌─────────────────────────────────────┐
│  7. 应用层    │ HTTP、MQTT、DNS     │  ← 用户接口
├─────────────────────────────────────┤
│  6. 表示层    │ SSL/TLS、编码       │  ← 数据格式
├─────────────────────────────────────┤
│  5. 会话层    │ Socket、RPC         │  ← 连接管理
├─────────────────────────────────────┤
│  4. 传输层    │ TCP、UDP            │  ← 端到端
├─────────────────────────────────────┤
│  3. 网络层    │ IP、ICMP、路由      │  ← 路由转发
├─────────────────────────────────────┤
│  2. 数据链路层│ Ethernet、WiFi      │  ← 帧传输
├─────────────────────────────────────┤
│  1. 物理层    │ 网线、无线电        │  ← 比特传输
└─────────────────────────────────────┘
```

### 1.2 封装与解封装

```c
// 数据封装过程
/*
 * 应用层数据: "Hello"
 *     ↓ + TCP头(20字节)
 * 传输层段: [TCP头][数据]
 *     ↓ + IP头(20字节)
 * 网络层包: [IP头][TCP头][数据]
 *     ↓ + 以太网头(14字节) + FCS(4字节)
 * 链路层帧: [以太网头][IP头][TCP头][数据][FCS]
 */

// 以太网帧
typedef struct {
    uint8_t  dst_mac[6];
    uint8_t  src_mac[6];
    uint16_t ethertype;     // 0x0800=IPv4, 0x0806=ARP
} ethernet_header_t;

// ARP协议
typedef struct {
    uint16_t hw_type;       // 硬件类型(1=以太网)
    uint16_t proto_type;    // 协议类型(0x0800=IPv4)
    uint8_t  hw_len;        // 硬件地址长度(6)
    uint8_t  proto_len;     // 协议地址长度(4)
    uint16_t opcode;        // 1=请求, 2=回复
    uint8_t  sender_mac[6];
    uint32_t sender_ip;
    uint8_t  target_mac[6];
    uint32_t target_ip;
} arp_header_t;

// ARP请求
void arp_request(uint32_t target_ip) {
    arp_header_t arp;
    arp.hw_type = htons(1);
    arp.proto_type = htons(0x0800);
    arp.hw_len = 6;
    arp.proto_len = 4;
    arp.opcode = htons(1);  // ARP请求
    memcpy(arp.sender_mac, local_mac, 6);
    arp.sender_ip = local_ip;
    memset(arp.target_mac, 0, 6);
    arp.target_ip = target_ip;

    // 广播发送
    uint8_t broadcast[6] = {0xFF, 0xFF, 0xFF, 0xFF, 0xFF, 0xFF};
    ethernet_send(broadcast, 0x0806, &arp, sizeof(arp));
}
```

## 二、IP路由

### 2.1 路由表

```c
// 路由表条目
typedef struct {
    uint32_t network;     // 网络地址
    uint32_t netmask;     // 子网掩码
    uint32_t gateway;     // 网关
    char interface[8];    // 接口
    uint32_t metric;      // 优先级
} route_entry_t;

// 路由表
route_entry_t routing_table[] = {
    {0xC0A80100, 0xFFFFFF00, 0, "eth0", 10},      // 192.168.1.0/24 直连
    {0x00000000, 0x00000000, 0xC0A80101, "eth0", 100},  // 默认路由
};

// 路由查找
uint32_t route_lookup(uint32_t dest_ip) {
    uint32_t best_match = 0;
    int best_metric = 999999;

    for (int i = 0; i < sizeof(routing_table)/sizeof(routing_table[0]); i++) {
        uint32_t network = dest_ip & routing_table[i].netmask;
        if (network == routing_table[i].network) {
            if (routing_table[i].metric < best_metric) {
                best_metric = routing_table[i].metric;
                best_match = routing_table[i].gateway;
            }
        }
    }

    return best_match;
}

// IP转发
void ip_forward(uint8_t *packet, int len) {
    ip_header_t *ip = (ip_header_t *)packet;

    // TTL检查
    if (ip->ttl <= 1) {
        icmp_send_time_exceeded(ip->src_addr);
        return;
    }

    // 路由查找
    uint32_t next_hop = route_lookup(ip->dst_addr);
    if (next_hop == 0) {
        icmp_send_dest_unreachable(ip->src_addr);
        return;
    }

    // 转发
    ip->ttl--;
    ip->checksum = 0;
    ip->checksum = ip_checksum((uint16_t *)ip, ip->ihl * 4);

    // ARP查找下一跳MAC
    uint8_t next_mac[6];
    arp_resolve(next_hop, next_mac);

    // 发送
    ethernet_send(next_mac, 0x0800, packet, len);
}
```

### 2.2 子网划分

```c
// 子网计算
typedef struct {
    uint32_t network;
    uint32_t broadcast;
    uint32_t first_host;
    uint32_t last_host;
    uint32_t num_hosts;
    uint32_t netmask;
} subnet_info_t;

subnet_info_t calc_subnet(uint32_t ip, int prefix_len) {
    subnet_info_t info;

    // 子网掩码
    info.netmask = htonl(~((1 << (32 - prefix_len)) - 1));

    // 网络地址
    info.network = ip & info.netmask;

    // 广播地址
    info.broadcast = info.network | ~info.netmask;

    // 主机范围
    info.first_host = info.network + 1;
    info.last_host = info.broadcast - 1;
    info.num_hosts = (1 << (32 - prefix_len)) - 2;

    return info;
}

// CIDR表示
void print_subnet(const char *cidr) {
    uint32_t ip;
    int prefix;
    sscanf(cidr, "%u.%u.%u.%u/%d",
           &ip, &prefix);

    subnet_info_t info = calc_subnet(ip, prefix);
    printf("Network: %08X\n", info.network);
    printf("Mask: %08X\n", info.netmask);
    printf("Hosts: %u\n", info.num_hosts);
}
```

## 三、NAT(网络地址转换)

### 3.1 NAT类型

```c
// NAT类型
typedef enum {
    NAT_STATIC,          // 静态NAT(1:1映射)
    NAT_DYNAMIC,         // 动态NAT(地址池)
    NAT_PAT,             // 端口地址转换(N:1)
} nat_type_t;

// NAT表条目
typedef struct {
    uint32_t local_ip;
    uint16_t local_port;
    uint32_t global_ip;
    uint16_t global_port;
    uint8_t protocol;
    uint32_t timestamp;
    bool active;
} nat_entry_t;

#define MAX_NAT_ENTRIES 256
static nat_entry_t nat_table[MAX_NAT_ENTRIES];

// PAT实现(最常用)
uint16_t nat_translate_outgoing(uint32_t local_ip, uint16_t local_port,
                                uint8_t protocol) {
    // 查找已有映射
    for (int i = 0; i < MAX_NAT_ENTRIES; i++) {
        if (nat_table[i].active &&
            nat_table[i].local_ip == local_ip &&
            nat_table[i].local_port == local_port &&
            nat_table[i].protocol == protocol) {
            nat_table[i].timestamp = get_tick();
            return nat_table[i].global_port;
        }
    }

    // 创建新映射
    uint16_t global_port = find_free_port();
    for (int i = 0; i < MAX_NAT_ENTRIES; i++) {
        if (!nat_table[i].active) {
            nat_table[i].local_ip = local_ip;
            nat_table[i].local_port = local_port;
            nat_table[i].global_ip = wan_ip;
            nat_table[i].global_port = global_port;
            nat_table[i].protocol = protocol;
            nat_table[i].timestamp = get_tick();
            nat_table[i].active = true;
            break;
        }
    }

    return global_port;
}

// NAT入站翻译
nat_entry_t *nat_translate_incoming(uint16_t global_port, uint8_t protocol) {
    for (int i = 0; i < MAX_NAT_ENTRIES; i++) {
        if (nat_table[i].active &&
            nat_table[i].global_port == global_port &&
            nat_table[i].protocol == protocol) {
            return &nat_table[i];
        }
    }
    return NULL;
}
```

## 四、网络安全

### 4.1 防火墙

```c
// 防火墙规则
typedef enum {
    FW_ALLOW,
    FW_DENY,
    FW_LOG,
} fw_action_t;

typedef struct {
    uint32_t src_ip;
    uint32_t src_mask;
    uint32_t dst_ip;
    uint32_t dst_mask;
    uint16_t src_port;
    uint16_t dst_port;
    uint8_t protocol;
    fw_action_t action;
    bool enabled;
} firewall_rule_t;

#define MAX_FW_RULES 32
static firewall_rule_t fw_rules[MAX_FW_RULES];

// 防火墙检查
fw_action_t firewall_check(uint32_t src_ip, uint32_t dst_ip,
                           uint16_t src_port, uint16_t dst_port,
                           uint8_t protocol) {
    for (int i = 0; i < MAX_FW_RULES; i++) {
        if (!fw_rules[i].enabled) continue;

        // 匹配规则
        if ((src_ip & fw_rules[i].src_mask) !=
            (fw_rules[i].src_ip & fw_rules[i].src_mask)) continue;
        if ((dst_ip & fw_rules[i].dst_mask) !=
            (fw_rules[i].dst_ip & fw_rules[i].dst_mask)) continue;
        if (fw_rules[i].src_port && fw_rules[i].src_port != src_port) continue;
        if (fw_rules[i].dst_port && fw_rules[i].dst_port != dst_port) continue;
        if (fw_rules[i].protocol && fw_rules[i].protocol != protocol) continue;

        // 匹配成功
        if (fw_rules[i].action == FW_LOG) {
            printf("Firewall log: %08X:%d → %08X:%d proto=%d\n",
                   src_ip, src_port, dst_ip, dst_port, protocol);
            continue;
        }
        return fw_rules[i].action;
    }

    return FW_DENY;  // 默认拒绝
}
```

### 4.2 入侵检测

```c
// 简单入侵检测
typedef struct {
    uint32_t ip;
    uint32_t failed_attempts;
    uint32_t last_attempt;
    bool blocked;
} ids_entry_t;

#define IDS_BLOCK_THRESHOLD 5
#define IDS_BLOCK_DURATION  300000  // 5分钟

bool ids_check(uint32_t src_ip) {
    // 查找记录
    for (int i = 0; i < MAX_IDS_ENTRIES; i++) {
        if (ids_table[i].ip == src_ip) {
            if (ids_table[i].blocked) {
                if (get_tick() - ids_table[i].last_attempt > IDS_BLOCK_DURATION) {
                    ids_table[i].blocked = false;
                    ids_table[i].failed_attempts = 0;
                    return true;  // 解除封锁
                }
                return false;  // 仍然封锁
            }
            return true;  // 正常
        }
    }
    return true;
}

void ids_record_failure(uint32_t src_ip) {
    for (int i = 0; i < MAX_IDS_ENTRIES; i++) {
        if (ids_table[i].ip == src_ip) {
            ids_table[i].failed_attempts++;
            ids_table[i].last_attempt = get_tick();
            if (ids_table[i].failed_attempts >= IDS_BLOCK_THRESHOLD) {
                ids_table[i].blocked = true;
                printf("IDS: Blocked %08X\n", src_ip);
            }
            return;
        }
    }
}
```

## 五、DHCP

### 5.1 DHCP客户端

```c
// DHCP消息
typedef struct {
    uint8_t  op;           // 1=请求, 2=回复
    uint8_t  htype;        // 硬件类型
    uint8_t  hlen;         // 硬件地址长度
    uint8_t  hops;
    uint32_t xid;          // 事务ID
    uint16_t secs;
    uint16_t flags;
    uint32_t ciaddr;       // 客户端IP
    uint32_t yiaddr;       // 你的IP(服务器分配)
    uint32_t siaddr;       // 服务器IP
    uint32_t giaddr;       // 网关IP
    uint8_t  chaddr[16];   // 客户端MAC
    uint8_t  sname[64];    // 服务器名
    uint8_t  file[128];    // 启动文件
    uint8_t  options[312]; // 选项
} dhcp_message_t;

// DHCP状态机
typedef enum {
    DHCP_INIT,
    DHCP_SELECTING,
    DHCP_REQUESTING,
    DHCP_BOUND,
    DHCP_RENEWING,
    DHCP_REBINDING,
} dhcp_state_t;

// DHCP发现
void dhcp_discover(void) {
    dhcp_message_t msg = {0};
    msg.op = 1;  // BOOTREQUEST
    msg.htype = 1;  // 以太网
    msg.hlen = 6;
    msg.xid = random_u32();
    memcpy(msg.chaddr, local_mac, 6);

    // 添加选项
    int opt_offset = 0;
    // DHCP消息类型
    msg.options[opt_offset++] = 53;  // Type
    msg.options[opt_offset++] = 1;
    msg.options[opt_offset++] = 1;   // Discover
    // 客户端标识符
    msg.options[opt_offset++] = 61;
    msg.options[opt_offset++] = 7;
    msg.options[opt_offset++] = 1;   // 以太网
    memcpy(msg.options + opt_offset, local_mac, 6);
    opt_offset += 6;
    // 结束
    msg.options[opt_offset++] = 0xFF;

    // 广播发送
    udp_send(0xFFFFFFFF, 67, 68, &msg, sizeof(msg));
}
```

## 六、ICMP

### 6.1 Ping实现

```c
// ICMP Echo请求
void ping(uint32_t target_ip, int count) {
    for (int i = 0; i < count; i++) {
        icmp_header_t icmp = {
            .type = 8,  // Echo Request
            .code = 0,
            .id = htons(getpid()),
            .sequence = htons(i),
        };

        // 添加时间戳
        uint32_t timestamp = get_tick();
        memcpy(&icmp + 1, &timestamp, 4);

        icmp.checksum = 0;
        icmp.checksum = ip_checksum((uint16_t *)&icmp, sizeof(icmp));

        uint32_t start = get_tick();
        ip_send(target_ip, IP_PROTO_ICMP, &icmp, sizeof(icmp));

        // 等待回复
        uint8_t reply[64];
        int len = ip_recv(reply, sizeof(reply), 1000);
        if (len > 0) {
            uint32_t rtt = get_tick() - start;
            printf("Reply from %s: time=%ums\n",
                   ip_to_str(target_ip), rtt);
        } else {
            printf("Request timeout\n");
        }

        vTaskDelay(pdMS_TO_TICKS(1000));
    }
}
```
