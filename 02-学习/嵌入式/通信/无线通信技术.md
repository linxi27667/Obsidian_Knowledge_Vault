# 无线通信技术

## 核心概念

- **无线通信** - 电磁波传输数据
- **频段** - 工作频率范围
- **调制** - 将数据加载到载波
- **协议** - 通信规则

---

## 一、无线通信基础

### 1.1 频段

| 频段 | 频率 | 特点 | 应用 |
|------|------|------|------|
| 2.4GHz | 2.4-2.5GHz | 全球通用 | WiFi/BLE/Zigbee |
| Sub-GHz | 433/868/915MHz | 穿透强 | LoRa/ISM |
| 5GHz | 5.1-5.8GHz | 速率高 | WiFi 5/6 |
| 60GHz | 57-71GHz | 超高速 | WiFi 7 |

---

### 1.2 调制方式

| 调制 | 说明 | 速率 |
|------|------|------|
| ASK | 幅移键控 | 低 |
| FSK | 频移键控 | 中 |
| PSK | 相移键控 | 中高 |
| QAM | 正交幅度调制 | 高 |

---

### 1.3 关键参数

| 参数 | 说明 |
|------|------|
| 灵敏度 | 能接收的最小信号 |
| 发射功率 | 发送信号强度 |
| 链路预算 | 发射功率-灵敏度 |
| 路径损耗 | 信号衰减 |

### 2.2 ESP32 WiFi

```c
#include "esp_wifi.h"
#include "esp_event.h"

void wifi_init_sta(void) {
    esp_netif_init();
    esp_event_loop_create_default();
    esp_netif_create_default_wifi_sta();

    wifi_init_config_t cfg = WIFI_INIT_CONFIG_DEFAULT();
    esp_wifi_init(&cfg);

    wifi_config_t wifi_config = {
        .sta = {
            .ssid = "MyWiFi",
            .password = "password123",
        },
    };

    esp_wifi_set_mode(WIFI_MODE_STA);
    esp_wifi_set_config(WIFI_IF_STA, &wifi_config);
    esp_wifi_start();
    esp_wifi_connect();
}

// 事件处理
static void wifi_event_handler(void *arg, esp_event_base_t base,
                               int32_t id, void *data) {
    if (base == WIFI_EVENT && id == WIFI_EVENT_STA_CONNECTED) {
        printf("Connected\n");
    } else if (base == IP_EVENT && id == IP_EVENT_STA_GOT_IP) {
        ip_event_got_ip_t *event = data;
        printf("IP: " IPSTR, IP2STR(&event->ip_info.ip));
    }
}
```

### 2.3 WiFi扫描

```c
void wifi_scan(void) {
    wifi_scan_config_t scan_config = {
        .ssid = NULL,
        .bssid = NULL,
        .channel = 0,
        .show_hidden = true,
    };

    esp_wifi_scan_start(&scan_config, true);

    uint16_t ap_count = 0;
    esp_wifi_scan_get_ap_num(&ap_count);

    wifi_ap_record_t *ap_list = malloc(sizeof(wifi_ap_record_t) * ap_count);
    esp_wifi_scan_get_ap_records(&ap_count, ap_list);

    for (int i = 0; i < ap_count; i++) {
        printf("SSID: %s, RSSI: %d\n", ap_list[i].ssid, ap_list[i].rssi);
    }
    free(ap_list);
}
```

### 3.3 GATT服务

```c
// 属性表
enum {
    IDX_SVC,
    IDX_CHAR,
    IDX_CHAR_VAL,
    IDX_CHAR_CFG,
    HRS_IDX_NB,
};

static uint16_t handle_table[HRS_IDX_NB];

// 服务UUID
static const uint16_t GATTS_SERVICE_UUID = 0x180D;  // Heart Rate
static const uint16_t GATTS_CHAR_UUID = 0x2A37;     // Heart Rate Measurement

// GATT事件处理
static void gatts_event_handler(esp_gatts_cb_event_t event, esp_gatt_if_t gatts_if,
                                esp_ble_gatts_cb_param_t *param) {
    switch (event) {
        case ESP_GATTS_REG_EVT:
            // 创建服务
            esp_gatt_srvc_id_t service_id = {
                .is_primary = true,
                .id.inst_id = 0,
                .id.uuid.len = ESP_UUID_LEN_16,
                .id.uuid.uuid.uuid16 = GATTS_SERVICE_UUID,
            };
            esp_ble_gatts_create_service(gatts_if, &service_id, HRS_IDX_NB);
            break;

        case ESP_GATTS_CREATE_EVT:
            // 添加特征值
            esp_ble_gatts_add_char(handle_table[IDX_SVC],
                                   &char_uuid,
                                   ESP_GATT_PERM_READ | ESP_GATT_PERM_WRITE,
                                   ESP_GATT_CHAR_PROP_BIT_READ | ESP_GATT_CHAR_PROP_BIT_NOTIFY,
                                   &char_val, NULL);
            break;

        case ESP_GATTS_READ_EVT:
            // 读取处理
            esp_gatt_rsp_t rsp = {
                .attr_value.handle = param->read.handle,
                .attr_value.len = 1,
                .attr_value.value[0] = heart_rate,
            };
            esp_ble_gatts_send_response(gatts_if, param->read.trans_id,
                                         ESP_GATT_OK, &rsp);
            break;
    }
}
```

### 5.2 SX1276驱动

```c
#include "driver/spi_master.h"

#define REG_FIFO           0x00
#define REG_OP_MODE        0x01
#define REG_FRF_MSB        0x06
#define REG_FRF_MID        0x07
#define REG_FRF_LSB        0x08
#define REG_PA_CONFIG      0x09
#define REG_LNA            0x0C
#define REG_FIFO_ADDR_PTR  0x0D

#define MODE_LORA          0x80
#define MODE_SLEEP         0x00
#define MODE_STDBY         0x01
#define MODE_TX            0x03
#define MODE_RX_CONTINUOUS 0x05

void sx1276_init(void) {
    // SPI初始化
    spi_device_handle_t spi;
    // ...

    // 进入睡眠模式
    sx1276_write(REG_OP_MODE, MODE_LORA | MODE_SLEEP);

    // 设置频率 433MHz
    uint32_t frf = (uint32_t)(433000000.0 / 32000000.0 * (1 << 19));
    sx1276_write(REG_FRF_MSB, (frf >> 16) & 0xFF);
    sx1276_write(REG_FRF_MID, (frf >> 8) & 0xFF);
    sx1276_write(REG_FRF_LSB, frf & 0xFF);

    // 设置PA
    sx1276_write(REG_PA_CONFIG, 0x8F);  // 17dBm

    // 进入待机模式
    sx1276_write(REG_OP_MODE, MODE_LORA | MODE_STDBY);
}

void sx1276_send(uint8_t *data, int len) {
    // 设置FIFO地址
    sx1276_write(REG_FIFO_ADDR_PTR, 0x00);

    // 写入FIFO
    for (int i = 0; i < len; i++) {
        sx1276_write(REG_FIFO, data[i]);
    }

    // 设置包长度
    sx1276_write(0x22, len);

    // 发送
    sx1276_write(REG_OP_MODE, MODE_LORA | MODE_TX);

    // 等待发送完成
    while (!(sx1276_read(0x12) & 0x08)) {
        vTaskDelay(pdMS_TO_TICKS(10));
    }

    // 清除中断
    sx1276_write(0x12, 0xFF);
}
```

### 6.2 ESP-NOW通信

```c
#include "esp_now.h"
#include "esp_wifi.h"

// 对端MAC地址
uint8_t peer_mac[] = {0xAA, 0xBB, 0xCC, 0xDD, 0xEE, 0xFF};

void on_data_recv(const uint8_t *mac, const uint8_t *data, int len) {
    printf("Received from %02X:%02X:%02X:%02X:%02X:%02X: ",
           mac[0], mac[1], mac[2], mac[3], mac[4], mac[5]);
    printf("%.*s\n", len, data);
}

void espnow_init(void) {
    // 初始化WiFi
    wifi_init_config_t cfg = WIFI_INIT_CONFIG_DEFAULT();
    esp_wifi_init(&cfg);
    esp_wifi_set_mode(WIFI_MODE_STA);
    esp_wifi_start();

    // 初始化ESP-NOW
    esp_now_init();

    // 注册回调
    esp_now_register_recv_cb(on_data_recv);

    // 添加对端
    esp_now_peer_info_t peer = {
        .channel = 0,
        .encrypt = false,
    };
    memcpy(peer.peer_addr, peer_mac, 6);
    esp_now_add_peer(&peer);
}

void espnow_send(const char *message) {
    esp_now_send(peer_mac, (uint8_t *)message, strlen(message));
}
```

## 七、MQTT over Wireless

### 7.1 WiFi + MQTT

```c
#include "mqtt_client.h"

esp_mqtt_client_handle_t client;

void mqtt_event_handler(void *arg, esp_event_base_t base,
                        int32_t id, void *data) {
    esp_mqtt_event_handle_t event = data;

    switch (event->event_id) {
        case MQTT_EVENT_CONNECTED:
            printf("MQTT Connected\n");
            esp_mqtt_client_subscribe(client, "device/command", 1);
            break;
        case MQTT_EVENT_DATA:
            printf("Topic: %.*s\n", event->topic_len, event->topic);
            printf("Data: %.*s\n", event->data_len, event->data);
            break;
        case MQTT_EVENT_DISCONNECTED:
            printf("MQTT Disconnected\n");
            break;
    }
}

void mqtt_init(void) {
    esp_mqtt_client_config_t mqtt_cfg = {
        .broker.address.uri = "mqtt://broker.hivemq.com",
    };

    client = esp_mqtt_client_init(&mqtt_cfg);
    esp_mqtt_client_register_event(client, ESP_EVENT_ANY_ID, mqtt_event_handler, NULL);
    esp_mqtt_client_start(client);
}

void mqtt_publish_data(const char *topic, const char *data) {
    esp_mqtt_client_publish(client, topic, data, 0, 1, 0);
}
```
