## 一、Modbus协议

### 1.1 Modbus RTU

```c
// Modbus RTU帧格式
// [地址(1B)] [功能码(1B)] [数据(NB)] [CRC(2B)]

// Modbus功能码
#define MB_FC_READ_COILS           0x01
#define MB_FC_READ_DISCRETE        0x02
#define MB_FC_READ_HOLDING         0x03
#define MB_FC_READ_INPUT           0x04
#define MB_FC_WRITE_SINGLE_COIL    0x05
#define MB_FC_WRITE_SINGLE_REG     0x06
#define MB_FC_WRITE_MULTIPLE_COILS 0x0F
#define MB_FC_WRITE_MULTIPLE_REGS  0x10

// Modbus请求构建
typedef struct {
    uint8_t slave_addr;
    uint8_t function;
    uint16_t start_addr;
    uint16_t quantity;
    uint8_t *data;
    uint16_t data_len;
} mb_request_t;

// CRC-16计算
uint16_t modbus_crc16(uint8_t *data, int len) {
    uint16_t crc = 0xFFFF;

    for (int i = 0; i < len; i++) {
        crc ^= data[i];
        for (int j = 0; j < 8; j++) {
            if (crc & 1) {
                crc = (crc >> 1) ^ 0xA001;
            } else {
                crc >>= 1;
            }
        }
    }

    return crc;
}

// 读保持寄存器
int mb_read_holding_regs(uint8_t slave, uint16_t start, uint16_t count,
                         uint16_t *values) {
    uint8_t tx_buf[8];
    tx_buf[0] = slave;
    tx_buf[1] = MB_FC_READ_HOLDING;
    tx_buf[2] = start >> 8;
    tx_buf[3] = start & 0xFF;
    tx_buf[4] = count >> 8;
    tx_buf[5] = count & 0xFF;

    uint16_t crc = modbus_crc16(tx_buf, 6);
    tx_buf[6] = crc & 0xFF;
    tx_buf[7] = crc >> 8;

    uart_send(tx_buf, 8);

    // 接收响应
    uint8_t rx_buf[256];
    int len = uart_recv(rx_buf, sizeof(rx_buf), 100);

    // 验证CRC
    uint16_t rx_crc = (rx_buf[len-1] << 8) | rx_buf[len-2];
    if (modbus_crc16(rx_buf, len-2) != rx_crc) {
        return -1;  // CRC错误
    }

    // 解析数据
    int byte_count = rx_buf[2];
    for (int i = 0; i < count; i++) {
        values[i] = (rx_buf[3 + i*2] << 8) | rx_buf[4 + i*2];
    }

    return count;
}
```

### 1.2 Modbus TCP

```c
// Modbus TCP帧格式
// [MBAP头(7B)] [功能码(1B)] [数据(NB)]

// MBAP头
typedef struct {
    uint16_t transaction_id;  // 事务标识
    uint16_t protocol_id;     // 协议标识(0x0000)
    uint16_t length;          // 后续字节数
    uint8_t  unit_id;         // 单元标识
} mbap_header_t;

// Modbus TCP客户端
int mb_tcp_read_holding(const char *host, int port,
                        uint8_t unit_id, uint16_t start, uint16_t count,
                        uint16_t *values) {
    int sock = socket(AF_INET, SOCK_STREAM, 0);

    struct sockaddr_in addr;
    addr.sin_family = AF_INET;
    addr.sin_port = htons(port);
    inet_pton(AF_INET, host, &addr.sin_addr);

    connect(sock, (struct sockaddr *)&addr, sizeof(addr));

    // 构建请求
    uint8_t buf[260];
    mbap_header_t *mbap = (mbap_header_t *)buf;
    mbap->transaction_id = htons(next_txn_id++);
    mbap->protocol_id = htons(0);
    mbap->length = htons(6);
    mbap->unit_id = unit_id;

    buf[7] = MB_FC_READ_HOLDING;
    buf[8] = start >> 8;
    buf[9] = start & 0xFF;
    buf[10] = count >> 8;
    buf[11] = count & 0xFF;

    send(sock, buf, 12, 0);

    // 接收响应
    int len = recv(sock, buf, sizeof(buf), 0);

    // 解析
    int byte_count = buf[8];
    for (int i = 0; i < count; i++) {
        values[i] = (buf[9 + i*2] << 8) | buf[10 + i*2];
    }

    close(sock);
    return count;
}
```

    // 读取端点
    *len = usb_read_endpoint(CDC_EP_OUT, data, CDC_MAX_PACKET);

void cdc_control_request(usb_setup_packet_t *setup) {
    switch (setup->bRequest) {

            // 设置波特率等参数

            memcpy(&coding, setup->data, sizeof(coding));
            configure_uart(coding.dwDTERate, coding.bCharFormat,
                          coding.bParityType, coding.bDataBits);
            break;
        }
        case CDC_REQ_SET_CONTROL_LINE_STATE:
            // DTR/RTS信号
            break;
    }
}

## 三、以太网

### 3.1 以太网帧

```c
// 以太网帧结构
typedef struct {
    uint8_t  dst_mac[6];
    uint8_t  src_mac[6];
    uint16_t ethertype;
    // 数据(46-1500字节)
    // FCS(4字节)
} ethernet_frame_t;

// 常用Ethertype
#define ETHERTYPE_IPV4  0x0800
#define ETHERTYPE_ARP   0x0806
#define ETHERTYPE_IPV6  0x86DD
#define ETHERTYPE_VLAN  0x8100

// VLAN标签
typedef struct {
    uint16_t tpid;      // 0x8100
    uint16_t tci;       // 优先级(3) + CFI(1) + VLAN ID(12)
} vlan_tag_t;

// 以太网发送
int ethernet_send(uint8_t *dst_mac, uint16_t ethertype,
                  void *payload, int len) {
    uint8_t frame[1518];
    ethernet_frame_t *hdr = (ethernet_frame_t *)frame;

    memcpy(hdr->dst_mac, dst_mac, 6);
    memcpy(hdr->src_mac, local_mac, 6);
    hdr->ethertype = htons(ethertype);
    memcpy(frame + 14, payload, len);

    // 计算FCS
    uint32_t fcs = crc32(frame, 14 + len);
    memcpy(frame + 14 + len, &fcs, 4);

    return phy_send(frame, 14 + len + 4);
}
```

### 3.2 PHY驱动

```c
// MII接口
typedef struct {
    uint8_t phy_addr;
    int (*mdio_read)(uint8_t phy, uint8_t reg);
    void (*mdio_write)(uint8_t phy, uint8_t reg, uint16_t val);
} mii_t;

// PHY寄存器
#define PHY_REG_BMCR      0x00  // 基本模式控制
#define PHY_REG_BMSR      0x01  // 基本模式状态
#define PHY_REG_ID1       0x02  // PHY ID 1
#define PHY_REG_ID2       0x03  // PHY ID 2
#define PHY_REG_ANAR      0x04  // 自协商广告
#define PHY_REG_ANLPAR    0x05  // 自协商链路伙伴
#define PHY_REG_ANER      0x06  // 自协商扩展

// PHY初始化
void phy_init(mii_t *mii) {
    uint16_t bmcr;

    // 软复位
    mii->mdio_write(mii->phy_addr, PHY_REG_BMCR, 0x8000);
    while (mii->mdio_read(mii->phy_addr, PHY_REG_BMCR) & 0x8000);

    // 自协商
    bmcr = mii->mdio_read(mii->phy_addr, PHY_REG_BMCR);
    bmcr |= 0x1000;  // 自协商使能
    bmcr |= 0x0200;  // 重启自协商
    mii->mdio_write(mii->phy_addr, PHY_REG_BMCR, bmcr);

    // 等待自协商完成
    while (!(mii->mdio_read(mii->phy_addr, PHY_REG_BMSR) & 0x0020));
}

// 获取链路状态
bool phy_link_up(mii_t *mii) {
    return (mii->mdio_read(mii->phy_addr, PHY_REG_BMSR) & 0x0004) != 0;
}

// 获取速度和双工
void phy_get_speed_duplex(mii_t *mii, int *speed, bool *full_duplex) {
    uint16_t bmcr = mii->mdio_read(mii->phy_addr, PHY_REG_BMCR);

    if (bmcr & 0x2000) {  // 速度选择位
        *speed = 100;
    } else {
        *speed = 10;
    }

    *full_duplex = (bmcr & 0x0100) != 0;
}
```

## 四、RS-485

### 4.1 RS-485驱动

```c
// RS-485半双工控制
typedef struct {
    uart_port_t uart;
    gpio_num_t de_pin;  // 发送使能
    gpio_num_t re_pin;  // 接收使能
} rs485_t;

void rs485_init(rs485_t *rs) {
    // 配置UART
    uart_config_t uart_cfg = {
        .baud_rate = 9600,
        .data_bits = UART_DATA_8_BITS,
        .parity = UART_PARITY_DISABLE,
        .stop_bits = UART_STOP_BITS_1,
        .flow_ctrl = UART_HW_FLOWCTRL_DISABLE,
    };
    uart_param_config(rs->uart, &uart_cfg);

    // 配置方向控制引脚
    gpio_set_direction(rs->de_pin, GPIO_MODE_OUTPUT);
    gpio_set_direction(rs->re_pin, GPIO_MODE_OUTPUT);

    // 默认接收模式
    gpio_set_level(rs->de_pin, 0);
    gpio_set_level(rs->re_pin, 0);
}

// 发送数据
void rs485_send(rs485_t *rs, const uint8_t *data, int len) {
    // 切换到发送模式
    gpio_set_level(rs->re_pin, 1);  // 禁用接收
    gpio_set_level(rs->de_pin, 1);  // 启用发送

    vTaskDelay(pdMS_TO_TICKS(1));  // 方向切换延时

    // 发送
    uart_write_bytes(rs->uart, data, len);
    uart_wait_tx_done(rs->uart, pdMS_TO_TICKS(100));

    // 切换回接收模式
    gpio_set_level(rs->de_pin, 0);
    gpio_set_level(rs->re_pin, 0);
}

// 接收数据
int rs485_recv(rs485_t *rs, uint8_t *buf, int max_len, uint32_t timeout_ms) {
    return uart_read_bytes(rs->uart, buf, max_len, pdMS_TO_TICKS(timeout_ms));
}
```
