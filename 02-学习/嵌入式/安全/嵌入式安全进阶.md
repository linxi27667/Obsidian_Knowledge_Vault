```c
typedef struct {
    uint32_t smc_id;      // 功能ID
    uint32_t param[4];    // 参数
} smc_args_t;

// SMC调用
uint32_t smc_call(smc_args_t *args) {
    register uint32_t r0 __asm__("r0") = args->smc_id;
    register uint32_t r1 __asm__("r1") = args->param[0];
    register uint32_t r2 __asm__("r2") = args->param[1];
    register uint32_t r3 __asm__("r3") = args->param[2];
    register uint32_t r4 __asm__("r4") = args->param[3];

    __asm__ volatile(
        "smc #0"
        : "+r"(r0), "+r"(r1), "+r"(r2), "+r"(r3), "+r"(r4)
        :
        : "memory"
    );

    return r0;
}

// 安全世界函数ID
#define SMC_ID_CRYPTO_INIT      0x1000
#define SMC_ID_CRYPTO_ENCRYPT   0x1001
#define SMC_ID_CRYPTO_DECRYPT   0x1002
#define SMC_ID_KEY_GENERATE     0x1003
#define SMC_ID_KEY_STORE        0x1004
#define SMC_ID_RANDOM_GENERATE  0x1005

// 安全世界加密服务

    smc_args_t args = {
        .smc_id = SMC_ID_CRYPTO_ENCRYPT,
        .param[0] = (uint32_t)data,
        .param[1] = len,
        .param[2] = (uint32_t)key,
    };
    return smc_call(&args);
}
```

```c
    TEE_FreeOperation(op);
```

### 3.2 RSA签名验证

```c
#include <stdint.h>
#include <string.h>

// RSA公钥(简化)
typedef struct {
    uint8_t n[256];   // 模数
    uint8_t e[4];     // 公钥指数
} rsa_public_key_t;

// SHA-256哈希(简化)
typedef struct {
    uint8_t hash[32];
} sha256_hash_t;

// PKCS#1 v1.5签名验证
bool verify_signature(const uint8_t *data, size_t data_len,
                      const uint8_t *signature, size_t sig_len,
                      const rsa_public_key_t *pub_key) {
    // 计算数据哈希
    sha256_hash_t hash;
    sha256_compute(data, data_len, &hash);

    // RSA解密签名
    uint8_t decrypted[256];
    rsa_public_decrypt(signature, sig_len, decrypted, pub_key);

    // 验证PKCS#1 v1.5填充
    // 填充格式: 0x00 0x01 [0xFF...] 0x00 [DigestInfo] [Hash]
    if (decrypted[0] != 0x00 || decrypted[1] != 0x01) {
        return false;
    }

    // 查找分隔符0x00
    int i = 2;
    while (i < 256 && decrypted[i] == 0xFF) {
        i++;
    }
    if (i >= 256 || decrypted[i] != 0x00) {
        return false;
    }
    i++;

    // 验证DigestInfo
    const uint8_t digest_info[] = {
        0x30, 0x31, 0x30, 0x0D, 0x06, 0x09, 0x60, 0x86,
        0x48, 0x01, 0x65, 0x03, 0x04, 0x02, 0x01, 0x05,
        0x00, 0x04, 0x20
    };
    if (memcmp(decrypted + i, digest_info, sizeof(digest_info)) != 0) {
        return false;
    }
    i += sizeof(digest_info);

    // 验证哈希
    if (memcmp(decrypted + i, hash.hash, 32) != 0) {
        return false;
    }

    return true;
}

// 安全启动验证
bool secure_boot_verify(const uint8_t *firmware, size_t fw_size,
                        const uint8_t *signature) {
    // 获取存储的公钥(OTP/Flash)
    const rsa_public_key_t *pub_key = get_boot_public_key();

    return verify_signature(firmware, fw_size, signature, 256, pub_key);
}
```

### 3.3 ECDSA签名

```c
// ECDSA P-256签名验证
typedef struct {
    uint8_t x[32];
    uint8_t y[32];
} ec_point_t;

typedef struct {
    uint8_t r[32];
    uint8_t s[32];
} ecdsa_signature_t;

bool ecdsa_verify(const uint8_t *data, size_t len,
                  const ecdsa_signature_t *sig,
                  const ec_point_t *pub_key) {
    // 计算哈希
    sha256_hash_t hash;
    sha256_compute(data, len, &hash);

    // ECDSA验证算法
    // 1. 计算 s^(-1) mod n
    uint8_t s_inv[32];
    modular_inverse(sig->s, s_inv, curve_n);

    // 2. 计算 u1 = hash * s^(-1) mod n
    uint8_t u1[32];
    modular_multiply(hash.hash, s_inv, u1, curve_n);

    // 3. 计算 u2 = r * s^(-1) mod n
    uint8_t u2[32];
    modular_multiply(sig->r, s_inv, u2, curve_n);

    // 4. 计算点 R = u1*G + u2*pub_key
    ec_point_t R;
    ec_point_t p1, p2;
    scalar_multiply(&p1, u1, &curve_G);      // u1*G
    scalar_multiply(&p2, u2, pub_key);       // u2*Q
    point_add(&R, &p1, &p2);                 // R = p1 + p2

    // 5. 验证 R.x mod n == r
    uint8_t r_check[32];
    modular_reduce(R.x, r_check, curve_n);

    return memcmp(r_check, sig->r, 32) == 0;
}
```

### 4.1 密钥存储

```c
// 密钥存储结构
typedef struct {
    uint32_t key_id;
    uint8_t key_type;      // AES/RSA/ECDSA
    uint8_t key_size;
    uint8_t key_data[64];
    uint32_t usage_count;
    uint32_t max_usage;
    uint32_t creation_time;
    uint32_t expiry_time;
    uint8_t permissions;
} key_entry_t;

// 密钥存储区域(OTP/安全Flash)
#define KEY_STORE_BASE   0x080FF000
#define KEY_STORE_SIZE   4096
#define MAX_KEYS         32

key_entry_t *key_store = (key_entry_t *)KEY_STORE_BASE;

// 密钥写入(一次性可编程)
int key_store_write(uint32_t key_id, const uint8_t *key, size_t len) {
    // 查找空闲槽位
    for (int i = 0; i < MAX_KEYS; i++) {
        if (key_store[i].key_id == 0) {
            // OTP写入
            otp_write(&key_store[i].key_id, key_id, sizeof(uint32_t));
            otp_write(key_store[i].key_data, key, len);
            key_store[i].key_size = len;
            return 0;
        }
    }
    return -1;  // 存储满
}

// 密钥读取(运行时)
int key_store_read(uint32_t key_id, uint8_t *buf, size_t *len) {
    for (int i = 0; i < MAX_KEYS; i++) {
        if (key_store[i].key_id == key_id) {
            memcpy(buf, key_store[i].key_data, key_store[i].key_size);
            *len = key_store[i].key_size;
            return 0;
        }
    }
    return -1;  // 未找到
}

// 密钥派生(HKDF)
int derive_key(const uint8_t *master_key, size_t master_len,
               const uint8_t *salt, size_t salt_len,
               const uint8_t *info, size_t info_len,
               uint8_t *derived_key, size_t derived_len) {
    // HKDF-Extract
    uint8_t prk[32];
    hmac_sha256(salt, salt_len, master_key, master_len, prk);

    // HKDF-Expand
    uint8_t t[32];
    uint8_t counter = 1;
    size_t offset = 0;

    while (offset < derived_len) {
        // T(i) = HMAC-Hash(PRK, T(i-1) | info | i)
        hmac_sha256_ctx ctx;
        hmac_sha256_init(&ctx, prk, 32);
        if (offset > 0) {
            hmac_sha256_update(&ctx, t, 32);
        }
        hmac_sha256_update(&ctx, info, info_len);
        hmac_sha256_update(&ctx, &counter, 1);
        hmac_sha256_final(&ctx, t);

        size_t copy_len = (derived_len - offset) > 32 ? 32 : (derived_len - offset);
        memcpy(derived_key + offset, t, copy_len);
        offset += copy_len;
        counter++;
    }

    return 0;
}
```

### 4.2 密钥轮换

```c
// 密钥版本管理
typedef struct {
    uint32_t key_id;
    uint8_t version;
    uint8_t current_key[32];
    uint8_t previous_key[32];
    uint32_t rotation_time;
} key_rotation_t;

// 密钥轮换
int rotate_key(key_rotation_t *key, const uint8_t *new_key) {
    // 保存旧密钥
    memcpy(key->previous_key, key->current_key, 32);

    // 设置新密钥
    memcpy(key->current_key, new_key, 32);
    key->version++;
    key->rotation_time = get_timestamp();

    return 0;
}

// 使用当前密钥加密
int encrypt_with_current_key(key_rotation_t *key, const uint8_t *data, size_t len, uint8_t *out) {
    return aes_gcm_encrypt(key->current_key, 32, data, len, out);
}

// 使用版本号解密(支持旧密钥)
int decrypt_with_version(key_rotation_t *key, uint8_t version,
                          const uint8_t *data, size_t len, uint8_t *out) {
    uint8_t *dec_key;

    if (version == key->version) {
        dec_key = key->current_key;
    } else if (version == key->version - 1) {
        dec_key = key->previous_key;
    } else {
        return -1;  // 版本不支持
    }

    return aes_gcm_decrypt(dec_key, 32, data, len, out);
}
```

```c
int dtls_encrypt_record(dtls_session_t *session,
                         const uint8_t *plaintext, size_t pt_len,
                         uint8_t *record, size_t *record_len) {
    // 添加DTLS头部
    record[0] = 0x17;  // Application Data
    record[1] = 0xFE, 0xFD;  // DTLS 1.2

    // 序列号
    static uint64_t seq = 0;
    memcpy(record + 3, &seq, 8);
    seq++;

    // 加密
    uint8_t nonce[12];
    memcpy(nonce, session->session_key, 8);
    memcpy(nonce + 8, &seq, 4);

    aes_gcm_encrypt(session->session_key, 32,
                    plaintext, pt_len,
                    record + 13, nonce);

    *record_len = 13 + pt_len + 16;  // 头+数据+Tag
    return 0;
}
```

### 5.2 安全OTA

```c
// 安全OTA流程
typedef struct {
    uint32_t version;
    uint32_t size;
    uint8_t hash[32];       // SHA-256
    uint8_t signature[256]; // RSA签名
} ota_manifest_t;

// 安全OTA验证
bool secure_ota_verify(const uint8_t *firmware, size_t fw_size,
                       const ota_manifest_t *manifest) {
    // 1. 验证版本号(防回滚)
    uint32_t current_version = get_current_version();
    if (manifest->version <= current_version) {
        return false;  // 版本回滚攻击
    }

    // 2. 验证大小
    if (fw_size != manifest->size) {
        return false;
    }

    // 3. 验证哈希
    uint8_t computed_hash[32];
    sha256_compute(firmware, fw_size, computed_hash);
    if (memcmp(computed_hash, manifest->hash, 32) != 0) {
        return false;
    }

    // 4. 验证签名
    const rsa_public_key_t *ota_key = get_ota_public_key();
    if (!verify_signature((uint8_t *)manifest, sizeof(ota_manifest_t) - 256,
                          manifest->signature, 256, ota_key)) {
        return false;
    }

    return true;
}

// 安全OTA流程
int secure_ota_update(const uint8_t *firmware, size_t fw_size,
                      const ota_manifest_t *manifest) {
    // 验证固件
    if (!secure_ota_verify(firmware, fw_size, manifest)) {
        return -1;
    }

    // 写入备份分区
    flash_erase(BACKUP_PARTITION);
    flash_write(BACKUP_PARTITION, firmware, fw_size);

    // 设置启动标志
    set_boot_flag(BOOT_FROM_BACKUP);

    // 重启
    system_reset();

    return 0;
}
```

```c
// 常数时间比较(防时序攻击)
int constant_time_compare(const uint8_t *a, const uint8_t *b, size_t len) {
    volatile uint8_t result = 0;

    for (size_t i = 0; i < len; i++) {
        result |= a[i] ^ b[i];
    }

    return result == 0;
}

// 常数时间条件选择
uint32_t constant_time_select(uint32_t a, uint32_t b, uint32_t condition) {
    // condition = 0: 返回a
    // condition = 1: 返回b
    uint32_t mask = ~((condition & 1) - 1);
    return (a & ~mask) | (b & mask);
}

// 掩码操作(防功耗分析)
typedef struct {
    uint32_t masked_value;
    uint32_t mask;
} masked_value_t;

masked_value_t mask_value(uint32_t value) {
    masked_value_t result;
    result.mask = random_uint32();
    result.masked_value = value ^ result.mask;
    return result;
}

uint32_t unmask_value(masked_value_t masked) {
    return masked.masked_value ^ masked.mask;
}

// 掩码加法
masked_value_t masked_add(masked_value_t a, masked_value_t b) {
    masked_value_t result;
    result.mask = a.mask ^ b.mask;
    result.masked_value = a.masked_value ^ b.masked_value;
    return result;
}
```

```c
    volatile uint32_t dummy;
    dummy = random_uint32();
    dummy ^= random_uint32();
    (void)dummy;
}

// 洗牌操作(防DPA)
void shuffled_operation(uint32_t *data, size_t count) {
    // 随机打乱操作顺序
    for (size_t i = count - 1; i > 0; i--) {
        size_t j = random_uint32() % (i + 1);
        uint32_t temp = data[i];
        data[i] = data[j];
        data[j] = temp;
    }

    // 执行操作
    for (size_t i = 0; i < count; i++) {
        process(data[i]);
        dummy_operation();  // 假操作
    }
}
```

```c
// 安全调试状态机
typedef enum {
    DEBUG_STATE_LOCKED,
    DEBUG_STATE_AUTHENTICATING,
    DEBUG_STATE_UNLOCKED,
} debug_state_t;

debug_state_t debug_state = DEBUG_STATE_LOCKED;

int debug_authenticate(const uint8_t *challenge_response) {
    if (debug_state != DEBUG_STATE_AUTHENTICATING) {
        return -1;
    }

    // 验证挑战-响应
    uint8_t expected[32];
    hmac_sha256(debug_key, 32, challenge, 32, expected);

    if (constant_time_compare(challenge_response, expected, 32)) {
        debug_state = DEBUG_STATE_UNLOCKED;
        enable_debug_port();
        return 0;
    }

    debug_state = DEBUG_STATE_LOCKED;
    return -1;
}

// 安全擦除(防物理攻击)
void secure_erase(void) {
    // 擦除密钥
    memset(key_store, 0xFF, KEY_STORE_SIZE);
    flash_erase(KEY_STORE_BASE);

    // 擦除敏感RAM
    memset(sensitive_ram, 0, SENSITIVE_RAM_SIZE);

    // 擦除栈
    uint8_t *stack_start = get_stack_start();
    uint8_t *stack_end = get_stack_end();
    memset(stack_start, 0, stack_end - stack_start);
}
```
