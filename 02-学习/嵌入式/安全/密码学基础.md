# 密码学基础

## 核心概念

- **对称加密** - AES、DES
- **非对称加密** - RSA、ECC
- **哈希函数** - SHA-256、MD5

## 一、哈希函数

### 1.1 SHA-256

```c
// SHA-256算法
typedef struct {
    uint32_t state[8];
    uint64_t count;
    uint8_t  buffer[64];
} sha256_context_t;

// SHA-256常量
static const uint32_t sha256_k[64] = {
    0x428a2f98, 0x71374491, 0xb5c0fbcf, 0xe9b5dba5,
    0x3956c25b, 0x59f111f1, 0x923f82a4, 0xab1c5ed5,
    // ... 全部64个常量
};

void sha256_init(sha256_context_t *ctx) {
    ctx->state[0] = 0x6a09e667;
    ctx->state[1] = 0xbb67ae85;
    ctx->state[2] = 0x3c6ef372;
    ctx->state[3] = 0xa54ff53a;
    ctx->state[4] = 0x510e527f;
    ctx->state[5] = 0x9b05688c;
    ctx->state[6] = 0x1f83d9ab;
    ctx->state[7] = 0x5be0cd19;
    ctx->count = 0;
}

// SHA-256压缩函数
void sha256_transform(sha256_context_t *ctx, const uint8_t block[64]) {
    uint32_t w[64];
    uint32_t a, b, c, d, e, f, g, h;

    // 消息调度
    for (int i = 0; i < 16; i++) {
        w[i] = (block[i*4] << 24) | (block[i*4+1] << 16) |
               (block[i*4+2] << 8) | block[i*4+3];
    }
    for (int i = 16; i < 64; i++) {
        uint32_t s0 = ROTR(w[i-15], 7) ^ ROTR(w[i-15], 18) ^ (w[i-15] >> 3);
        uint32_t s1 = ROTR(w[i-2], 17) ^ ROTR(w[i-2], 19) ^ (w[i-2] >> 10);
        w[i] = w[i-16] + s0 + w[i-7] + s1;
    }

    // 初始化工作变量
    a = ctx->state[0]; b = ctx->state[1];
    c = ctx->state[2]; d = ctx->state[3];
    e = ctx->state[4]; f = ctx->state[5];
    g = ctx->state[6]; h = ctx->state[7];

    // 主循环
    for (int i = 0; i < 64; i++) {
        uint32_t S1 = ROTR(e, 6) ^ ROTR(e, 11) ^ ROTR(e, 25);
        uint32_t ch = (e & f) ^ (~e & g);
        uint32_t temp1 = h + S1 + ch + sha256_k[i] + w[i];
        uint32_t S0 = ROTR(a, 2) ^ ROTR(a, 13) ^ ROTR(a, 22);
        uint32_t maj = (a & b) ^ (a & c) ^ (b & c);
        uint32_t temp2 = S0 + maj;

        h = g; g = f; f = e; e = d + temp1;
        d = c; c = b; b = a; a = temp1 + temp2;
    }

    ctx->state[0] += a; ctx->state[1] += b;
    ctx->state[2] += c; ctx->state[3] += d;
    ctx->state[4] += e; ctx->state[5] += f;
    ctx->state[6] += g; ctx->state[7] += h;
}

// SHA-256完整计算
void sha256(const uint8_t *data, int len, uint8_t hash[32]) {
    sha256_context_t ctx;
    sha256_init(&ctx);
    sha256_update(&ctx, data, len);
    sha256_final(&ctx, hash);
}
```

### 1.2 HMAC

```c
// HMAC-SHA256
void hmac_sha256(const uint8_t *key, int key_len,
                 const uint8_t *data, int data_len,
                 uint8_t mac[32]) {
    uint8_t k[64] = {0};
    uint8_t ipad[64], opad[64];
    uint8_t hash[32];

    // 如果key超过块大小，先哈希
    if (key_len > 64) {
        sha256(key, key_len, k);
    } else {
        memcpy(k, key, key_len);
    }

    // 生成ipad和opad
    for (int i = 0; i < 64; i++) {
        ipad[i] = k[i] ^ 0x36;
        opad[i] = k[i] ^ 0x5C;
    }

    // 内层哈希: H(ipad || data)
    sha256_context_t ctx;
    sha256_init(&ctx);
    sha256_update(&ctx, ipad, 64);
    sha256_update(&ctx, data, data_len);
    sha256_final(&ctx, hash);

    // 外层哈希: H(opad || hash)
    sha256_init(&ctx);
    sha256_update(&ctx, opad, 64);
    sha256_update(&ctx, hash, 32);
    sha256_final(&ctx, mac);
}
```

## 二、对称加密

### 2.1 AES

```c
// AES-128加密
typedef struct {
    uint8_t round_keys[176];  // 11轮密钥
} aes_context_t;

// AES S-Box
static const uint8_t aes_sbox[256] = {
    0x63, 0x7c, 0x77, 0x7b, 0xf2, 0x6b, 0x6f, 0xc5,
    // ... 完整S-Box
};

void aes_init(aes_context_t *ctx, const uint8_t key[16]) {
    // 密钥扩展
    memcpy(ctx->round_keys, key, 16);

    for (int i = 4; i < 44; i++) {
        uint32_t temp = ((uint32_t *)ctx->round_keys)[i - 1];

        if (i % 4 == 0) {
            temp = (temp << 8) | (temp >> 24);  // RotWord
            temp = (aes_sbox[(temp >> 24) & 0xFF] << 24) |
                   (aes_sbox[(temp >> 16) & 0xFF] << 16) |
                   (aes_sbox[(temp >> 8) & 0xFF] << 8) |
                   aes_sbox[temp & 0xFF];
            temp ^= (uint32_t)aes_rcon[i / 4 - 1] << 24;
        }

        ((uint32_t *)ctx->round_keys)[i] =
            ((uint32_t *)ctx->round_keys)[i - 4] ^ temp;
    }
}

// AES加密单块
void aes_encrypt_block(aes_context_t *ctx, uint8_t input[16], uint8_t output[16]) {
    uint8_t state[16];
    memcpy(state, input, 16);

    // 初始轮密钥加
    add_round_key(state, ctx->round_keys);

    // 9轮
    for (int round = 1; round < 10; round++) {
        sub_bytes(state);
        shift_rows(state);
        mix_columns(state);
        add_round_key(state, ctx->round_keys + round * 16);
    }

    // 最后一轮(无MixColumns)
    sub_bytes(state);
    shift_rows(state);
    add_round_key(state, ctx->round_keys + 10 * 16);

    memcpy(output, state, 16);
}
```

### 2.2 AES-GCM

```c
// AES-GCM认证加密
typedef struct {
    aes_context_t aes;
    uint8_t h[16];         // 哈希子密钥
    uint8_t j0[16];        // 初始计数器
} aes_gcm_context_t;

void aes_gcm_init(aes_gcm_context_t *ctx, const uint8_t key[16],
                  const uint8_t iv[12]) {
    aes_init(&ctx->aes, key);

    // 计算H = E(K, 0^128)
    uint8_t zero[16] = {0};
    aes_encrypt_block(&ctx->aes, zero, ctx->h);

    // 计算J0
    memcpy(ctx->j0, iv, 12);
    ctx->j0[12] = 0;
    ctx->j0[13] = 0;
    ctx->j0[14] = 0;
    ctx->j0[15] = 1;
}

// AES-GCM加密+认证
void aes_gcm_encrypt(aes_gcm_context_t *ctx,
                     const uint8_t *plaintext, int len,
                     const uint8_t *aad, int aad_len,
                     uint8_t *ciphertext, uint8_t tag[16]) {
    uint8_t counter[16];
    memcpy(counter, ctx->j0, 16);

    // 加密
    for (int i = 0; i < len; i += 16) {
        uint8_t keystream[16];
        increment_counter(counter);
        aes_encrypt_block(&ctx->aes, counter, keystream);

        int block_len = (len - i < 16) ? len - i : 16;
        for (int j = 0; j < block_len; j++) {
            ciphertext[i + j] = plaintext[i + j] ^ keystream[j];
        }
    }

    // 计算GHASH
    uint8_t ghash[16] = {0};
    // 处理AAD
    ghash_update(ghash, aad, aad_len, ctx->h);
    // 处理密文
    ghash_update(ghash, ciphertext, len, ctx->h);
    // 处理长度
    uint8_t len_block[16];
    ((uint64_t *)len_block)[0] = 0;
    ((uint64_t *)len_block)[1] = (uint64_t)len * 8;
    ghash_update(ghash, len_block, 16, ctx->h);

    // 计算Tag
    uint8_t enc_j0[16];
    aes_encrypt_block(&ctx->aes, ctx->j0, enc_j0);
    for (int i = 0; i < 16; i++) {
        tag[i] = ghash[i] ^ enc_j0[i];
    }
}
```

## 三、非对称加密

### 3.1 RSA

```c
// RSA密钥
typedef struct {
    uint8_t n[256];     // 模数
    uint8_t e[4];       // 公钥指数
    uint8_t d[256];     // 私钥指数
} rsa_key_t;

// RSA加密: C = M^e mod N
void rsa_encrypt(rsa_key_t *key, const uint8_t *plaintext, int len,
                 uint8_t *ciphertext) {
    // 转换为大整数
    bignum_t m, c;
    bignum_from_bytes(&m, plaintext, len);
    bignum_from_bytes(&c, ciphertext, 256);

    // C = M^e mod N
    bignum_t n, e;
    bignum_from_bytes(&n, key->n, 256);
    bignum_from_bytes(&e, key->e, 4);

    bignum_mod_exp(&c, &m, &e, &n);
    bignum_to_bytes(&c, ciphertext, 256);
}

// RSA签名: S = M^d mod N
void rsa_sign(rsa_key_t *key, const uint8_t *hash, uint8_t *signature) {
    // PKCS#1 v1.5 填充
    uint8_t padded[256];
    padded[0] = 0x00;
    padded[1] = 0x01;
    memset(padded + 2, 0xFF, 256 - 32 - 3);
    padded[256 - 32 - 1] = 0x00;
    memcpy(padded + 256 - 32, hash, 32);

    // 签名
    bignum_t m, s;
    bignum_from_bytes(&m, padded, 256);
    bignum_from_bytes(&s, signature, 256);

    bignum_t n, d;
    bignum_from_bytes(&n, key->n, 256);
    bignum_from_bytes(&d, key->d, 256);

    bignum_mod_exp(&s, &m, &d, &n);
    bignum_to_bytes(&s, signature, 256);
}
```

### 3.2 ECC(椭圆曲线)

```c
// ECC密钥(ECDH)
typedef struct {
    uint8_t private_key[32];
    uint8_t public_key[64];  // 未压缩点(x,y)
} ecdh_key_t;

// P-256曲线参数
typedef struct {
    bignum_t p;     // 素数模数
    bignum_t a;     // 曲线参数a
    bignum_t b;     // 曲线参数b
    bignum_t n;     // 基点阶
    ecc_point_t G;  // 基点
} ecc_curve_t;

// ECDH密钥交换
void ecdh_generate_keypair(ecdh_key_t *key) {
    // 生成随机私钥
    random_bytes(key->private_key, 32);

    // 计算公钥: Q = d * G
    ecc_point_t Q;
    ecc_scalar_mult(&Q, key->private_key, &curve.G, &curve);
    ecc_point_to_bytes(&Q, key->public_key);
}

void ecdh_compute_shared(const ecdh_key_t *my_key,
                         const uint8_t *peer_public_key,
                         uint8_t shared_secret[32]) {
    // 解析对方公钥
    ecc_point_t peer_point;
    ecc_point_from_bytes(&peer_point, peer_public_key);

    // 计算共享密钥: S = d * Q_peer
    ecc_point_t S;
    ecc_scalar_mult(&S, my_key->private_key, &peer_point, &curve);

    // 取x坐标作为共享密钥
    bignum_to_bytes(&S.x, shared_secret, 32);
}
```

## 四、随机数生成

### 4.1 真随机数(TRNG)

```c
// 硬件随机数生成器
void trng_init(void) {
    // 配置硬件RNG
    RNG->CR = RNG_CR_RNGEN;
}

uint32_t trng_read(void) {
    while (!(RNG->SR & RNG_SR_DRDY));
    return RNG->DR;
}

// 随机字节
void random_bytes(uint8_t *buf, int len) {
    for (int i = 0; i < len; i += 4) {
        uint32_t val = trng_read();
        int remaining = len - i;
        int copy = (remaining < 4) ? remaining : 4;
        memcpy(buf + i, &val, copy);
    }
}
```

### 4.2 伪随机数(PRNG)

```c
// ChaCha20 PRNG
typedef struct {
    uint32_t state[16];
    uint64_t counter;
} chacha20_prng_t;

void chacha20_prng_init(chacha20_prng_t *prng, const uint8_t seed[32],
                        const uint8_t nonce[12]) {
    // "expand 32-byte k"
    prng->state[0] = 0x61707865;
    prng->state[1] = 0x3320646e;
    prng->state[2] = 0x79622d32;
    prng->state[3] = 0x6b206574;

    memcpy(&prng->state[4], seed, 32);
    prng->state[12] = 0;  // counter
    prng->state[13] = 0;
    memcpy(&prng->state[14], nonce, 12);

    prng->counter = 0;
}

void chacha20_prng_generate(chacha20_prng_t *prng, uint8_t *output, int len) {
    while (len > 0) {
        uint32_t working[16];
        memcpy(working, prng->state, 64);
        working[12] = (uint32_t)prng->counter;
        working[13] = (uint32_t)(prng->counter >> 32);

        chacha20_block(working);

        int block_len = (len < 64) ? len : 64;
        memcpy(output, working, block_len);

        prng->counter++;
        output += block_len;
        len -= block_len;
    }
}
```

## 五、密钥管理

### 5.1 密钥派生(HKDF)

```c
// HKDF-SHA256
void hkdf_extract(const uint8_t *salt, int salt_len,
                  const uint8_t *ikm, int ikm_len,
                  uint8_t prk[32]) {
    hmac_sha256(salt, salt_len, ikm, ikm_len, prk);
}

void hkdf_expand(const uint8_t *prk, int prk_len,
                 const uint8_t *info, int info_len,
                 uint8_t *okm, int okm_len) {
    uint8_t t[32];
    int t_len = 0;
    int offset = 0;
    uint8_t counter = 1;

    while (offset < okm_len) {
        // T(i) = HMAC-Hash(PRK, T(i-1) | info | i)
        uint8_t hmac_input[32 + info_len + 1];
        int input_len = 0;
        if (t_len > 0) {
            memcpy(hmac_input, t, t_len);
            input_len += t_len;
        }
        memcpy(hmac_input + input_len, info, info_len);
        input_len += info_len;
        hmac_input[input_len++] = counter;

        hmac_sha256(prk, prk_len, hmac_input, input_len, t);
        t_len = 32;

        int copy_len = (okm_len - offset < 32) ? okm_len - offset : 32;
        memcpy(okm + offset, t, copy_len);
        offset += copy_len;
        counter++;
    }
}
```
