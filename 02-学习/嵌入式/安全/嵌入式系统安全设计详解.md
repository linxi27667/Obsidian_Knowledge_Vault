# 嵌入式系统安全设计详解

## 核心概念

- **安全启动** - 固件完整性验证
- **安全存储** - 密钥和敏感数据保护
- **安全通信** - 加密和认证
- **安全更新** - OTA固件升级

## 一、安全启动

### 1.1 安全启动链

```c
// 安全启动流程
typedef struct {
    uint32_t magic;
    uint32_t version;
    uint32_t header_size;
    uint32_t payload_size;
    uint8_t hash[32];      // SHA-256
    uint8_t signature[64]; // ECDSA
} secure_boot_header_t;

// 启动验证
bool secure_boot_verify(const uint8_t *firmware, size_t size) {
    secure_boot_header_t *header = (secure_boot_header_t *)firmware;

    // 检查魔数
    if (header->magic != SECURE_BOOT_MAGIC) {
        return false;
    }

    // 计算哈希
    uint8_t computed_hash[32];
    sha256(firmware + header->header_size, header->payload_size, computed_hash);

    // 验证哈希
    if (memcmp(computed_hash, header->hash, 32) != 0) {
        return false;
    }

    // 验证签名
    if (!ecdsa_verify(public_key, header->hash, 32, header->signature, 64)) {
        return false;
    }

    return true;
}

// 安全启动执行
void secure_boot(void) {
    const uint8_t *firmware = (const uint8_t *)FIRMWARE_ADDRESS;

    if (!secure_boot_verify(firmware, FIRMWARE_SIZE)) {
        // 验证失败，进入恢复模式
        enter_recovery_mode();
        return;
    }

    // 跳转到应用程序
    uint32_t *app_vector = (uint32_t *)FIRMWARE_ADDRESS;
    uint32_t sp = app_vector[0];
    uint32_t pc = app_vector[1];

    // 设置栈指针
    __set_MSP(sp);

    // 跳转
    ((void (*)(void))pc)();
}
```

### 1.2 密钥管理

```c
// 密钥存储
typedef struct {
    uint8_t device_id[16];
    uint8_t aes_key[32];
    uint8_t hmac_key[32];
    uint8_t rsa_private[256];
    uint8_t rsa_public[256];
} key_store_t;

// 密钥派生
void derive_key(const uint8_t *master_key, const uint8_t *salt,
                const char *info, uint8_t *derived_key, size_t key_len) {
    // HKDF密钥派生
    uint8_t prk[32];
    hmac_sha256(salt, 32, master_key, 32, prk);

    uint8_t prev[32];
    uint8_t counter = 1;
    size_t offset = 0;

    while (offset < key_len) {
        hmac_sha256_ctx ctx;
        hmac_sha256_init(&ctx, prk, 32);

        if (offset > 0) {
            hmac_sha256_update(&ctx, prev, 32);
        }
        hmac_sha256_update(&ctx, info, strlen(info));
        hmac_sha256_update(&ctx, &counter, 1);
        hmac_sha256_final(&ctx, prev);

        size_t copy_len = key_len - offset;
        if (copy_len > 32) copy_len = 32;
        memcpy(derived_key + offset, prev, copy_len);
        offset += copy_len;
        counter++;
    }
}

// 密钥轮换
typedef struct {
    uint8_t current_key[32];
    uint8_t previous_key[32];
    uint32_t version;
    uint32_t created_at;
} key_rotation_t;

void key_rotate(key_rotation_t *kr, const uint8_t *new_key) {
    memcpy(kr->previous_key, kr->current_key, 32);
    memcpy(kr->current_key, new_key, 32);
    kr->version++;
    kr->created_at = get_timestamp();
}
```

## 二、安全存储

### 2.1 加密存储

```c
// 加密存储
typedef struct {
    uint8_t iv[16];
    uint8_t data[256];
    uint8_t tag[16];  // GCM认证标签
    uint32_t size;
} encrypted_data_t;

// 加密写入
int secure_write(const uint8_t *key, const uint8_t *data, size_t size) {
    encrypted_data_t enc;

    // 生成随机IV
    random_bytes(enc.iv, 16);

    // AES-GCM加密
    aes_gcm_encrypt(key, enc.iv, 16, data, size, enc.data, enc.tag);

    enc.size = size;

    // 写入Flash
    flash_write(SECURE_STORAGE_ADDR, &enc, sizeof(enc));

    return 0;
}

// 解密读取
int secure_read(const uint8_t *key, uint8_t *data, size_t *size) {
    encrypted_data_t enc;

    // 从Flash读取
    flash_read(SECURE_STORAGE_ADDR, &enc, sizeof(enc));

    // AES-GCM解密和验证
    if (!aes_gcm_decrypt(key, enc.iv, 16, enc.data, enc.size, data, enc.tag)) {
        return -1;  // 认证失败
    }

    *size = enc.size;
    return 0;
}
```

### 2.2 安全密钥存储

```c
// OTP密钥存储
typedef struct {
    uint32_t otp_addr;
    size_t size;
} otp_key_t;

// 写入OTP
int otp_write_key(const otp_key_t *otp, const uint8_t *key, size_t size) {
    if (size > otp->size) return -1;

    // OTP只能写一次
    for (size_t i = 0; i < size; i++) {
        if (otp_read_byte(otp->otp_addr + i) != 0xFF) {
            return -2;  // 已写入
        }
    }

    // 写入
    for (size_t i = 0; i < size; i++) {
        otp_write_byte(otp->otp_addr + i, key[i]);
    }

    return 0;
}

// 读取OTP
int otp_read_key(const otp_key_t *otp, uint8_t *key, size_t size) {
    if (size > otp->size) return -1;

    for (size_t i = 0; i < size; i++) {
        key[i] = otp_read_byte(otp->otp_addr + i);
    }

    return 0;
}
```

## 三、安全通信

### 3.1 TLS配置

```c
// TLS配置
typedef struct {
    const char *ca_cert;
    const char *client_cert;
    const char *client_key;
    const char *server_name;
    bool verify_peer;
} tls_config_t;

// TLS会话
typedef struct {
    mbedtls_ssl_context ssl;
    mbedtls_ssl_config conf;
    mbedtls_x509_crt ca_crt;
    mbedtls_x509_crt client_crt;
    mbedtls_pk_context client_key;
    int socket;
} tls_session_t;

// TLS初始化
int tls_init(tls_session_t *session, tls_config_t *config) {
    mbedtls_ssl_init(&session->ssl);
    mbedtls_ssl_config_init(&session->conf);
    mbedtls_x509_crt_init(&session->ca_crt);
    mbedtls_x509_crt_init(&session->client_crt);
    mbedtls_pk_init(&session->client_key);

    // 加载证书
    mbedtls_x509_crt_parse(&session->ca_crt,
                            (const uint8_t *)config->ca_cert,
                            strlen(config->ca_cert) + 1);

    if (config->client_cert) {
        mbedtls_x509_crt_parse(&session->client_crt,
                                (const uint8_t *)config->client_cert,
                                strlen(config->client_cert) + 1);
        mbedtls_pk_parse_key(&session->client_key,
                              (const uint8_t *)config->client_key,
                              strlen(config->client_key) + 1, NULL, 0);
    }

    // 配置TLS
    mbedtls_ssl_config_defaults(&session->conf,
                                 MBEDTLS_SSL_IS_CLIENT,
                                 MBEDTLS_SSL_TRANSPORT_STREAM,
                                 MBEDTLS_SSL_PRESET_DEFAULT);

    mbedtls_ssl_conf_authmode(&session->conf,
                               config->verify_peer ?
                               MBEDTLS_SSL_VERIFY_REQUIRED :
                               MBEDTLS_SSL_VERIFY_NONE);

    mbedtls_ssl_conf_ca_chain(&session->conf, &session->ca_crt, NULL);

    if (config->client_cert) {
        mbedtls_ssl_conf_own_cert(&session->conf, &session->client_crt,
                                   &session->client_key);
    }

    mbedtls_ssl_setup(&session->ssl, &session->conf);
    mbedtls_ssl_set_hostname(&session->ssl, config->server_name);

    return 0;
}

// TLS连接
int tls_connect(tls_session_t *session, const char *host, int port) {
    // 建立TCP连接
    session->socket = tcp_connect(host, port);
    if (session->socket < 0) return -1;

    mbedtls_ssl_set_bio(&session->ssl, &session->socket,
                         mbedtls_net_send, mbedtls_net_recv, NULL);

    // TLS握手
    int ret;
    while ((ret = mbedtls_ssl_handshake(&session->ssl)) != 0) {
        if (ret != MBEDTLS_ERR_SSL_WANT_READ && ret != MBEDTLS_ERR_SSL_WANT_WRITE) {
            return -1;
        }
    }

    return 0;
}
```

### 3.2 安全认证

```c
// 设备认证
typedef struct {
    uint8_t device_id[16];
    uint8_t challenge[32];
    uint8_t response[32];
} device_auth_t;

// 生成认证响应
void auth_generate_response(device_auth_t *auth, const uint8_t *secret_key) {
    // HMAC-SHA256
    hmac_sha256(secret_key, 32, auth->challenge, 32, auth->response);
}

// 验证认证响应
bool auth_verify_response(device_auth_t *auth, const uint8_t *expected_response) {
    // 常量时间比较
    return secure_compare(auth->response, expected_response, 32);
}

// 安全比较（防时序攻击）
bool secure_compare(const uint8_t *a, const uint8_t *b, size_t len) {
    volatile uint8_t result = 0;
    for (size_t i = 0; i < len; i++) {
        result |= a[i] ^ b[i];
    }
    return result == 0;
}

// 令牌认证
typedef struct {
    uint8_t token[32];
    uint32_t expires_at;
    uint8_t signature[32];
} auth_token_t;

bool auth_token_verify(auth_token_t *token, const uint8_t *secret) {
    // 检查过期
    if (get_timestamp() > token->expires_at) {
        return false;
    }

    // 验证签名
    uint8_t expected_sig[32];
    hmac_sha256(secret, 32, token->token, 32, expected_sig);

    return secure_compare(token->signature, expected_sig, 32);
}
```

## 四、安全更新

### 4.1 安全OTA

```c
// OTA元数据
typedef struct {
    uint32_t magic;
    uint32_t version;
    uint32_t size;
    uint8_t hash[32];
    uint8_t signature[64];
    uint32_t crc;
} ota_metadata_t;

// OTA验证
bool ota_verify(const uint8_t *firmware, size_t size, ota_metadata_t *meta) {
    // 检查魔数
    if (meta->magic != OTA_MAGIC) {
        return false;
    }

    // 检查大小
    if (meta->size != size) {
        return false;
    }

    // 检查CRC
    uint32_t computed_crc = crc32(firmware, size);
    if (computed_crc != meta->crc) {
        return false;
    }

    // 计算哈希
    uint8_t computed_hash[32];
    sha256(firmware, size, computed_hash);
    if (memcmp(computed_hash, meta->hash, 32) != 0) {
        return false;
    }

    // 验证签名
    if (!ecdsa_verify(ota_public_key, meta->hash, 32, meta->signature, 64)) {
        return false;
    }

    return true;
}

// 安全OTA流程
int secure_ota_update(const uint8_t *firmware, size_t size, ota_metadata_t *meta) {
    // 验证固件
    if (!ota_verify(firmware, size, meta)) {
        return -1;
    }

    // 检查版本
    uint32_t current_version = get_current_version();
    if (meta->version <= current_version) {
        return -2;  // 版本过低
    }

    // 写入备用分区
    if (!flash_write(BACKUP_PARTITION_ADDR, firmware, size)) {
        return -3;
    }

    // 设置启动标志
    boot_flags_t flags = {
        .active_partition = BACKUP_PARTITION,
        .pending_update = true,
        .version = meta->version,
    };
    write_boot_flags(&flags);

    // 重启
    NVIC_SystemReset();

    return 0;
}
```

### 4.2 回滚保护

```c
// 回滚保护
typedef struct {
    uint32_t min_version;
    uint32_t max_version;
    uint8_t version_hash[32];
} rollback_protection_t;

bool rollback_check(uint32_t new_version) {
    rollback_protection_t protection;

    // 读取保护配置
    read_rollback_protection(&protection);

    // 检查版本范围
    if (new_version < protection.min_version) {
        return false;  // 版本过低
    }

    if (new_version > protection.max_version) {
        return false;  // 版本过高
    }

    // 检查版本哈希
    uint8_t version_hash[32];
    sha256(&new_version, sizeof(new_version), version_hash);
    if (memcmp(version_hash, protection.version_hash, 32) != 0) {
        return false;
    }

    return true;
}

// 更新版本记录
void update_version_record(uint32_t version) {
    rollback_protection_t protection;
    read_rollback_protection(&protection);

    protection.min_version = version;
    sha256(&version, sizeof(version), protection.version_hash);

    write_rollback_protection(&protection);
}
```

## 五、安全审计

### 5.1 安全日志

```c
// 安全日志
typedef struct {
    uint32_t timestamp;
    uint32_t event_id;
    uint8_t severity;
    uint8_t data[32];
    uint8_t hash[32];
} security_log_t;

#define MAX_SECURITY_LOGS 100
static security_log_t security_logs[MAX_SECURITY_LOGS];
static int log_count = 0;

// 记录安全事件
void security_log(uint32_t event_id, uint8_t severity,
                  const uint8_t *data, size_t data_len) {
    if (log_count >= MAX_SECURITY_LOGS) {
        // 滚动日志
        memmove(&security_logs[0], &security_logs[1],
                (MAX_SECURITY_LOGS - 1) * sizeof(security_log_t));
        log_count = MAX_SECURITY_LOGS - 1;
    }

    security_log_t *log = &security_logs[log_count];
    log->timestamp = get_timestamp();
    log->event_id = event_id;
    log->severity = severity;

    if (data && data_len > 0) {
        size_t copy_len = data_len > 32 ? 32 : data_len;
        memcpy(log->data, data, copy_len);
    }

    // 计算哈希（链式）
    if (log_count > 0) {
        sha256_ctx ctx;
        sha256_init(&ctx);
        sha256_update(&ctx, &security_logs[log_count - 1], sizeof(security_log_t) - 32);
        sha256_update(&ctx, log, sizeof(security_log_t) - 32);
        sha256_final(&ctx, log->hash);
    } else {
        sha256(log, sizeof(security_log_t) - 32, log->hash);
    }

    log_count++;
}

// 验证日志完整性
bool security_log_verify(void) {
    for (int i = 1; i < log_count; i++) {
        uint8_t expected_hash[32];
        sha256_ctx ctx;
        sha256_init(&ctx);
        sha256_update(&ctx, &security_logs[i - 1], sizeof(security_log_t) - 32);
        sha256_update(&ctx, &security_logs[i], sizeof(security_log_t) - 32);
        sha256_final(&ctx, expected_hash);

        if (memcmp(security_logs[i].hash, expected_hash, 32) != 0) {
            return false;
        }
    }
    return true;
}
```
