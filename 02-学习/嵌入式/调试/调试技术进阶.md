# 调试技术进阶

## 核心概念

- **GDB** - GNU调试器

## 一、GDB调试

### 1.1 GDB命令速查

```bash
# 启动GDB
arm-none-eabi-gdb firmware.elf

# 连接目标
target remote :3333  # OpenOCD
target remote /dev/ttyUSB0  # 串口

# 程序控制
load                  # 下载程序
run                   # 运行
continue              # 继续
step                  # 单步(进入函数)
next                  # 单步(不进入)
finish                # 执行到返回
until 100             # 执行到第100行

# 断点
break main            # 函数断点
break file.c:42       # 行断点
break *0x08001000     # 地址断点
watch *0x20000000     # 数据观察点
info breakpoints      # 查看断点
delete 1              # 删除断点1

# 查看信息
info registers        # 寄存器
info locals           # 局部变量
print variable        # 打印变量
print /x variable     # 十六进制
x/10x 0x20000000      # 查看内存
backtrace             # 调用栈
info threads          # 线程列表

# 修改
set variable x = 42   # 修改变量
set {int}0x20000000 = 1  # 修改内存
```

### 1.2 GDB脚本

```python
# GDB Python脚本
import gdb

class RegisterDump(gdb.Command):
    """Dump all registers"""
    def __init__(self):
        super().__init__("reg-dump", gdb.COMMAND_USER)

    def invoke(self, arg, from_tty):
        regs = ["r0", "r1", "r2", "r3", "r4", "r5", "r6", "r7",
                "r8", "r9", "r10", "r11", "r12", "sp", "lr", "pc"]
        for reg in regs:
            val = gdb.parse_and_eval(f"${reg}")
            print(f"{reg:>4} = 0x{int(val):08x}")

RegisterDump()

# FreeRTOS任务栈分析
class TaskStackAnalysis(gdb.Command):
    """Analyze FreeRTOS task stacks"""
    def __init__(self):
        super().__init__("task-stacks", gdb.COMMAND_USER)

    def invoke(self, arg, from_tty):
        # 遍历任务列表
        task = gdb.parse_and_eval("pxCurrentTCB")
        while task:
            name = task["pcTaskName"].string()
            stack = task["pxStack"]
            top = task["pxTopOfStack"]
            size = int(task["uxStackSize"]) * 4

            usage = int(top) - int(stack)
            percent = (usage / size) * 100

            print(f"Task: {name:20s} Stack: {usage:6d}/{size:6d} ({percent:.1f}%)")

            task = task["pxNext"]

TaskStackAnalysis()
```

## 三、RTT(实时传输)

### 3.1 SEGGER RTT

```c
#include "SEGGER_RTT.h"

// RTT初始化
void rtt_init(void) {
    SEGGER_RTT_Init();
}

// RTT输出(替代printf)
void rtt_printf(const char *fmt, ...) {
    va_list args;
    va_start(args, fmt);
    SEGGER_RTT_vprintf(0, fmt, &args);
    va_end(args);
}

// RTT输入
int rtt_read(void *buf, int len) {
    return SEGGER_RTT_Read(0, buf, len);
}

// 使用示例
void debug_task(void *param) {
    while (1) {
        float temp = read_temperature();
        float humidity = read_humidity();

        rtt_printf("T=%.1f H=%.1f\n", temp, humidity);

        vTaskDelay(pdMS_TO_TICKS(1000));
    }
}
```

## 四、CoreDump

### 4.1 崩溃信息收集

```c
// CoreDump结构
typedef struct {
    uint32_t magic;          // 魔数
    uint32_t version;        // 版本
    uint32_t crash_pc;       // 崩溃PC
    uint32_t crash_lr;       // 崩溃LR
    uint32_t psr;            // 程序状态
    uint32_t registers[16];  // R0-R15
    uint32_t msp;            // 主栈指针
    uint32_t psp;            // 进程栈指针
    uint8_t  stack[1024];    // 栈内容
    char     task_name[32];  // 任务名
} coredump_t;

// HardFault处理
void HardFault_Handler(void) {
    coredump_t dump;
    dump.magic = 0x434F5245;  // "CORE"

    // 保存寄存器
    __asm__ volatile (
        "mov %0, r0\n"
        "mov %1, r1\n"
        "mov %2, r2\n"
        "mov %3, r3\n"
        "mov %4, r12\n"
        "mov %5, lr\n"
        "mov %6, pc\n"
        "mrs %7, psr\n"
        : "=r"(dump.registers[0]), "=r"(dump.registers[1]),
          "=r"(dump.registers[2]), "=r"(dump.registers[3]),
          "=r"(dump.registers[12]), "=r"(dump.crash_lr),
          "=r"(dump.crash_pc), "=r"(dump.psr)
    );

    // 保存栈
    uint32_t sp;
    __asm__ volatile ("mov %0, sp" : "=r"(sp));
    memcpy(dump.stack, (void *)sp, sizeof(dump.stack));

    // 保存到Flash
    save_coredump_to_flash(&dump);

    // 复位
    NVIC_SystemReset();
}
```

### 4.2 崩溃分析

```python
# CoreDump分析脚本
import struct

def analyze_coredump(filename):
    with open(filename, 'rb') as f:
        data = f.read()

    magic = struct.unpack('<I', data[0:4])[0]
    if magic != 0x434F5245:
        print("Invalid coredump")
        return

    version = struct.unpack('<I', data[4:8])[0]
    crash_pc = struct.unpack('<I', data[8:12])[0]
    crash_lr = struct.unpack('<I', data[12:16])[0]
    psr = struct.unpack('<I', data[16:20])[0]

    print(f"Version: {version}")
    print(f"Crash PC: 0x{crash_pc:08X}")
    print(f"Crash LR: 0x{crash_lr:08X}")
    print(f"PSR: 0x{psr:08X}")

    # 解析故障类型
    fault_type = (psr >> 0) & 0x1F
    print(f"Exception: {fault_type}")

    # 使用addr2line定位
    import subprocess
    result = subprocess.run(
        ['arm-none-eabi-addr2line', '-e', 'firmware.elf', '-f', '-p', hex(crash_pc)],
        capture_output=True, text=True
    )
    print(f"Location: {result.stdout}")

# 使用
analyze_coredump("coredump.bin")
```

## 五、逻辑分析仪

### 5.1 协议解码

```python
# Saleae逻辑分析仪脚本
# I2C解码示例

def decode_i2c(sda, scl, sample_rate):
    """解码I2C信号"""
    transactions = []
    current_byte = 0
    bit_count = 0
    in_transaction = False

    for i in range(len(sda)):
        # 检测起始条件: SCL高时SDA下降沿
        if scl[i] == 1 and sda[i] == 0 and sda[i-1] == 1:
            in_transaction = True
            bit_count = 0
            current_byte = 0
            continue

        # 检测停止条件: SCL高时SDA上升沿
        if scl[i] == 1 and sda[i] == 1 and sda[i-1] == 0:
            in_transaction = False
            continue

        # SCL上升沿采样数据
        if in_transaction and scl[i] == 1 and scl[i-1] == 0:
            current_byte = (current_byte << 1) | sda[i]
            bit_count += 1

            if bit_count == 8:
                # ACK/NACK
                pass
            elif bit_count == 9:
                transactions.append(current_byte & 0xFF)
                bit_count = 0
                current_byte = 0

    return transactions
```

## 六、性能分析

### 6.1 执行时间测量

```c
// DWT周期计数器
void dwt_init(void) {
    CoreDebug->DEMCR |= CoreDebug_DEMCR_TRCENA_Msk;
    DWT->CYCCNT = 0;
    DWT->CTRL |= DWT_CTRL_CYCCNTENA_Msk;
}

// 测量执行时间
#define MEASURE_START() uint32_t _start = DWT->CYCCNT
#define MEASURE_END()   uint32_t _end = DWT->CYCCNT
#define MEASURE_US()    ((_end - _start) / (SystemCoreClock / 1000000))

void measure_function(void) {
    MEASURE_START();

    // 被测代码
    function_to_measure();

    MEASURE_END();

    printf("Execution time: %lu us\n", MEASURE_US());
}
```

### 6.2 栈使用分析

```c
// 栈溢出检测
#define STACK_FILL_BYTE 0xA5

void fill_stack_pattern(uint32_t *stack, size_t size) {
    for (size_t i = 0; i < size / sizeof(uint32_t); i++) {
        stack[i] = STACK_FILL_BYTE;
    }
}

size_t measure_stack_usage(uint32_t *stack, size_t size) {
    size_t unused = 0;
    uint32_t *p = stack;

    while (*p == STACK_FILL_BYTE && unused < size) {
        unused += sizeof(uint32_t);
        p++;
    }

    return size - unused;
}
```
