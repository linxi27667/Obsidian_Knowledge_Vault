## 二、医疗安全标准

### 2.1 IEC 60601

```c
// IEC 60601安全要求
/*
 * 通用要求:
 * - 电气安全(绝缘、接地)
 * - 机械安全(稳定性、锐边)
 * - 温度安全(表面温度)
 * - 电磁兼容性
 *
 * 特殊要求:
 * - 心脏接触设备(漏电流<10μA)
 * - 防除颤器保护
 * - 防火外壳
 */

// 漏电流检测
typedef struct {
    float earth_leakage_mA;
    float enclosure_leakage_mA;
    float patient_leakage_mA;
    bool alarm;
} leakage_monitor_t;

void check_leakage(leakage_monitor_t *mon) {
    // 患者漏电流限值
    float limit = 0.01f;  // 10μA (CF型)

    mon->alarm = (mon->patient_leakage_mA > limit);
    if (mon->alarm) {
        trigger_alarm(ALARM_LEAKAGE);
    }
}
```

## 三、医疗传感器

### 3.1 心电(ECG)

```c
// ECG信号采集
typedef struct {
    int16_t samples[1000];
    int sample_count;
    float heart_rate;
    bool lead_off;
} ecg_data_t;

void ecg_init(void) {
    // ADS1293配置
    spi_write_reg(ADS1293_ADDR, 0x01, 0x01);  // 复位
    vTaskDelay(pdMS_TO_TICKS(10));

    // 配置通道
    spi_write_reg(ADS1293_ADDR, 0x02, 0x30);  // 使能3通道
    spi_write_reg(ADS1293_ADDR, 0x03, 0x02);  // 500SPS
}

ecg_data_t ecg_read(void) {
    ecg_data_t data = {0};

    // 读取采样
    uint8_t buf[9];
    spi_read_regs(ADS1293_ADDR, 0x12, buf, 9);

    // 解析3通道数据
    data.samples[0] = (buf[0] << 8) | buf[1];
    data.samples[1] = (buf[3] << 8) | buf[4];
    data.samples[2] = (buf[6] << 8) | buf[7];
    data.sample_count = 3;

    // 导联脱落检测
    uint8_t status = spi_read_reg(ADS1293_ADDR, 0x0A);
    data.lead_off = (status & 0x04) != 0;

    return data;
}

// 心率计算(基于R波检测)
float detect_heart_rate(int16_t *ecg, int count, int sample_rate) {
    int peaks = 0;
    int last_peak = 0;

    // 简化阈值检测
    int16_t threshold = 500;  // 需要自适应调整
    int min_interval = sample_rate * 0.3f;  // 最小300ms

    for (int i = 1; i < count - 1; i++) {
        if (ecg[i] > threshold &&
            ecg[i] > ecg[i-1] && ecg[i] > ecg[i+1] &&
            i - last_peak > min_interval) {
            peaks++;
            last_peak = i;
        }
    }

    float duration = (float)count / sample_rate;
    return peaks / duration * 60;  // BPM
}
```

### 3.2 血氧(SpO2)

```c
// SpO2测量
typedef struct {
    float spo2;      // 血氧饱和度(%)
    float heart_rate; // 心率(BPM)
    bool valid;
} spo2_data_t;

spo2_data_t calc_spo2(uint32_t *red_buffer, uint32_t *ir_buffer, int count) {
    spo2_data_t data = {0};

    // 计算AC/DC
    float red_ac = 0, red_dc = 0;
    float ir_ac = 0, ir_dc = 0;

    for (int i = 0; i < count; i++) {
        red_dc += red_buffer[i];
        ir_dc += ir_buffer[i];
    }
    red_dc /= count;
    ir_dc /= count;

    for (int i = 0; i < count; i++) {
        red_ac += fabsf(red_buffer[i] - red_dc);
        ir_ac += fabsf(ir_buffer[i] - ir_dc);
    }
    red_ac /= count;
    ir_ac /= count;

    // 计算R值
    float R = (red_ac / red_dc) / (ir_ac / ir_dc);

    // 查表或公式转换
    data.spo2 = 110.0f - 25.0f * R;  // 简化公式
    if (data.spo2 > 100) data.spo2 = 100;
    if (data.spo2 < 0) data.spo2 = 0;

    data.valid = (R > 0.4f && R < 3.0f);

    return data;
}
```

### 3.3 血压监测

```c
// 无创血压测量(示波法)
typedef struct {
    int systolic;    // 收缩压(mmHg)
    int diastolic;   // 舒张压(mmHg)
    int mean_bp;     // 平均压(mmHg)
    int heart_rate;  // 心率
} blood_pressure_t;

blood_pressure_t measure_bp(void) {
    blood_pressure_t bp = {0};
    float pressure[500];
    float oscillation[500];
    int count = 0;

    // 充气
    pump_on();
    while (get_pressure() < 200) {
        pressure[count++] = get_pressure();
        vTaskDelay(pdMS_TO_TICKS(10));
    }

    // 放气
    pump_off();
    valve_open(3);  // 慢放气

    while (get_pressure() > 40) {
        pressure[count++] = get_pressure();
        vTaskDelay(pdMS_TO_TICKS(10));
    }

    valve_open(0);  // 快放气

    // 提取振荡波
    for (int i = 1; i < count - 1; i++) {
        oscillation[i] = fabsf(pressure[i+1] - pressure[i-1]);
    }

    // 找最大振荡点(平均压)
    int max_idx = 0;
    float max_osc = 0;
    for (int i = 0; i < count; i++) {
        if (oscillation[i] > max_osc) {
            max_osc = oscillation[i];
            max_idx = i;
        }
    }
    bp.mean_bp = (int)pressure[max_idx];

    // 收缩压: 最大振荡前50%处
    float threshold = max_osc * 0.5f;
    for (int i = max_idx; i >= 0; i--) {
        if (oscillation[i] < threshold) {
            bp.systolic = (int)pressure[i];
            break;
        }
    }

    // 舒张压: 最大振荡后75%处
    threshold = max_osc * 0.75f;
    for (int i = max_idx; i < count; i++) {
        if (oscillation[i] < threshold) {
            bp.diastolic = (int)pressure[i];
            break;
        }
    }

    return bp;
}
```

/*
 * MSH|^~\&|Device|Hospital|EMR|Hospital|20230621||ORU^R01|MSG001|P|2.5

    strcpy(obs->status, "final");

    strcpy(obs->unit, unit);

                                heart_rate_handle, 2, data, false);

typedef struct {
    char patient_id[32];

    char dob[16];  // 出生日期

// AES加密患者数据
void encrypt_patient_data(patient_data_t *data, uint8_t *key, uint8_t *encrypted) {
    // 序列化
    uint8_t plaintext[sizeof(patient_data_t)];
    memcpy(plaintext, data, sizeof(patient_data_t));

    // AES-256-CBC加密
    aes_cbc_encrypt(plaintext, sizeof(plaintext), key, iv, encrypted);
