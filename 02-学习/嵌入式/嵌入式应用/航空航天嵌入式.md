### 2.2 安全关键代码

```c
// 看门狗监控
typedef struct {
    uint32_t magic;
    uint32_t counter;
    uint32_t last_feed;
    bool alive;
} watchdog_t;

#define WATCHDOG_MAGIC 0x5A5A5A5A

void watchdog_init(watchdog_t *wdg) {
    wdg->magic = WATCHDOG_MAGIC;
    wdg->counter = 0;
    wdg->last_feed = 0;
    wdg->alive = true;
}

void watchdog_feed(watchdog_t *wdg) {
    if (wdg->magic != WATCHDOG_MAGIC) {
        // 内存损坏
        system_reset();
    }
    wdg->counter++;
    wdg->last_feed = get_tick();
}

bool watchdog_check(watchdog_t *wdg) {
    if (wdg->magic != WATCHDOG_MAGIC) return false;
    if (get_tick() - wdg->last_feed > WATCHDOG_TIMEOUT) {
        wdg->alive = false;
        return false;
    }
    return true;
}

// 程序流监控
typedef enum {
    FLOW_INIT = 0x1234,
    FLOW_STEP1 = 0x2345,
    FLOW_STEP2 = 0x3456,
    FLOW_STEP3 = 0x4567,
    FLOW_DONE = 0x5678
} program_flow_t;

static program_flow_t current_flow = FLOW_INIT;

void flow_check(program_flow_t expected) {
    if (current_flow != expected) {
        // 程序流错误
        report_error(ERROR_FLOW);
    }
}

void flow_advance(program_flow_t next) {
    current_flow = next;
}
```

### 2.3 冗余设计

```c
// 三模冗余(TMR)
typedef struct {
    float value[3];
    int voter_id;
} tmr_t;

float tmr_vote(tmr_t *tmr) {
    // 中值表决
    float sorted[3];
    memcpy(sorted, tmr->value, sizeof(float) * 3);

    // 简单排序
    for (int i = 0; i < 2; i++) {
        for (int j = i + 1; j < 3; j++) {
            if (sorted[j] < sorted[i]) {
                float t = sorted[i];
                sorted[i] = sorted[j];
                sorted[j] = t;
            }
        }
    }

    return sorted[1];  // 中值
}

// 双通道比较
typedef struct {
    float channel_a;
    float channel_b;
    float threshold;
    bool agree;
} dual_channel_t;

bool dual_channel_check(dual_channel_t *ch) {
    float diff = fabsf(ch->channel_a - ch->channel_b);
    ch->agree = (diff < ch->threshold);
    return ch->agree;
}
```

## 三、航天嵌入式

### 3.1 抗辐射设计

```c
// 抗辐射措施
/*
 * 1. 三模冗余(TMR)
 * 2. EDAC(Error Detection And Correction)
 * 3. 刷新(Scrubbing)
 * 4. 看门狗
 * 5. 状态恢复
 */

// SRAM刷新
void sram_scrub(void) {
    uint32_t *addr = (uint32_t *)SRAM_START;
    uint32_t *end = (uint32_t *)SRAM_END;

    while (addr < end) {
        // 读取-检验-回写
        uint32_t data = *addr;
        if (check_ecc(data) != ECC_OK) {
            data = correct_ecc(data);
            *addr = data;
        }
        addr++;
    }
}

// TMR寄存器
typedef struct {
    uint32_t value_a;
    uint32_t value_b;
    uint32_t value_c;
} tmr_register_t;

uint32_t tmr_read(tmr_register_t *reg) {
    // 三取二表决
    if (reg->value_a == reg->value_b) return reg->value_a;
    if (reg->value_a == reg->value_c) return reg->value_a;
    if (reg->value_b == reg->value_c) return reg->value_b;

    // 三个都不同，报告错误
    report_error(ERROR_TMR);
    return reg->value_a;
}

void tmr_write(tmr_register_t *reg, uint32_t value) {
    reg->value_a = value;
    reg->value_b = value;
    reg->value_c = value;
}
```

## 四、飞控系统

### 4.1 PID控制器

```c
// 飞控PID
typedef struct {
    float kp, ki, kd;
    float integral;
    float prev_error;
    float output_min, output_max;
} flight_pid_t;

float flight_pid_update(flight_pid_t *pid, float setpoint, float measurement, float dt) {
    float error = setpoint - measurement;

    // 积分(带抗饱和)
    pid->integral += error * dt;
    if (pid->integral > pid->output_max) pid->integral = pid->output_max;
    if (pid->integral < pid->output_min) pid->integral = pid->output_min;

    // 微分(带滤波)
    float derivative = (error - pid->prev_error) / dt;
    pid->prev_error = error;

    // 输出
    float output = pid->kp * error + pid->ki * pid->integral + pid->kd * derivative;

    // 限幅
    if (output > pid->output_max) output = pid->output_max;
    if (output < pid->output_min) output = pid->output_min;

    return output;
}

// 四轴PID
typedef struct {
    flight_pid_t roll;
    flight_pid_t pitch;
    flight_pid_t yaw;
    flight_pid_t altitude;
} quadrotor_pid_t;

void quadrotor_control(quadrotor_pid_t *pid, float *setpoints, float *measurements,
                       float *motor_outputs) {
    float roll_out = flight_pid_update(&pid->roll, setpoints[0], measurements[0], 0.01f);
    float pitch_out = flight_pid_update(&pid->pitch, setpoints[1], measurements[1], 0.01f);
    float yaw_out = flight_pid_update(&pid->yaw, setpoints[2], measurements[2], 0.01f);
    float alt_out = flight_pid_update(&pid->altitude, setpoints[3], measurements[3], 0.01f);

    // 混控
    motor_outputs[0] = alt_out + roll_out - pitch_out + yaw_out;  // 前左
    motor_outputs[1] = alt_out - roll_out - pitch_out - yaw_out;  // 前右
    motor_outputs[2] = alt_out - roll_out + pitch_out + yaw_out;  // 后左
    motor_outputs[3] = alt_out + roll_out + pitch_out - yaw_out;  // 后右

    // 限幅
    for (int i = 0; i < 4; i++) {
        if (motor_outputs[i] > 1.0f) motor_outputs[i] = 1.0f;
        if (motor_outputs[i] < 0.0f) motor_outputs[i] = 0.0f;
    }
}
```
