## 二、运动学

### 2.1 正运动学

```c
// 2D正运动学(两连杆)
typedef struct {
    float x;
    float y;
} point_2d_t;

point_2d_t forward_kinematics(float l1, float l2, float theta1, float theta2) {
    point_2d_t p;
    p.x = l1 * cosf(theta1) + l2 * cosf(theta1 + theta2);
    p.y = l1 * sinf(theta1) + l2 * sinf(theta1 + theta2);
    return p;
}

// 3D正运动学(DH参数)
typedef struct {
    float theta;  // 关节角
    float d;      // 连杆偏移
    float a;      // 连杆长度
    float alpha;  // 连杆扭转
} dh_param_t;

// DH变换矩阵
void dh_matrix(dh_param_t *param, float T[4][4]) {
    float ct = cosf(param->theta);
    float st = sinf(param->theta);
    float ca = cosf(param->alpha);
    float sa = sinf(param->alpha);

    T[0][0] = ct;    T[0][1] = -st*ca; T[0][2] =  st*sa; T[0][3] = param->a*ct;
    T[1][0] = st;    T[1][1] =  ct*ca; T[1][2] = -ct*sa; T[1][3] = param->a*st;
    T[2][0] = 0;     T[2][1] =  sa;    T[2][2] =  ca;    T[2][3] = param->d;
    T[3][0] = 0;     T[3][1] =  0;     T[3][2] =  0;     T[3][3] = 1;
}
```

### 2.2 逆运动学

```c
// 2D逆运动学(两连杆)
typedef struct {
    float theta1;
    float theta2;
} joint_angles_t;

joint_angles_t inverse_kinematics(float l1, float l2, float x, float y) {
    joint_angles_t angles;

    float c2 = (x*x + y*y - l1*l1 - l2*l2) / (2 * l1 * l2);
    float s2 = sqrtf(1 - c2*c2);

    angles.theta2 = atan2f(s2, c2);
    angles.theta1 = atan2f(y, x) - atan2f(l2*s2, l1 + l2*c2);

    return angles;
}

// 6轴机器人逆运动学(解析法)
bool ik_6dof(float T[4][4], float joints[6], dh_param_t *dh) {
    // 计算腕部位置
    float px = T[0][3] - dh[5].a * T[0][2];
    float py = T[1][3] - dh[5].a * T[1][2];
    float pz = T[2][3] - dh[5].a * T[2][2];

    // 关节1
    joints[0] = atan2f(py, px);

    // 关节2,3 (平面几何)
    float r = sqrtf(px*px + py*py);
    float s = pz - dh[0].d;
    float D = (r*r + s*s - dh[1].a*dh[1].a - dh[2].a*dh[2].a) / (2 * dh[1].a * dh[2].a);

    if (fabsf(D) > 1) return false;  // 无解

    joints[2] = atan2f(sqrtf(1 - D*D), D);
    joints[1] = atan2f(s, r) - atan2f(dh[2].a*sinf(joints[2]),
                                       dh[1].a + dh[2].a*cosf(joints[2]));

    // 关节4,5,6 (欧拉角)
    // ... 省略

    return true;
}
```

## 三、路径规划

### 3.1 A*算法

```c
// A*路径规划
typedef struct {
    int x, y;
    float g;  // 起点到当前点
    float h;  // 启发式估计
    float f;  // g + h
    int parent;
} astar_node_t;

typedef struct {
    astar_node_t nodes[1000];
    int count;
} open_list_t;

float heuristic(int x1, int y1, int x2, int y2) {
    return sqrtf((x2-x1)*(x2-x1) + (y2-y1)*(y2-y1));
}

bool astar_search(int grid[][GRID_W], int sx, int sy, int ex, int ey,
                  int *path_x, int *path_y, int *path_len) {
    open_list_t open = {0};
    bool closed[GRID_H][GRID_W] = {0};

    // 起点
    open.nodes[0] = (astar_node_t){sx, sy, 0, heuristic(sx,sy,ex,ey), 0, -1};
    open.count = 1;

    while (open.count > 0) {
        // 找f最小的节点
        int best = 0;
        for (int i = 1; i < open.count; i++) {
            if (open.nodes[i].f < open.nodes[best].f) best = i;
        }

        astar_node_t current = open.nodes[best];

        // 到达终点
        if (current.x == ex && current.y == ey) {
            // 回溯路径
            int idx = best;
            *path_len = 0;
            while (idx >= 0) {
                path_x[*path_len] = open.nodes[idx].x;
                path_y[*path_len] = open.nodes[idx].y;
                (*path_len)++;
                idx = open.nodes[idx].parent;
            }
            return true;
        }

        // 移除当前节点
        open.nodes[best] = open.nodes[--open.count];
        closed[current.y][current.x] = true;

        // 扩展邻居
        int dx[] = {0, 0, 1, -1};
        int dy[] = {1, -1, 0, 0};
        for (int i = 0; i < 4; i++) {
            int nx = current.x + dx[i];
            int ny = current.y + dy[i];

            if (nx < 0 || nx >= GRID_W || ny < 0 || ny >= GRID_H) continue;
            if (closed[ny][nx] || grid[ny][nx]) continue;

            float ng = current.g + 1;
            float nh = heuristic(nx, ny, ex, ey);

            open.nodes[open.count++] = (astar_node_t){nx, ny, ng, nh, ng+nh, best};
        }
    }

    return false;
}
```

### 3.2 RRT算法

```c
// RRT(快速随机树)
typedef struct {
    float x, y;
    int parent;
} rrt_node_t;

rrt_node_t rrt_nodes[1000];
int rrt_count = 0;

void rrt_expand(float goal_x, float goal_y, float step_size) {
    // 随机采样
    float rx = (float)rand() / RAND_MAX * MAP_W;
    float ry = (float)rand() / RAND_MAX * MAP_H;

    // 找最近节点
    int nearest = 0;
    float min_dist = INFINITY;
    for (int i = 0; i < rrt_count; i++) {
        float d = sqrtf((rx-rrt_nodes[i].x)*(rx-rrt_nodes[i].x) +
                       (ry-rrt_nodes[i].y)*(ry-rrt_nodes[i].y));
        if (d < min_dist) { min_dist = d; nearest = i; }
    }

    // 扩展
    float angle = atan2f(ry - rrt_nodes[nearest].y, rx - rrt_nodes[nearest].x);
    float nx = rrt_nodes[nearest].x + step_size * cosf(angle);
    float ny = rrt_nodes[nearest].y + step_size * sinf(angle);

    // 碰撞检测
    if (!is_collision(rrt_nodes[nearest].x, rrt_nodes[nearest].y, nx, ny)) {
        rrt_nodes[rrt_count] = (rrt_node_t){nx, ny, nearest};
        rrt_count++;
    }
}
```

## 四、SLAM

### 4.1 激光SLAM

```c
// 简化ICP(迭代最近点)
typedef struct {
    float x, y;
} point_2d_t;

void icp_match(point_2d_t *source, int src_count,
               point_2d_t *target, int tgt_count,
               float *R, float *t) {
    // 初始化
    R[0] = 1; R[1] = 0; R[2] = 0; R[3] = 1;
    t[0] = 0; t[1] = 0;

    for (int iter = 0; iter < MAX_ITER; iter++) {
        // 1. 找最近点
        int matches[src_count];
        for (int i = 0; i < src_count; i++) {
            float min_d = INFINITY;
            for (int j = 0; j < tgt_count; j++) {
                float dx = source[i].x - target[j].x;
                float dy = source[i].y - target[j].y;
                float d = dx*dx + dy*dy;
                if (d < min_d) { min_d = d; matches[i] = j; }
            }
        }

        // 2. 计算质心
        float cx_s = 0, cy_s = 0, cx_t = 0, cy_t = 0;
        for (int i = 0; i < src_count; i++) {
            cx_s += source[i].x; cy_s += source[i].y;
            cx_t += target[matches[i]].x; cy_t += target[matches[i]].y;
        }
        cx_s /= src_count; cy_s /= src_count;
        cx_t /= src_count; cy_t /= src_count;

        // 3. 计算旋转和平移
        float H[4] = {0};
        for (int i = 0; i < src_count; i++) {
            float sx = source[i].x - cx_s;
            float sy = source[i].y - cy_s;
            float tx = target[matches[i]].x - cx_t;
            float ty = target[matches[i]].y - cy_t;
            H[0] += sx*tx; H[1] += sx*ty;
            H[2] += sy*tx; H[3] += sy*ty;
        }

        // SVD分解(简化)
        float theta = atan2f(H[1]-H[2], H[0]+H[3]);
        R[0] = cosf(theta); R[1] = -sinf(theta);
        R[2] = sinf(theta); R[3] = cosf(theta);

        t[0] = cx_t - (R[0]*cx_s + R[1]*cy_s);
        t[1] = cy_t - (R[2]*cx_s + R[3]*cy_s);
    }
}
```

### 4.2 视觉SLAM

```c
// 特征提取(简化ORB)
typedef struct {
    float x, y;
    float angle;
    uint8_t descriptor[32];
} feature_t;

void extract_features(uint8_t *image, int w, int h, feature_t *features, int *count) {
    *count = 0;

    // 简化: 使用FAST角点检测
    for (int y = 3; y < h - 3; y++) {
        for (int x = 3; x < w - 3; x++) {
            if (is_fast_corner(image, w, x, y, 20)) {
                features[*count].x = x;
                features[*count].y = y;
                (*count)++;
            }
        }
    }
}

// 特征匹配
int match_features(feature_t *f1, int n1, feature_t *f2, int n2, int *matches) {
    int match_count = 0;

    for (int i = 0; i < n1; i++) {
        int best = -1, second = -1;
        int best_dist = INT_MAX, second_dist = INT_MAX;

        for (int j = 0; j < n2; j++) {
            int dist = hamming_distance(f1[i].descriptor, f2[j].descriptor, 32);
            if (dist < best_dist) {
                second_dist = best_dist; second = best;
                best_dist = dist; best = j;
            } else if (dist < second_dist) {
                second_dist = dist; second = j;
            }
        }

        // 比率测试
        if (best_dist < 0.7 * second_dist) {
            matches[match_count++] = i;
            matches[match_count++] = best;
        }
    }

    return match_count / 2;
}
```

```c
// 差速驱动运动学
typedef struct {
    float linear;   // 线速度 m/s
    float angular;  // 角速度 rad/s
} velocity_t;

typedef struct {
    float left;   // 左轮速度 m/s
    float right;  // 右轮速度 m/s
} wheel_vel_t;

wheel_vel_t diff_drive_ik(velocity_t vel, float wheel_base) {
    wheel_vel_t wv;
    wv.left = vel.linear - vel.angular * wheel_base / 2;
    wv.right = vel.linear + vel.angular * wheel_base / 2;
    return wv;
}

velocity_t diff_drive_fk(wheel_vel_t wv, float wheel_base) {
    velocity_t vel;
    vel.linear = (wv.left + wv.right) / 2;
    vel.angular = (wv.right - wv.left) / wheel_base;
    return vel;
}
```

### 5.2 PID运动控制

```c
// 位置PID
typedef struct {
    float kp, ki, kd;
    float integral;
    float prev_error;
} pid_t;

float pid_update(pid_t *pid, float error, float dt) {
    pid->integral += error * dt;
    float derivative = (error - pid->prev_error) / dt;
    pid->prev_error = error;

    return pid->kp * error + pid->ki * pid->integral + pid->kd * derivative;
}

// 路径跟踪(Pure Pursuit)
float pure_pursuit(float target_x, float target_y, float robot_x, float robot_y,
                   float robot_theta, float look_ahead) {
    float dx = target_x - robot_x;
    float dy = target_y - robot_y;

    // 转换到机器人坐标系
    float local_x =  cosf(robot_theta)*dx + sinf(robot_theta)*dy;
    float local_y = -sinf(robot_theta)*dx + cosf(robot_theta)*dy;

    // 计算曲率
    float curvature = 2 * local_y / (look_ahead * look_ahead);

    return curvature;
}
```

```c
// 激光雷达数据结构
typedef struct {
    float angle;      // 角度(rad)
    float distance;   // 距离(m)
    float intensity;  // 强度
} lidar_point_t;

// 障碍物检测
typedef struct {
    float x, y;
    float radius;
} obstacle_t;

int detect_obstacles(lidar_point_t *points, int count, obstacle_t *obstacles) {
    int obs_count = 0;

    // 简单聚类
    int cluster_start = 0;
    for (int i = 1; i <= count; i++) {
        if (i == count || fabsf(points[i].distance - points[i-1].distance) > 0.5f) {
            // 计算聚类中心
            float cx = 0, cy = 0;
            int cluster_size = i - cluster_start;
            for (int j = cluster_start; j < i; j++) {
                cx += points[j].distance * cosf(points[j].angle);
                cy += points[j].distance * sinf(points[j].angle);
            }
            cx /= cluster_size;
            cy /= cluster_size;

            obstacles[obs_count].x = cx;
            obstacles[obs_count].y = cy;
            obstacles[obs_count].radius = 0.3f;
            obs_count++;

            cluster_start = i;
        }
    }

    return obs_count;
}
```
