# 音频编解码技术

## 核心概念

- **PCM** - 脉冲编码调制
- **采样率** - 每秒采样数
- **比特深度** - 每个采样位数
- **编解码** - 压缩与解压缩

## 一、音频基础

### 1.1 PCM格式

```c
// PCM参数
typedef struct {
    uint32_t sample_rate;    // 采样率(8000/16000/44100/48000)
    uint8_t channels;        // 通道数(1=单声道, 2=立体声)
    uint8_t bits_per_sample; // 比特深度(8/16/24/32)
    uint32_t samples;        // 采样数
} pcm_params_t;

// PCM帧大小计算
uint32_t pcm_frame_size(pcm_params_t *params) {
    return params->channels * (params->bits_per_sample / 8);
}

// PCM缓冲区大小
uint32_t pcm_buffer_size(pcm_params_t *params, float duration_sec) {
    uint32_t samples = params->sample_rate * duration_sec;
    return samples * pcm_frame_size(params);
}

// 常见PCM格式
// 8kHz/16bit/Mono - 电话语音(16KB/s)
// 16kHz/16bit/Mono - 宽带语音(32KB/s)
// 44.1kHz/16bit/Stereo - CD质量(176.4KB/s)
// 48kHz/24bit/Stereo - 专业音频(288KB/s)

// PCM数据转换
int16_t float_to_pcm16(float sample) {
    if (sample > 1.0f) sample = 1.0f;
    if (sample < -1.0f) sample = -1.0f;
    return (int16_t)(sample * 32767.0f);
}

float pcm16_to_float(int16_t sample) {
    return (float)sample / 32768.0f;
}

// 音量调节
void adjust_volume(int16_t *buffer, size_t samples, float gain) {
    for (size_t i = 0; i < samples; i++) {
        float sample = pcm16_to_float(buffer[i]) * gain;
        buffer[i] = float_to_pcm16(sample);
    }
}

// 静音检测
bool is_silence(const int16_t *buffer, size_t samples, int16_t threshold) {
    for (size_t i = 0; i < samples; i++) {
        if (abs(buffer[i]) > threshold) {
            return false;
        }
    }
    return true;
}
```

### 1.2 采样率转换

```c
// 简单下采样(2倍)
void downsample_2x(const int16_t *input, int16_t *output, size_t input_len) {
    for (size_t i = 0; i < input_len / 2; i++) {
        // 简单抽取
        output[i] = input[i * 2];
    }
}

// 带抗混叠滤波的下采样
void downsample_with_filter(const int16_t *input, int16_t *output,
                            size_t input_len, int ratio) {
    // 简单移动平均抗混叠
    for (size_t i = 0; i < input_len / ratio; i++) {
        int32_t sum = 0;
        for (int j = 0; j < ratio; j++) {
            sum += input[i * ratio + j];
        }
        output[i] = (int16_t)(sum / ratio);
    }
}

// 线性插值上采样
void upsample_linear(const int16_t *input, int16_t *output,
                     size_t input_len, int ratio) {
    for (size_t i = 0; i < input_len - 1; i++) {
        output[i * ratio] = input[i];
        for (int j = 1; j < ratio; j++) {
            float t = (float)j / ratio;
            output[i * ratio + j] = (int16_t)(input[i] * (1 - t) + input[i + 1] * t);
        }
    }
}
```

## 二、音频编解码

### 2.1 ADPCM压缩

```c
// IMA ADPCM编解码
typedef struct {
    int16_t predicted_sample;
    int8_t step_index;
} adpcm_state_t;

// ADPCM索引表
static const int adpcm_index_table[16] = {
    -1, -1, -1, -1, 2, 4, 6, 8,
    -1, -1, -1, -1, 2, 4, 6, 8
};

// ADPCM步长表
static const int adpcm_step_table[89] = {
    7, 8, 9, 10, 11, 12, 13, 14, 16, 17,
    19, 21, 23, 25, 28, 31, 34, 37, 41, 45,
    50, 55, 60, 66, 73, 80, 88, 97, 107, 118,
    130, 143, 157, 173, 190, 209, 230, 253, 279, 307,
    337, 371, 408, 449, 494, 544, 598, 658, 724, 796,
    876, 963, 1060, 1166, 1282, 1411, 1552, 1707, 1878, 2066,
    2272, 2499, 2749, 3024, 3327, 3660, 4026, 4428, 4871, 5358,
    5894, 6484, 7132, 7845, 8630, 9493, 10442, 11487, 12635, 13899,
    15289, 16818, 18500, 20350, 22385, 24623, 27086, 29794, 32767
};

// ADPCM编码
uint8_t adpcm_encode(adpcm_state_t *state, int16_t sample) {
    int step = adpcm_step_table[state->step_index];
    int diff = sample - state->predicted_sample;

    uint8_t code = 0;
    if (diff < 0) {
        code = 8;
        diff = -diff;
    }

    if (diff >= step) { code |= 4; diff -= step; }
    if (diff >= step/2) { code |= 2; diff -= step/4; }
    if (diff >= step/4) { code |= 1; }

    // 更新预测
    int diffq = step / 8;
    if (code & 4) diffq += step;
    if (code & 2) diffq += step / 2;
    if (code & 1) diffq += step / 4;

    if (code & 8) {
        state->predicted_sample -= diffq;
    } else {
        state->predicted_sample += diffq;
    }

    // 饱和
    if (state->predicted_sample > 32767) state->predicted_sample = 32767;
    if (state->predicted_sample < -32768) state->predicted_sample = -32768;

    // 更新步长索引
    state->step_index += adpcm_index_table[code];
    if (state->step_index < 0) state->step_index = 0;
    if (state->step_index > 88) state->step_index = 88;

    return code;
}

// ADPCM解码
int16_t adpcm_decode(adpcm_state_t *state, uint8_t code) {
    int step = adpcm_step_table[state->step_index];

    int diffq = step / 8;
    if (code & 4) diffq += step;
    if (code & 2) diffq += step / 2;
    if (code & 1) diffq += step / 4;

    if (code & 8) {
        state->predicted_sample -= diffq;
    } else {
        state->predicted_sample += diffq;
    }

    if (state->predicted_sample > 32767) state->predicted_sample = 32767;
    if (state->predicted_sample < -32768) state->predicted_sample = -32768;

    state->step_index += adpcm_index_table[code];
    if (state->step_index < 0) state->step_index = 0;
    if (state->step_index > 88) state->step_index = 88;

    return state->predicted_sample;
}

// 批量编码
size_t adpcm_encode_block(adpcm_state_t *state,
                          const int16_t *input, size_t input_len,
                          uint8_t *output) {
    size_t out_idx = 0;
    for (size_t i = 0; i < input_len; i += 2) {
        uint8_t lo = adpcm_encode(state, input[i]);
        uint8_t hi = adpcm_encode(state, input[i + 1]);
        output[out_idx++] = (hi << 4) | lo;
    }
    return out_idx;
}

// 批量解码
size_t adpcm_decode_block(adpcm_state_t *state,
                          const uint8_t *input, size_t input_len,
                          int16_t *output) {
    size_t out_idx = 0;
    for (size_t i = 0; i < input_len; i++) {
        output[out_idx++] = adpcm_decode(state, input[i] & 0x0F);
        output[out_idx++] = adpcm_decode(state, (input[i] >> 4) & 0x0F);
    }
    return out_idx;
}
```

```c
                       int profile, int sample_rate, int channels) {
    int sample_rates[] = {96000, 88200, 64000, 48000, 44100, 32000,
                          24000, 22050, 16000, 12000, 11025, 8000, 7350};
    int sr_index = 0;
    for (int i = 0; i < 13; i++) {
        if (sample_rates[i] == sample_rate) {
            sr_index = i;
            break;
        }
    }

    memset(header, 0, sizeof(adts_header_t));
    header->syncword = 0xFF;
    header->id = 0;  // MPEG4
    header->profile = profile - 1;
    header->sampling_freq = sr_index;
    header->channel_config = channels;
    header->frame_length = (data_len + 7) & 0x1FFF;
    header->frames = 1;
}
```

## 三、音频处理

### 3.1 混音

```c
// 多路混音
void mix_audio(const int16_t **inputs, int input_count,
               int16_t *output, size_t samples) {
    for (size_t i = 0; i < samples; i++) {
        int32_t sum = 0;
        for (int ch = 0; ch < input_count; ch++) {
            sum += inputs[ch][i];
        }

        // 饱和处理
        if (sum > 32767) sum = 32767;
        if (sum < -32768) sum = -32768;

        output[i] = (int16_t)sum;
    }
}

// 带权重的混音
void mix_audio_weighted(const int16_t **inputs, const float *weights,
                        int input_count, int16_t *output, size_t samples) {
    for (size_t i = 0; i < samples; i++) {
        float sum = 0;
        for (int ch = 0; ch < input_count; ch++) {
            sum += pcm16_to_float(inputs[ch][i]) * weights[ch];
        }
        output[i] = float_to_pcm16(sum);
    }
}
```

### 3.2 回声消除

```c
// 简单回声消除器
typedef struct {
    int16_t *ref_buffer;     // 参考信号缓冲
    int16_t *echo_buffer;    // 回声估计缓冲
    int buffer_size;
    int delay_samples;       // 回声延迟
    float suppression;       // 抑制强度
} echo_canceller_t;

// 回声消除
void echo_cancel(echo_canceller_t *ec, const int16_t *near_end,
                 const int16_t *far_end, int16_t *output, size_t samples) {
    for (size_t i = 0; i < samples; i++) {
        // 更新参考信号缓冲
        ec->ref_buffer[ec->buffer_size - 1] = far_end[i];

        // 估计回声(简化: 使用延迟后的远端信号)
        int echo_idx = (ec->buffer_size - ec->delay_samples + i) % ec->buffer_size;
        int16_t echo_estimate = (int16_t)(ec->ref_buffer[echo_idx] * ec->suppression);

        // 消除回声
        int32_t result = near_end[i] - echo_estimate;
        if (result > 32767) result = 32767;
        if (result < -32768) result = -32768;
        output[i] = (int16_t)result;

        // 移动缓冲
        memmove(ec->ref_buffer, ec->ref_buffer + 1,
                (ec->buffer_size - 1) * sizeof(int16_t));
    }
}
```

### 3.3 噪声抑制

```c>
// 谱减法噪声抑制
typedef struct {
    float *noise_estimate;
    int fft_size;
    float alpha;             // 平滑因子
    float floor;             // 噪声下限
} noise_suppressor_t;

// FFT(简化声明)
void fft(const float *input, float *output_real, float *output_imag, int size);
void ifft(const float *input_real, const float *input_imag, float *output, int size);

// 噪声抑制
void suppress_noise(noise_suppressor_t *ns, const int16_t *input,
                    int16_t *output, size_t samples) {
    float *windowed = malloc(ns->fft_size * sizeof(float));
    float *mag = malloc(ns->fft_size * sizeof(float));
    float *phase = malloc(ns->fft_size * sizeof(float));
    float *out_real = malloc(ns->fft_size * sizeof(float));
    float *out_imag = malloc(ns->fft_size * sizeof(float));

    for (size_t frame = 0; frame < samples; frame += ns->fft_size / 2) {
        // 加窗
        for (int i = 0; i < ns->fft_size; i++) {
            float w = 0.54f - 0.46f * cosf(2 * M_PI * i / (ns->fft_size - 1));
            windowed[i] = pcm16_to_float(input[frame + i]) * w;
        }

        // FFT
        fft(windowed, out_real, out_imag, ns->fft_size);

        // 计算幅度和相位
        for (int i = 0; i < ns->fft_size; i++) {
            mag[i] = sqrtf(out_real[i] * out_real[i] + out_imag[i] * out_imag[i]);
            phase[i] = atan2f(out_imag[i], out_real[i]);
        }

        // 更新噪声估计
        for (int i = 0; i < ns->fft_size; i++) {
            ns->noise_estimate[i] = ns->alpha * ns->noise_estimate[i] +
                                    (1 - ns->alpha) * mag[i];
        }

        // 谱减
        for (int i = 0; i < ns->fft_size; i++) {
            float gain = mag[i] - ns->noise_estimate[i];
            if (gain < ns->floor * mag[i]) {
                gain = ns->floor * mag[i];
            }
            gain /= (mag[i] + 1e-10f);

            out_real[i] *= gain;
            out_imag[i] *= gain;
        }

        // IFFT
        ifft(out_real, out_imag, windowed, ns->fft_size);

        // 重叠相加
        for (int i = 0; i < ns->fft_size / 2; i++) {
            int32_t sample = float_to_pcm16(windowed[i]);
            output[frame + i] = (int16_t)sample;
        }
    }

    free(windowed);
    free(mag);
    free(phase);
    free(out_real);
    free(out_imag);
}
```

## 四、语音活动检测(VAD)

```c
// VAD参数
typedef struct {
    float energy_threshold;
    float zcr_threshold;      // 过零率阈值
    int hangover_frames;      // 保持帧数
    int min_speech_frames;    // 最小语音帧数
} vad_params_t;

// VAD状态
typedef struct {
    bool is_speech;
    int hangover_count;
    int speech_frame_count;
    float energy;
    float zcr;
} vad_state_t;

// 计算帧能量
float frame_energy(const int16_t *frame, size_t samples) {
    float energy = 0;
    for (size_t i = 0; i < samples; i++) {
        float sample = pcm16_to_float(frame[i]);
        energy += sample * sample;
    }
    return energy / samples;
}

// 计算过零率
float zero_crossing_rate(const int16_t *frame, size_t samples) {
    int count = 0;
    for (size_t i = 1; i < samples; i++) {
        if ((frame[i] >= 0 && frame[i-1] < 0) ||
            (frame[i] < 0 && frame[i-1] >= 0)) {
            count++;
        }
    }
    return (float)count / (samples - 1);
}

// VAD检测
bool vad_detect(vad_state_t *state, vad_params_t *params,
                const int16_t *frame, size_t samples) {
    state->energy = frame_energy(frame, samples);
    state->zcr = zero_crossing_rate(frame, samples);

    bool current_speech = (state->energy > params->energy_threshold) &&
                          (state->zcr < params->zcr_threshold);

    if (current_speech) {
        state->speech_frame_count++;
        if (state->speech_frame_count >= params->min_speech_frames) {
            state->is_speech = true;
        }
        state->hangover_count = params->hangover_frames;
    } else {
        state->speech_frame_count = 0;
        if (state->hangover_count > 0) {
            state->hangover_count--;
        } else {
            state->is_speech = false;
        }
    }

    return state->is_speech;
}
```

### 5.1 音频管道

```c
// 音频管道
typedef struct {
    // 采集
    struct {
        uint32_t sample_rate;
        int bits;
        int channels;
        void *i2s_handle;
    } capture;

    // 处理
    struct {
        noise_suppressor_t *ns;
        echo_canceller_t *aec;
        vad_state_t *vad;
    } process;

    // 编码
    struct {
        void *encoder;
        uint8_t *encoded_buffer;
        size_t encoded_size;
    } encode;

    // 传输
    struct {
        void *transport;
        uint8_t *send_buffer;
    } transmit;
} audio_pipeline_t;

// 音频处理任务
void audio_task(void *param) {
    audio_pipeline_t *pipe = (audio_pipeline_t *)param;

    int16_t capture_buf[320];   // 20ms @ 16kHz
    int16_t process_buf[320];
    uint8_t encode_buf[128];

    while (1) {
        // 采集
        i2s_read(pipe->capture.i2s_handle, capture_buf, sizeof(capture_buf));

        // 处理
        suppress_noise(pipe->process.ns, capture_buf, process_buf, 320);

        // VAD
        bool speech = vad_detect(pipe->process.vad, NULL, process_buf, 320);

        if (speech) {
            // 编码
            int encoded = opus_encode_frame(pipe->encode.encoder,
                                           process_buf, 320,
                                           encode_buf, sizeof(encode_buf));

            // 传输
            transport_send(pipe->transmit.transport, encode_buf, encoded);
        }
    }
}
```
