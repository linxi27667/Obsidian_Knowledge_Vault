# 信号处理基础

## 核心概念

- **信号** - 随时间变化的物理量
- **采样** - 连续信号离散化
- **频域** - 信号的频率表示
- **滤波** - 去除不需要的频率成分

---

## 一、信号基础

### 1.1 信号分类

| 分类 | 类型 | 示例 |
|------|------|------|
| 连续/离散 | 连续时间/离散时间 | 模拟/数字信号 |
| 周期/非周期 | 重复/不重复 | 正弦波/语音 |
| 确定/随机 | 可预测/不可预测 | 方波/噪声 |

---

### 1.2 基本信号

**正弦信号：**
$$x(t) = A \sin(2\pi f t + \phi)$$

**单位阶跃：**
$$u(t) = \begin{cases} 1, & t \geq 0 \\ 0, & t < 0 \end{cases}$$

**单位冲激：**
$$\delta(t) = \begin{cases} \infty, & t = 0 \\ 0, & t \neq 0 \end{cases}, \quad \int_{-\infty}^{\infty} \delta(t) dt = 1$$

## 二、采样定理

### 2.1 奈奎斯特定理

**采样定理：** 采样频率 ≥ 2 × 信号最高频率

$$f_s \geq 2 f_{max}$$

**混叠：** 采样率不足导致高频信号伪装成低频

---

### 2.2 采样与重建

```c
// 采样
float sample_signal(float (*signal)(float), float fs, int n) {
    return signal(n / fs);
}

// 重建(理想低通)
float reconstruct(float *samples, int n_samples, float fs, float t) {
    float result = 0;
    for (int n = 0; n < n_samples; n++) {
        float sinc_arg = M_PI * fs * (t - n / fs);
        float sinc_val = (sinc_arg == 0) ? 1.0f : sinf(sinc_arg) / sinc_arg;
        result += samples[n] * sinc_val;
    }
    return result;
}
```

## 三、傅里叶变换

### 3.1 DFT/FFT

**离散傅里叶变换(DFT)：**
$$X[k] = \sum_{n=0}^{N-1} x[n] e^{-j2\pi kn/N}$$

**逆DFT：**
$$x[n] = \frac{1}{N} \sum_{k=0}^{N-1} X[k] e^{j2\pi kn/N}$$

---

### 3.2 FFT实现

```c
#include <math.h>

#define FFT_SIZE 256

// 蝶形运算
void fft(float *real, float *imag, int n) {
    if (n <= 1) return;

    // 位反转
    for (int i = 1, j = 0; i < n; i++) {
        int bit = n >> 1;
        for (; j & bit; bit >>= 1) {
            j ^= bit;
        }
        j ^= bit;

        if (i < j) {
            float temp;
            temp = real[i]; real[i] = real[j]; real[j] = temp;
            temp = imag[i]; imag[i] = imag[j]; imag[j] = temp;
        }
    }

    // FFT
    for (int len = 2; len <= n; len <<= 1) {
        float angle = -2.0f * M_PI / len;
        float w_real = cosf(angle);
        float w_imag = sinf(angle);

        for (int i = 0; i < n; i += len) {
            float cur_real = 1.0f, cur_imag = 0.0f;

            for (int j = 0; j < len / 2; j++) {
                int u = i + j;
                int v = i + j + len / 2;

                float t_real = cur_real * real[v] - cur_imag * imag[v];
                float t_imag = cur_real * imag[v] + cur_imag * real[v];

                real[v] = real[u] - t_real;
                imag[v] = imag[u] - t_imag;
                real[u] += t_real;
                imag[u] += t_imag;

                float new_real = cur_real * w_real - cur_imag * w_imag;
                cur_imag = cur_real * w_imag + cur_imag * w_real;
                cur_real = new_real;
            }
        }
    }
}

// 计算幅度谱
void magnitude_spectrum(float *real, float *imag, float *mag, int n) {
    for (int i = 0; i < n; i++) {
        mag[i] = sqrtf(real[i] * real[i] + imag[i] * imag[i]) / n;
    }
}
```

### 3.3 窗函数

```c
// Hanning窗
void hanning_window(float *signal, int n) {
    for (int i = 0; i < n; i++) {
        float w = 0.5f * (1.0f - cosf(2.0f * M_PI * i / (n - 1)));
        signal[i] *= w;
    }
}

// Hamming窗
void hamming_window(float *signal, int n) {
    for (int i = 0; i < n; i++) {
        float w = 0.54f - 0.46f * cosf(2.0f * M_PI * i / (n - 1));
        signal[i] *= w;
    }
}

// Blackman窗
void blackman_window(float *signal, int n) {
    for (int i = 0; i < n; i++) {
        float w = 0.42f - 0.5f * cosf(2.0f * M_PI * i / (n - 1))
                  + 0.08f * cosf(4.0f * M_PI * i / (n - 1));
        signal[i] *= w;
    }
}
```

## 四、数字滤波器

### 4.1 FIR滤波器

**有限冲激响应：**
$$y[n] = \sum_{k=0}^{M} b_k x[n-k]$$

```c
#define FIR_ORDER 32

typedef struct {
    float coeffs[FIR_ORDER];
    float buffer[FIR_ORDER];
    int index;
} fir_filter_t;

float fir_process(fir_filter_t *f, float input) {
    f->buffer[f->index] = input;

    float output = 0;
    int idx = f->index;
    for (int i = 0; i < FIR_ORDER; i++) {
        output += f->coeffs[i] * f->buffer[idx];
        idx = (idx == 0) ? FIR_ORDER - 1 : idx - 1;
    }

    f->index = (f->index + 1) % FIR_ORDER;
    return output;
}

// 设计低通FIR滤波器(窗函数法)
void design_lowpass_fir(fir_filter_t *f, float fc, float fs) {
    int M = FIR_ORDER - 1;
    float wc = 2.0f * M_PI * fc / fs;

    for (int i = 0; i < FIR_ORDER; i++) {
        int n = i - M / 2;
        if (n == 0) {
            f->coeffs[i] = wc / M_PI;
        } else {
            f->coeffs[i] = sinf(wc * n) / (M_PI * n);
        }
        // Hanning窗
        f->coeffs[i] *= 0.5f * (1.0f - cosf(2.0f * M_PI * i / M));
    }
}
```

### 4.2 IIR滤波器

**无限冲激响应：**
$$y[n] = \sum_{k=0}^{M} b_k x[n-k] - \sum_{k=1}^{N} a_k y[n-k]$$

```c
#define IIR_ORDER 2

typedef struct {
    float a[IIR_ORDER + 1];
    float b[IIR_ORDER + 1];
    float x[IIR_ORDER + 1];
    float y[IIR_ORDER + 1];
} iir_filter_t;

float iir_process(iir_filter_t *f, float input) {
    // 移位
    for (int i = IIR_ORDER; i > 0; i--) {
        f->x[i] = f->x[i - 1];
        f->y[i] = f->y[i - 1];
    }
    f->x[0] = input;

    // 计算输出
    float output = 0;
    for (int i = 0; i <= IIR_ORDER; i++) {
        output += f->b[i] * f->x[i];
    }
    for (int i = 1; i <= IIR_ORDER; i++) {
        output -= f->a[i] * f->y[i];
    }
    f->y[0] = output;

    return output;
}

// 二阶低通Butterworth滤波器系数
void design_butterworth_lowpass(iir_filter_t *f, float fc, float fs) {
    float wc = 2.0f * M_PI * fc / fs;
    float k = tanf(wc / 2.0f);
    float k2 = k * k;
    float sqrt2 = 1.41421356f;

    float norm = 1.0f + sqrt2 * k + k2;

    f->b[0] = k2 / norm;
    f->b[1] = 2.0f * k2 / norm;
    f->b[2] = k2 / norm;
    f->a[0] = 1.0f;
    f->a[1] = 2.0f * (k2 - 1.0f) / norm;
    f->a[2] = (1.0f - sqrt2 * k + k2) / norm;
}
```

## 五、频谱分析

### 5.1 功率谱密度

```c
void power_spectrum(float *signal, float *psd, int n) {
    float real[FFT_SIZE], imag[FFT_SIZE];

    // 复制信号
    for (int i = 0; i < n; i++) {
        real[i] = signal[i];
        imag[i] = 0;
    }

    // 加窗
    hanning_window(real, n);

    // FFT
    fft(real, imag, n);

    // 功率谱
    for (int i = 0; i < n; i++) {
        psd[i] = (real[i] * real[i] + imag[i] * imag[i]) / (n * n);
    }
}
```

---

### 5.2 频谱特征

```c
// 主频率
float dominant_frequency(float *psd, int n, float fs) {
    int max_idx = 1;  // 跳过DC
    for (int i = 2; i < n / 2; i++) {
        if (psd[i] > psd[max_idx]) {
            max_idx = i;
        }
    }
    return (float)max_idx * fs / n;
}

// 频谱质心
float spectral_centroid(float *psd, int n, float fs) {
    float sum_weighted = 0, sum = 0;
    for (int i = 1; i < n / 2; i++) {
        float freq = (float)i * fs / n;
        sum_weighted += freq * psd[i];
        sum += psd[i];
    }
    return (sum > 0) ? sum_weighted / sum : 0;
}
```

## 六、自适应滤波

### 6.1 LMS算法

```c
typedef struct {
    float *weights;
    float *buffer;
    int order;
    float mu;  // 步长
} lms_filter_t;

float lms_process(lms_filter_t *f, float input, float desired) {
    // 更新缓冲
    for (int i = f->order - 1; i > 0; i--) {
        f->buffer[i] = f->buffer[i - 1];
    }
    f->buffer[0] = input;

    // 计算输出
    float output = 0;
    for (int i = 0; i < f->order; i++) {
        output += f->weights[i] * f->buffer[i];
    }

    // 误差
    float error = desired - output;

    // 更新权重
    for (int i = 0; i < f->order; i++) {
        f->weights[i] += f->mu * error * f->buffer[i];
    }

    return output;
}
```

## 七、相关与卷积

### 7.1 互相关

```c
void cross_correlation(float *x, float *y, float *r, int n) {
    for (int lag = 0; lag < n; lag++) {
        r[lag] = 0;
        for (int i = 0; i < n - lag; i++) {
            r[lag] += x[i] * y[i + lag];
        }
        r[lag] /= (n - lag);
    }
}
```

---

### 7.2 卷积

```c
void convolution(float *x, int nx, float *h, int nh, float *y) {
    int ny = nx + nh - 1;
    for (int i = 0; i < ny; i++) {
        y[i] = 0;
        for (int j = 0; j < nh; j++) {
            if (i - j >= 0 && i - j < nx) {
                y[i] += x[i - j] * h[j];
            }
        }
    }
}
```

## 八、MFCC特征

### 8.1 梅尔频率倒谱系数

```c
#define N_MFCC 13
#define N_MEL_FILTERS 26

float hz_to_mel(float hz) {
    return 2595.0f * log10f(1.0f + hz / 700.0f);
}

float mel_to_hz(float mel) {
    return 700.0f * (powf(10.0f, mel / 2595.0f) - 1.0f);
}

void extract_mfcc(float *signal, int n_samples, float fs, float mfcc[][N_MFCC]) {
    float frame_size = 0.025f;  // 25ms
    float frame_step = 0.010f;  // 10ms
    int frame_len = (int)(frame_size * fs);
    int step_len = (int)(frame_step * fs);
    int n_frames = (n_samples - frame_len) / step_len + 1;

    for (int f = 0; f < n_frames; f++) {
        float frame[FFT_SIZE] = {0};
        float real[FFT_SIZE], imag[FFT_SIZE];

        // 提取帧
        for (int i = 0; i < frame_len; i++) {
            frame[i] = signal[f * step_len + i];
        }

        // 预加重
        for (int i = frame_len - 1; i > 0; i--) {
            frame[i] -= 0.97f * frame[i - 1];
        }

        // 加窗
        hamming_window(frame, frame_len);

        // FFT
        for (int i = 0; i < FFT_SIZE; i++) {
            real[i] = frame[i];
            imag[i] = 0;
        }
        fft(real, imag, FFT_SIZE);

        // 功率谱
        float power[FFT_SIZE / 2 + 1];
        for (int i = 0; i <= FFT_SIZE / 2; i++) {
            power[i] = (real[i] * real[i] + imag[i] * imag[i]) / FFT_SIZE;
        }

        // Mel滤波
        float mel_energies[N_MEL_FILTERS] = {0};
        // ... (Mel滤波器组计算)

        // DCT得到MFCC
        for (int i = 0; i < N_MFCC; i++) {
            float sum = 0;
            for (int j = 0; j < N_MEL_FILTERS; j++) {
                sum += logf(mel_energies[j] + 1e-10f) *
                       cosf(M_PI * i * (2 * j + 1) / (2 * N_MEL_FILTERS));
            }
            mfcc[f][i] = sum;
        }
    }
}
```
