## 一、音频技术

### 1.1 I2S接口

```c
// I2S配置
typedef struct {
    uint32_t sample_rate;    // 采样率
    uint8_t  bits_per_sample;// 位深
    uint8_t  channels;       // 通道数
    i2s_mode_t mode;         // 主/从模式
} i2s_config_t;

// I2S初始化
void i2s_init(i2s_config_t *config) {
    // 配置I2S外设
    I2S->CR1 = I2S_CR1_I2SMOD |     // I2S模式
               I2S_CR1_I2SE;         // 使能

    // 采样率配置
    if (config->sample_rate == 44100) {
        I2S->PR = 0x00000002;  // 分频系数
    }

    // 位深配置
    switch (config->bits_per_sample) {
        case 16: I2S->CR2 |= I2S_CR2_DS_0 | I2S_CR2_DS_1; break;
        case 24: I2S->CR2 |= I2S_CR2_DS_0 | I2S_CR2_DS_1 | I2S_CR2_DS_2; break;
        case 32: I2S->CR2 |= I2S_CR2_DS_2; break;
    }
}

// I2S音频流
typedef struct {
    int16_t left;
    int16_t right;
} audio_sample_t;

// 音频缓冲区
#define AUDIO_BUFFER_SIZE 1024
audio_sample_t audio_buffer[AUDIO_BUFFER_SIZE];

// DMA传输
void audio_dma_init(void) {
    // 配置DMA
    DMA1_Stream5->CR = DMA_SxCR_MINC |   // 内存递增
                       DMA_SxCR_CIRC |    // 循环模式
                       DMA_SxCR_DIR_0 |   // 内存到外设
                       DMA_SxCR_PSIZE_0 | // 16位外设
                       DMA_SxCR_MSIZE_0;  // 16位内存

    DMA1_Stream5->NDTR = AUDIO_BUFFER_SIZE;
    DMA1_Stream5->PAR = (uint32_t)&I2S->DR;
    DMA1_Stream5->M0AR = (uint32_t)audio_buffer;
}
```

### 1.2 音频编解码

```c
// 音频编解码器接口
typedef struct {
    int (*init)(void);
    int (*encode)(int16_t *pcm, int samples, uint8_t *output);
    int (*decode)(uint8_t *input, int len, int16_t *pcm);
    int (*set_bitrate)(int bitrate);
} audio_codec_t;

// PCM数据处理
void process_audio(audio_sample_t *buffer, int samples) {
    for (int i = 0; i < samples; i++) {
        // 音量调节
        buffer[i].left = (int16_t)(buffer[i].left * volume_factor);
        buffer[i].right = (int16_t)(buffer[i].right * volume_factor);

        // 限幅
        if (buffer[i].left > 32767) buffer[i].left = 32767;
        if (buffer[i].left < -32768) buffer[i].left = -32768;
        if (buffer[i].right > 32767) buffer[i].right = 32767;
        if (buffer[i].right < -32768) buffer[i].right = -32768;
    }
}
```

### 2.1 摄像头接口

```c
// DCMI接口(数字摄像头)
typedef struct {
    uint16_t width;
    uint16_t height;
    uint8_t  format;      // RGB565/YUV422
    uint32_t frame_rate;
} camera_config_t;

// DCMI初始化
void dcmi_init(camera_config_t *config) {
    // 配置DCMI
    DCMI->CR = DCMI_CR_CM |          // 连续模式
               DCMI_CR_PCKPOL |       // 像素时钟极性
               DCMI_CR_VSPOL |        // 垂直同步极性
               DCMI_CR_HSPOL;         // 水平同步极性

    // 配置DMA
    DMA2_Stream1->CR = DMA_SxCR_MINC |
                       DMA_SxCR_CIRC |
                       DMA_SxCR_PSIZE_1 |  // 32位外设
                       DMA_SxCR_MSIZE_1;   // 32位内存
}

// 帧缓冲
#define FRAME_WIDTH  320
#define FRAME_HEIGHT 240
uint16_t frame_buffer[FRAME_WIDTH * FRAME_HEIGHT];

// 摄像头启动
void camera_start(void) {
    DMA2_Stream1->NDTR = FRAME_WIDTH * FRAME_HEIGHT;
    DMA2_Stream1->M0AR = (uint32_t)frame_buffer;

    DCMI->CR |= DCMI_CR_CAPTURE;
}
```

### 2.2 图像处理

```c
// RGB565图像处理
typedef struct {
    uint16_t *data;
    uint16_t width;
    uint16_t height;
} image_t;

// 灰度转换
void rgb_to_grayscale(image_t *src, uint8_t *gray) {
    for (int i = 0; i < src->width * src->height; i++) {
        uint16_t pixel = src->data[i];
        uint8_t r = (pixel >> 11) & 0x1F;
        uint8_t g = (pixel >> 5) & 0x3F;
        uint8_t b = pixel & 0x1F;

        // 灰度公式
        gray[i] = (uint8_t)(0.299f * r * 8 + 0.587f * g * 4 + 0.114f * b * 8);
    }
}

// 高斯模糊
void gaussian_blur(uint8_t *input, uint8_t *output,
                   int width, int height) {
    // 3x3高斯核
    float kernel[3][3] = {
        {1, 2, 1},
        {2, 4, 2},
        {1, 2, 1}
    };

    for (int y = 1; y < height - 1; y++) {
        for (int x = 1; x < width - 1; x++) {
            float sum = 0;
            for (int ky = -1; ky <= 1; ky++) {
                for (int kx = -1; kx <= 1; kx++) {
                    sum += input[(y+ky)*width + (x+kx)] * kernel[ky+1][kx+1];
                }
            }
            output[y*width + x] = (uint8_t)(sum / 16);
        }
    }
}

// 边缘检测(Sobel)
void sobel_edge(uint8_t *input, uint8_t *output,
                int width, int height) {
    for (int y = 1; y < height - 1; y++) {
        for (int x = 1; x < width - 1; x++) {
            // 水平梯度
            int gx = -input[(y-1)*width + (x-1)] + input[(y-1)*width + (x+1)]
                     -2*input[y*width + (x-1)] + 2*input[y*width + (x+1)]
                     -input[(y+1)*width + (x-1)] + input[(y+1)*width + (x+1)];

            // 垂直梯度
            int gy = -input[(y-1)*width + (x-1)] - 2*input[(y-1)*width + x] - input[(y-1)*width + (x+1)]
                     +input[(y+1)*width + (x-1)] + 2*input[(y+1)*width + x] + input[(y+1)*width + (x+1)];

            // 梯度幅值
            int magnitude = abs(gx) + abs(gy);
            if (magnitude > 255) magnitude = 255;

            output[y*width + x] = (uint8_t)magnitude;
        }
    }
}
```

## 三、显示技术

### 3.1 LCD驱动

```c
// SPI LCD驱动
typedef struct {
    uint16_t width;
    uint16_t height;
    uint8_t  format;
} lcd_config_t;

// LCD初始化
void lcd_init(lcd_config_t *config) {
    // 硬件复位
    gpio_set_level(LCD_RST_PIN, 0);
    delay_ms(10);
    gpio_set_level(LCD_RST_PIN, 1);
    delay_ms(10);

    // 软件复位
    lcd_write_cmd(0x01);
    delay_ms(120);

    // 像素格式
    lcd_write_cmd(0x3A);
    lcd_write_data(0x55);  // RGB565

    // 显示方向
    lcd_write_cmd(0x36);
    lcd_write_data(0x00);  // 竖屏

    // 显示开
    lcd_write_cmd(0x29);
}

// 区域填充
void lcd_fill_rect(uint16_t x, uint16_t y, uint16_t w, uint16_t h,
                   uint16_t color) {
    // 设置窗口
    lcd_write_cmd(0x2A);  // 列地址
    lcd_write_data(x >> 8);
    lcd_write_data(x & 0xFF);
    lcd_write_data((x + w - 1) >> 8);
    lcd_write_data((x + w - 1) & 0xFF);

    lcd_write_cmd(0x2B);  // 行地址
    lcd_write_data(y >> 8);
    lcd_write_data(y & 0xFF);
    lcd_write_data((y + h - 1) >> 8);
    lcd_write_data((y + h - 1) & 0xFF);

    // 填充颜色
    lcd_write_cmd(0x2C);
    for (int i = 0; i < w * h; i++) {
        lcd_write_data(color >> 8);
        lcd_write_data(color & 0xFF);
    }
}
```

### 4.1 RTSP流媒体

```c
// RTSP简化实现
typedef enum {
    RTSP_OPTIONS,
    RTSP_DESCRIBE,
    RTSP_SETUP,
    RTSP_PLAY,
    RTSP_PAUSE,
    RTSP_TEARDOWN,
} rtsp_method_t;

// RTSP请求构建
int rtsp_build_request(char *buf, rtsp_method_t method,
                       const char *url, int cseq) {
    const char *methods[] = {
        "OPTIONS", "DESCRIBE", "SETUP", "PLAY", "PAUSE", "TEARDOWN"
    };

    return sprintf(buf,
                   "%s %s RTSP/1.0\r\n"
                   "CSeq: %d\r\n"
                   "\r\n",
                   methods[method], url, cseq);
}
```

### 4.2 音频流

```c
// RTP音频包
typedef struct {
    uint8_t  version;     // 2
    uint8_t  padding;     // 填充标志
    uint8_t  extension;   // 扩展标志
    uint8_t  csrc_count;  // CSRC计数
    uint8_t  marker;      // 标记
    uint8_t  payload_type;// 载荷类型
    uint16_t sequence;    // 序列号
    uint32_t timestamp;   // 时间戳
    uint32_t ssrc;        // 同步源
} rtp_header_t;

// RTP音频发送
void rtp_send_audio(int16_t *samples, int count, uint32_t timestamp) {
    uint8_t packet[1500];
    rtp_header_t *header = (rtp_header_t *)packet;

    header->version = 2;
    header->payload_type = 11;  // L16
    header->sequence = htons(seq_num++);
    header->timestamp = htonl(timestamp);
    header->ssrc = htonl(local_ssrc);

    memcpy(packet + sizeof(rtp_header_t), samples, count * 2);

    udp_send(rtp_dest_ip, RTP_PORT, packet,
             sizeof(rtp_header_t) + count * 2);
}
```

## 五、压缩算法

### 5.1 音频压缩

```c
// 简单音频压缩(差分编码)
void compress_audio_diff(int16_t *input, int16_t *output, int samples) {
    int16_t prev = 0;

    for (int i = 0; i < samples; i++) {
        int16_t diff = input[i] - prev;

        // 限制差分范围到8位
        if (diff > 127) diff = 127;
        if (diff < -128) diff = -128;

        output[i] = diff;
        prev = input[i];
    }
}

// 解压
void decompress_audio_diff(int16_t *input, int16_t *output, int samples) {
    int16_t prev = 0;

    for (int i = 0; i < samples; i++) {
        output[i] = prev + input[i];
        prev = output[i];
    }
}
```
