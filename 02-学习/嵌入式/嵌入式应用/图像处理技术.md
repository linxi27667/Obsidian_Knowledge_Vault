# 图像处理技术

## 核心概念

- **图像** - 像素矩阵
- **分辨率** - 宽×高(像素)
- **色彩空间** - 颜色表示方式
- **ISP** - 图像信号处理器

## 一、图像基础

### 1.1 图像参数

| 参数 | 说明 | 常用值 |
|------|------|--------|
| 分辨率 | 像素数量 | 320×240, 640×480, 1920×1080 |
| 色彩深度 | 每像素位数 | 8/16/24 bit |
| 帧率 | 每秒帧数(FPS) | 15/30/60 |

---

### 1.2 色彩空间

| 色彩空间 | 说明 | 应用 |
|----------|------|------|
| RGB | 红绿蓝 | 显示 |
| YUV | 亮度+色度 | 视频编码 |
| HSV | 色相饱和度明度 | 颜色识别 |
| Grayscale | 灰度 | 图像处理 |

**RGB565(16bit)：**
```
Bit:  15..11  10..5   4..0
      R(5)    G(6)    B(5)
```

**YUV422：**
```
每4字节表示2像素: Y0 U Y1 V
```

**YUV420：**
```
每4个Y共享1个U和1个V
```

### 1.3 图像存储

```c
// RGB565图像缓冲
uint8_t framebuffer[320 * 240 * 2];  // 320×240 RGB565

// 获取像素
uint16_t get_pixel(uint8_t *fb, int x, int y, int width) {
    int offset = (y * width + x) * 2;
    return (fb[offset] << 8) | fb[offset + 1];
}

// 设置像素
void set_pixel(uint8_t *fb, int x, int y, int width, uint16_t color) {
    int offset = (y * width + x) * 2;
    fb[offset] = (color >> 8) & 0xFF;
    fb[offset + 1] = color & 0xFF;
}
```

## 三、图像处理算法

### 3.1 灰度转换

```c
// RGB565 → 灰度
void rgb565_to_gray(uint16_t *src, uint8_t *dst, int pixels) {
    for (int i = 0; i < pixels; i++) {
        uint8_t r = (src[i] >> 11) & 0x1F;
        uint8_t g = (src[i] >> 5) & 0x3F;
        uint8_t b = src[i] & 0x1F;
        // 扩展到8bit
        r = (r << 3) | (r >> 2);
        g = (g << 2) | (g >> 4);
        b = (b << 3) | (b >> 2);
        // 加权平均
        dst[i] = (uint8_t)(0.299f * r + 0.587f * g + 0.114f * b);
    }
}
```

### 3.2 二值化

```c
// 固定阈值二值化
void threshold(uint8_t *src, uint8_t *dst, int pixels, uint8_t thresh) {
    for (int i = 0; i < pixels; i++) {
        dst[i] = (src[i] > thresh) ? 255 : 0;
    }
}

// Otsu自适应阈值
uint8_t otsu_threshold(uint8_t *image, int pixels) {
    int hist[256] = {0};
    for (int i = 0; i < pixels; i++) hist[image[i]]++;

    float sum = 0;
    for (int i = 0; i < 256; i++) sum += i * hist[i];

    float sumB = 0, wB = 0, wF = 0;
    float max_var = 0;
    uint8_t best_thresh = 0;

    for (int t = 0; t < 256; t++) {
        wB += hist[t];
        if (wB == 0) continue;
        wF = pixels - wB;
        if (wF == 0) break;

        sumB += t * hist[t];
        float mB = sumB / wB;
        float mF = (sum - sumB) / wF;
        float var = wB * wF * (mB - mF) * (mB - mF);

        if (var > max_var) {
            max_var = var;
            best_thresh = t;
        }
    }
    return best_thresh;
}
```

### 3.3 滤波

**均值滤波：**
```c
void blur_3x3(uint8_t *src, uint8_t *dst, int w, int h) {
    for (int y = 1; y < h-1; y++) {
        for (int x = 1; x < w-1; x++) {
            int sum = 0;
            for (int dy = -1; dy <= 1; dy++) {
                for (int dx = -1; dx <= 1; dx++) {
                    sum += src[(y+dy)*w + (x+dx)];
                }
            }
            dst[y*w + x] = sum / 9;
        }
    }
}
```

**中值滤波：**
```c
void median_3x3(uint8_t *src, uint8_t *dst, int w, int h) {
    for (int y = 1; y < h-1; y++) {
        for (int x = 1; x < w-1; x++) {
            uint8_t buf[9];
            int idx = 0;
            for (int dy = -1; dy <= 1; dy++) {
                for (int dx = -1; dx <= 1; dx++) {
                    buf[idx++] = src[(y+dy)*w + (x+dx)];
                }
            }
            // 排序取中值
            for (int i = 0; i < 5; i++) {
                for (int j = i+1; j < 9; j++) {
                    if (buf[j] < buf[i]) {
                        uint8_t t = buf[i]; buf[i] = buf[j]; buf[j] = t;
                    }
                }
            }
            dst[y*w + x] = buf[4];
        }
    }
}
```

### 3.4 边缘检测

**Sobel算子：**
```c
void sobel(uint8_t *src, uint8_t *dst, int w, int h) {
    int gx_kernel[3][3] = {{-1,0,1},{-2,0,2},{-1,0,1}};
    int gy_kernel[3][3] = {{-1,-2,-1},{0,0,0},{1,2,1}};

    for (int y = 1; y < h-1; y++) {
        for (int x = 1; x < w-1; x++) {
            int gx = 0, gy = 0;
            for (int dy = -1; dy <= 1; dy++) {
                for (int dx = -1; dx <= 1; dx++) {
                    uint8_t val = src[(y+dy)*w + (x+dx)];
                    gx += val * gx_kernel[dy+1][dx+1];
                    gy += val * gy_kernel[dy+1][dx+1];
                }
            }
            int mag = abs(gx) + abs(gy);  // 近似梯度
            dst[y*w + x] = (mag > 255) ? 255 : mag;
        }
    }
}
```

### 3.5 形态学操作

```c
void erode(uint8_t *src, uint8_t *dst, int w, int h) {
    for (int y = 1; y < h-1; y++) {
        for (int x = 1; x < w-1; x++) {
            uint8_t min_val = 255;
            for (int dy = -1; dy <= 1; dy++) {
                for (int dx = -1; dx <= 1; dx++) {
                    uint8_t val = src[(y+dy)*w + (x+dx)];
                    if (val < min_val) min_val = val;
                }
            }
            dst[y*w + x] = min_val;
        }
    }
}
```

## 四、颜色识别

### 4.1 RGB转HSV

```c
typedef struct {
    uint8_t h;  // 0-179
    uint8_t s;  // 0-255
    uint8_t v;  // 0-255
} hsv_t;

hsv_t rgb_to_hsv(uint8_t r, uint8_t g, uint8_t b) {
    hsv_t hsv;
    uint8_t max_val = r > g ? (r > b ? r : b) : (g > b ? g : b);
    uint8_t min_val = r < g ? (r < b ? r : b) : (g < b ? g : b);
    uint8_t diff = max_val - min_val;

    hsv.v = max_val;

    if (max_val == 0) {
        hsv.s = 0;
        hsv.h = 0;
        return hsv;
    }
    hsv.s = (uint8_t)(255 * diff / max_val);

    if (diff == 0) {
        hsv.h = 0;
    } else if (max_val == r) {
        hsv.h = 60 * (g - b) / diff;
    } else if (max_val == g) {
        hsv.h = 120 + 60 * (b - r) / diff;
    } else {
        hsv.h = 240 + 60 * (r - g) / diff;
    }

    // OpenCV HSV: H 0-179
    hsv.h = hsv.h / 2;
    return hsv;
}
```

### 4.2 颜色区域检测

```c
typedef struct {
    int x, y, w, h;
    int area;
} color_region_t;

int detect_color_region(uint16_t *rgb565, int img_w, int img_h,
                        uint8_t h_min, uint8_t h_max,
                        uint8_t s_min, uint8_t s_max,
                        color_region_t *region) {
    int sum_x = 0, sum_y = 0, count = 0;
    int min_x = img_w, max_x = 0, min_y = img_h, max_y = 0;

    for (int y = 0; y < img_h; y++) {
        for (int x = 0; x < img_w; x++) {
            uint16_t pixel = rgb565[y * img_w + x];
            uint8_t r = ((pixel >> 11) & 0x1F) << 3;
            uint8_t g = ((pixel >> 5) & 0x3F) << 2;
            uint8_t b = (pixel & 0x1F) << 3;

            hsv_t hsv = rgb_to_hsv(r, g, b);

            if (hsv.h >= h_min && hsv.h <= h_max &&
                hsv.s >= s_min && hsv.s <= s_max) {
                sum_x += x;
                sum_y += y;
                count++;
                if (x < min_x) min_x = x;
                if (x > max_x) max_x = x;
                if (y < min_y) min_y = y;
                if (y > max_y) max_y = y;
            }
        }
    }

    if (count > 100) {  // 最小面积阈值
        region->x = sum_x / count;  // 质心
        region->y = sum_y / count;
        region->w = max_x - min_x;
        region->h = max_y - min_y;
        region->area = count;
        return 1;
    }
    return 0;
}
```

## 五、二维码与条形码

### 5.1 二维码解码

```c
// 使用quirc库
#include "quirc.h"

struct quirc *qr;

void qr_decode(uint8_t *gray, int w, int h) {
    qr = quirc_new();
    quirc_resize(qr, w, h);

    uint8_t *buf = quirc_begin(qr, &w, &h);
    memcpy(buf, gray, w * h);
    quirc_end(qr);

    int count = quirc_count(qr);
    for (int i = 0; i < count; i++) {
        struct quirc_code code;
        quirc_extract(qr, i, &code);

        struct quirc_data data;
        if (quirc_decode(&code, &data) == QUIRC_SUCCESS) {
            printf("QR: %s\n", data.payload);
        }
    }
    quirc_destroy(qr);
}
```

### 6.2 SPI显示驱动

```c
#include "driver/spi_master.h"

#define LCD_WIDTH  240
#define LCD_HEIGHT 240

spi_device_handle_t spi;

void lcd_init(void) {
    spi_bus_config_t bus_cfg = {
        .mosi_io_num = GPIO_NUM_23,
        .miso_io_num = -1,
        .sclk_io_num = GPIO_NUM_18,
        .quadwp_io_num = -1,
        .quadhd_io_num = -1,
        .max_transfer_sz = LCD_WIDTH * LCD_HEIGHT * 2,
    };
    spi_bus_initialize(SPI2_HOST, &bus_cfg, SPI_DMA_CH_AUTO);

    spi_device_interface_config_t dev_cfg = {
        .clock_speed_hz = 40 * 1000 * 1000,  // 40MHz
        .mode = 0,
        .spics_io_num = GPIO_NUM_5,
        .queue_size = 7,
    };
    spi_bus_add_device(SPI2_HOST, &dev_cfg, &spi);

    // 初始化命令序列
    lcd_cmd(0x01);  // 复位
    vTaskDelay(pdMS_TO_TICKS(150));
    lcd_cmd(0x11);  // 退出睡眠
    vTaskDelay(pdMS_TO_TICKS(120));
    lcd_cmd(0x36, 0x00);  // 显示方向
    lcd_cmd(0x3A, 0x55);  // 16bit颜色
    lcd_cmd(0x29);  // 开显示
}

// 刷新屏幕
void lcd_flush(uint16_t *buffer) {
    lcd_set_window(0, 0, LCD_WIDTH-1, LCD_HEIGHT-1);

    spi_transaction_t trans = {
        .length = LCD_WIDTH * LCD_HEIGHT * 2 * 8,
        .tx_buffer = buffer,
    };
    spi_device_polling_transmit(spi, &trans);
}
```
