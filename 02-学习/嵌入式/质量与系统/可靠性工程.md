# 可靠性工程

## 核心概念

- **可靠性** - 规定条件下完成规定功能的能力
- **MTBF** - 平均故障间隔时间
- **MTTF** - 平均故障前时间
- **失效率** - 单位时间内故障概率

---

## 一、可靠性基础

### 1.1 可靠性指标

| 指标 | 公式 | 说明 |
|------|------|------|
| 可靠度 R(t) | R(t) = e^(-λt) | t时刻正常概率 |
| 失效率 λ | λ = 1/MTBF | 单位时间故障率 |
| MTBF | MTBF = 1/λ | 平均故障间隔 |
| MTTF | MTTF = Σtᵢ/n | 平均寿命 |
| 可用性 A | A = MTBF/(MTBF+MTTR) | 正常运行时间比例 |

---

### 1.2 浴盆曲线

```
失效率
  ↑
  │  ┌──┐
  │  │  │
  │  │  └─────────────────────
  │  │                        ──┐
  │  │                          │
  └──┴──────────────────────────┴──→ 时间
     早期    有用寿命期    磨损期
     故障    (恒定失效率)   故障
```

---

### 1.3 串并联系统

**串联系统(任一故障则系统故障)：**
$$R_s = R_1 \times R_2 \times ... \times R_n$$

**并联系统(全部故障才系统故障)：**
$$R_p = 1 - (1-R_1)(1-R_2)...(1-R_n)$$

**k/n系统(k个正常则系统正常)：**
$$R_{k/n} = \sum_{i=k}^{n} C_n^i R^i (1-R)^{n-i}$$

### 2.1 FMEA(失效模式与影响分析)

| 项目 | 失效模式 | 影响 | 严重度S | 发生度O | 探测度D | RPN |

```c
// 失效模式枚举
typedef enum {
    FAILURE_NONE = 0,
    FAILURE_OVER_VOLTAGE,
    FAILURE_UNDER_VOLTAGE,
    FAILURE_OVER_CURRENT,
    FAILURE_OVER_TEMP,
    FAILURE_SENSOR_DRIFT,
    FAILURE_COMM_TIMEOUT,
    FAILURE_WATCHDOG,
    FAILURE_MEMORY_ERROR,
    FAILURE_SOFTWARE_ERROR,
} failure_mode_t;

// 失效记录
typedef struct {
    failure_mode_t mode;
    uint32_t timestamp;
    uint32_t count;
    float severity;  // 0-1
} failure_record_t;

#define MAX_FAILURES 50
static failure_record_t failure_log[MAX_FAILURES];
static int failure_count = 0;

void log_failure(failure_mode_t mode, float severity) {
    if (failure_count < MAX_FAILURES) {
        failure_log[failure_count].mode = mode;
        failure_log[failure_count].timestamp = get_tick();
        failure_log[failure_count].count = 1;
        failure_log[failure_count].severity = severity;
        failure_count++;
    }
}
```

### 3.2 双机热备

```c
typedef struct {
    bool primary_active;
    bool backup_active;
    uint32_t primary_heartbeat;
    uint32_t backup_heartbeat;
    uint32_t last_check;
} redundancy_state_t;

void redundancy_check(redundancy_state_t *state) {
    uint32_t now = get_tick();

    // 检查主机心跳
    if (now - state->primary_heartbeat > HEARTBEAT_TIMEOUT) {
        state->primary_active = false;
        // 切换到备机
        switch_to_backup();
    }

    // 检查备机心跳
    if (now - state->backup_heartbeat > HEARTBEAT_TIMEOUT) {
        state->backup_active = false;
        // 报警
        raise_alarm(BACKUP_FAILED);
    }

    state->last_check = now;
}
```

### 3.3 ECC纠错

```c
// 海明码(简单示例)
uint8_t hamming_encode(uint8_t data) {
    uint8_t code = 0;

    // 数据位
    code |= (data & 0x01) << 2;  // D1 -> bit 2
    code |= (data & 0x02) << 2;  // D2 -> bit 3
    code |= (data & 0x04) << 2;  // D3 -> bit 4
    code |= (data & 0x08) << 2;  // D4 -> bit 5

    // 校验位
    uint8_t p1 = ((code >> 2) ^ (code >> 4) ^ (code >> 6)) & 1;
    uint8_t p2 = ((code >> 2) ^ (code >> 5) ^ (code >> 6)) & 1;
    uint8_t p4 = ((code >> 4) ^ (code >> 5) ^ (code >> 6)) & 1;

    code |= (p1 << 0) | (p2 << 1) | (p4 << 3);

    return code;
}
```

## 四、容错设计

### 4.1 看门狗

```c
// 窗口看门狗
typedef struct {
    uint32_t min_time;  // 最小喂狗时间
    uint32_t max_time;  // 最大喂狗时间
    uint32_t last_feed;
} window_wdg_t;

bool window_wdg_feed(window_wdg_t *wdg) {
    uint32_t now = get_tick();
    uint32_t elapsed = now - wdg->last_feed;

    if (elapsed < wdg->min_time) {
        // 喂狗太快
        return false;
    }

    if (elapsed > wdg->max_time) {
        // 喂狗太慢
        return false;
    }

    wdg->last_feed = now;
    return true;
}

// 独立看门狗
void iwdg_init(uint32_t timeout_ms) {
    // STM32独立看门狗
    IWDG->KR = 0x5555;  // 允许写入
    IWDG->PR = 4;       // 分频
    IWDG->RLR = timeout_ms * 40 / 256;  // 重装载值
    IWDG->KR = 0xCCCC;  // 启动看门狗
}

void iwdg_feed(void) {
    IWDG->KR = 0xAAAA;  // 喂狗
}
```

### 4.2 检查点(Checkpoint)

```c
typedef enum {
    CHECKPOINT_INIT = 0,
    CHECKPOINT_SENSOR_READ,
    CHECKPOINT_COMPUTE,
    CHECKPOINT_OUTPUT,
    CHECKPOINT_DONE
} checkpoint_t;

static checkpoint_t current_checkpoint = CHECKPOINT_INIT;
static uint32_t checkpoint_counter = 0;

void checkpoint_update(checkpoint_t cp) {
    current_checkpoint = cp;
    checkpoint_counter++;

    // 存储到非易失性存储器
    backup_register_write(BACKUP_REG_CHECKPOINT, cp);
    backup_register_write(BACKUP_REG_COUNTER, checkpoint_counter);
}

checkpoint_t checkpoint_restore(void) {
    current_checkpoint = backup_register_read(BACKUP_REG_CHECKPOINT);
    checkpoint_counter = backup_register_read(BACKUP_REG_COUNTER);
    return current_checkpoint;
}

// 从断点恢复
void system_recover(void) {
    checkpoint_t cp = checkpoint_restore();

    switch (cp) {
        case CHECKPOINT_SENSOR_READ:
            // 从传感器读取阶段恢复
            goto label_sensor_read;
        case CHECKPOINT_COMPUTE:
            // 从计算阶段恢复
            goto label_compute;
        case CHECKPOINT_OUTPUT:
            // 从输出阶段恢复
            goto label_output;
        default:
            // 从头开始
            break;
    }
}
```

### 5.2 温度补偿

```c
// NTC温度补偿
float temperature_compensate(float raw_value, float temperature) {
    // 补偿公式: V_comp = V_raw * (1 + α * (T - T_ref))
    float alpha = 0.001f;  // 温度系数
    float T_ref = 25.0f;   // 参考温度

    return raw_value * (1.0f + alpha * (temperature - T_ref));
}

// ADC温度校准
typedef struct {
    float offset;
    float gain;
    float temp_coeff;
} calibration_t;

float apply_calibration(float raw, calibration_t *cal, float temperature) {
    return (raw + cal->offset) * cal->gain * (1.0f + cal->temp_coeff * (temperature - 25.0f));
}
```

### 5.3 EMC防护

```c
// 软件滤波抗干扰
float emc_filter(float *samples, int count) {
    // 中值滤波
    float sorted[count];
    memcpy(sorted, samples, count * sizeof(float));

    // 简单排序
    for (int i = 0; i < count - 1; i++) {
        for (int j = i + 1; j < count; j++) {
            if (sorted[j] < sorted[i]) {
                float temp = sorted[i];
                sorted[i] = sorted[j];
                sorted[j] = temp;
            }
        }
    }

    return sorted[count / 2];  // 返回中值
}

// 数字滤波
float moving_average_filter(float new_sample) {
    static float buffer[16];
    static int index = 0;
    static float sum = 0;

    sum -= buffer[index];
    buffer[index] = new_sample;
    sum += new_sample;
    index = (index + 1) % 16;

    return sum / 16.0f;
}
```

## 六、寿命预测

### 6.1 Arrhenius模型(温度)

$$AF = e^{\frac{E_a}{k}(\frac{1}{T_use} - \frac{1}{T_test})}$$

- AF: 加速因子
- Ea: 活化能(eV)
- k: 玻尔兹曼常数(8.617×10⁻⁵ eV/K)
- T: 绝对温度(K)

---

### 6.2 寿命计算

```c
// Arrhenius加速因子
float arrhenius_af(float t_use, float t_test, float ea) {
    float k = 8.617e-5f;  // eV/K
    float t_use_k = t_use + 273.15f;
    float t_test_k = t_test + 273.15f;

    return expf(ea / k * (1.0f / t_use_k - 1.0f / t_test_k));
}

// 预测寿命
float predict_lifetime(float test_hours, float test_temp,
                       float use_temp, float ea) {
    float af = arrhenius_af(use_temp, test_temp, ea);
    return test_hours * af;
}
```

```c
    float eta;   // 尺度参数

float weibull_reliability(weibull_t *w, float t) {
```
