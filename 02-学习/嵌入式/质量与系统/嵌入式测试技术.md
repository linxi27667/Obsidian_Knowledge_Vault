# 嵌入式测试技术

## 核心概念

- **单元测试** - 函数级测试
- **集成测试** - 模块间测试
- **系统测试** - 系统级测试
- **HIL测试** - 硬件在环测试

## 二、单元测试

### 2.1 Unity测试框架

```c
// Unity测试示例
#include "unity.h"

// 测试固件
void setUp(void) {
    // 每个测试前执行
}

void tearDown(void) {
    // 每个测试后执行
}

// 测试用例
void test_add(void) {
    TEST_ASSERT_EQUAL_INT(5, add(2, 3));
    TEST_ASSERT_EQUAL_INT(0, add(-1, 1));
    TEST_ASSERT_EQUAL_INT(-3, add(-1, -2));
}

void test_string(void) {
    char buf[32];
    format_string(buf, "Hello %s", "World");
    TEST_ASSERT_EQUAL_STRING("Hello World", buf);
}

void test_float(void) {
    float result = calculate_pi();
    TEST_ASSERT_FLOAT_WITHIN(0.001, 3.14159, result);
}

// 测试套件
int main(void) {
    UNITY_BEGIN();

    RUN_TEST(test_add);
    RUN_TEST(test_string);
    RUN_TEST(test_float);

    return UNITY_END();
}
```

// mock_sensor.h
void sensor_init_Expect(void);
int sensor_read_ExpectAndReturn(int return_val);

// 测试使用Mock
void test_sensor_processing(void) {
    // 设置Mock期望
    sensor_init_Expect();
    sensor_read_ExpectAndReturn(25);

    // 执行被测函数
    float result = process_sensor_data();

    // 验证结果
    TEST_ASSERT_FLOAT_WITHIN(0.1, 25.0, result);

// 回调Mock
int custom_sensor_read(void) {
    return 30;  // 自定义返回值
}

void test_with_callback(void) {
    sensor_read_StubWithCallback(custom_sensor_read);
    float result = process_sensor_data();
    TEST_ASSERT_FLOAT_WITHIN(0.1, 30.0, result);
}

## 三、集成测试

### 3.1 模块集成测试

```c
// 集成测试示例
void test_communication_stack(void) {
    // 初始化通信栈
    uart_init();
    protocol_init();
    app_init();

    // 发送测试数据
    uint8_t test_data[] = {0x01, 0x02, 0x03};
    app_send(test_data, sizeof(test_data));

    // 验证UART输出
    uint8_t uart_output[32];
    int len = uart_get_output(uart_output, sizeof(uart_output));

    // 验证协议封装
    TEST_ASSERT_EQUAL_UINT8(0xAA, uart_output[0]);  // 帧头
    TEST_ASSERT_EQUAL_UINT8(sizeof(test_data), uart_output[1]);  // 长度
    TEST_ASSERT_EQUAL_UINT8_ARRAY(test_data, &uart_output[2], sizeof(test_data));
}

// 接收测试
void test_receive_path(void) {
    // 模拟UART输入
    uint8_t input[] = {0xAA, 0x03, 0x01, 0x02, 0x03, 0xCC};
    uart_put_input(input, sizeof(input));

    // 处理接收
    app_process();

    // 验证应用层数据
    uint8_t received[32];
    int len = app_get_received(received, sizeof(received));
    TEST_ASSERT_EQUAL_INT(3, len);
    TEST_ASSERT_EQUAL_UINT8_ARRAY(&input[2], received, 3);
}
```

## 四、HIL测试

### 4.1 硬件在环测试

```c
// HIL测试框架
typedef struct {
    // 模拟输入
    float adc_values[8];
    uint32_t gpio_inputs;
    uint8_t uart_rx[256];
    int uart_rx_len;

    // 预期输出
    float expected_pwm[4];
    uint32_t expected_gpio;
    uint8_t expected_tx[256];
    int expected_tx_len;
} hil_test_case_t;

void hil_run_test(hil_test_case_t *test) {
    // 设置模拟输入
    for (int i = 0; i < 8; i++) {
        hil_set_adc(i, test->adc_values[i]);
    }
    hil_set_gpio(test->gpio_inputs);
    hil_send_uart(test->uart_rx, test->uart_rx_len);

    // 运行一个周期
    app_one_cycle();

    // 读取实际输出
    float actual_pwm[4];
    for (int i = 0; i < 4; i++) {
        actual_pwm[i] = hil_get_pwm(i);
    }
    uint32_t actual_gpio = hil_read_gpio();

    // 验证
    for (int i = 0; i < 4; i++) {
        TEST_ASSERT_FLOAT_WITHIN(0.01, test->expected_pwm[i], actual_pwm[i]);
    }
    TEST_ASSERT_EQUAL_HEX32(test->expected_gpio, actual_gpio);
}
```

### 4.2 自动化HIL

```c
// 自动化测试脚本
typedef struct {
    const char *test_name;
    void (*setup)(void);
    void (*execute)(void);
    void (*verify)(void);
    void (*cleanup)(void);
} hil_test_suite_t;

hil_test_suite_t test_suite[] = {
    {"ADC Test", adc_setup, adc_execute, adc_verify, adc_cleanup},
    {"PWM Test", pwm_setup, pwm_execute, pwm_verify, pwm_cleanup},
    {"UART Test", uart_setup, uart_execute, uart_verify, uart_cleanup},
    {"GPIO Test", gpio_setup, gpio_execute, gpio_verify, gpio_cleanup},
};

void run_all_tests(void) {
    int pass = 0, fail = 0;

    for (int i = 0; i < sizeof(test_suite)/sizeof(test_suite[0]); i++) {
        printf("Running: %s\n", test_suite[i].test_name);

        test_suite[i].setup();

        bool result = true;
        // try-catch简化
        test_suite[i].execute();
        test_suite[i].verify();

        test_suite[i].cleanup();

        if (result) {
            printf("  PASS\n");
            pass++;
        } else {
            printf("  FAIL\n");
            fail++;
        }
    }

    printf("\nResults: %d pass, %d fail\n", pass, fail);
}
```

## 五、性能测试

### 5.1 实时性测试

```c
// 响应时间测试
typedef struct {
    uint32_t min_us;
    uint32_t max_us;
    uint32_t avg_us;
    uint32_t count;
} timing_stats_t;

void timing_test(void (*func)(void), timing_stats_t *stats, int iterations) {
    stats->min_us = UINT32_MAX;
    stats->max_us = 0;
    stats->avg_us = 0;
    stats->count = iterations;

    for (int i = 0; i < iterations; i++) {
        uint32_t start = get_microseconds();

        func();

        uint32_t elapsed = get_microseconds() - start;

        if (elapsed < stats->min_us) stats->min_us = elapsed;
        if (elapsed > stats->max_us) stats->max_us = elapsed;
        stats->avg_us += elapsed;
    }

    stats->avg_us /= iterations;
}

// 打印统计
void print_timing_stats(const char *name, timing_stats_t *stats) {
    printf("%s: min=%u, max=%u, avg=%u us\n",
           name, stats->min_us, stats->max_us, stats->avg_us);
}
```

## 六、测试自动化

### 6.1 CI集成

```yaml
# .github/workflows/test.yml
name: Tests

on: [push, pull_request]

jobs:
  unit-tests:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v2

      - name: Install dependencies
        run: |
          sudo apt-get install gcc-arm-none-eabi
          pip install gcovr

      - name: Build and test
        run: |
          make test
          gcovr --xml -o coverage.xml

      - name: Upload coverage
        uses: codecov/codecov-action@v1
        with:
          file: coverage.xml
```

### 6.2 测试报告

```c
// JUnit格式测试报告
typedef struct {
    char name[64];
    bool passed;
    float time_s;
    char error_msg[256];
} test_result_t;

void generate_junit_report(test_result_t *results, int count, const char *filename) {
    FILE *f = fopen(filename, "w");
    fprintf(f, "<?xml version=\"1.0\" encoding=\"UTF-8\"?>\n");
    fprintf(f, "<testsuites>\n");
    fprintf(f, "  <testsuite name=\"EmbeddedTests\" tests=\"%d\">\n", count);

    for (int i = 0; i < count; i++) {
        fprintf(f, "    <testcase name=\"%s\" time=\"%.3f\"",
                results[i].name, results[i].time_s);
        if (!results[i].passed) {
            fprintf(f, ">\n");
            fprintf(f, "      <failure message=\"%s\"/>\n", results[i].error_msg);
            fprintf(f, "    </testcase>\n");
        } else {
            fprintf(f, "/>\n");
        }
    }

    fprintf(f, "  </testsuite>\n");
    fprintf(f, "</testsuites>\n");
    fclose(f);
}
```
