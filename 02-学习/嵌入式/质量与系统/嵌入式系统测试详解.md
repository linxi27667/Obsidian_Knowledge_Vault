# 嵌入式系统测试详解

## 核心概念

- **单元测试** - 模块级测试
- **集成测试** - 系统级测试
- **硬件在环** - HIL测试
- **覆盖率** - 代码覆盖度量

```c
#include "unity.h"

// 测试夹具

    // 每个测试前执行

void tearDown(void) {

void test_add_function(void) {

    TEST_ASSERT_EQUAL_INT(0, add(-1, 1));

void test_buffer_overflow(void) {
    uint8_t buf[10];
    memset(buf, 0, sizeof(buf));

    // 测试边界
    TEST_ASSERT_EQUAL(10, write_buffer(buf, sizeof(buf)));
    TEST_ASSERT_EQUAL(-1, write_buffer(buf, sizeof(buf) + 1));
}

// 测试运行

    RUN_TEST(test_add_function);
```

```c
// Mock接口
typedef struct {
    int call_count;
    int return_value;
    void *last_arg;
} mock_t;

static mock_t mock_gpio_read = {0};
static mock_t mock_uart_send = {0};

// Mock实现
int gpio_read_mock(int pin) {
    mock_gpio_read.call_count++;
    mock_gpio_read.last_arg = (void *)(intptr_t)pin;
    return mock_gpio_read.return_value;
}

void uart_send_mock(const uint8_t *data, size_t len) {
    mock_uart_send.call_count++;
    mock_uart_send.last_arg = (void *)data;
}

// 设置Mock返回值
void mock_gpio_read_set_return(int value) {
    mock_gpio_read.return_value = value;
    mock_gpio_read.call_count = 0;
}

// 验证Mock调用
void mock_verify_call_count(mock_t *mock, int expected) {
    TEST_ASSERT_EQUAL_INT(expected, mock->call_count);
}

// 测试使用Mock
void test_button_read(void) {
    mock_gpio_read_set_return(1);

    bool pressed = is_button_pressed();

    TEST_ASSERT_TRUE(pressed);
    mock_verify_call_count(&mock_gpio_read, 1);
}
```

## 二、集成测试

### 2.1 系统测试

```c
// 系统测试框架
typedef struct {
    const char *name;
    void (*setup)(void);
    void (*teardown)(void);
    void (*test)(void);
    bool passed;
    char error_msg[256];
} system_test_t;

#define MAX_SYSTEM_TESTS 32
static system_test_t system_tests[MAX_SYSTEM_TESTS];
static int system_test_count = 0;

// 注册测试
void system_test_register(const char *name, void (*setup)(void),
                           void (*teardown)(void), void (*test)(void)) {
    if (system_test_count >= MAX_SYSTEM_TESTS) return;

    system_test_t *t = &system_tests[system_test_count];
    t->name = name;
    t->setup = setup;
    t->teardown = teardown;
    t->test = test;
    t->passed = false;
    system_test_count++;
}

// 运行所有测试
int system_test_run_all(void) {
    int passed = 0;
    int failed = 0;

    for (int i = 0; i < system_test_count; i++) {
        system_test_t *t = &system_tests[i];

        printf("Running: %s... ", t->name);

        if (t->setup) t->setup();

        // 执行测试
        t->test();

        if (t->teardown) t->teardown();

        if (t->passed) {
            printf("PASSED\n");
            passed++;
        } else {
            printf("FAILED: %s\n", t->error_msg);
            failed++;
        }
    }

    printf("\nResults: %d passed, %d failed\n", passed, failed);
    return failed;
}

// 断言宏
#define SYSTEM_ASSERT(cond, msg) do { \
    if (!(cond)) { \
        t->passed = false; \
        snprintf(t->error_msg, sizeof(t->error_msg), "%s:%d: %s", \
                 __FILE__, __LINE__, msg); \
        return; \
    } \
} while(0)
```

```c
// UART回环测试
void test_uart_loopback(void) {
    uint8_t tx_data[] = "Hello, World!";
    uint8_t rx_data[32];

    // 发送数据
    uart_send(UART1, tx_data, sizeof(tx_data));

    // 接收数据
    int rx_len = uart_receive(UART1, rx_data, sizeof(rx_data), 1000);

    // 验证
    TEST_ASSERT_EQUAL_INT(sizeof(tx_data), rx_len);
    TEST_ASSERT_EQUAL_MEMORY(tx_data, rx_data, sizeof(tx_data));
}

// SPI测试
void test_spi_transfer(void) {
    uint8_t tx[] = {0x9F, 0x00, 0x00, 0x00};  // JEDEC ID
    uint8_t rx[4];

    spi_transfer(SPI1, tx, rx, 4);

    // 验证Flash ID
    TEST_ASSERT_EQUAL_HEX8(0xEF, rx[1]);  // Winbond
    TEST_ASSERT_EQUAL_HEX8(0x40, rx[2]);  // 25Q
}

// I2C测试
void test_i2c_device(void) {
    uint8_t who_am_i;

    // 读取设备ID
    i2c_read_reg(I2C1, 0x68, 0x75, &who_am_i, 1);

    // 验证MPU6050
    TEST_ASSERT_EQUAL_HEX8(0x68, who_am_i);
}
```

```c
// HIL测试配置
typedef struct {
    const char *test_name;
    void (*setup_hardware)(void);
    void (*run_test)(void);
    void (*cleanup)(void);
    uint32_t timeout_ms;
} hil_test_t;

// GPIO测试
void hil_test_gpio(void) {
    // 配置测试引脚
    gpio_set_direction(TEST_PIN, GPIO_MODE_OUTPUT);

    // 测试高电平
    gpio_set_level(TEST_PIN, 1);
    delay_ms(10);
    int level = gpio_get_level(TEST_PIN);
    TEST_ASSERT_EQUAL_INT(1, level);

    // 测试低电平
    gpio_set_level(TEST_PIN, 0);
    delay_ms(10);
    level = gpio_get_level(TEST_PIN);
    TEST_ASSERT_EQUAL_INT(0, level);
}

// ADC测试
void hil_test_adc(void) {
    // 配置ADC
    adc_config_t config = {
        .channel = ADC_CHANNEL_0,
        .atten = ADC_ATTEN_DB_11,
        .width = ADC_WIDTH_BIT_12,
    };
    adc_init(&config);

    // 读取参考电压
    uint32_t adc_value = adc_read();
    float voltage = adc_value * 3.3f / 4096;

    // 验证参考电压（假设2.5V参考）
    TEST_ASSERT_FLOAT_WITHIN(0.1, 2.5, voltage);
}

// PWM测试
void hil_test_pwm(void) {
    // 配置PWM
    pwm_config_t config = {
        .pin = PWM_PIN,
        .freq_hz = 1000,
        .duty = 50,
    };
    pwm_init(&config);

    // 测量频率和占空比
    float freq = measure_frequency(PWM_PIN);
    float duty = measure_duty_cycle(PWM_PIN);

    TEST_ASSERT_FLOAT_WITHIN(10, 1000, freq);
    TEST_ASSERT_FLOAT_WITHIN(2, 50, duty);
}
```

### 3.2 自动化测试

```python
# Python HIL测试脚本
import serial
import time
import pytest

class EmbeddedTester:
    def __init__(self, port, baudrate=115200):
        self.serial = serial.Serial(port, baudrate, timeout=1)

    def send_command(self, cmd):
        self.serial.write(f"{cmd}\n".encode())
        time.sleep(0.1)
        return self.serial.read_all().decode()

    def test_led(self):
        """测试LED控制"""
        response = self.send_command("led on")
        assert "OK" in response

        response = self.send_command("led off")
        assert "OK" in response

    def test_sensor(self):
        """测试传感器读取"""
        response = self.send_command("sensor read temp")
        temp = float(response.strip())
        assert 20 < temp < 30  # 室温范围

    def test_motor(self):
        """测试电机控制"""
        response = self.send_command("motor speed 100")
        assert "OK" in response

        time.sleep(1)

        response = self.send_command("motor speed 0")
        assert "OK" in response

# pytest运行
@pytest.fixture
def tester():
    t = EmbeddedTester("/dev/ttyUSB0")
    yield t
    t.serial.close()

def test_full_system(tester):
    """完整系统测试"""
    tester.test_led()
    tester.test_sensor()
    tester.test_motor()
```

## 四、性能测试

### 4.1 基准测试

```c
// 基准测试框架
typedef struct {
    const char *name;
    void (*func)(void);
    uint32_t iterations;
    uint32_t elapsed_us;
} benchmark_t;

#define MAX_BENCHMARKS 16
static benchmark_t benchmarks[MAX_BENCHMARKS];
static int benchmark_count = 0;

// 注册基准测试
void benchmark_register(const char *name, void (*func)(void), uint32_t iterations) {
    if (benchmark_count >= MAX_BENCHMARKS) return;

    benchmarks[benchmark_count].name = name;
    benchmarks[benchmark_count].func = func;
    benchmarks[benchmark_count].iterations = iterations;
    benchmark_count++;
}

// 运行基准测试
void benchmark_run(benchmark_t *bench) {
    uint32_t start = get_tick_us();

    for (uint32_t i = 0; i < bench->iterations; i++) {
        bench->func();
    }

    bench->elapsed_us = get_tick_us() - start;
}

// 运行所有基准测试
void benchmark_run_all(void) {
    printf("Benchmark Results:\n");
    printf("%-30s %10s %10s %10s\n", "Name", "Iterations", "Time(us)", "Per Call");
    printf("--------------------------------------------------------------\n");

    for (int i = 0; i < benchmark_count; i++) {
        benchmark_run(&benchmarks[i]);

        float per_call = (float)benchmarks[i].elapsed_us / benchmarks[i].iterations;
        printf("%-30s %10lu %10lu %10.2f\n",
               benchmarks[i].name,
               benchmarks[i].iterations,
               benchmarks[i].elapsed_us,
               per_call);
    }
}

// 示例基准测试
void benchmark_crc32(void) {
    uint8_t data[1024];
    memset(data, 0xAA, sizeof(data));
    crc32(data, sizeof(data));
}

void benchmark_aes_encrypt(void) {
    uint8_t key[16] = {0};
    uint8_t data[16] = {0};
    aes_encrypt(data, key);
}
```

### 4.2 内存测试

```c
// 内存测试
typedef struct {
    size_t total_heap;
    size_t used_heap;
    size_t free_heap;
    size_t max_block;
    int allocation_count;
} memory_stats_t;

memory_stats_t memory_get_stats(void) {
    memory_stats_t stats;
    stats.total_heap = configTOTAL_HEAP_SIZE;
    stats.free_heap = xPortGetFreeHeapSize();
    stats.used_heap = stats.total_heap - stats.free_heap;
    stats.max_block = xPortGetMinimumEverFreeHeapSize();
    stats.allocation_count = 0;  // 需要自定义跟踪
    return stats;
}

// 内存泄漏检测
void memory_leak_test(void) {
    memory_stats_t before = memory_get_stats();

    // 执行被测功能
    for (int i = 0; i < 100; i++) {
        do_some_work();
    }

    memory_stats_t after = memory_get_stats();

    // 检查内存是否泄漏
    if (after.used_heap > before.used_heap) {
        printf("Memory leak detected: %zu bytes\n",
               after.used_heap - before.used_heap);
    }
}

// 栈溢出检测
void stack_overflow_test(void) {
    uint32_t stack_marker;
    uint32_t *stack_ptr = &stack_marker;

    // 记录栈起始位置
    uint32_t stack_start = (uint32_t)stack_ptr;

    // 执行被测功能
    recursive_function(100);

    // 检查栈使用
    uint32_t stack_used = stack_start - (uint32_t)stack_ptr;
    printf("Stack used: %lu bytes\n", stack_used);
}
```

## 五、回归测试

### 5.1 测试套件

```c
// 测试套件
typedef struct {
    const char *suite_name;
    void (*tests[])(void);
    int test_count;
} test_suite_t;

// 定义测试套件
#define DEFINE_TEST_SUITE(name, ...) \
    static void (*name##_tests[])(void) = { __VA_ARGS__ }; \
    static test_suite_t name##_suite = { \
        .suite_name = #name, \
        .tests = name##_tests, \
        .test_count = sizeof(name##_tests) / sizeof(name##_tests[0]) \
    }

// 运行测试套件
void run_test_suite(test_suite_t *suite) {
    printf("Running suite: %s\n", suite->suite_name);

    int passed = 0;
    int failed = 0;

    for (int i = 0; i < suite->test_count; i++) {
        // 运行测试
        suite->tests[i]();

        // 检查结果（简化）
        if (unity_current_test_failed) {
            failed++;
        } else {
            passed++;
        }
    }

    printf("Suite %s: %d passed, %d failed\n",
           suite->suite_name, passed, failed);
}

// 测试报告
typedef struct {
    const char *test_name;
    bool passed;
    const char *error_msg;
    uint32_t duration_ms;
} test_report_t;

#define MAX_REPORTS 100
static test_report_t reports[MAX_REPORTS];
static int report_count = 0;

void test_report_add(const char *name, bool passed, const char *error, uint32_t duration) {
    if (report_count >= MAX_REPORTS) return;

    test_report_t *r = &reports[report_count];
    r->test_name = name;
    r->passed = passed;
    r->error_msg = error;
    r->duration_ms = duration;
    report_count++;
}

void test_report_print(void) {
    printf("\n=== Test Report ===\n");
    printf("%-40s %-10s %-10s\n", "Test", "Status", "Duration");
    printf("--------------------------------------------------\n");

    int passed = 0, failed = 0;
    for (int i = 0; i < report_count; i++) {
        printf("%-40s %-10s %-10lu ms\n",
               reports[i].test_name,
               reports[i].passed ? "PASS" : "FAIL",
               reports[i].duration_ms);

        if (reports[i].passed) passed++;
        else failed++;
    }

    printf("\nTotal: %d passed, %d failed\n", passed, failed);
}
```

## 附录：测试类型对比

| 测试类型 | 范围 | 工具 | 时机 |
|----------|------|------|------|
| 单元测试 | 函数/模块 | Unity, CMock | 开发中 |
| 集成测试 | 模块交互 | pytest | 集成时 |
| HIL测试 | 硬件交互 | 自定义 | 部署前 |
| 性能测试 | 响应时间 | 基准工具 | 优化时 |
| 回归测试 | 全部功能 | CI/CD | 每次提交 |
