### 1.2 边缘节点架构

```c
// 边缘节点结构
typedef struct {
    char node_id[32];
    char location[64];
    float latitude;
    float longitude;

    // 计算资源
    uint32_t cpu_cores;
    uint32_t memory_mb;
    uint32_t storage_gb;

    // 网络
    char ip_address[16];
    uint16_t port;
    float bandwidth_mbps;

    // 状态
    float cpu_usage;
    float memory_usage;
    float temperature;
    bool online;
} edge_node_t;

// 边缘应用
typedef struct {
    char app_id[32];
    char name[32];
    char version[16];
    enum { APP_TYPE_INFER, APP_TYPE_AGGREGATE, APP_TYPE_FILTER } type;
    uint32_t cpu_required;
    uint32_t memory_required;
    bool running;
} edge_app_t;

// 资源调度
int schedule_app(edge_node_t *nodes, int node_count, edge_app_t *app) {
    // 找到资源最充足的节点
    int best_node = -1;
    float best_score = 0;

    for (int i = 0; i < node_count; i++) {
        if (!nodes[i].online) continue;

        float available_cpu = 1.0f - nodes[i].cpu_usage;
        float available_mem = 1.0f - nodes[i].memory_usage;
        float score = available_cpu * 0.5f + available_mem * 0.5f;

        if (score > best_score &&
            available_cpu * nodes[i].cpu_cores * 1000 >= app->cpu_required &&
            available_mem * nodes[i].memory_mb >= app->memory_required) {
            best_score = score;
            best_node = i;
        }
    }

    return best_node;
}
```

## 二、数据处理

### 2.1 数据聚合

```c
// 数据聚合器
typedef struct {
    uint32_t count;
    float sum;
    float min;
    float max;
    float avg;
    float stddev;
} aggregator_t;

// 滑动窗口聚合
#define WINDOW_SIZE 60
typedef struct {
    float buffer[WINDOW_SIZE];
    int index;
    int count;
    float sum;
} sliding_window_t;

void window_init(sliding_window_t *w) {
    memset(w, 0, sizeof(sliding_window_t));
}

float window_add(sliding_window_t *w, float value) {
    // 移除旧值
    if (w->count >= WINDOW_SIZE) {
        w->sum -= w->buffer[w->index];
    } else {
        w->count++;
    }

    // 添加新值
    w->buffer[w->index] = value;
    w->sum += value;
    w->index = (w->index + 1) % WINDOW_SIZE;

    return w->sum / w->count;
}

// 多传感器聚合
typedef struct {
    float temperature;
    float humidity;
    float pressure;
    uint32_t timestamp;
} sensor_data_t;

typedef struct {
    sensor_data_t min;
    sensor_data_t max;
    sensor_data_t avg;
    uint32_t count;
} aggregate_result_t;

void aggregate_sensor_data(const sensor_data_t *data, int count,
                           aggregate_result_t *result) {
    result->count = count;
    result->min = data[0];
    result->max = data[0];

    float sum_temp = 0, sum_hum = 0, sum_pres = 0;

    for (int i = 0; i < count; i++) {
        // 最小值
        if (data[i].temperature < result->min.temperature) result->min.temperature = data[i].temperature;
        if (data[i].humidity < result->min.humidity) result->min.humidity = data[i].humidity;
        if (data[i].pressure < result->min.pressure) result->min.pressure = data[i].pressure;

        // 最大值
        if (data[i].temperature > result->max.temperature) result->max.temperature = data[i].temperature;
        if (data[i].humidity > result->max.humidity) result->max.humidity = data[i].humidity;
        if (data[i].pressure > result->max.pressure) result->max.pressure = data[i].pressure;

        // 累加
        sum_temp += data[i].temperature;
        sum_hum += data[i].humidity;
        sum_pres += data[i].pressure;
    }

    // 平均值
    result->avg.temperature = sum_temp / count;
    result->avg.humidity = sum_hum / count;
    result->avg.pressure = sum_pres / count;
}
```

### 2.2 数据过滤

```c
// 异常值过滤
bool is_outlier(float value, float mean, float stddev, float threshold) {
    float z_score = fabsf(value - mean) / stddev;
    return z_score > threshold;
}

// 变化检测
typedef struct {
    float last_value;
    float threshold;
    bool changed;
} change_detector_t;

bool detect_change(change_detector_t *detector, float new_value) {
    float delta = fabsf(new_value - detector->last_value);
    detector->changed = (delta > detector->threshold);
    detector->last_value = new_value;
    return detector->changed;
}

// 数据压缩(死区压缩)
typedef struct {
    float last_sent;
    float dead_band;
} dead_band_compressor_t;

bool should_send(dead_band_compressor_t *comp, float value) {
    if (fabsf(value - comp->last_sent) >= comp->dead_band) {
        comp->last_sent = value;
        return true;
    }
    return false;
}

// 旋转门压缩(Swinging Door)
typedef struct {
    float last_point;
    float compression_factor;
    float upper_slope;
    float lower_slope;
    bool first_point;
} swc_t;

bool swc_compress(swc_t *swc, float time, float value) {
    if (swc->first_point) {
        swc->last_point = value;
        swc->upper_slope = -INFINITY;
        swc->lower_slope = INFINITY;
        swc->first_point = false;
        return true;  // 保留第一个点
    }

    float slope = (value - swc->last_point) / time;

    if (slope > swc->upper_slope) {
        swc->upper_slope = slope;
        if (swc->upper_slope - swc->lower_slope > swc->compression_factor) {
            swc->last_point = value;
            swc->upper_slope = -INFINITY;
            swc->lower_slope = INFINITY;
            return true;  // 保留
        }
    }

    if (slope < swc->lower_slope) {
        swc->lower_slope = slope;
        if (swc->upper_slope - swc->lower_slope > swc->compression_factor) {
            swc->last_point = value;
            swc->upper_slope = -INFINITY;
            swc->lower_slope = INFINITY;
            return true;  // 保留
        }
    }

    return false;  // 压缩掉
}
```

### 3.1 模型部署

```c
// 模型管理
typedef struct {
    char model_id[32];
    char version[16];
    uint8_t *model_data;
    size_t model_size;
    enum { MODEL_TYPE_TFLITE, MODEL_TYPE_ONNX, MODEL_TYPE_CUSTOM } type;
    uint32_t input_size;
    uint32_t output_size;
    bool loaded;
} model_t;

// 模型加载
int load_model(model_t *model, const char *path) {
    FILE *f = fopen(path, "rb");
    if (!f) return -1;

    fseek(f, 0, SEEK_END);
    model->model_size = ftell(f);
    fseek(f, 0, SEEK_SET);

    model->model_data = malloc(model->model_size);
    fread(model->model_data, 1, model->model_size, f);
    fclose(f);

    model->loaded = true;
    return 0;
}

// 推理结果
typedef struct {
    float confidence;
    int class_id;
    char label[32];
    float bbox[4];  // x, y, w, h
} inference_result_t;

// 边缘推理
int edge_inference(model_t *model, const float *input, inference_result_t *result) {
    if (!model->loaded) return -1;

    // 运行推理(简化)
    float output[100];
    run_inference(model->model_data, input, output);

    // 后处理
    int max_idx = 0;
    float max_val = output[0];
    for (int i = 1; i < model->output_size; i++) {
        if (output[i] > max_val) {
            max_val = output[i];
            max_idx = i;
        }
    }

    result->class_id = max_idx;
    result->confidence = max_val;

    return 0;
}
```

```c
// 流式推理管道
typedef struct {
    model_t *model;
    float *input_buffer;
    float *output_buffer;
    uint32_t frame_count;
    uint32_t window_size;
} streaming_inference_t;

// 滑动窗口推理
int streaming_infer(streaming_inference_t *ctx, const float *new_frame,
                    inference_result_t *result) {
    // 移入新帧
    memmove(ctx->input_buffer, ctx->input_buffer + ctx->frame_size,
            (ctx->window_size - 1) * ctx->frame_size * sizeof(float));
    memcpy(ctx->input_buffer + (ctx->window_size - 1) * ctx->frame_size,
           new_frame, ctx->frame_size * sizeof(float));

    // 运行推理
    run_inference(ctx->model->model_data, ctx->input_buffer, ctx->output_buffer);

    // 后处理
    postprocess(ctx->output_buffer, result);

    return 0;
}
```

### 4.1 协议转换

```c
// 协议转换器
typedef struct {
    enum { PROTO_MQTT, PROTO_MODBUS, PROTO_OPCUA, PROTO_HTTP } src_proto;
    enum { PROTO_MQTT, PROTO_MODBUS, PROTO_OPCUA, PROTO_HTTP } dst_proto;
    void *src_handle;
    void *dst_handle;
} protocol_converter_t;

// Modbus到MQTT转换
void modbus_to_mqtt(protocol_converter_t *conv) {
    // 读取Modbus数据
    uint16_t registers[10];
    modbus_read_registers(conv->src_handle, 0, 10, registers);

    // 转换为JSON
    char json[256];
    snprintf(json, sizeof(json),
             "{\"temp\":%.1f,\"humid\":%.1f,\"press\":%.1f}",
             registers[0] / 10.0f, registers[1] / 10.0f, registers[2] / 10.0f);

    // 发布到MQTT
    mqtt_publish(conv->dst_handle, "sensors/data", json);
}

// OPC UA到MQTT转换
void opcua_to_mqtt(protocol_converter_t *conv) {
    // 读取OPC UA节点
    opcua_node_t nodes[3];
    nodes[0].node_id = 1001;  // 温度
    nodes[1].node_id = 1002;  // 湿度
    nodes[2].node_id = 1003;  // 压力

    opcua_read_multiple(conv->src_handle, nodes, 3);

    // 转换并发布
    char json[256];
    snprintf(json, sizeof(json),
             "{\"temp\":%.2f,\"humid\":%.2f,\"press\":%.2f}",
             nodes[0].value.float_val,
             nodes[1].value.float_val,
             nodes[2].value.float_val);

    mqtt_publish(conv->dst_handle, "opcua/data", json);
}
```

### 4.2 数据缓存

```c
// 边缘数据缓存
#define CACHE_SIZE 1000
typedef struct {
    char key[32];
    char value[256];
    uint32_t timestamp;
    uint32_t ttl;
} cache_entry_t;

typedef struct {
    cache_entry_t entries[CACHE_SIZE];
    int count;
    int head;
} edge_cache_t;

// 缓存写入
int cache_put(edge_cache_t *cache, const char *key, const char *value, uint32_t ttl) {
    int idx = cache->head;
    strncpy(cache->entries[idx].key, key, sizeof(cache->entries[idx].key) - 1);
    strncpy(cache->entries[idx].value, value, sizeof(cache->entries[idx].value) - 1);
    cache->entries[idx].timestamp = get_timestamp();
    cache->entries[idx].ttl = ttl;

    cache->head = (cache->head + 1) % CACHE_SIZE;
    if (cache->count < CACHE_SIZE) cache->count++;

    return 0;
}

// 缓存读取
int cache_get(edge_cache_t *cache, const char *key, char *value, size_t len) {
    for (int i = 0; i < cache->count; i++) {
        if (strcmp(cache->entries[i].key, key) == 0) {
            // 检查TTL
            if (get_timestamp() - cache->entries[i].timestamp > cache->entries[i].ttl) {
                return -1;  // 过期
            }
            strncpy(value, cache->entries[i].value, len);
            return 0;
        }
    }
    return -1;  // 未找到
}

// 离线缓存(断网时存储)
typedef struct {
    edge_cache_t cache;
    bool offline_mode;
    uint32_t sync_threshold;
} offline_cache_t;

void offline_store(offline_cache_t *oc, const char *topic, const char *data) {
    if (oc->offline_mode) {
        cache_put(&oc->cache, topic, data, 3600);  // 1小时TTL
    }
}

// 恢复连接后同步
void sync_offline_data(offline_cache_t *oc, mqtt_client_t *client) {
    for (int i = 0; i < oc->cache.count; i++) {
        if (strlen(oc->cache.entries[i].key) > 0) {
            mqtt_publish(client, oc->cache.entries[i].key,
                        oc->cache.entries[i].value);
        }
    }
    oc->cache.count = 0;
    oc->cache.head = 0;
}
```

### 5.1 设备认证

```c
// 设备证书
typedef struct {
    char device_id[32];
    uint8_t cert[512];
    uint8_t private_key[256];
    uint32_t valid_from;
    uint32_t valid_to;
} device_cert_t;

// TLS握手
typedef struct {
    int socket;
    device_cert_t *cert;
    uint8_t session_key[32];
    bool connected;
} tls_session_t;

int tls_connect(tls_session_t *session, const char *host, int port) {
    // 创建TCP连接
    session->socket = tcp_connect(host, port);
    if (session->socket < 0) return -1;

    // TLS握手
    // 1. ClientHello
    send_client_hello(session);

    // 2. ServerHello + Certificate
    recv_server_hello(session);

    // 3. 验证服务器证书
    if (!verify_server_cert(session)) {
        close(session->socket);
        return -1;
    }

    // 4. 客户端认证(双向TLS)
    send_client_cert(session, session->cert);

    // 5. 密钥交换
    derive_session_key(session);

    session->connected = true;
    return 0;
}

// 设备注册
int device_register(tls_session_t *session, const char *device_id) {
    char registration[256];
    snprintf(registration, sizeof(registration),
             "{\"device_id\":\"%s\",\"timestamp\":%lu}",
             device_id, get_timestamp());

    // 签名
    uint8_t signature[64];
    ecdsa_sign(session->cert->private_key,
               (uint8_t *)registration, strlen(registration),
               signature);

    // 发送注册请求
    send_registration(session, registration, signature);

    return 0;
}
```

## 六、边缘编排

### 6.1 应用编排

```c
// 应用描述符
typedef struct {
    char app_id[32];
    char image[64];
    uint32_t cpu_limit;
    uint32_t memory_limit;
    uint32_t replicas;
    char env_vars[10][64];
    int env_count;
    char volumes[5][128];
    int volume_count;
} app_descriptor_t;

// 编排决策
typedef struct {
    int node_id;
    char app_id[32];
    enum { ACTION_DEPLOY, ACTION_SCALE, ACTION_MIGRATE, ACTION_STOP } action;
    int replicas;
} orchestration_decision_t;

// 基于规则的编排
int orchestrate(edge_node_t *nodes, int node_count,
                app_descriptor_t *apps, int app_count,
                orchestration_decision_t *decisions) {
    int decision_count = 0;

    for (int i = 0; i < app_count; i++) {
        // 检查资源不足
        int running_count = count_running_replicas(&apps[i], nodes, node_count);

        if (running_count < apps[i].replicas) {
            // 需要扩容
            int target_node = find_best_node(nodes, node_count, &apps[i]);
            if (target_node >= 0) {
                decisions[decision_count].node_id = target_node;
                strncpy(decisions[decision_count].app_id, apps[i].app_id, 32);
                decisions[decision_count].action = ACTION_SCALE;
                decisions[decision_count].replicas = apps[i].replicas - running_count;
                decision_count++;
            }
        }

        // 检查节点过载
        for (int j = 0; j < node_count; j++) {
            if (nodes[j].cpu_usage > 0.9f) {
                // 迁移应用到其他节点
                decisions[decision_count].node_id = j;
                strncpy(decisions[decision_count].app_id, apps[i].app_id, 32);
                decisions[decision_count].action = ACTION_MIGRATE;
                decision_count++;
            }
        }
    }

    return decision_count;
}
```
