# 物联网平台

## 核心概念

- **IoT平台** - 设备管理、数据处理
- **设备影子** - 虚拟设备状态
- **规则引擎** - 数据路由和处理
- **OTA** - 远程固件更新

## 二、阿里云IoT

### 2.1 设备接入

```c
// 阿里云IoT MQTT连接
#define PRODUCT_KEY    "a1xxxxx"
#define DEVICE_NAME    "device01"
#define DEVICE_SECRET  "xxxxxxxx"
#define REGION         "cn-shanghai"

void aliyun_connect(void) {
    // 构建MQTT参数
    char client_id[128];
    char username[128];
    char password[128];

    snprintf(client_id, sizeof(client_id), "%s.%s|securemode=2,signmethod=hmacsha256|",
             PRODUCT_KEY, DEVICE_NAME);

    snprintf(username, sizeof(username), "%s&%s", DEVICE_NAME, PRODUCT_KEY);

    // 计算密码(HMAC-SHA256)
    char content[256];
    snprintf(content, sizeof(content), "clientId%s.%sdeviceName%sproductKey%s",
             PRODUCT_KEY, DEVICE_NAME, DEVICE_NAME, PRODUCT_KEY);
    hmac_sha256(DEVICE_SECRET, strlen(DEVICE_SECRET), content, strlen(content), password);

    // MQTT连接
    char host[128];
    snprintf(host, sizeof(host), "%s.iot-as-mqtt.%s.aliyuncs.com", PRODUCT_KEY, REGION);

    mqtt_connect(host, 1883, client_id, username, password);
}
```

## 三、AWS IoT

### 3.1 AWS IoT Core

```c
// AWS IoT MQTT连接
void aws_iot_connect(void) {
    // TLS配置
    tls_config_t tls = {
        .ca_cert = aws_root_ca,
        .client_cert = device_cert,
        .client_key = device_key,
    };

    // MQTT连接
    mqtt_config_t config = {
        .host = "xxxxxxxxxxxx-ats.iot.us-east-1.amazonaws.com",
        .port = 8883,
        .client_id = "device01",
        .tls = &tls,
    };

    mqtt_connect(&config);
}

// 设备影子
void aws_update_shadow(float temp) {
    char payload[256];
    snprintf(payload, sizeof(payload),
             "{\"state\":{\"reported\":{\"temperature\":%.1f}}}",
             temp);

    mqtt_publish("$aws/things/device01/shadow/update", payload);
}

// 订阅命令
void aws_subscribe_commands(void) {
    mqtt_subscribe("$aws/things/device01/shadow/update/accepted");
    mqtt_subscribe("$aws/things/device01/shadow/update/delta");
}
```

void write_to_influxdb(const char *device_id, float temp, float humidity) {

    char query[512];
    snprintf(query, sizeof(query),
             "from(bucket: \"iot\") "
             "|> range(start: %u, stop: %u) "
             "|> filter(fn: (r) => r.device == \"%s\") "
             "|> filter(fn: (r) => r._field == \"temperature\")",
             start, end, device_id);

    http_post("http://localhost:8086/api/v2/query?org=myorg", query);
}

## 五、规则引擎

### 5.1 数据路由

```c
// 规则引擎配置
typedef struct {
    char topic_pattern[128];
    char condition[256];
    char action_type[32];
    char action_target[128];
} rule_t;

rule_t rules[] = {
    {
        .topic_pattern = "/sensor/+/temperature",
        .condition = "temperature > 35",
        .action_type = "alert",
        .action_target = "high_temp_alert"
    },
    {
        .topic_pattern = "/sensor/+/battery",
        .condition = "battery < 20",
        .action_type = "alert",
        .action_target = "low_battery_alert"
    },
};

// 规则匹配
void process_rule(const char *topic, json_t *data) {
    for (int i = 0; i < sizeof(rules)/sizeof(rules[0]); i++) {
        if (match_topic(topic, rules[i].topic_pattern)) {
            if (evaluate_condition(rules[i].condition, data)) {
                execute_action(rules[i].action_type, rules[i].action_target, data);
            }
        }
    }
}
```

## 六、设备管理

### 6.1 设备注册

```c
// 设备注册流程
typedef struct {
    char device_id[32];
    char product_key[32];
    char device_secret[64];
    bool activated;
} device_info_t;

device_info_t register_device(const char *product_key, const char *device_name) {
    // 1. 生成设备密钥
    char secret[64];
    generate_random_secret(secret, sizeof(secret));

    // 2. 注册到平台
    char payload[256];
    snprintf(payload, sizeof(payload),
             "{\"productKey\":\"%s\",\"deviceName\":\"%s\",\"deviceSecret\":\"%s\"}",
             product_key, device_name, secret);

    http_post("http://platform/api/device/register", payload);

    // 3. 返回设备信息
    device_info_t info;
    strcpy(info.product_key, product_key);
    strcpy(info.device_secret, secret);
    info.activated = true;

    return info;
}
```
