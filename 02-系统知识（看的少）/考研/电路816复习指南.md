## 一、电路分析基础

### 1.1 基尔霍夫定律

```c
// KCL: 节点电流和为零
// ΣI = 0 (流入为正，流出为负)

// KVL: 回路电压和为零
// ΣV = 0 (顺时针为正，逆时针为负)

// 节点分析法示例
// 3个节点，2个已知电压源
typedef struct {
    double conductance[3][3];  // 电导矩阵
    double current[3];         // 电流向量
    double voltage[3];         // 电压解
} node_analysis_t;

// 建立节点方程 G*V = I
void setup_node_equations(node_analysis_t *circuit) {
    // 节点1: (G1+G2)*V1 - G2*V2 = Is1
    // 节点2: -G2*V1 + (G2+G3)*V2 = Is2

    double G1 = 1.0/100;  // 100Ω
    double G2 = 1.0/200;  // 200Ω
    double G3 = 1.0/300;  // 300Ω

    circuit->conductance[0][0] = G1 + G2;
    circuit->conductance[0][1] = -G2;
    circuit->conductance[1][0] = -G2;
    circuit->conductance[1][1] = G2 + G3;

    circuit->current[0] = 0.01;  // 10mA
    circuit->current[1] = 0.02;  // 20mA
}

// 求解节点电压(高斯消元)
void solve_node_voltages(node_analysis_t *circuit, int n) {
    for (int i = 0; i < n; i++) {
        // 选主元
        int max_row = i;
        for (int k = i + 1; k < n; k++) {
            if (fabs(circuit->conductance[k][i]) > fabs(circuit->conductance[max_row][i])) {
                max_row = k;
            }
        }
        // 交换行
        for (int j = 0; j < n; j++) {
            double temp = circuit->conductance[i][j];
            circuit->conductance[i][j] = circuit->conductance[max_row][j];
            circuit->conductance[max_row][j] = temp;
        }
        double temp = circuit->current[i];
        circuit->current[i] = circuit->current[max_row];
        circuit->current[max_row] = temp;

        // 消元
        for (int k = i + 1; k < n; k++) {
            double factor = circuit->conductance[k][i] / circuit->conductance[i][i];
            for (int j = i; j < n; j++) {
                circuit->conductance[k][j] -= factor * circuit->conductance[i][j];
            }
            circuit->current[k] -= factor * circuit->current[i];
        }
    }

    // 回代
    for (int i = n - 1; i >= 0; i--) {
        circuit->voltage[i] = circuit->current[i];
        for (int j = i + 1; j < n; j++) {
            circuit->voltage[i] -= circuit->conductance[i][j] * circuit->voltage[j];
        }
        circuit->voltage[i] /= circuit->conductance[i][i];
    }
}
```

### 1.2 戴维南定理

```c
// 戴维南等效电路
typedef struct {
    double vth;    // 开路电压
    double rth;    // 等效电阻
} thevenin_t;

// 求戴维南等效
thevenin_t thevenin_equivalent(double *voltage, double *current, int n) {
    thevenin_t eq;

    // 开路电压 = 电压源电压
    eq.vth = voltage[0];

    // 等效电阻 = 开路电压 / 短路电流
    eq.rth = voltage[0] / current[0];

    return eq;
}

// 诺顿等效电路
typedef struct {
    double in;     // 短路电流
    double rn;     // 等效电阻
} norton_t;

// 戴维南转诺顿
norton_t thevenin_to_norton(thevenin_t th) {
    norton_t n;
    n.in = th.vth / th.rth;
    n.rn = th.rth;
    return n;
}

// 最大功率传输
double max_power_transfer(double vs, double rs, double *rl_optimal) {
    // 当RL = RS时，功率最大
    *rl_optimal = rs;
    double i = vs / (rs + *rl_optimal);
    return i * i * *rl_optimal;
}
```

## 二、正弦稳态分析

### 2.1 相量法

```c
// 复数表示
typedef struct {
    double real;
    double imag;
} complex_t;

// 极坐标形式
typedef struct {
    double magnitude;
    double angle;  // 弧度
} phasor_t;

// 直角坐标转极坐标
phasor_t to_polar(complex_t c) {
    phasor_t p;
    p.magnitude = sqrt(c.real * c.real + c.imag * c.imag);
    p.angle = atan2(c.imag, c.real);
    return p;
}

// 极坐标转直角坐标
complex_t to_rectangular(phasor_t p) {
    complex_t c;
    c.real = p.magnitude * cos(p.angle);
    c.imag = p.magnitude * sin(p.angle);
    return c;
}

// 复数运算
complex_t complex_add(complex_t a, complex_t b) {
    complex_t result;
    result.real = a.real + b.real;
    result.imag = a.imag + b.imag;
    return result;
}

complex_t complex_multiply(complex_t a, complex_t b) {
    complex_t result;
    result.real = a.real * b.real - a.imag * b.imag;
    result.imag = a.real * b.imag + a.imag * b.real;
    return result;
}

complex_t complex_divide(complex_t a, complex_t b) {
    complex_t result;
    double denom = b.real * b.real + b.imag * b.imag;
    result.real = (a.real * b.real + a.imag * b.imag) / denom;
    result.imag = (a.imag * b.real - a.real * b.imag) / denom;
    return result;
}

// 阻抗计算
// 电阻: Z = R
// 电感: Z = jωL
// 电容: Z = 1/(jωC)

// RLC串联阻抗
complex_t rlc_impedance(double r, double l, double c, double omega) {
    complex_t z;
    z.real = r;
    z.imag = omega * l - 1.0 / (omega * c);
    return z;
}

// 阻抗幅值和相角
phasor_t impedance_phasor(complex_t z) {
    return to_polar(z);
}
```

### 2.2 功率计算

```c
// 交流功率
typedef struct {
    double active;    // 有功功率 P (W)
    double reactive;  // 无功功率 Q (var)
    double apparent;  // 视在功率 S (VA)
    double power_factor;  // 功率因数
} ac_power_t;

// 功率计算
ac_power_t calculate_ac_power(complex_t voltage, complex_t current) {
    ac_power_t power;

    // 复功率 S = V * I*
    complex_t current_conj;
    current_conj.real = current.real;
    current_conj.imag = -current.imag;

    complex_t s = complex_multiply(voltage, current_conj);

    power.active = s.real;        // P = Re(S)
    power.reactive = s.imag;      // Q = Im(S)
    power.apparent = sqrt(s.real * s.real + s.imag * s.imag);  // |S|
    power.power_factor = s.real / power.apparent;  // cos(φ)

    return power;
}

// 功率因数校正
double power_factor_correction(double p, double pf_before, double pf_target,
                                double v, double omega) {
    // 需要补偿的无功功率
    double q_before = p * tan(acos(pf_before));
    double q_after = p * tan(acos(pf_target));
    double q_compensation = q_before - q_after;

    // 并联电容值
    double c = q_compensation / (omega * v * v);
    return c;
}

// 三相电路功率
typedef struct {
    double p_total;  // 三相总有功
    double q_total;  // 三相总无功
    double s_total;  // 三相总视在
} three_phase_power_t;

three_phase_power_t three_phase_power(double v_line, double i_line,
                                       double pf, bool is_balanced) {
    three_phase_power_t power;

    if (is_balanced) {
        power.p_total = sqrt(3) * v_line * i_line * pf;
        power.q_total = sqrt(3) * v_line * i_line * sin(acos(pf));
        power.s_total = sqrt(3) * v_line * i_line;
    }

    return power;
}
```

## 三、动态电路

### 3.1 一阶电路

```c
// RC电路响应
typedef struct {
    double r;       // 电阻
    double c;       // 电容
    double v0;      // 初始电压
    double vs;      // 电源电压
    double tau;     // 时间常数 τ = RC
} rc_circuit_t;

// RC充电响应
double rc_charging(rc_circuit_t *circuit, double t) {
    circuit->tau = circuit->r * circuit->c;
    return circuit->vs * (1 - exp(-t / circuit->tau)) +
           circuit->v0 * exp(-t / circuit->tau);
}

// RC放电响应
double rc_discharging(rc_circuit_t *circuit, double t) {
    circuit->tau = circuit->r * circuit->c;
    return circuit->v0 * exp(-t / circuit->tau);
}

// RL电路响应
typedef struct {
    double r;       // 电阻
    double l;       // 电感
    double i0;      // 初始电流
    double is;      // 稳态电流
    double tau;     // 时间常数 τ = L/R
} rl_circuit_t;

// RL响应
double rl_response(rl_circuit_t *circuit, double t) {
    circuit->tau = circuit->l / circuit->r;
    return circuit->is + (circuit->i0 - circuit->is) * exp(-t / circuit->tau);
}

// 三要素法
// f(t) = f(∞) + [f(0+) - f(∞)] * e^(-t/τ)
typedef struct {
    double f_inf;     // 稳态值 f(∞)
    double f_0_plus;  // 初始值 f(0+)
    double tau;       // 时间常数
} three_element_t;

double three_element_response(three_element_t *elem, double t) {
    return elem->f_inf + (elem->f_0_plus - elem->f_inf) * exp(-t / elem->tau);
}
```

### 3.2 二阶电路

```c
// RLC串联电路
typedef struct {
    double r;       // 电阻
    double l;       // 电感
    double c;       // 电容
    double v0;      // 电容初始电压
    double i0;      // 电感初始电流
} rlc_circuit_t;

// 特征方程: s^2 + (R/L)s + 1/(LC) = 0
typedef struct {
    double s1;      // 特征根1
    double s2;      // 特征根2
    enum { OVERDAMPED, CRITICALLY_DAMPED, UNDERDAMPED } type;
} characteristic_t;

characteristic_t solve_characteristic(rlc_circuit_t *circuit) {
    characteristic_t result;

    double alpha = circuit->r / (2 * circuit->l);  // 衰减系数
    double omega0 = 1.0 / sqrt(circuit->l * circuit->c);  // 谐振频率
    double discriminant = alpha * alpha - omega0 * omega0;

    if (discriminant > 0) {
        // 过阻尼
        result.type = OVERDAMPED;
        result.s1 = -alpha + sqrt(discriminant);
        result.s2 = -alpha - sqrt(discriminant);
    } else if (discriminant == 0) {
        // 临界阻尼
        result.type = CRITICALLY_DAMPED;
        result.s1 = result.s2 = -alpha;
    } else {
        // 欠阻尼
        result.type = UNDERDAMPED;
        double omega_d = sqrt(-discriminant);
        result.s1 = -alpha;
        result.s2 = omega_d;
    }

    return result;
}

// RLC响应
double rlc_response(rlc_circuit_t *circuit, double t) {
    characteristic_t roots = solve_characteristic(circuit);

    switch (roots.type) {
        case OVERDAMPED: {
            double a1 = (circuit->v0 - roots.s2 * circuit->v0) / (roots.s1 - roots.s2);
            double a2 = circuit->v0 - a1;
            return a1 * exp(roots.s1 * t) + a2 * exp(roots.s2 * t);
        }
        case CRITICALLY_DAMPED: {
            return (circuit->v0 + (circuit->i0 / circuit->c + circuit->v0 * roots.s1) * t)
                   * exp(roots.s1 * t);
        }
        case UNDERDAMPED: {
            double alpha = -roots.s1;
            double omega_d = roots.s2;
            return exp(-alpha * t) * (circuit->v0 * cos(omega_d * t) +
                   ((circuit->i0 / circuit->c + alpha * circuit->v0) / omega_d) * sin(omega_d * t));
        }
    }

    return 0;
}
```

## 四、频率响应

### 4.1 网络函数

```c
// 网络函数 H(jω)
typedef struct {
    double *numerator;      // 分子系数
    int num_order;
    double *denominator;    // 分母系数
    int den_order;
} transfer_function_t;

// 计算频率响应
complex_t evaluate_tf(transfer_function_t *tf, double omega) {
    complex_t jw;
    jw.real = 0;
    jw.imag = omega;

    // 计算分子
    complex_t num = {tf->numerator[0], 0};
    for (int i = 1; i <= tf->num_order; i++) {
        complex_t term = {tf->numerator[i], 0};
        for (int j = 0; j < i; j++) {
            term = complex_multiply(term, jw);
        }
        num = complex_add(num, term);
    }

    // 计算分母
    complex_t den = {tf->denominator[0], 0};
    for (int i = 1; i <= tf->den_order; i++) {
        complex_t term = {tf->denominator[i], 0};
        for (int j = 0; j < i; j++) {
            term = complex_multiply(term, jw);
        }
        den = complex_add(den, term);
    }

    return complex_divide(num, den);
}

// 波特图
typedef struct {
    double *frequencies;
    double *magnitude_db;
    double *phase_deg;
    int points;
} bode_plot_t;

bode_plot_t compute_bode(transfer_function_t *tf, double f_min, double f_max, int points) {
    bode_plot_t bode;
    bode.points = points;
    bode.frequencies = malloc(points * sizeof(double));
    bode.magnitude_db = malloc(points * sizeof(double));
    bode.phase_deg = malloc(points * sizeof(double));

    double log_min = log10(f_min);
    double log_max = log10(f_max);
    double log_step = (log_max - log_min) / (points - 1);

    for (int i = 0; i < points; i++) {
        double log_f = log_min + i * log_step;
        bode.frequencies[i] = pow(10, log_f);
        double omega = 2 * M_PI * bode.frequencies[i];

        complex_t h = evaluate_tf(tf, omega);
        phasor_t p = to_polar(h);

        bode.magnitude_db[i] = 20 * log10(p.magnitude);
        bode.phase_deg[i] = p.angle * 180 / M_PI;
    }

    return bode;
}
```

### 4.2 谐振电路

```c
// 串联谐振
typedef struct {
    double r;
    double l;
    double c;
    double omega_res;    // 谐振角频率
    double f_res;        // 谐振频率
    double q_factor;     // 品质因数
    double bandwidth;    // 带宽
} series_resonance_t;

void series_resonance_analysis(series_resonance_t *circuit) {
    circuit->omega_res = 1.0 / sqrt(circuit->l * circuit->c);
    circuit->f_res = circuit->omega_res / (2 * M_PI);
    circuit->q_factor = circuit->omega_res * circuit->l / circuit->r;
    circuit->bandwidth = circuit->f_res / circuit->q_factor;
}

// 并联谐振
typedef struct {
    double r;
    double l;
    double c;
    double omega_res;
    double q_factor;
    double bandwidth;
} parallel_resonance_t;

void parallel_resonance_analysis(parallel_resonance_t *circuit) {
    circuit->omega_res = 1.0 / sqrt(circuit->l * circuit->c);
    circuit->q_factor = circuit->r / (circuit->omega_res * circuit->l);
    circuit->bandwidth = circuit->omega_res / (2 * M_PI * circuit->q_factor);
}
```

## 五、二端口网络

### 5.1 参数矩阵

```c
// Z参数(阻抗参数)
typedef struct {
    double z11, z12;
    double z21, z22;
} z_params_t;

// Y参数(导纳参数)
typedef struct {
    double y11, y12;
    double y21, y22;
} y_params_t;

// A参数(传输参数)
typedef struct {
    double a11, a12;
    double a21, a22;
} a_params_t;

// Z参数转Y参数
y_params_t z_to_y(z_params_t *z) {
    y_params_t y;
    double det_z = z->z11 * z->z22 - z->z12 * z->z21;
    y.y11 = z->z22 / det_z;
    y.y12 = -z->z12 / det_z;
    y.y21 = -z->z21 / det_z;
    y.y22 = z->z11 / det_z;
    return y;
}

// Y参数转Z参数
z_params_t y_to_z(y_params_t *y) {
    z_params_t z;
    double det_y = y->y11 * y->y22 - y->y12 * y->y21;
    z.z11 = y->y22 / det_y;
    z.z12 = -y->y12 / det_y;
    z.z21 = -y->y21 / det_y;
    z.z22 = y->y11 / det_y;
    return z;
}

// 级联网络
a_params_t cascade_a_params(a_params_t *a1, a_params_t *a2) {
    a_params_t result;
    result.a11 = a1->a11 * a2->a11 + a1->a12 * a2->a21;
    result.a12 = a1->a11 * a2->a12 + a1->a12 * a2->a22;
    result.a21 = a1->a21 * a2->a11 + a1->a22 * a2->a21;
    result.a22 = a1->a21 * a2->a12 + a1->a22 * a2->a22;
    return result;
}
```
