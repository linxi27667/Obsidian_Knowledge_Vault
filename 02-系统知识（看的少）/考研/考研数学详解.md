```c
double sequence_limit(double (*f)(int), int n_max) {
    double prev = f(1);
    for (int n = 2; n <= n_max; n++) {
        double curr = f(n);
        if (fabs(curr - prev) < 1e-10) {
            return curr;
        }
        prev = curr;
    }
    return prev;
}

// 泰勒展开
// e^x = 1 + x + x²/2! + x³/3! + ...
double taylor_exp(double x, int terms) {
    double result = 1.0;
    double term = 1.0;
    for (int n = 1; n < terms; n++) {
        term *= x / n;
        result += term;
    }
    return result;
}

// sin(x) = x - x³/3! + x⁵/5! - ...
double taylor_sin(double x, int terms) {
    double result = 0;
    double term = x;
    for (int n = 0; n < terms; n++) {
        result += term;
        term *= -x * x / ((2*n+2) * (2*n+3));
    }
    return result;
}

// 连续性检查
typedef struct {
    double left_limit;
    double right_limit;
    double value;
    bool continuous;
} continuity_check_t;

continuity_check_t check_continuity(double (*f)(double), double x0, double eps) {
    continuity_check_t result;
    result.left_limit = f(x0 - eps);
    result.right_limit = f(x0 + eps);
    result.value = f(x0);
    result.continuous = (fabs(result.left_limit - result.right_limit) < 1e-10) &&
                        (fabs(result.left_limit - result.value) < 1e-10);
    return result;
}
```

```c
double trapezoidal_rule(double (*f)(double), double a, double b, int n) {
    double h = (b - a) / n;
    double sum = 0.5 * (f(a) + f(b));
    for (int i = 1; i < n; i++) {

// 辛普森法则
double simpson_rule(double (*f)(double), double a, double b, int n) {
    if (n % 2 != 0) n++;  // n必须为偶数
```

```c
// 泰勒级数展开
typedef struct {
    double *coefficients;
    int order;
} taylor_series_t;

// e^x 泰勒展开
taylor_series_t taylor_exp_series(int order) {
    taylor_series_t s;
    s.order = order;
    s.coefficients = malloc((order + 1) * sizeof(double));

    double factorial = 1;
    for (int n = 0; n <= order; n++) {
        if (n > 0) factorial *= n;
        s.coefficients[n] = 1.0 / factorial;
    }

    return s;
}

// 计算级数值
double evaluate_series(taylor_series_t *s, double x) {
    double result = 0;
    double x_power = 1;
    for (int n = 0; n <= s->order; n++) {
        result += s->coefficients[n] * x_power;
        x_power *= x;
    }
    return result;
}

// 傅里叶级数
typedef struct {
    double a0;
    double *an;  // 余弦系数
    double *bn;  // 正弦系数
    int n_terms;
    double period;
} fourier_series_t;

// 计算傅里叶系数
fourier_series_t compute_fourier(double (*f)(double), double T, int n_terms) {
    fourier_series_t fs;
    fs.n_terms = n_terms;
    fs.period = T;
    fs.an = calloc(n_terms, sizeof(double));
    fs.bn = calloc(n_terms, sizeof(double));

    double omega = 2 * M_PI / T;

    // a0
    fs.a0 = 0;
    for (int i = 0; i < 1000; i++) {
        double t = i * T / 1000;
        fs.a0 += f(t);
    }
    fs.a0 *= 2.0 / 1000;

    // an, bn
    for (int n = 1; n <= n_terms; n++) {
        fs.an[n-1] = 0;
        fs.bn[n-1] = 0;
        for (int i = 0; i < 1000; i++) {
            double t = i * T / 1000;
            fs.an[n-1] += f(t) * cos(n * omega * t);
            fs.bn[n-1] += f(t) * sin(n * omega * t);
        }
        fs.an[n-1] *= 2.0 / 1000;
        fs.bn[n-1] *= 2.0 / 1000;
    }

    return fs;
}
```

```c
// 一阶常微分方程(欧拉法)
void euler_method(double (*f)(double, double), double x0, double y0,
                  double h, int steps, double *x, double *y) {
    x[0] = x0;
    y[0] = y0;

    for (int i = 0; i < steps; i++) {
        x[i+1] = x[i] + h;
        y[i+1] = y[i] + h * f(x[i], y[i]);
    }
}

// 龙格-库塔法(四阶)
void runge_kutta_4(double (*f)(double, double), double x0, double y0,
                   double h, int steps, double *x, double *y) {
    x[0] = x0;
    y[0] = y0;

    for (int i = 0; i < steps; i++) {
        double k1 = h * f(x[i], y[i]);
        double k2 = h * f(x[i] + h/2, y[i] + k1/2);
        double k3 = h * f(x[i] + h/2, y[i] + k2/2);
        double k4 = h * f(x[i] + h, y[i] + k3);

        x[i+1] = x[i] + h;
        y[i+1] = y[i] + (k1 + 2*k2 + 2*k3 + k4) / 6;
    }
}

// 可分离变量方程
// dy/dx = f(x)g(y)
// ∫dy/g(y) = ∫f(x)dx

// 一阶线性方程
// y' + P(x)y = Q(x)
// y = e^(-∫Pdx) * [∫Q*e^(∫Pdx)dx + C]

// 二阶常系数齐次方程
// y'' + py' + qy = 0
// 特征方程: r² + pr + q = 0
typedef struct {
    double r1, r2;
    enum { REAL_DISTINCT, REAL_EQUAL, COMPLEX } type;
} characteristic_roots_t;

characteristic_roots_t solve_characteristic(double p, double q) {
    characteristic_roots_t roots;
    double discriminant = p * p - 4 * q;

    if (discriminant > 0) {
        roots.type = REAL_DISTINCT;
        roots.r1 = (-p + sqrt(discriminant)) / 2;
        roots.r2 = (-p - sqrt(discriminant)) / 2;
    } else if (discriminant == 0) {
        roots.type = REAL_EQUAL;
        roots.r1 = roots.r2 = -p / 2;
    } else {
        roots.type = COMPLEX;
        roots.r1 = -p / 2;
        roots.r2 = sqrt(-discriminant) / 2;
    }

    return roots;
}
```

```c
typedef struct {

// 创建矩阵
matrix_t *matrix_create(int rows, int cols) {
    matrix_t *m = malloc(sizeof(matrix_t));
    m->rows = rows;
    m->cols = cols;
    m->data = malloc(rows * sizeof(double *));
    for (int i = 0; i < rows; i++) {
        m->data[i] = calloc(cols, sizeof(double));
    }
    return m;
}

// 矩阵加法
matrix_t *matrix_add(matrix_t *a, matrix_t *b) {
    matrix_t *result = matrix_create(a->rows, a->cols);
    for (int i = 0; i < a->rows; i++) {
        for (int j = 0; j < a->cols; j++) {
            result->data[i][j] = a->data[i][j] + b->data[i][j];
        }
    }
    return result;
}

// 矩阵乘法
matrix_t *matrix_multiply(matrix_t *a, matrix_t *b) {
    matrix_t *result = matrix_create(a->rows, b->cols);
    for (int i = 0; i < a->rows; i++) {
        for (int j = 0; j < b->cols; j++) {
            for (int k = 0; k < a->cols; k++) {
                result->data[i][j] += a->data[i][k] * b->data[k][j];
            }
        }
    }
    return result;
}

// 转置
matrix_t *matrix_transpose(matrix_t *m) {
    matrix_t *result = matrix_create(m->cols, m->rows);
    for (int i = 0; i < m->rows; i++) {
        for (int j = 0; j < m->cols; j++) {
            result->data[j][i] = m->data[i][j];
        }
    }
    return result;
}

// 行列式
double matrix_determinant(matrix_t *m) {
    if (m->rows == 1) return m->data[0][0];
    if (m->rows == 2) {
        return m->data[0][0] * m->data[1][1] - m->data[0][1] * m->data[1][0];
    }

    double det = 0;
    for (int j = 0; j < m->cols; j++) {
        matrix_t *minor = matrix_create(m->rows - 1, m->cols - 1);
        for (int i = 1; i < m->rows; i++) {
            int col = 0;
            for (int k = 0; k < m->cols; k++) {
                if (k == j) continue;
                minor->data[i-1][col++] = m->data[i][k];
            }
        }
        det += (j % 2 == 0 ? 1 : -1) * m->data[0][j] * matrix_determinant(minor);
    }

    return det;
}
```

```c
// 高斯消元法
int gaussian_elimination(matrix_t *A, double *b, double *x, int n) {
    // 增广矩阵
    double **aug = malloc(n * sizeof(double *));
    for (int i = 0; i < n; i++) {
        aug[i] = malloc((n + 1) * sizeof(double));
        for (int j = 0; j < n; j++) {
            aug[i][j] = A->data[i][j];
        }
        aug[i][n] = b[i];
    }

    // 前向消元
    for (int i = 0; i < n; i++) {
        // 选主元
        int max_row = i;
        for (int k = i + 1; k < n; k++) {
            if (fabs(aug[k][i]) > fabs(aug[max_row][i])) {
                max_row = k;
            }
        }
        // 交换行
        double *temp = aug[i];
        aug[i] = aug[max_row];
        aug[max_row] = temp;

        // 检查奇异
        if (fabs(aug[i][i]) < 1e-10) {
            return -1;  // 奇异矩阵
        }

        // 消元
        for (int k = i + 1; k < n; k++) {
            double factor = aug[k][i] / aug[i][i];
            for (int j = i; j <= n; j++) {
                aug[k][j] -= factor * aug[i][j];
            }
        }
    }

    // 回代
    for (int i = n - 1; i >= 0; i--) {
        x[i] = aug[i][n];
        for (int j = i + 1; j < n; j++) {
            x[i] -= aug[i][j] * x[j];
        }
        x[i] /= aug[i][i];
    }

    return 0;
}

// LU分解
typedef struct {
    matrix_t *L;
    matrix_t *U;
} lu_decomposition_t;

lu_decomposition_t lu_decompose(matrix_t *A) {
    int n = A->rows;
    lu_decomposition_t lu;
    lu.L = matrix_create(n, n);
    lu.U = matrix_create(n, n);

    for (int i = 0; i < n; i++) {
        // U的上三角
        for (int k = i; k < n; k++) {
            double sum = 0;
            for (int j = 0; j < i; j++) {
                sum += lu.L->data[i][j] * lu.U->data[j][k];
            }
            lu.U->data[i][k] = A->data[i][k] - sum;
        }

        // L的下三角
        for (int k = i; k < n; k++) {
            if (i == k) {
                lu.L->data[i][i] = 1;
            } else {
                double sum = 0;
                for (int j = 0; j < i; j++) {
                    sum += lu.L->data[k][j] * lu.U->data[j][i];
                }
                lu.L->data[k][i] = (A->data[k][i] - sum) / lu.U->data[i][i];
            }
        }
    }

    return lu;
}
```

```c
// 幂法求最大特征值
double power_method(matrix_t *A, double *eigenvector, int max_iter, double tol) {
    int n = A->rows;
    double *x = malloc(n * sizeof(double));
    double *y = malloc(n * sizeof(double));

    // 初始向量
    for (int i = 0; i < n; i++) x[i] = 1.0;

    double eigenvalue = 0;
    for (int iter = 0; iter < max_iter; iter++) {
        // y = Ax
        for (int i = 0; i < n; i++) {
            y[i] = 0;
            for (int j = 0; j < n; j++) {
                y[i] += A->data[i][j] * x[j];
            }
        }

        // 找最大分量
        double max_val = 0;
        for (int i = 0; i < n; i++) {
            if (fabs(y[i]) > fabs(max_val)) max_val = y[i];
        }

        // 归一化
        for (int i = 0; i < n; i++) x[i] = y[i] / max_val;

        // 检查收敛
        if (fabs(max_val - eigenvalue) < tol) {
            eigenvalue = max_val;
            break;
        }
        eigenvalue = max_val;
    }

    for (int i = 0; i < n; i++) eigenvector[i] = x[i];

    free(x);
    free(y);
    return eigenvalue;
}
```

```c
// 正态分布
typedef struct {
    double mu;      // 均值
    double sigma;   // 标准差
} normal_dist_t;

// 正态分布概率密度
double normal_pdf(normal_dist_t *dist, double x) {
    double z = (x - dist->mu) / dist->sigma;
    return exp(-0.5 * z * z) / (dist->sigma * sqrt(2 * M_PI));
}

// 正态分布累积分布(近似)
double normal_cdf(normal_dist_t *dist, double x) {
    double z = (x - dist->mu) / dist->sigma;
    // 使用误差函数近似
    return 0.5 * (1 + erf(z / sqrt(2)));
}

// 标准正态分布
double standard_normal_cdf(double z) {
    normal_dist_t std = {0, 1};
    return normal_cdf(&std, z);
}

// 二项分布
typedef struct {
    int n;          // 试验次数
    double p;       // 成功概率
} binomial_dist_t;

// 二项分布概率
double binomial_pmf(binomial_dist_t *dist, int k) {
    if (k < 0 || k > dist->n) return 0;

    // C(n,k) * p^k * (1-p)^(n-k)
    double coeff = 1;
    for (int i = 0; i < k; i++) {
        coeff *= (dist->n - i) / (i + 1.0);
    }

    return coeff * pow(dist->p, k) * pow(1 - dist->p, dist->n - k);
}

// 泊松分布
typedef struct {
    double lambda;  // 参数λ
} poisson_dist_t;

double poisson_pmf(poisson_dist_t *dist, int k) {
    return pow(dist->lambda, k) * exp(-dist->lambda) / tgamma(k + 1);
}
```

```c
// 样本统计
typedef struct {
    double *data;
    int n;
    double mean;
    double variance;
    double stddev;
} sample_stats_t;

// 计算统计量
void compute_statistics(sample_stats_t *stats) {
    // 均值
    double sum = 0;
    for (int i = 0; i < stats->n; i++) {
        sum += stats->data[i];
    }
    stats->mean = sum / stats->n;

    // 方差
    double sum_sq = 0;
    for (int i = 0; i < stats->n; i++) {
        double diff = stats->data[i] - stats->mean;
        sum_sq += diff * diff;
    }
    stats->variance = sum_sq / (stats->n - 1);  // 无偏估计
    stats->stddev = sqrt(stats->variance);
}

// 置信区间
typedef struct {
    double lower;
    double upper;
    double confidence;
} confidence_interval_t;

// 均值的置信区间
confidence_interval_t mean_confidence_interval(sample_stats_t *stats,
                                                double confidence) {
    confidence_interval_t ci;
    ci.confidence = confidence;

    // z值(95%置信度约1.96)
    double z = 1.96;
    if (confidence == 0.99) z = 2.576;
    if (confidence == 0.90) z = 1.645;

    double margin = z * stats->stddev / sqrt(stats->n);
    ci.lower = stats->mean - margin;
    ci.upper = stats->mean + margin;

    return ci;
}

// 假设检验
typedef struct {
    double test_statistic;
    double p_value;
    bool reject_null;
} hypothesis_test_t;

// 单样本t检验
hypothesis_test_t one_sample_t_test(sample_stats_t *stats, double mu0) {
    hypothesis_test_t result;

    // t统计量
    result.test_statistic = (stats->mean - mu0) / (stats->stddev / sqrt(stats->n));

    // p值(简化)
    double df = stats->n - 1;
    result.p_value = 2 * (1 - standard_normal_cdf(fabs(result.test_statistic)));

    // 在0.05显著性水平下
    result.reject_null = (result.p_value < 0.05);

    return result;
}

// 线性回归
typedef struct {
    double slope;
    double intercept;
    double r_squared;
} regression_t;

regression_t linear_regression(double *x, double *y, int n) {
    regression_t reg;

    double sum_x = 0, sum_y = 0, sum_xy = 0, sum_x2 = 0, sum_y2 = 0;
    for (int i = 0; i < n; i++) {
        sum_x += x[i];
        sum_y += y[i];
        sum_xy += x[i] * y[i];
        sum_x2 += x[i] * x[i];
        sum_y2 += y[i] * y[i];
    }

    double mean_x = sum_x / n;
    double mean_y = sum_y / n;

    reg.slope = (sum_xy - n * mean_x * mean_y) / (sum_x2 - n * mean_x * mean_x);
    reg.intercept = mean_y - reg.slope * mean_x;

    // R²
    double ss_res = 0, ss_tot = 0;
    for (int i = 0; i < n; i++) {
        double y_pred = reg.slope * x[i] + reg.intercept;
        ss_res += (y[i] - y_pred) * (y[i] - y_pred);
        ss_tot += (y[i] - mean_y) * (y[i] - mean_y);
    }
    reg.r_squared = 1 - ss_res / ss_tot;

    return reg;
}
```

| 积分 | 结果 |
|------|------|
| ∫x^n dx | x^(n+1)/(n+1) + C |
| ∫1/x dx | ln\|x\| + C |
| ∫e^x dx | e^x + C |
| ∫a^x dx | a^x/ln a + C |
| ∫sin x dx | -cos x + C |
| ∫cos x dx | sin x + C |
| ∫tan x dx | -ln\|cos x\| + C |
| ∫sec²x dx | tan x + C |
| ∫1/(1+x²) dx | arctan x + C |
| ∫1/√(1-x²) dx | arcsin x + C |
