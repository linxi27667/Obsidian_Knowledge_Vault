## 一、以太网基础

### 1.1 以太网帧格式

```
┌─────────────────────────────────────────────────────────────┐
│ 前导码(7B) │ SFD(1B) │ 目的MAC(6B) │ 源MAC(6B) │ 类型(2B) │
├─────────────────────────────────────────────────────────────┤
│                    数据(46-1500B)                            │
├─────────────────────────────────────────────────────────────┤
│                    FCS(4B)                                   │
└─────────────────────────────────────────────────────────────┘

SFD: Start Frame Delimiter (0x5D)
类型: 0x0800=IPv4, 0x0806=ARP, 0x86DD=IPv6
FCS: CRC-32校验
```

### 1.2 MAC地址

```c
// MAC地址格式
typedef struct {
    uint8_t bytes[6];
} mac_addr_t;

// MAC地址操作
void mac_to_string(mac_addr_t *mac, char *str) {
    sprintf(str, "%02X:%02X:%02X:%02X:%02X:%02X",
            mac->bytes[0], mac->bytes[1], mac->bytes[2],
            mac->bytes[3], mac->bytes[4], mac->bytes[5]);
}

bool mac_is_broadcast(mac_addr_t *mac) {
    for (int i = 0; i < 6; i++) {
        if (mac->bytes[i] != 0xFF) return false;
    }
    return true;
}

bool mac_is_multicast(mac_addr_t *mac) {
    return (mac->bytes[0] & 0x01) != 0;
}
```

## 二、MAC驱动

### 2.1 STM32以太网MAC

```c
// STM32 ETH初始化
void eth_init(void) {
    // 使能时钟
    RCC->AHB1ENR |= RCC_AHB1ENR_ETHMACEN |
                    RCC_AHB1ENR_ETHMACTXEN |
                    RCC_AHB1ENR_ETHMACRXEN;

    // MAC配置
    ETH->MACCR = ETH_MACCR_DM        |  // 全双工
                 ETH_MACCR_SPEED_100M |  // 100Mbps
                 ETH_MACCR_IPCO       |  // IPv4校验
                 ETH_MACCR_APCS;         // 自动校验

    // MAC地址
    ETH->MACA0HR = (mac_addr.bytes[5] << 8) | mac_addr.bytes[4];
    ETH->MACA0LR = (mac_addr.bytes[3] << 24) | (mac_addr.bytes[2] << 16) |
                   (mac_addr.bytes[1] << 8) | mac_addr.bytes[0];

    // DMA配置
    ETH->DMABMR = ETH_DMABMR_AAB |   // 地址对齐
                  ETH_DMABMR_RDP_32 | // RX描述符突发长度
                  ETH_DMABMR_PBL_32;  // TX描述符突发长度

    // 初始化DMA描述符
    eth_init_descriptors();

    // 启动
    ETH->DMAOMR = ETH_DMAOMR_ST |  // 启动TX
                  ETH_DMAOMR_SR;    // 启动RX
    ETH->MACCR |= ETH_MACCR_TE |   // 使能TX
                  ETH_MACCR_RE;     // 使能RX
}

// DMA描述符
typedef struct {
    uint32_t Status;
    uint32_t ControlBufferSize;
    uint32_t Buffer1Addr;
    uint32_t Buffer2NextDescAddr;
} ETH_DMADescTypeDef;

// 发送帧
int eth_send_frame(uint8_t *data, int len) {
    ETH_DMADescTypeDef *dma_tx_desc = tx_desc;

    // 设置缓冲区
    dma_tx_desc->Buffer1Addr = (uint32_t)data;
    dma_tx_desc->ControlBufferSize = len & 0x1FFF;

    // 设置控制位
    dma_tx_desc->Status = ETH_DMATXDESC_OWN |  // DMA拥有
                          ETH_DMATXDESC_LS |    // 最后一段
                          ETH_DMATXDESC_FS |    // 第一段
                          ETH_DMATXDESC_TCH;    // 第二个地址是下一个描述符

    // 触发发送
    ETH->DMATPDR = 0;

    // 等待完成
    while (dma_tx_desc->Status & ETH_DMATXDESC_OWN);

    return len;
}
```

## 三、PHY驱动

### 3.1 MDIO接口

```c
// MDIO读取
uint16_t mdio_read(uint8_t phy_addr, uint8_t reg_addr) {
    // 等待MDIO空闲
    while (ETH->MACMIIAR & ETH_MACMIIAR_MB);

    // 配置地址和寄存器
    ETH->MACMIIAR = (phy_addr << 11) |
                    (reg_addr << 6) |
                    ETH_MACMIIAR_CR_Div42 |  // 时钟分频
                    ETH_MACMIIAR_MB;          // 启动传输

    // 等待完成
    while (ETH->MACMIIAR & ETH_MACMIIAR_MB);

    return (uint16_t)(ETH->MACMIIDR & 0xFFFF);
}

// MDIO写入
void mdio_write(uint8_t phy_addr, uint8_t reg_addr, uint16_t data) {
    while (ETH->MACMIIAR & ETH_MACMIIAR_MB);

    ETH->MACMIIDR = data;
    ETH->MACMIIAR = (phy_addr << 11) |
                    (reg_addr << 6) |
                    ETH_MACMIIAR_CR_Div42 |
                    ETH_MACMIIAR_MW |         // 写操作
                    ETH_MACMIIAR_MB;

    while (ETH->MACMIIAR & ETH_MACMIIAR_MB);
}

// PHY初始化
void phy_init(uint8_t phy_addr) {
    // 软复位
    mdio_write(phy_addr, 0, 0x8000);
    while (mdio_read(phy_addr, 0) & 0x8000);

    // 自协商
    mdio_write(phy_addr, 0, 0x1000 | 0x0200);

    // 等待自协商完成
    while (!(mdio_read(phy_addr, 1) & 0x0020));
}
```

### 3.2 常见PHY芯片

```c
// LAN8720A配置
void lan8720_init(uint8_t phy_addr) {
    // 基本初始化
    phy_init(phy_addr);

    // 读取PHY ID
    uint16_t id1 = mdio_read(phy_addr, 2);
    uint16_t id2 = mdio_read(phy_addr, 3);
    printf("PHY ID: %04X:%04X\n", id1, id2);

    // 配置LED
    mdio_write(phy_addr, 31, 0x0001);  // LED配置寄存器
    mdio_write(phy_addr, 30, 0x0400);  // LED0=Link, LED1=Activity
}

// 获取链路状态
bool phy_link_status(uint8_t phy_addr) {
    return (mdio_read(phy_addr, 1) & 0x0004) != 0;
}

// 获取速度和双工
void phy_speed_duplex(uint8_t phy_addr, int *speed, bool *full_duplex) {
    uint16_t bcr = mdio_read(phy_addr, 0);

    if (bcr & 0x2000) {  // 速度选择位
        *speed = 100;
    } else {
        *speed = 10;
    }

    *full_duplex = (bcr & 0x0100) != 0;
}
```

## 四、RMII接口

### 4.1 RMII信号

```c
// RMII引脚定义
/*
 * REF_CLK  - 50MHz参考时钟
 * CRS_DV   - 载波侦听/数据有效
 * RXD0     - 接收数据0
 * RXD1     - 接收数据1
 * TX_EN    - 发送使能
 * TXD0     - 发送数据0
 * TXD1     - 发送数据1
 * MDIO     - 管理数据
 * MDC      - 管理时钟
 */

// RMII vs MII
/*
 * MII:  4位数据宽度, 25MHz时钟(100Mbps)
 * RMII: 2位数据宽度, 50MHz时钟(100Mbps)
 * RMII引脚更少，更常用
 */

// RMII配置
void rmii_init(void) {
    // 配置GPIO为AF模式
    // REF_CLK: PA1
    // CRS_DV:  PA7
    // RXD0:    PC4
    // RXD1:    PC5
    // TX_EN:   PB11
    // TXD0:    PB12
    // TXD1:    PB13
    // MDIO:    PA2
    // MDC:     PC1

    GPIO_InitTypeDef GPIO_InitStruct = {0};

    // 配置所有RMII引脚
    GPIO_InitStruct.Mode = GPIO_MODE_AF_PP;
    GPIO_InitStruct.Pull = GPIO_NOPULL;
    GPIO_InitStruct.Speed = GPIO_SPEED_FREQ_VERY_HIGH;
    GPIO_InitStruct.Alternate = GPIO_AF11_ETH;

    // PA1, PA2, PA7
    GPIO_InitStruct.Pin = GPIO_PIN_1 | GPIO_PIN_2 | GPIO_PIN_7;
    HAL_GPIO_Init(GPIOA, &GPIO_InitStruct);

    // PB11, PB12, PB13
    GPIO_InitStruct.Pin = GPIO_PIN_11 | GPIO_PIN_12 | GPIO_PIN_13;
    HAL_GPIO_Init(GPIOB, &GPIO_InitStruct);

    // PC1, PC4, PC5
    GPIO_InitStruct.Pin = GPIO_PIN_1 | GPIO_PIN_4 | GPIO_PIN_5;
    HAL_GPIO_Init(GPIOC, &GPIO_InitStruct);
}
```

def packet_handler(pkt):
