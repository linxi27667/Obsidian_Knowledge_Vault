# 功耗优化进阶

## 核心概念

- **功耗** - 电能消耗速率
- **静态功耗** - 漏电流
- **动态功耗** - 开关活动
- **占空比** - 活跃时间比例

---

## 一、功耗分析

### 1.1 功耗组成

```
总功耗 = 静态功耗 + 动态功耗

静态功耗(Pstatic):
  Pstatic = Vdd × Ileak
  - 晶体管漏电流
  - 温度相关(指数增长)

动态功耗(Pdynamic):
  Pdynamic = α × C × Vdd² × f
  - α: 翻转率
  - C: 负载电容
  - Vdd: 供电电压
  - f: 时钟频率
```

```c
 * 1. 串联电阻法:

 * 2. 霍尔传感器:

// INA219读取电流
float read_current(void) {
    uint16_t raw = i2c_read_reg(INA219_ADDR, INA219_REG_CURRENT);
    // 电流 LSB = 0.1mA (取决于配置)
    return (int16_t)raw * 0.1f;  // mA
}

// 功耗计算
float calculate_power(float voltage, float current_mA) {
    return voltage * current_mA / 1000.0f;  // mW
}
```

### 1.3 功耗预算

```c
// 功耗预算表
typedef struct {
    const char *name;
    float active_current_mA;
    float sleep_current_uA;
    float duty_cycle;  // 0-1
    float avg_current_mA;
} power_budget_item_t;

void calculate_budget(power_budget_item_t *items, int count) {
    float total = 0;
    for (int i = 0; i < count; i++) {
        items[i].avg_current_mA =
            items[i].active_current_mA * items[i].duty_cycle +
            items[i].sleep_current_uA / 1000.0f * (1 - items[i].duty_cycle);
        total += items[i].avg_current_mA;
        printf("%s: %.2f mA avg\n", items[i].name, items[i].avg_current_mA);
    }
    printf("Total: %.2f mA\n", total);
}
```

### 2.3 外设功耗管理

```c
// 外设功耗控制
typedef struct {
    volatile uint32_t *clock_reg;
    uint32_t clock_bit;
    float current_mA;
    bool enabled;
} peripheral_power_t;

peripheral_power_t peripherals[] = {
    {&RCC->APB1ENR, RCC_APB1ENR_TIM2EN, 2.0, false},
    {&RCC->APB1ENR, RCC_APB1ENR_USART2EN, 1.5, false},
    {&RCC->APB2ENR, RCC_APB2ENR_ADC1EN, 3.0, false},
    {&RCC->APB1ENR, RCC_APB1ENR_SPI2EN, 1.0, false},
};

void enable_peripheral(int idx) {
    *peripherals[idx].clock_reg |= peripherals[idx].clock_bit;
    peripherals[idx].enabled = true;
}

void disable_peripheral(int idx) {
    *peripherals[idx].clock_reg &= ~peripherals[idx].clock_bit;
    peripherals[idx].enabled = false;
}

float get_total_peripheral_power(void) {
    float total = 0;
    for (int i = 0; i < sizeof(peripherals)/sizeof(peripherals[0]); i++) {
        if (peripherals[i].enabled) total += peripherals[i].current_mA;
    }
    return total;
}
```

## 三、传感器功耗优化

### 3.1 传感器占空比

```c
// 传感器电源控制
typedef struct {
    gpio_num_t power_pin;
    uint32_t warmup_time_ms;
    uint32_t sample_time_ms;
    uint32_t interval_ms;
    float active_current_mA;
    float standby_current_uA;
} sensor_power_config_t;

sensor_power_config_t sensor = {
    .power_pin = GPIO_NUM_5,
    .warmup_time_ms = 50,
    .sample_time_ms = 10,
    .interval_ms = 60000,  // 1分钟采样一次
    .active_current_mA = 5.0,
    .standby_current_uA = 1.0
};

void sensor_power_cycle(void) {
    // 开启传感器
    gpio_set_level(sensor.power_pin, 1);
    vTaskDelay(pdMS_TO_TICKS(sensor.warmup_time_ms));

    // 采样
    float value = read_sensor();

    // 关闭传感器
    gpio_set_level(sensor.power_pin, 0);

    // 计算平均功耗
    float duty = (sensor.warmup_time_ms + sensor.sample_time_ms) / (float)sensor.interval_ms;
    float avg_current = sensor.active_current_mA * duty +
                        sensor.standby_current_uA / 1000.0f * (1 - duty);
    printf("Sensor avg current: %.3f mA\n", avg_current);
}
```

### 3.2 ADC功耗优化

```c
// ADC低功耗配置
void adc_low_power_init(void) {
    // 使用最低采样率
    ADC1->SMPR = ADC_SAMPLE_84CYCLES;  // 较长采样时间，降低时钟需求

    // 单次转换模式(非连续)
    ADC1->CR2 &= ~ADC_CR2_CONT;

    // 使用DMA减少CPU唤醒
    ADC1->CR2 |= ADC_CR2_DMA;
}

// 间歇性ADC读取
uint16_t adc_read_low_power(uint32_t channel) {
    // 配置通道
    ADC1->SQR3 = channel;

    // 开始转换
    ADC1->CR2 |= ADC_CR2_ADON;

    // 等待完成
    while (!(ADC1->SR & ADC_SR_EOC));

    uint16_t value = ADC1->DR;

    // 关闭ADC
    ADC1->CR2 &= ~ADC_CR2_ADON;

    return value;
}
```

### 6.2 状态机功耗管理

```c
// 功耗状态机
typedef enum {
    POWER_MODE_ACTIVE,
    POWER_MODE_IDLE,
    POWER_MODE_LIGHT_SLEEP,
    POWER_MODE_DEEP_SLEEP,
    POWER_MODE_SHUTDOWN
} power_mode_t;

typedef struct {
    power_mode_t mode;
    uint32_t idle_timeout_ms;
    uint32_t sleep_timeout_ms;
    uint32_t last_activity;
} power_manager_t;

static power_manager_t pm = {
    .mode = POWER_MODE_ACTIVE,
    .idle_timeout_ms = 5000,
    .sleep_timeout_ms = 30000,
    .last_activity = 0
};

void power_manager_update(void) {
    uint32_t now = get_tick();
    uint32_t idle_time = now - pm.last_activity;

    switch (pm.mode) {
        case POWER_MODE_ACTIVE:
            if (idle_time > pm.idle_timeout_ms) {
                // 降低频率
                set_cpu_frequency(80);
                pm.mode = POWER_MODE_IDLE;
            }
            break;

        case POWER_MODE_IDLE:
            if (idle_time > pm.sleep_timeout_ms) {
                // 进入轻度睡眠
                pm.mode = POWER_MODE_LIGHT_SLEEP;
                enter_light_sleep();
            }
            break;

        case POWER_MODE_LIGHT_SLEEP:
            // 唤醒后回到ACTIVE
            pm.mode = POWER_MODE_ACTIVE;
            set_cpu_frequency(240);
            break;
    }
}

void power_manager_activity(void) {
    pm.last_activity = get_tick();
    if (pm.mode != POWER_MODE_ACTIVE) {
        pm.mode = POWER_MODE_ACTIVE;
        set_cpu_frequency(240);
    }
}
```

## 七、电池管理

### 7.1 电池电量估算

```c
// 电池电量表
typedef struct {
    float voltage;
    int soc_percent;  // 0-100
} battery_soc_point_t;

battery_soc_point_t li_ion_soc[] = {
    {4.20, 100},
    {4.03, 90},
    {3.86, 80},
    {3.83, 70},
    {3.79, 60},
    {3.75, 50},
    {3.73, 40},
    {3.71, 30},
    {3.69, 20},
    {3.61, 10},
    {3.27, 0},
};

int estimate_soc(float voltage) {
    for (int i = 0; i < sizeof(li_ion_soc)/sizeof(li_ion_soc[0]) - 1; i++) {
        if (voltage >= li_ion_soc[i+1].voltage) {
            float ratio = (voltage - li_ion_soc[i+1].voltage) /
                         (li_ion_soc[i].voltage - li_ion_soc[i+1].voltage);
            return li_ion_soc[i+1].soc_percent +
                   ratio * (li_ion_soc[i].soc_percent - li_ion_soc[i+1].soc_percent);
        }
    }
    return 0;
}

// 运行时间估算
float estimate_runtime_hours(float battery_mah, float avg_current_ma) {
    return battery_mah / avg_current_ma;
}
```
