# 电源管理技术详解

## 核心概念

- **LDO** - 低压差线性稳压器

## 一、LDO设计

### 1.1 LDO原理

```c
// LDO特性
/*
 * 优点:
 * - 低噪声
 * - 简单
 * - 快速响应
 * - 低成本
 *
 * 缺点:
 * - 效率低(压差大时)
 * - 发热
 *
 * 效率 = Vout / Vin × 100%
 *
 * 例: 5V → 3.3V
 * 效率 = 3.3 / 5 × 100% = 66%
 * 损耗 = (5 - 3.3) × I
 */

// LDO选型参数
typedef struct {
    float vin_min;
    float vin_max;
    float vout;
    float iout_max;
    float dropout_mv;
    float noise_uv;
    float psrr_db;
} ldo_params_t;
```

### 1.2 LDO应用电路

```c
// LDO电路设计
/*
 *        VIN
 *         │
 *     ┌───┴───┐
 *     │  LDO  │
 *     │       │

 *     └───┬───┘
 *         │
 *    C1   ├── VOUT

 * 输入电容C1: 1-10μF(陶瓷)
```

## 二、DC-DC设计

### 2.1 Buck变换器

```c
// Buck变换器参数计算
typedef struct {
    float vin;
    float vout;
    float iout;
    float fsw;      // 开关频率
    float ripple;   // 纹波要求
} buck_params_t;

void calc_buck_components(buck_params_t *p, float *L, float *C) {
    float duty = p->vout / p->vin;

    // 电感计算
    float delta_i = 0.3f * p->iout;  // 30%纹波电流
    *L = (p->vin - p->vout) * duty / (p->fsw * delta_i);

    // 电容计算
    float delta_v = p->ripple * p->vout;
    *C = delta_i / (8 * p->fsw * delta_v);
}

// Buck效率估算
float calc_buck_efficiency(float vin, float vout, float iout) {
    // 典型损耗
    float rds_on = 0.05f;    // MOSFET导通电阻
    float dcr = 0.01f;       // 电感DCR
    float switching_loss = 0.02f;  // 开关损耗(比例)

    float conduction_loss = iout * iout * (rds_on + dcr);
    float switching_total = vin * iout * switching_loss;

    float total_loss = conduction_loss + switching_total;
    float input_power = vin * iout;

    return (input_power - total_loss) / input_power;
}
```

### 2.2 Boost变换器

```c
// Boost变换器参数
typedef struct {
    float vin;
    float vout;
    float iout;
    float fsw;
} boost_params_t;

void calc_boost_components(boost_params_t *p, float *L, float *C) {
    float duty = 1 - p->vin / p->vout;

    // 电感计算
    float delta_i = 0.4f * p->vin * duty / (p->fsw * *L);
    *L = p->vin * duty / (p->fsw * 0.4f * p->iout * p->vout / p->vin);

    // 电容计算
    *C = p->iout * duty / (p->fsw * 0.01f * p->vout);
}
```

## 三、电池管理

### 3.1 锂电池充电

```c
// 锂电池充电曲线
/*
 * 阶段1: 预充电(涓流)
 *   - 电压 < 3.0V
 *   - 电流 = 0.1C
 *
 * 阶段2: 恒流充电(CC)
 *   - 电压 3.0-4.2V
 *   - 电流 = 0.5-1C
 *
 * 阶段3: 恒压充电(CV)
 *   - 电压 = 4.2V
 *   - 电流逐渐减小
 *
 * 阶段4: 充满
 *   - 电流 < 0.05C
 */

typedef enum {
    CHG_PRECHARGE,
    CHG_CC,
    CHG_CV,
    CHG_DONE,
    CHG_FAULT
} charge_state_t;

typedef struct {
    charge_state_t state;
    float voltage;
    float current;
    float target_voltage;
    float target_current;
    uint32_t start_time;
} charger_t;

void charger_update(charger_t *chg) {
    switch (chg->state) {
        case CHG_PRECHARGE:
            if (chg->voltage >= 3.0f) {
                chg->state = CHG_CC;
                chg->target_current = 1.0f;  // 1C
            }
            break;

        case CHG_CC:
            if (chg->voltage >= 4.2f) {
                chg->state = CHG_CV;
                chg->target_voltage = 4.2f;
            }
            break;

        case CHG_CV:
            if (chg->current < 0.05f) {
                chg->state = CHG_DONE;
            }
            break;

        case CHG_DONE:
            // 维持浮充
            break;
    }
}
```

### 3.2 电量计

```c
// 库仑计数法
typedef struct {
    float capacity_mah;
    float remaining_mah;
    float current_ma;
    uint32_t last_time;
    float soc;
} fuel_gauge_t;

void fuel_gauge_update(fuel_gauge_t *fg, float current_ma) {
    uint32_t now = get_tick();
    float dt_hours = (now - fg->last_time) / 3600000.0f;

    fg->remaining_mah -= current_ma * dt_hours;

    // 限幅
    if (fg->remaining_mah > fg->capacity_mah)
        fg->remaining_mah = fg->capacity_mah;
    if (fg->remaining_mah < 0)
        fg->remaining_mah = 0;

    fg->soc = fg->remaining_mah / fg->capacity_mah * 100;
    fg->current_ma = current_ma;
    fg->last_time = now;
}

// 电压修正
float voltage_corrected_soc(float voltage) {
    // 锂电池OCV-SOC曲线
    if (voltage >= 4.20f) return 100;
    if (voltage >= 4.03f) return 90;
    if (voltage >= 3.86f) return 80;
    if (voltage >= 3.83f) return 70;
    if (voltage >= 3.79f) return 60;
    if (voltage >= 3.75f) return 50;
    if (voltage >= 3.73f) return 40;
    if (voltage >= 3.71f) return 30;
    if (voltage >= 3.69f) return 20;
    if (voltage >= 3.61f) return 10;
    return 0;
}
```

## 四、PMIC设计

### 4.1 多路电源时序

```c
// 电源时序控制
typedef struct {
    const char *name;
    gpio_num_t enable_pin;
    float voltage;
    uint32_t delay_ms;
    float max_current_ma;
} power_rail_t;

power_rail_t power_rails[] = {
    {"1.2V_CORE",  GPIO_NUM_5,  1.2f, 0,    500},
    {"1.8V_IO",    GPIO_NUM_6,  1.8f, 10,   300},
    {"3.3V_PERI",  GPIO_NUM_7,  3.3f, 20,   800},
    {"5V_USB",     GPIO_NUM_8,  5.0f, 30,   1000},
};

void power_sequence_on(void) {
    for (int i = 0; i < sizeof(power_rails)/sizeof(power_rails[0]); i++) {
        vTaskDelay(pdMS_TO_TICKS(power_rails[i].delay_ms));
        gpio_set_level(power_rails[i].enable_pin, 1);
        printf("Power ON: %s (%.1fV)\n", power_rails[i].name, power_rails[i].voltage);
    }
}

void power_sequence_off(void) {
    for (int i = sizeof(power_rails)/sizeof(power_rails[0]) - 1; i >= 0; i--) {
        gpio_set_level(power_rails[i].enable_pin, 0);
        printf("Power OFF: %s\n", power_rails[i].name);
        vTaskDelay(pdMS_TO_TICKS(10));
    }
}
```

## 五、电源保护

### 5.1 过压保护

```c
// 过压保护电路
/*
 * VIN ──┤TVS├──┬── 稳压器
 *              │
 *             R1
 *              │
 *              ├── 检测点
 *              │
 *             R2
 *              │
 *             GND
 *
 * 检测电压 = VIN × R2 / (R1 + R2)
 * 当检测电压 > 阈值时触发保护
 */

// 软件过压保护
void overvoltage_check(float voltage, float limit) {
    if (voltage > limit) {
        // 关断输出
        gpio_set_level(POWER_EN_PIN, 0);
        // 报警
        trigger_alarm(ALARM_OVERVOLTAGE);
    }
}
```

---

### 5.2 过流保护

```c
// 过流保护
typedef struct {
    float current_limit;
    float trip_time_ms;
    uint32_t fault_start;
    bool tripped;
} overcurrent_prot_t;

bool overcurrent_check(overcurrent_prot_t *ocp, float current) {
    if (current > ocp->current_limit) {
        if (ocp->fault_start == 0) {
            ocp->fault_start = get_tick();
        } else if (get_tick() - ocp->fault_start > ocp->trip_time_ms) {
            ocp->tripped = true;
            gpio_set_level(POWER_EN_PIN, 0);
            return true;
        }
    } else {
        ocp->fault_start = 0;
    }
    return false;
}
```
