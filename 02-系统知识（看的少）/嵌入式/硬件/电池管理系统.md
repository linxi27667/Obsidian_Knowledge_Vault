## 二、SOC估算

### 2.1 库仑计数法

```c
// 库仑计数
typedef struct {
    float capacity_mah;      // 总容量
    float remaining_mah;     // 剩余容量
    float current_ma;        // 当前电流
    uint32_t last_tick;      // 上次更新时间
    float soc;               // 荷电状态
    bool calibrated;         // 是否已校准
} coulomb_counter_t;

void coulomb_counter_update(coulomb_counter_t *cc, float current_ma) {
    uint32_t now = get_tick();
    float dt_hours = (now - cc->last_tick) / 3600000.0f;

    // 累计电量变化
    cc->remaining_mah += current_ma * dt_hours;

    // 限幅
    if (cc->remaining_mah > cc->capacity_mah)
        cc->remaining_mah = cc->capacity_mah;
    if (cc->remaining_mah < 0)
        cc->remaining_mah = 0;

    // 计算SOC
    cc->soc = (cc->remaining_mah / cc->capacity_mah) * 100.0f;

    cc->current_ma = current_ma;
    cc->last_tick = now;
}

// SOC校准(满充满放)
void calibrate_soc(coulomb_counter_t *cc, float voltage) {
    // 充满校准
    if (voltage >= 4.2f && cc->current_ma < 0) {
        cc->remaining_mah = cc->capacity_mah;
        cc->soc = 100.0f;
        cc->calibrated = true;
    }

    // 放空校准
    if (voltage <= 2.5f) {
        cc->remaining_mah = 0;
        cc->soc = 0.0f;
        cc->calibrated = true;
    }
}
```

### 2.2 电压法

```c
// 电压-SOC查表
float voltage_to_soc(float voltage) {
    int size = sizeof(ocv_curve) / sizeof(ocv_curve[0]);

    // 查找区间
    for (int i = 0; i < size - 1; i++) {
        if (voltage >= ocv_curve[i+1].voltage &&
            voltage <= ocv_curve[i].voltage) {
            // 线性插值
            float t = (voltage - ocv_curve[i+1].voltage) /
                      (ocv_curve[i].voltage - ocv_curve[i+1].voltage);
            return ocv_curve[i+1].soc +
                   t * (ocv_curve[i].soc - ocv_curve[i+1].soc);
        }
    }

    if (voltage > ocv_curve[0].voltage) return 100.0f;
    return 0.0f;
}

// 综合SOC估算
float estimate_soc(coulomb_counter_t *cc, float voltage, float current) {
    // 静置时用电压法校准
    if (fabsf(current) < 10.0f) {  // 小于10mA认为静置
        float voltage_soc = voltage_to_soc(voltage);
        // 缓慢修正库仑计数
        cc->soc = 0.9f * cc->soc + 0.1f * voltage_soc;
    }

    return cc->soc;
}
```

## 三、SOH估算

### 3.1 电池老化

```c
// SOH估算
typedef struct {
    float initial_capacity;   // 初始容量
    float current_capacity;   // 当前容量
    int cycle_count;          // 循环次数
    float internal_resistance;// 内阻
    float soh;                // 健康状态
} soh_estimator_t;

void update_soh(soh_estimator_t *soh, float full_charge_capacity) {
    // 容量衰减
    soh->current_capacity = full_charge_capacity;
    soh->soh = (soh->current_capacity / soh->initial_capacity) * 100.0f;

    // 循环次数记录
    soh->cycle_count++;
}

// 容量衰减模型
float capacity_fade_model(int cycles, float temperature) {
    // Arrhenius模型简化
    float k = 0.0001f;  // 衰减系数
    float Ea = 20000.0f; // 活化能
    float R = 8.314f;     // 气体常数
    float T = temperature + 273.15f;

    float rate = k * expf(-Ea / (R * T));
    float capacity_retention = 1.0f - rate * sqrtf(cycles);

    return capacity_retention * 100.0f;
}
```

## 四、电池均衡

### 4.1 被动均衡

```c
// 被动均衡(电阻放电)
typedef struct {
    float cell_voltages[8];   // 单体电压
    bool balance_switches[8]; // 均衡开关
    float balance_current;    // 均衡电流
    float threshold;          // 压差阈值
} passive_balancer_t;

void passive_balance(passive_balancer_t *bal) {
    // 找到最低电压
    float min_voltage = bal->cell_voltages[0];
    for (int i = 1; i < 8; i++) {
        if (bal->cell_voltages[i] < min_voltage) {
            min_voltage = bal->cell_voltages[i];
        }
    }

    // 开启高压单体的均衡
    for (int i = 0; i < 8; i++) {
        float diff = bal->cell_voltages[i] - min_voltage;
        if (diff > bal->threshold) {
            bal->balance_switches[i] = true;
            // 放电电阻消耗多余能量
            // P = V²/R
        } else {
            bal->balance_switches[i] = false;
        }
    }
}
```

### 4.2 主动均衡

```c
// 主动均衡(能量转移)
typedef enum {
    BALANCE_COUK,      // Cuk变换器
    BALANCE_FLYBACK,   // 反激变换器
    BALANCE_CAPACITOR, // 电容切换
} balance_method_t;

// 电容切换均衡
void capacitor_balance(float *cell_voltages, int num_cells) {
    // 使用飞电容在相邻单体间转移能量
    static float flying_cap_voltage = 0;
    static int phase = 0;

    switch (phase) {
        case 0:
            // 从高电压单体充电飞电容
            int high_cell = find_highest(cell_voltages, num_cells);
            connect_flying_cap(high_cell);
            phase = 1;
            break;

        case 1:
            // 等待充电完成
            if (flying_cap_voltage >= cell_voltages[high_cell] - 0.1f) {
                phase = 2;
            }
            break;

        case 2:
            // 向低电压单体放电
            int low_cell = find_lowest(cell_voltages, num_cells);
            connect_flying_cap(low_cell);
            phase = 3;
            break;

        case 3:
            // 等待放电完成
            if (flying_cap_voltage <= cell_voltages[low_cell] + 0.1f) {
                phase = 0;
            }
            break;
    }
}
```

## 五、充放电管理

### 5.1 充电控制

```c
// CC-CV充电状态机
typedef enum {
    CHG_IDLE,
    CHG_PRECHARGE,
    CHG_CC,
    CHG_CV,
    CHG_DONE,
    CHG_FAULT,
} charge_state_t;

typedef struct {
    charge_state_t state;
    float voltage;
    float current;
    float target_voltage;
    float target_current;
    uint32_t start_time;
    uint32_t timeout;
} charger_t;

void charger_update(charger_t *chg) {
    switch (chg->state) {
        case CHG_IDLE:
            if (chg->voltage < 2.5f) {
                chg->state = CHG_PRECHARGE;
                chg->target_current = 0.1f * chg->target_current;
            } else {
                chg->state = CHG_CC;
            }
            break;

        case CHG_PRECHARGE:
            if (chg->voltage >= 2.5f) {
                chg->state = CHG_CC;
            }
            break;

        case CHG_CC:
            // 恒流充电
            set_charge_current(chg->target_current);
            if (chg->voltage >= chg->target_voltage) {
                chg->state = CHG_CV;
            }
            break;

        case CHG_CV:
            // 恒压充电
            set_charge_voltage(chg->target_voltage);
            if (chg->current < 0.05f * chg->target_current) {
                chg->state = CHG_DONE;
            }
            break;

        case CHG_DONE:
            // 充电完成
            set_charge_current(0);
            break;
    }
}
```

### 5.2 放电保护

```c
// 放电保护
typedef struct {
    float min_voltage;
    float max_current;
    float max_temperature;
    bool protection_active;
} discharge_protection_t;

bool check_discharge_protection(discharge_protection_t *prot,
                                float voltage, float current,
                                float temperature) {
    // 欠压保护
    if (voltage < prot->min_voltage) {
        prot->protection_active = true;
        printf("Under-voltage protection: %.2fV\n", voltage);
        return false;
    }

    // 过流保护
    if (current > prot->max_current) {
        prot->protection_active = true;
        printf("Over-current protection: %.2fA\n", current);
        return false;
    }

    // 温度保护
    if (temperature > prot->max_temperature) {
        prot->protection_active = true;
        printf("Over-temperature protection: %.1f°C\n", temperature);
        return false;
    }

    return true;
}
```

## 六、电池安全

### 6.1 安全机制

```c
// 电池安全等级
typedef enum {
    SAFETY_LEVEL_NORMAL,
    SAFETY_LEVEL_WARNING,
    SAFETY_LEVEL_PROTECTION,
    SAFETY_LEVEL_SHUTDOWN,
} safety_level_t;

safety_level_t evaluate_safety(float voltage, float current,
                                float temperature, int cycle_count) {
    // 电压检查
    if (voltage < 2.0f || voltage > 4.5f)
        return SAFETY_LEVEL_SHUTDOWN;

    // 温度检查
    if (temperature > 60.0f || temperature < -20.0f)
        return SAFETY_LEVEL_SHUTDOWN;

    // 电流检查
    if (fabsf(current) > 10.0f)
        return SAFETY_LEVEL_PROTECTION;

    // 老化检查
    if (cycle_count > 1000)
        return SAFETY_LEVEL_WARNING;

    return SAFETY_LEVEL_NORMAL;
}
```
