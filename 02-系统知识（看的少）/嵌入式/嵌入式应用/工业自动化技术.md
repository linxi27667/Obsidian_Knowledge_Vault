## 一、PLC技术

### 1.1 PLC编程

```c
// PLC编程语言(IEC 61131-3)
// 1. LD(梯形图)
// 2. FBD(功能块图)
// 3. ST(结构化文本)
// 4. IL(指令表)
// 5. SFC(顺序功能图)

// ST语言示例
void plc_program(void) {
    // 输入
    bool start_btn = read_digital_input(0);
    bool stop_btn = read_digital_input(1);
    bool sensor = read_digital_input(2);

    // 内部变量
    static bool motor_running = false;
    static int counter = 0;

    // 逻辑
    if (start_btn && !stop_btn && sensor) {
        motor_running = true;
    }
    if (stop_btn || !sensor) {
        motor_running = false;
    }

    // 输出
    write_digital_output(0, motor_running);

    // 计数器
    if (sensor && !last_sensor) {
        counter++;
    }
    last_sensor = sensor;
}
```

### 1.2 功能块

```c
// 定时器功能块
typedef struct {
    bool enable;
    bool output;
    uint32_t preset_time;
    uint32_t elapsed_time;
    uint32_t start_time;
} timer_t;

bool timer_ton(timer_t *timer, bool input) {
    if (input) {
        if (!timer->enable) {
            timer->enable = true;
            timer->start_time = get_tick();
        }
        timer->elapsed_time = get_tick() - timer->start_time;
        timer->output = (timer->elapsed_time >= timer->preset_time);
    } else {
        timer->enable = false;
        timer->elapsed_time = 0;
        timer->output = false;
    }
    return timer->output;
}

// 计数器功能块
typedef struct {
    bool output;
    int preset_value;
    int current_value;
    bool last_input;
} counter_t;

bool counter_ctu(counter_t *counter, bool input) {
    if (input && !counter->last_input) {
        counter->current_value++;
        if (counter->current_value >= counter->preset_value) {
            counter->output = true;
        }
    }
    counter->last_input = input;
    return counter->output;
}
```

### 2.2 数据采集

```c
// SCADA数据点
typedef struct {
    char name[32];
    char description[64];
    enum { TYPE_BOOL, TYPE_INT, TYPE_FLOAT, TYPE_STRING } type;
    union {
        bool bool_val;
        int int_val;
        float float_val;
        char string_val[64];
    } value;
    float min_value;
    float max_value;
    char unit[8];
    uint32_t timestamp;
    bool alarm_enabled;
    float alarm_high;
    float alarm_low;
} scada_point_t;

// 数据采集循环
void scada_poll_task(void *param) {
    while (1) {
        for (int i = 0; i < point_count; i++) {
            // 读取数据
            switch (points[i].type) {
                case TYPE_BOOL:
                    points[i].value.bool_val = modbus_read_coil(i);
                    break;
                case TYPE_INT:
                    points[i].value.int_val = modbus_read_register(i);
                    break;
                case TYPE_FLOAT:
                    points[i].value.float_val = modbus_read_float(i);
                    break;
            }
            points[i].timestamp = get_tick();

            // 报警检查
            if (points[i].alarm_enabled) {
                check_alarm(&points[i]);
            }
        }

        vTaskDelay(pdMS_TO_TICKS(1000));
    }
}
```

## 四、工业通信

### 4.1 OPC UA

```c
// OPC UA客户端
typedef struct {
    char endpoint[128];
    int session_id;
    int subscription_id;
} opcua_client_t;

// OPC UA节点
typedef struct {
    uint32_t node_id;
    char name[64];
    int data_type;
    union {
        bool bool_val;
        int int_val;
        float float_val;
    } value;
} opcua_node_t;

// OPC UA读取
int opcua_read(opcua_client_t *client, opcua_node_t *node) {
    // 构建Read请求
    uint8_t request[256];
    int len = build_read_request(request, node->node_id);

    // 发送请求
    send(client->session_id, request, len, 0);

    // 接收响应
    uint8_t response[256];
    int recv_len = recv(client->session_id, response, sizeof(response), 0);

    // 解析响应
    parse_read_response(response, node);

    return 0;
}

// OPC UA订阅
void opcua_subscribe(opcua_client_t *client, opcua_node_t *nodes,
                     int count, int interval_ms) {
    // 创建订阅
    client->subscription_id = create_subscription(client, interval_ms);

    // 添加监控项
    for (int i = 0; i < count; i++) {
        add_monitored_item(client, client->subscription_id,
                           nodes[i].node_id);
    }
}
```

## 六、功能安全

### 6.1 IEC 61508

```c
// 安全完整性等级(SIL)
typedef enum {
    SIL_1,  // 10^-5 - 10^-4
    SIL_2,  // 10^-6 - 10^-5
    SIL_3,  // 10^-7 - 10^-6
    SIL_4,  // 10^-8 - 10^-7
} sil_level_t;

// 安全功能
typedef struct {
    sil_level_t sil_level;
    bool (*safe_state)(void);
    uint32_t test_interval;
    uint32_t last_test;
} safety_function_t;

// 安全功能测试
bool test_safety_function(safety_function_t *func) {
    // 执行安全功能测试
    bool result = func->safe_state();
    func->last_test = get_tick();

    if (!result) {
        // 进入安全状态
        enter_safe_state();
    }

    return result;
}
```
