## 一、汽车电子概述

### 1.1 汽车电子系统

### 2.2 SWC设计

```c
// 软件组件(SWC)接口定义
typedef struct {
    // 输入端口
    float throttle_input;
    float brake_input;
    float steering_angle;

    // 输出端口
    float engine_torque;
    float brake_pressure;
    float steering_assist;

    // 状态
    int system_state;
} swc_vehicle_control_t;

// SWC运行函数
void SWC_VehicleControl_Run(swc_vehicle_control_t *swc) {
    // 读取输入(通过RTE)
    Rte_Read_ThrottleInput(&swc->throttle_input);
    Rte_Read_BrakeInput(&swc->brake_input);

    // 处理逻辑
    if (swc->brake_input > 0.1f) {
        swc->engine_torque = 0.0f;
        swc->brake_pressure = swc->brake_input * 100.0f;
    } else {
        swc->engine_torque = swc->throttle_input * 100.0f;
        swc->brake_pressure = 0.0f;
    }

    // 写出输出(通过RTE)
    Rte_Write_EngineTorque(swc->engine_torque);
    Rte_Write_BrakePressure(swc->brake_pressure);
}
```

```c
void Com_SendSignal(Com_SignalIdType signalId, const void *signalData) {
    // 将信号写入IPDU缓冲区
    Com_IpduGroupType ipduId = Com_GetIpduOfSignal(signalId);
    uint8_t *ipduBuffer = Com_GetIpduBuffer(ipduId);

    // 信号打包
    Com_PackSignal(signalId, signalData, ipduBuffer);

    // 触发发送
    Com_TriggerIpduSend(ipduId);
}

// CAN信号接收
void Com_ReceiveSignal(Com_SignalIdType signalId, void *signalData) {
    Com_IpduGroupType ipduId = Com_GetIpduOfSignal(signalId);
    uint8_t *ipduBuffer = Com_GetIpduBuffer(ipduId);

    // 信号解包
    Com_UnpackSignal(signalId, ipduBuffer, signalData);
}
```

### 3.2 安全机制

```c
// 看门狗监控
typedef struct {
    uint32_t counter;
    uint32_t expected;
    bool alive;
} watchdog_monitor_t;

void watchdog_check(watchdog_monitor_t *w) {
    w->counter++;
    if (w->counter != w->expected) {
        // 安全响应: 进入安全状态
        enter_safe_state();
    }
    w->expected = w->counter + 1;
    w->alive = true;
}

// 程序流监控
void program_flow_check(void) {
    static uint32_t checkpoint = 0;

    switch (checkpoint) {
        case 0:
            // 执行任务1
            task1();
            checkpoint = 1;
            break;
        case 1:
            // 执行任务2
            task2();
            checkpoint = 2;
            break;
        case 2:
            // 执行任务3
            task3();
            checkpoint = 0;
            break;
        default:
            // 流程错误
            enter_safe_state();
            break;
    }
}

// 内存校验
uint32_t calculate_crc32(uint8_t *data, uint32_t len) {
    uint32_t crc = 0xFFFFFFFF;
    for (uint32_t i = 0; i < len; i++) {
        crc ^= data[i];
        for (int j = 0; j < 8; j++) {
            if (crc & 1) {
                crc = (crc >> 1) ^ 0xEDB88320;
            } else {
                crc >>= 1;
            }
        }
    }
    return ~crc;
}

bool verify_critical_data(uint8_t *data, uint32_t len, uint32_t expected_crc) {
    uint32_t crc = calculate_crc32(data, len);
    return (crc == expected_crc);
}
```

### 3.3 双通道监控

```c
// 双通道冗余计算
typedef struct {
    float (*compute_a)(float input);
    float (*compute_b)(float input);
    float tolerance;
} dual_channel_t;

bool dual_channel_check(dual_channel_t *ch, float input, float *result) {
    float result_a = ch->compute_a(input);
    float result_b = ch->compute_b(input);

    float diff = fabsf(result_a - result_b);

    if (diff <= ch->tolerance) {
        *result = (result_a + result_b) / 2.0f;
        return true;  // 一致
    } else {
        // 不一致，进入安全状态
        return false;
    }
}
```

## 四、UDS诊断

### 4.1 UDS服务

| SID | 服务 | 说明 |
|-----|------|------|
| 0x10 | DiagnosticSessionControl | 诊断会话 |
| 0x11 | ECUReset | ECU复位 |
| 0x22 | ReadDataByIdentifier | 读数据 |
| 0x23 | ReadMemoryByAddress | 读内存 |
| 0x27 | SecurityAccess | 安全访问 |
| 0x2E | WriteDataByIdentifier | 写数据 |
| 0x31 | RoutineControl | 例程控制 |
| 0x34-0x37 | Download/Upload | 下载/上传 |
| 0x3E | TesterPresent | 测试器在线 |

---

### 4.2 UDS实现

```c
// UDS请求处理
void uds_process_request(uint8_t *request, int len) {
    uint8_t sid = request[0];
    uint8_t response[256];
    int resp_len = 0;

    switch (sid) {
        case 0x10: {  // 诊断会话
            uint8_t session_type = request[1];
            response[0] = 0x50;  // 正响应
            response[1] = session_type;
            response[2] = 0x00;  // P2 high
            response[3] = 0x32;  // P2 low (50ms)
            response[4] = 0x03;  // P2* high
            response[5] = 0xE8;  // P2* low (1000ms)
            resp_len = 6;
            break;
        }

        case 0x22: {  // 读数据
            uint16_t did = (request[1] << 8) | request[2];
            response[0] = 0x62;  // 正响应
            response[1] = request[1];
            response[2] = request[2];

            // 读取DID数据
            int data_len = uds_read_did(did, &response[3]);
            resp_len = 3 + data_len;
            break;
        }

        case 0x27: {  // 安全访问
            uint8_t sub_func = request[1];
            if (sub_func == 0x01) {
                // 请求种子
                response[0] = 0x67;
                response[1] = 0x01;
                // 生成随机种子
                uint32_t seed = generate_seed();
                response[2] = (seed >> 24) & 0xFF;
                response[3] = (seed >> 16) & 0xFF;
                response[4] = (seed >> 8) & 0xFF;
                response[5] = seed & 0xFF;
                resp_len = 6;
            } else if (sub_func == 0x02) {
                // 验证密钥
                uint32_t key = (request[2] << 24) | (request[3] << 16) |
                               (request[4] << 8) | request[5];
                if (verify_key(key)) {
                    response[0] = 0x67;
                    response[1] = 0x02;
                    resp_len = 2;
                    security_unlocked = true;
                } else {
                    response[0] = 0x7F;
                    response[1] = 0x27;
                    response[2] = 0x35;  // 无效密钥
                    resp_len = 3;
                }
            }
            break;
        }

        default:
            // 否定响应
            response[0] = 0x7F;
            response[1] = sid;
            response[2] = 0x11;  // 不支持的服务
            resp_len = 3;
            break;
    }

    uds_send_response(response, resp_len);
}
```

```c
// XCP命令处理
void xcp_process_command(uint8_t *cmd, int len) {
    uint8_t cmd_code = cmd[0];

    switch (cmd_code) {
        case 0xFF:  // CONNECT
            xcp_connect(cmd[1]);
            break;
        case 0xF1:  // DISCONNECT
            xcp_disconnect();
            break;
        case 0xF4:  // GET_STATUS
            xcp_get_status();
            break;
        case 0xF5:  // SYNCH
            xcp_synch();
            break;
        case 0xF8:  // SHORT_UPLOAD
            xcp_short_upload(cmd);
            break;
    }
}
```
