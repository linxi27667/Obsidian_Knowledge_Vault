# 操作系统原理

## 核心概念

## 一、进程管理

### 1.1 进程控制块(PCB)

```c
// 进程控制块
typedef struct process {
    uint32_t pid;                // 进程ID
    char name[32];               // 进程名
    process_state_t state;       // 状态
    uint32_t priority;           // 优先级

    // 上下文
    uint32_t *stack_pointer;     // 栈指针
    uint32_t *page_table;        // 页表

    // 内存信息
    uint32_t code_start;
    uint32_t code_end;
    uint32_t data_start;
    uint32_t data_end;
    uint32_t heap_start;
    uint32_t stack_start;

    // 文件描述符
    int fd_table[MAX_FD];

    // 时间信息
    uint32_t create_time;
    uint32_t cpu_time;
    uint32_t wakeup_time;

    // 关系
    struct process *parent;
    struct process *children;
    struct process *next;
} process_t;

// 进程状态
typedef enum {
    PROC_NEW,
    PROC_READY,
    PROC_RUNNING,
    PROC_WAITING,
    PROC_TERMINATED,
    PROC_ZOMBIE,
} process_state_t;
```

### 1.2 进程调度

```c
// 调度队列
typedef struct {
    process_t *head;
    process_t *tail;
    int count;
} ready_queue_t;

// 调度算法
process_t *scheduler_round_robin(ready_queue_t *rq) {
    if (rq->head == NULL) return NULL;

    process_t *proc = rq->head;
    rq->head = proc->next;
    if (rq->head == NULL) rq->tail = NULL;
    rq->count--;

    proc->state = PROC_RUNNING;
    return proc;
}

// 上下文切换
void context_switch(process_t *prev, process_t *next) {
    // 保存当前进程上下文
    if (prev) {
        save_registers(prev);
        prev->state = PROC_READY;
        enqueue_ready(prev);
    }

    // 恢复下一个进程上下文
    next->state = PROC_RUNNING;
    current_process = next;

    // 切换页表
    set_page_table(next->page_table);

    // 切换栈指针
    set_stack_pointer(next->stack_pointer);

    // 恢复寄存器
    restore_registers(next);
}
```

### 1.3 进程间通信(IPC)

```c
// 管道
typedef struct {
    uint8_t buffer[PIPE_BUFFER_SIZE];
    int head;
    int tail;
    int count;
    process_t *reader;
    process_t *writer;
} pipe_t;

int pipe_write(pipe_t *pipe, const void *data, int len) {
    for (int i = 0; i < len; i++) {
        while (pipe->count >= PIPE_BUFFER_SIZE) {
            // 阻塞等待
            sleep_process(pipe->writer);
        }
        pipe->buffer[pipe->tail] = ((uint8_t *)data)[i];
        pipe->tail = (pipe->tail + 1) % PIPE_BUFFER_SIZE;
        pipe->count++;
    }
    // 唤醒读进程
    wakeup_process(pipe->reader);
    return len;
}

int pipe_read(pipe_t *pipe, void *data, int max_len) {
    int read = 0;
    while (read < max_len && pipe->count > 0) {
        ((uint8_t *)data)[read++] = pipe->buffer[pipe->head];
        pipe->head = (pipe->head + 1) % PIPE_BUFFER_SIZE;
        pipe->count--;
    }
    // 唤醒写进程
    wakeup_process(pipe->writer);
    return read;
}

// 共享内存
typedef struct {
    uint32_t key;
    uint32_t size;
    uint32_t *pages;
    int ref_count;
} shared_mem_t;

void *shm_get(uint32_t key, uint32_t size) {
    // 查找或创建共享内存
    shared_mem_t *shm = find_shared_mem(key);
    if (!shm) {
        shm = create_shared_mem(key, size);
    }

    // 映射到当前进程地址空间
    return map_shared_mem(shm);
}
```

## 二、内存管理

### 2.1 虚拟内存

```c
// 页表项(PTE)
typedef struct {
    uint32_t present    : 1;   // 在内存中
    uint32_t writable   : 1;   // 可写
    uint32_t user       : 1;   // 用户可访问
    uint32_t accessed   : 1;   // 已访问
    uint32_t dirty      : 1;   // 已修改
    uint32_t frame      : 20;  // 物理帧号
} pte_t;

// 页目录(PDE)
typedef struct {
    pte_t *page_table;
    uint32_t present : 1;
} pde_t;

// 虚拟地址解析
// 32位虚拟地址: [10位页目录索引][10位页表索引][12位偏移]
uint32_t translate_address(uint32_t virtual_addr) {
    uint32_t page_dir_idx = (virtual_addr >> 22) & 0x3FF;
    uint32_t page_table_idx = (virtual_addr >> 12) & 0x3FF;
    uint32_t offset = virtual_addr & 0xFFF;

    // 查找页目录
    pde_t *pde = &page_directory[page_dir_idx];
    if (!pde->present) {
        page_fault(virtual_addr);
        return 0;
    }

    // 查找页表
    pte_t *pte = &pde->page_table[page_table_idx];
    if (!pte->present) {
        page_fault(virtual_addr);
        return 0;
    }

    // 计算物理地址
    return (pte->frame << 12) | offset;
}
```

### 2.2 页面置换

```c
// 页面置换算法
// FIFO
typedef struct {
    uint32_t *frames;
    int head;
    int count;
    int capacity;
} fifo_queue_t;

uint32_t fifo_replace(fifo_queue_t *q, uint32_t new_page) {
    uint32_t victim = q->frames[q->head];
    q->frames[q->head] = new_page;
    q->head = (q->head + 1) % q->capacity;
    return victim;
}

// LRU(最近最少使用)
typedef struct {
    uint32_t page;
    uint32_t last_access;
    uint32_t frame;
} lru_entry_t;

uint32_t lru_replace(lru_entry_t *table, int count) {
    uint32_t oldest_time = 0xFFFFFFFF;
    int oldest_idx = 0;

    for (int i = 0; i < count; i++) {
        if (table[i].last_access < oldest_time) {
            oldest_time = table[i].last_access;
            oldest_idx = i;
        }
    }

    return oldest_idx;
}
```

```c
// VFS接口
typedef struct vfs_ops {
    int (*open)(const char *path, int flags);
    int (*close)(int fd);
    int (*read)(int fd, void *buf, int len);
    int (*write)(int fd, const void *buf, int len);
    int (*seek)(int fd, int offset, int whence);
    int (*stat)(const char *path, struct stat *st);
    int (*mkdir)(const char *path);
    int (*unlink)(const char *path);
} vfs_ops_t;

// 文件系统注册
typedef struct filesystem {
    char name[16];
    vfs_ops_t *ops;
    struct filesystem *next;
} filesystem_t;

// 挂载点
typedef struct mount_point {
    char path[64];
    filesystem_t *fs;
    void *private_data;
    struct mount_point *next;
} mount_point_t;

// VFS操作
int vfs_open(const char *path, int flags) {
    // 查找挂载点
    mount_point_t *mp = find_mount_point(path);
    if (!mp) return -ENOENT;

    // 调用文件系统特定操作
    const char *relative_path = path + strlen(mp->path);
    return mp->fs->ops->open(relative_path, flags);
}
```

## 四、中断处理

### 4.1 中断向量表

```c
// 中断向量
typedef void (*isr_t)(void);

isr_t interrupt_vector[256];

void register_interrupt(int irq, isr_t handler) {
    interrupt_vector[irq] = handler;
}

// 中断处理流程
void interrupt_handler(uint32_t irq) {
    // 1. 保存上下文
    save_context();

    // 2. 调用中断服务程序
    if (interrupt_vector[irq]) {
        interrupt_vector[irq]();
    }

    // 3. 中断EOI
    send_eoi(irq);

    // 4. 恢复上下文
    restore_context();
}

// 中断嵌套
volatile int interrupt_nesting = 0;

void interrupt_handler_nested(uint32_t irq) {
    interrupt_nesting++;

    save_context();

    if (interrupt_vector[irq]) {
        interrupt_vector[irq]();
    }

    send_eoi(irq);

    interrupt_nesting--;

    if (interrupt_nesting == 0) {
        // 检查是否有更高优先级任务就绪
        if (need_reschedule) {
            schedule();
        }
    }

    restore_context();
}
```

```c
// 字符设备接口
typedef struct cdev {
    char name[32];
    int major;
    int minor;
    struct cdev_ops *ops;
    void *private_data;

typedef struct cdev_ops {
    int (*open)(cdev_t *dev);
    int (*close)(cdev_t *dev);

int cdev_register(cdev_t *dev) {
    // 添加到设备表

// 设备操作
int cdev_read(int fd, char *buf, int len) {
    cdev_t *dev = get_device(fd);
    if (!dev || !dev->ops->read) return -EINVAL;
    return dev->ops->read(dev, buf, len);
```

```c
typedef struct bdev {

    uint32_t total_blocks;
    struct bdev_ops *ops;
    void *private_data;

typedef struct buffer_head {
    uint32_t block_no;
    uint8_t *data;
    bool dirty;
    bool valid;
    struct buffer_head *next;
} buffer_head_t;

// 缓存读取
buffer_head_t *bread(bdev_t *dev, uint32_t block) {
    // 查找缓存
    buffer_head_t *bh = find_in_cache(dev, block);
    if (bh && bh->valid) return bh;

    // 分配缓冲区
    bh = alloc_buffer_head();
    bh->data = alloc_page();
    bh->block_no = block;

    // 读取
    dev->ops->read_block(dev, block, bh->data);
    bh->valid = true;
    bh->dirty = false;

    // 添加到缓存
    add_to_cache(dev, bh);

    return bh;
}
```

### 6.1 自旋锁

```c
// 自旋锁
typedef struct {
    volatile uint32_t lock;
} spinlock_t;

void spin_lock(spinlock_t *sl) {
    while (__sync_lock_test_and_set(&sl->lock, 1)) {
        // 自旋等待
        __asm__ volatile("pause");
    }
}

void spin_unlock(spinlock_t *sl) {
    __sync_lock_release(&sl->lock);
}
```

### 6.2 读写锁

```c
// 读写锁
typedef struct {
    volatile int readers;
    volatile int writers;
    volatile int write_wait;
    mutex_t mutex;
} rwlock_t;

void read_lock(rwlock_t *rw) {
    mutex_lock(&rw->mutex);
    while (rw->writers > 0 || rw->write_wait > 0) {
        cond_wait(&rw->cond, &rw->mutex);
    }
    rw->readers++;
    mutex_unlock(&rw->mutex);
}

void write_lock(rwlock_t *rw) {
    mutex_lock(&rw->mutex);
    rw->write_wait++;
    while (rw->readers > 0 || rw->writers > 0) {
        cond_wait(&rw->cond, &rw->mutex);
    }
    rw->write_wait--;
    rw->writers++;
    mutex_unlock(&rw->mutex);
}
```
