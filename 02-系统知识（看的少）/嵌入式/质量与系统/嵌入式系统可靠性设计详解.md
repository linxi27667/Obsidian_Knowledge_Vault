## 一、可靠性分析

### 1.1 失效率计算

```c
// 失效率模型
typedef struct {
    float base_failure_rate;  // 基本失效率
    float temperature_factor; // 温度系数
    float voltage_factor;     // 电压系数
    float quality_factor;     // 质量系数
    float environment_factor; // 环境系数
} failure_rate_t;

// 计算失效率
float calculate_failure_rate(failure_rate_t *params) {
    return params->base_failure_rate *
           params->temperature_factor *
           params->voltage_factor *
           params->quality_factor *
           params->environment_factor;
}

// MTBF计算
float calculate_mtbf(float failure_rate) {
    if (failure_rate <= 0) return INFINITY;
    return 1.0f / failure_rate;
}

// 系统MTBF（串联）
float system_mtbf_series(float *component_mtbf, int count) {
    float total_failure_rate = 0;
    for (int i = 0; i < count; i++) {
        if (component_mtbf[i] > 0) {
            total_failure_rate += 1.0f / component_mtbf[i];
        }
    }
    return 1.0f / total_failure_rate;
}

// 系统MTBF（并联冗余）
float system_mtbf_parallel(float mtbf1, float mtbf2) {
    float lambda1 = 1.0f / mtbf1;
    float lambda2 = 1.0f / mtbf2;
    return (mtbf1 + mtbf2) / (mtbf1 * mtbf2 * (lambda1 + lambda2));
}
```

### 1.2 FMEA分析

```c
// FMEA记录
typedef struct {
    const char *component;
    const char *failure_mode;
    const char *effect;
    int severity;      // 严重度 (1-10)
    int occurrence;    // 发生度 (1-10)
    int detection;     // 检测度 (1-10)
    int rpn;           // 风险优先数
    const char *action;
} fmea_record_t;

// 计算RPN
int calculate_rpn(int severity, int occurrence, int detection) {
    return severity * occurrence * detection;
}

// FMEA分析
#define MAX_FMEA_RECORDS 50
static fmea_record_t fmea_records[MAX_FMEA_RECORDS];
static int fmea_count = 0;

void fmea_add_record(const char *component, const char *mode,
                     const char *effect, int sev, int occ, int det,
                     const char *action) {
    if (fmea_count >= MAX_FMEA_RECORDS) return;

    fmea_record_t *r = &fmea_records[fmea_count];
    r->component = component;
    r->failure_mode = mode;
    r->effect = effect;
    r->severity = sev;
    r->occurrence = occ;
    r->detection = det;
    r->rpn = calculate_rpn(sev, occ, det);
    r->action = action;
    fmea_count++;
}

// FMEA报告
void fmea_print_report(void) {
    printf("\n=== FMEA Analysis Report ===\n");
    printf("%-15s %-20s %-20s %5s %5s %5s %5s\n",
           "Component", "Failure Mode", "Effect", "S", "O", "D", "RPN");
    printf("------------------------------------------------------------------------\n");

    for (int i = 0; i < fmea_count; i++) {
        printf("%-15s %-20s %-20s %5d %5d %5d %5d\n",
               fmea_records[i].component,
               fmea_records[i].failure_mode,
               fmea_records[i].effect,
               fmea_records[i].severity,
               fmea_records[i].occurrence,
               fmea_records[i].detection,
               fmea_records[i].rpn);
    }

    // 找出高RPN项
    printf("\nHigh Risk Items (RPN > 100):\n");
    for (int i = 0; i < fmea_count; i++) {
        if (fmea_records[i].rpn > 100) {
            printf("  %s: %s (RPN=%d) - %s\n",
                   fmea_records[i].component,
                   fmea_records[i].failure_mode,
                   fmea_records[i].rpn,
                   fmea_records[i].action);
        }
    }
}
```

## 二、冗余设计

### 2.1 硬件冗余

```c
// 双机热备
typedef struct {
    bool is_primary;
    bool is_active;
    uint32_t heartbeat_count;
    uint32_t last_heartbeat;
} redundant_system_t;

// 主系统
static redundant_system_t primary = {
    .is_primary = true,
    .is_active = true,
};

// 备用系统
static redundant_system_t backup = {
    .is_primary = false,
    .is_active = false,
};

// 心跳检测
void heartbeat_check(void) {
    uint32_t now = get_tick_count();

    // 主系统发送心跳
    if (primary.is_active) {
        primary.heartbeat_count++;
        send_heartbeat_to_backup(primary.heartbeat_count);
    }

    // 备用系统检测心跳
    if (!backup.is_active) {
        if (now - primary.last_heartbeat > HEARTBEAT_TIMEOUT) {
            // 主系统超时，切换到备用
            switch_to_backup();
        }
    }
}

// 切换到备用系统
void switch_to_backup(void) {
    primary.is_active = false;
    backup.is_active = true;

    // 通知外部
    notify_failover();

    // 记录日志
    log_event("System failover to backup");
}

// 三模冗余表决
typedef struct {
    bool module_a;
    bool module_b;
    bool module_c;
} tmr_system_t;

// 表决器
bool tmr_vote(tmr_system_t *tmr) {
    int true_count = 0;
    if (tmr->module_a) true_count++;
    if (tmr->module_b) true_count++;
    if (tmr->module_c) true_count++;

    return true_count >= 2;  // 多数表决
}
```

### 2.2 软件冗余

```c
// 看门狗
typedef struct {
    uint32_t timeout_ms;
    uint32_t last_feed;
    bool enabled;
} watchdog_t;

void watchdog_init(watchdog_t *wdg, uint32_t timeout) {
    wdg->timeout_ms = timeout;
    wdg->last_feed = get_tick_count();
    wdg->enabled = true;

    // 配置硬件看门狗
    IWDG->KR = 0x5555;  // 允许写入
    IWDG->PR = 3;       // 分频
    IWDG->RLR = timeout * 40 / 256;  // 重载值
    IWDG->KR = 0xCCCC;  // 启动
}

void watchdog_feed(watchdog_t *wdg) {
    if (!wdg->enabled) return;

    wdg->last_feed = get_tick_count();
    IWDG->KR = 0xAAAA;  // 喂狗
}

void watchdog_check(watchdog_t *wdg) {
    if (!wdg->enabled) return;

    uint32_t now = get_tick_count();
    if (now - wdg->last_feed > wdg->timeout_ms) {
        // 超时，系统复位
        NVIC_SystemReset();
    }
}

// 数据校验
typedef struct {
    uint32_t magic;
    uint32_t version;
    uint8_t data[256];
    uint32_t crc;
} reliable_data_t;

bool reliable_data_write(const reliable_data_t *data) {
    // 写入主区域
    flash_write(PRIMARY_DATA_ADDR, data, sizeof(reliable_data_t));

    // 写入备份区域
    flash_write(BACKUP_DATA_ADDR, data, sizeof(reliable_data_t));

    return true;
}

bool reliable_data_read(reliable_data_t *data) {
    // 读取主区域
    flash_read(PRIMARY_DATA_ADDR, data, sizeof(reliable_data_t));

    // 验证CRC
    uint32_t computed_crc = crc32(data, sizeof(reliable_data_t) - 4);
    if (computed_crc == data->crc) {
        return true;
    }

    // 主区域损坏，尝试备份
    flash_read(BACKUP_DATA_ADDR, data, sizeof(reliable_data_t));
    computed_crc = crc32(data, sizeof(reliable_data_t) - 4);
    if (computed_crc == data->crc) {
        // 恢复主区域
        flash_write(PRIMARY_DATA_ADDR, data, sizeof(reliable_data_t));
        return true;
    }

    return false;  // 两个区域都损坏
}
```

## 三、降额设计

### 3.1 电压降额

```c
// 电压降额
typedef struct {
    float rated_voltage;      // 额定电压
    float operating_voltage;  // 工作电压
    float derating_factor;    // 降额系数
} voltage_derating_t;

float calculate_voltage_derating(voltage_derating_t *v) {
    return v->operating_voltage / v->rated_voltage;
}

// 电容降额
typedef struct {
    float rated_voltage;
    float operating_voltage;
    float temperature;
    float rated_temperature;
} capacitor_derating_t;

float capacitor_lifetime_estimation(capacitor_derating_t *cap) {
    // 电压降额系数
    float voltage_factor = powf(2, (cap->rated_voltage - cap->operating_voltage) / 10);

    // 温度降额系数
    float temp_factor = powf(2, (cap->rated_temperature - cap->temperature) / 10);

    // 额定寿命（小时）
    float rated_life = 5000;  // 假设5000小时

    return rated_life * voltage_factor * temp_factor;
}
```

### 3.2 电流降额

```c
// 电流降额
typedef struct {
    float rated_current;
    float operating_current;
    float temperature;
} current_derating_t;

bool check_current_derating(current_derating_t *dev) {
    // 温度降额
    float temp_derating = 1.0f;
    if (dev->temperature > 25) {
        temp_derating = 1.0f - (dev->temperature - 25) * 0.01f;
    }

    float max_allowed = dev->rated_current * temp_derating * 0.8f;  // 80%降额

    return dev->operating_current <= max_allowed;
}

// 功率降额
typedef struct {
    float rated_power;
    float operating_power;
    float ambient_temperature;
    float max_junction_temperature;
    float thermal_resistance;
} power_derating_t;

bool check_power_derating(power_derating_t *dev) {
    // 计算结温
    float junction_temp = dev->ambient_temperature +
                          dev->operating_power * dev->thermal_resistance;

    if (junction_temp > dev->max_junction_temperature) {
        return false;
    }

    // 功率降额
    float temp_margin = dev->max_junction_temperature - junction_temp;
    float derated_power = dev->rated_power * (temp_margin / (dev->max_junction_temperature - 25));

    return dev->operating_power <= derated_power;
}
```

## 四、故障检测

### 4.1 自检系统

```c
// 自检项目
typedef enum {
    BIST_RAM,
    BIST_FLASH,
    BIST_CPU,
    BIST_PERIPHERAL,
    BIST_COMMUNICATION,
    BIST_MAX,
} bist_item_t;

typedef struct {
    bist_item_t item;
    const char *name;
    bool (*test_func)(void);
    bool passed;
} bist_test_t;

// RAM测试
bool bist_ram_test(void) {
    volatile uint32_t *ram_start = (uint32_t *)0x20000000;
    uint32_t ram_size = 128 * 1024;  // 128KB

    // 写入测试模式
    for (uint32_t i = 0; i < ram_size / 4; i++) {
        ram_start[i] = i;
    }

    // 验证
    for (uint32_t i = 0; i < ram_size / 4; i++) {
        if (ram_start[i] != i) {
            return false;
        }
    }

    return true;
}

// Flash测试
bool bist_flash_test(void) {
    // 计算固件CRC
    uint32_t computed_crc = crc32((void *)FIRMWARE_ADDR, FIRMWARE_SIZE);

    // 读取存储的CRC
    uint32_t stored_crc;
    flash_read(CRC_ADDR, &stored_crc, sizeof(stored_crc));

    return computed_crc == stored_crc;
}

// CPU测试
bool bist_cpu_test(void) {
    // 算术测试
    volatile int a = 12345;
    volatile int b = 67890;
    volatile int c = a + b;
    if (c != 80235) return false;

    // 逻辑测试
    volatile uint32_t x = 0xAA55AA55;
    volatile uint32_t y = ~x;
    if ((x | y) != 0xFFFFFFFF) return false;

    // 移位测试
    volatile uint32_t z = 1;
    for (int i = 0; i < 32; i++) {
        z <<= 1;
    }
    if (z != 0) return false;

    return true;
}

// 运行自检
bool bist_run_all(void) {
    bist_test_t tests[] = {
        {BIST_RAM, "RAM", bist_ram_test, false},
        {BIST_FLASH, "Flash", bist_flash_test, false},
        {BIST_CPU, "CPU", bist_cpu_test, false},
    };

    bool all_passed = true;
    int test_count = sizeof(tests) / sizeof(tests[0]);

    for (int i = 0; i < test_count; i++) {
        tests[i].passed = tests[i].test_func();
        printf("BIST %s: %s\n", tests[i].name, tests[i].passed ? "PASS" : "FAIL");

        if (!tests[i].passed) {
            all_passed = false;
        }
    }

    return all_passed;
}
```

### 4.2 运行时监控

```c
// 运行时监控
typedef struct {
    uint32_t loop_count;
    uint32_t last_loop_time;
    uint32_t max_loop_time;
    uint32_t watchdog_count;
    bool is_alive;
} runtime_monitor_t;

static runtime_monitor_t monitor = {0};

void runtime_monitor_update(void) {
    uint32_t now = get_tick_count();

    // 计算循环时间
    uint32_t loop_time = now - monitor.last_loop_time;
    if (loop_time > monitor.max_loop_time) {
        monitor.max_loop_time = loop_time;
    }

    monitor.last_loop_time = now;
    monitor.loop_count++;
    monitor.is_alive = true;

    // 检测异常
    if (loop_time > MAX_LOOP_TIME) {
        log_event("Loop time exceeded");
    }
}

// 堆栈监控
void stack_monitor(void) {
    uint32_t stack_usage = get_stack_usage();
    uint32_t stack_size = get_stack_size();

    if (stack_usage > stack_size * 0.8) {
        log_event("Stack usage critical");
    }
}

// 内存监控
void memory_monitor(void) {
    size_t free_heap = xPortGetFreeHeapSize();
    size_t min_free_heap = xPortGetMinimumEverFreeHeapSize();

    if (free_heap < MIN_FREE_HEAP) {
        log_event("Low memory warning");
    }

    if (min_free_heap < MIN_FREE_HEAP) {
        log_event("Memory leak detected");
    }
}
```

## 五、环境适应性

### 5.1 温度监控

```c
// 温度监控
typedef struct {
    float current_temp;
    float min_temp;
    float max_temp;
    float warning_threshold;
    float critical_threshold;
    int over_temp_count;
} temp_monitor_t;

void temp_monitor_update(temp_monitor_t *mon, float temp) {
    mon->current_temp = temp;

    if (temp < mon->min_temp) mon->min_temp = temp;
    if (temp > mon->max_temp) mon->max_temp = temp;

    if (temp > mon->critical_threshold) {
        mon->over_temp_count++;
        if (mon->over_temp_count > 3) {
            // 过温保护
            enter_safe_mode();
        }
    } else if (temp > mon->warning_threshold) {
        log_event("Temperature warning");
    } else {
        mon->over_temp_count = 0;
    }
}

// 温度补偿
float temperature_compensate(float value, float temp, float ref_temp, float coeff) {
    return value * (1 + coeff * (temp - ref_temp));
}
```

```c
// 软件滤波（抗干扰）
typedef struct {
    float buffer[16];
    int index;
    int count;
} median_filter_t;

float median_filter_update(median_filter_t *f, float value) {
    f->buffer[f->index] = value;
    f->index = (f->index + 1) % 16;
    if (f->count < 16) f->count++;

    // 排序
    float sorted[16];
    memcpy(sorted, f->buffer, f->count * sizeof(float));
    for (int i = 0; i < f->count - 1; i++) {
        for (int j = i + 1; j < f->count; j++) {
            if (sorted[i] > sorted[j]) {
                float temp = sorted[i];
                sorted[i] = sorted[j];
                sorted[j] = temp;
            }
        }
    }

    return sorted[f->count / 2];
}
```
