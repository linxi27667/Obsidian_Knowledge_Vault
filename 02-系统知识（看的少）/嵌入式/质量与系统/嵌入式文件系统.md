# 嵌入式文件系统

## 核心概念

- **SPIFFS** - SPI Flash文件系统
- **LittleFS** - 小型日志文件系统
- **FAT** - 文件分配表
- **磨损均衡** - 延长Flash寿命

---

## 一、Flash存储特性

### 1.1 Flash特性

```c
// Flash特性
/*
 * NOR Flash:
 * - 擦除单位: 扇区(Sector, 4KB)
 * - 写入单位: 页(Page, 256B)
 * - 擦除后全为0xFF
 * - 不能直接覆写(需先擦除)
 * - 擦除次数: 10万-100万次
 *
 * NAND Flash:
 * - 擦除单位: 块(Block, 128KB+)
 * - 写入单位: 页(Page, 2KB+)
 * - 有坏块管理
 * - 需要ECC纠错
 */

// Flash操作接口
typedef struct {
    int (*read)(uint32_t addr, uint8_t *buf, int len);
    int (*write)(uint32_t addr, const uint8_t *buf, int len);
    int (*erase_sector)(uint32_t addr);
    uint32_t sector_size;
    uint32_t page_size;
    uint32_t total_size;
} flash_device_t;
```

## 二、SPIFFS

### 2.1 SPIFFS原理

```c
// SPIFFS结构
/*
 * 逻辑页 → 物理页映射
 * 每个页有: obj_id + span_index + data
 *
 * 目录项: 存储在普通数据页中
 * 删除: 标记页为删除(不立即擦除)
 * GC: 合并有效页，擦除旧扇区
 */

// SPIFFS配置
typedef struct {
    uint32_t phys_addr;       // 起始地址
    uint32_t phys_size;       // 总大小
    uint32_t phys_erase_block;// 擦除块大小
    uint32_t log_block_size;  // 逻辑块大小
    uint32_t log_page_size;   // 逻辑页大小
} spiffs_config_t;

// SPIFFS初始化
void spiffs_init_example(void) {
    spiffs_config_t config = {
        .phys_addr = 0x110000,
        .phys_size = 0xF0000,      // 960KB
        .phys_erase_block = 4096,
        .log_block_size = 4096,
        .log_page_size = 256,
    };

    spiffs_fs fs;
    spiffs_mount(&fs, &config);
}

// 文件操作
void spiffs_example(void) {
    // 写入
    spiffs_file f = spiffs_open(&fs, "config.json", SPIFFS_WRONLY | SPIFFS_CREAT);
    const char *data = "{\"ssid\":\"WiFi\",\"password\":\"123456\"}";
    spiffs_write(&fs, f, data, strlen(data));
    spiffs_close(&fs, f);

    // 读取
    f = spiffs_open(&fs, "config.json", SPIFFS_RDONLY);
    char buf[128];
    int len = spiffs_read(&fs, f, buf, sizeof(buf));
    spiffs_close(&fs, f);
    buf[len] = '\0';
}
```

## 三、LittleFS

### 3.1 LittleFS优势

```c
// LittleFS特点
/*
 * - 掉电安全: 日志结构，原子操作
 * - 磨损均衡: 动态分配
 * - 内存占用小: ~4KB RAM
 * - 支持目录
 * - CRC校验
 */

// LittleFS配置
#include "lfs.h"

lfs_t lfs;
lfs_config_t lfs_cfg;

void littlefs_init(void) {
    lfs_cfg.read = flash_read;
    lfs_cfg.prog = flash_write;
    lfs_cfg.erase = flash_erase;
    lfs_cfg.sync = flash_sync;

    lfs_cfg.read_size = 256;
    lfs_cfg.prog_size = 256;
    lfs_cfg.block_size = 4096;
    lfs_cfg.block_count = 240;     // 960KB / 4KB
    lfs_cfg.cache_size = 256;
    lfs_cfg.lookahead_size = 16;
    lfs_cfg.block_cycles = 500;    // 磨损均衡阈值

    int err = lfs_mount(&lfs, &lfs_cfg);
    if (err) {
        lfs_format(&lfs, &lfs_cfg);
        lfs_mount(&lfs, &lfs_cfg);
    }
}

// 文件操作
void littlefs_example(void) {
    // 写入
    lfs_file_t file;
    lfs_file_open(&lfs, &file, "data.bin", LFS_O_WRONLY | LFS_O_CREAT);
    uint8_t data[] = {0x01, 0x02, 0x03, 0x04};
    lfs_file_write(&lfs, &file, data, sizeof(data));
    lfs_file_close(&lfs, &file);

    // 读取
    lfs_file_open(&lfs, &file, "data.bin", LFS_O_RDONLY);
    uint8_t buf[32];
    lfs_ssize_t len = lfs_file_read(&lfs, &file, buf, sizeof(buf));
    lfs_file_close(&lfs, &file);

    // 目录操作
    lfs_mkdir(&lfs, "logs");

    // 遍历目录
    lfs_dir_t dir;
    lfs_dir_open(&lfs, &dir, "/");
    struct lfs_info info;
    while (lfs_dir_read(&lfs, &dir, &info) > 0) {
        printf("%s (%s, %u bytes)\n",
               info.name,
               info.type == LFS_TYPE_DIR ? "dir" : "file",
               info.size);
    }
    lfs_dir_close(&lfs, &dir);
}
```

### 3.2 磨损均衡

```c
// LittleFS磨损均衡原理
/*
 * 块分配策略:
 * 1. 优先选择擦除次数少的块
 * 2. block_cycles控制重新分配频率
 * 3. 每block_cycles次擦除，强制移动数据
 *
 * 例: block_cycles=500
 * - 每个块擦除500次后，数据会被移动到新块
 * - 确保所有块的擦除次数趋于一致
 */

// 磨损信息查看
void check_wear(void) {
    // LittleFS不直接暴露磨损信息
    // 但可以通过观察块分配来推断

    lfs_ssize_t used = lfs_fs_size(&lfs);
    printf("Used: %d blocks\n", used);
    printf("Total: %d blocks\n", lfs_cfg.block_count);
    printf("Free: %d blocks\n", lfs_cfg.block_count - used);
}
```

## 四、FAT文件系统

### 4.1 FAT实现

```c
// FatFs配置
#include "ff.h"

FATFS fs;
FIL file;

void fat_init(void) {
    // 挂载
    FRESULT res = f_mount(&fs, "0:", 1);
    if (res == FR_NO_FILESYSTEM) {
        // 格式化
        BYTE work[4096];
        f_mkfs("0:", FM_FAT32, 0, work, sizeof(work));
        f_mount(&fs, "0:", 1);
    }
}

void fat_example(void) {
    // 写入
    f_open(&file, "log.txt", FA_WRITE | FA_CREATE_ALWAYS);
    UINT bw;
    f_write(&file, "Hello World\n", 12, &bw);
    f_close(&file);

    // 追加
    f_open(&file, "log.txt", FA_WRITE | FA_OPEN_APPEND);
    f_write(&file, "New line\n", 9, &bw);
    f_close(&file);

    // 读取
    f_open(&file, "log.txt", FA_READ);
    char buf[256];
    UINT br;
    f_read(&file, buf, sizeof(buf) - 1, &br);
    buf[br] = '\0';
    f_close(&file);

    printf("File content: %s\n", buf);
}
```

## 六、数据库KV存储

### 6.1 简单KV实现

```c
// 基于Flash的KV存储
typedef struct {
    uint32_t magic;      // 0x4B564442 "KVDB"
    uint32_t version;
    uint32_t write_count;
    char key[32];
    uint8_t value[128];
    uint16_t value_len;
    uint16_t crc;
} kv_entry_t;

// KV写入(追加写入)
int kv_set(const char *key, const void *value, int len) {
    kv_entry_t entry = {
        .magic = 0x4B564442,
        .version = 1,
        .write_count = write_counter++,
        .value_len = len,
    };
    strcpy(entry.key, key);
    memcpy(entry.value, value, len);
    entry.crc = crc16((uint8_t *)&entry, sizeof(kv_entry_t) - 2);

    // 找到空闲位置写入
    uint32_t addr = find_free_slot();
    flash_write(addr, (uint8_t *)&entry, sizeof(kv_entry_t));

    return 0;
}

// KV读取(读取最新值)
int kv_get(const char *key, void *value, int max_len) {
    kv_entry_t newest = {0};

    // 遍历所有条目
    for (uint32_t addr = KV_START; addr < KV_END; addr += sizeof(kv_entry_t)) {
        kv_entry_t entry;
        flash_read(addr, (uint8_t *)&entry, sizeof(kv_entry_t));

        if (entry.magic != 0x4B564442) continue;
        if (strcmp(entry.key, key) != 0) continue;
        if (entry.write_count > newest.write_count) {
            newest = entry;
        }
    }

    if (newest.magic == 0) return -1;  // 未找到

    int copy_len = (newest.value_len < max_len) ? newest.value_len : max_len;
    memcpy(value, newest.value, copy_len);
    return copy_len;
}
```
