## 一、功耗分析

### 1.1 功耗模型

```c
// 功耗组件
typedef struct {
    float mcu_active_mw;      // MCU活跃功耗
    float mcu_sleep_mw;       // MCU睡眠功耗
    float sensor_active_mw;   // 传感器活跃功耗
    float sensor_idle_mw;     // 传感器空闲功耗
    float radio_tx_mw;        // 发射功耗
    float radio_rx_mw;        // 接收功耗
    float radio_sleep_mw;     // 无线电睡眠功耗
    float peripheral_mw;      // 外设功耗
} power_budget_t;

// 功耗计算
typedef struct {
    float total_energy_mwh;   // 总能量
    float average_power_mw;   // 平均功率
    float peak_power_mw;      // 峰值功率
    float battery_life_hours; // 电池寿命
} power_analysis_t;

power_analysis_t analyze_power(power_budget_t *budget,
                                float active_ratio,
                                float tx_ratio,
                                float rx_ratio,
                                float battery_mah,
                                float voltage) {
    power_analysis_t analysis;

    // 平均功率计算
    analysis.average_power_mw =
        budget->mcu_active_mw * active_ratio +
        budget->mcu_sleep_mw * (1 - active_ratio) +
        budget->sensor_active_mw * active_ratio +
        budget->sensor_idle_mw * (1 - active_ratio) +
        budget->radio_tx_mw * tx_ratio +
        budget->radio_rx_mw * rx_ratio +
        budget->radio_sleep_mw * (1 - tx_ratio - rx_ratio) +
        budget->peripheral_mw * active_ratio;

    // 峰值功率
    analysis.peak_power_mw = budget->mcu_active_mw +
                             budget->sensor_active_mw +
                             budget->radio_tx_mw +
                             budget->peripheral_mw;

    // 电池寿命
    float battery_wh = battery_mah * voltage / 1000;
    analysis.battery_life_hours = battery_wh / (analysis.average_power_mw / 1000);

    return analysis;
}
```

```c
// 电流采样
typedef struct {

    float gain;               // 放大倍数
    uint16_t adc_resolution;
    float vref;
} current_sensor_t;

typedef struct {
    float samples[1000];
    int sample_count;
    float min_current;
    float max_current;
    float avg_current;
    float total_charge_uah;
} power_measurement_t;

void power_measure_start(power_measurement_t *meas) {
    meas->sample_count = 0;
    meas->min_current = FLT_MAX;
    meas->max_current = FLT_MIN;
    meas->avg_current = 0;
    meas->total_charge_uah = 0;
}

void power_measure_sample(power_measurement_t *meas, float current_ma) {
    if (meas->sample_count < 1000) {
        meas->samples[meas->sample_count++] = current_ma;
    }

    if (current_ma < meas->min_current) meas->min_current = current_ma;
    if (current_ma > meas->max_current) meas->max_current = current_ma;
}

void power_measure_finish(power_measurement_t *meas, float duration_sec) {
    float sum = 0;
    for (int i = 0; i < meas->sample_count; i++) {
        sum += meas->samples[i];
    }
    meas->avg_current = sum / meas->sample_count;
    meas->total_charge_uah = meas->avg_current * duration_sec / 3600;
}
```

```c
// STM32低功耗模式
typedef enum {
    LP_MODE_RUN,          // 正常运行
    LP_MODE_LOW_POWER_RUN,// 低功耗运行
    LP_MODE_SLEEP,        // 睡眠
    LP_MODE_LOW_POWER_SLEEP, // 低功耗睡眠
    LP_MODE_STOP0,        // 停止0
    LP_MODE_STOP1,        // 停止1
    LP_MODE_STOP2,        // 停止2
    LP_MODE_STANDBY,      // 待机
    LP_MODE_SHUTDOWN,     // 关机
} lp_mode_t;

// 低功耗配置
typedef struct {
    lp_mode_t mode;
    uint32_t wakeup_pin;
    uint32_t wakeup_timer_ms;
    bool retain_sram;
    bool retain_io;
} lp_config_t;

// 进入低功耗
void enter_low_power(lp_config_t *config) {
    switch (config->mode) {
        case LP_MODE_SLEEP:
            __WFI();
            break;

        case LP_MODE_STOP0:
            HAL_SuspendTick();
            HAL_PWR_EnterSTOP0Mode(PWR_LOWPOWERREGULATOR_ON, PWR_STOPENTRY_WFI);
            break;

        case LP_MODE_STOP1:
            HAL_SuspendTick();
            HAL_PWR_EnterSTOP1Mode(PWR_LOWPOWERREGULATOR_ON, PWR_STOPENTRY_WFI);
            break;

        case LP_MODE_STOP2:
            HAL_SuspendTick();
            HAL_PWR_EnterSTOP2Mode(PWR_LOWPOWERREGULATOR_ON, PWR_STOPENTRY_WFI);
            break;

        case LP_MODE_STANDBY:
            HAL_PWR_EnterSTANDBYMode();
            break;

        case LP_MODE_SHUTDOWN:
            HAL_PWR_EnterSHUTDOWNMode();
            break;
    }
}

// 唤醒后恢复
void exit_low_power(lp_config_t *config) {
    // 恢复时钟
    SystemClock_Config();

    // 恢复Tick
    HAL_ResumeTick();

    // 恢复外设
    if (!config->retain_io) {
        MX_GPIO_Init();
    }
}
```

### 2.2 动态电压频率调整

```c
// DVFS配置
typedef struct {
    uint32_t freq_hz;
    float voltage_mv;
    const char *name;
} dvfs_level_t;

static const dvfs_level_t dvfs_levels[] = {
    {  2000000, 900,  "低功耗"},
    {  8000000, 1000, "正常"},
    { 16000000, 1100, "高性能"},
    { 32000000, 1200, "最大性能"},
};

// 当前等级
static int current_dvfs_level = 1;

// 设置DVFS等级
void dvfs_set_level(int level) {
    if (level < 0 || level >= 4) return;

    const dvfs_level_t *target = &dvfs_levels[level];

    // 先调电压
    if (target->voltage_mv > dvfs_levels[current_dvfs_level].voltage_mv) {
        set_voltage(target->voltage_mv);
        delay_ms(10);  // 等待稳定
    }

    // 再调频率
    set_system_clock(target->freq_hz);

    // 后降电压
    if (target->voltage_mv < dvfs_levels[current_dvfs_level].voltage_mv) {
        set_voltage(target->voltage_mv);
    }

    current_dvfs_level = level;
}

// 自适应DVFS
void adaptive_dvfs(float cpu_load) {
    if (cpu_load > 0.8f) {
        dvfs_set_level(3);  // 最大性能
    } else if (cpu_load > 0.5f) {
        dvfs_set_level(2);  // 高性能
    } else if (cpu_load > 0.2f) {
        dvfs_set_level(1);  // 正常
    } else {
        dvfs_set_level(0);  // 低功耗
    }
}
```

## 三、外设功耗管理

### 3.1 外设电源控制

```c
// 外设电源域
typedef struct {
    const char *name;
    gpio_num_t enable_pin;
    bool active_high;
    float current_ma;
    uint32_t startup_time_ms;
    int ref_count;
} peripheral_power_t;

#define MAX_PERIPHERALS 16
static peripheral_power_t peripherals[MAX_PERIPHERALS];
static int peripheral_count = 0;

// 注册外设电源
int peripheral_power_register(const char *name, gpio_num_t pin,
                               bool active_high, float current_ma,
                               uint32_t startup_time) {
    if (peripheral_count >= MAX_PERIPHERALS) return -1;

    peripheral_power_t *p = &peripherals[peripheral_count];
    p->name = name;
    p->enable_pin = pin;
    p->active_high = active_high;
    p->current_ma = current_ma;
    p->startup_time_ms = startup_time;
    p->ref_count = 0;

    // 默认关闭
    gpio_set_direction(pin, GPIO_MODE_OUTPUT);
    gpio_set_level(pin, active_high ? 0 : 1);

    peripheral_count++;
    return 0;
}

// 使能外设
int peripheral_enable(const char *name) {
    for (int i = 0; i < peripheral_count; i++) {
        if (strcmp(peripherals[i].name, name) == 0) {
            if (peripherals[i].ref_count == 0) {
                gpio_set_level(peripherals[i].enable_pin,
                              peripherals[i].active_high ? 1 : 0);
                delay_ms(peripherals[i].startup_time_ms);
            }
            peripherals[i].ref_count++;
            return 0;
        }
    }
    return -1;
}

// 禁用外设
int peripheral_disable(const char *name) {
    for (int i = 0; i < peripheral_count; i++) {
        if (strcmp(peripherals[i].name, name) == 0) {
            if (peripherals[i].ref_count > 0) {
                peripherals[i].ref_count--;
                if (peripherals[i].ref_count == 0) {
                    gpio_set_level(peripherals[i].enable_pin,
                                  peripherals[i].active_high ? 0 : 1);
                }
            }
            return 0;
        }
    }
    return -1;
}

// 计算总外设功耗
float peripheral_total_current(void) {
    float total = 0;
    for (int i = 0; i < peripheral_count; i++) {
        if (peripherals[i].ref_count > 0) {
            total += peripherals[i].current_ma;
        }
    }
    return total;
}
```

### 3.2 GPIO功耗优化

```c
// GPIO低功耗配置
void gpio_low_power_init(void) {
    // 未使用的GPIO配置为输出低电平
    for (int i = 0; i < 16; i++) {
        if (!is_gpio_used(i)) {
            HAL_GPIO_Init(GPIOA, &(GPIO_InitTypeDef){
                .Pin = (1 << i),
                .Mode = GPIO_MODE_OUTPUT_PP,
                .Pull = GPIO_NOPULL,
                .Speed = GPIO_SPEED_FREQ_LOW,
            });
            HAL_GPIO_WritePin(GPIOA, (1 << i), GPIO_PIN_RESET);
        }
    }

    // 模拟引脚配置为高阻
    HAL_GPIO_Init(GPIOA, &(GPIO_InitTypeDef){
        .Pin = GPIO_PIN_0 | GPIO_PIN_1,
        .Mode = GPIO_MODE_ANALOG,
        .Pull = GPIO_NOPULL,
    });
}

// 按键唤醒配置
void configure_wakeup_pin(gpio_num_t pin, bool high_level) {
    GPIO_InitTypeDef init = {
        .Pin = (1 << pin),
        .Mode = GPIO_MODE_IT_RISING,
        .Pull = GPIO_PULLDOWN,
        .Speed = GPIO_SPEED_FREQ_LOW,
    };

    if (!high_level) {
        init.Mode = GPIO_MODE_IT_FALLING;
        init.Pull = GPIO_PULLUP;
    }

    HAL_GPIO_Init(GPIOA, &init);
    HAL_NVIC_SetPriority(EXTI0_IRQn, 0, 0);
    HAL_NVIC_EnableIRQ(EXTI0_IRQn);
}
```

## 四、传感器低功耗

### 4.1 传感器电源管理

```c
// 传感器电源控制
typedef struct {
    const char *name;
    gpio_num_t power_pin;
    float active_current_ma;
    float sleep_current_ua;
    uint32_t wakeup_time_ms;
    uint32_t sample_time_ms;
    bool is_powered;
} sensor_power_t;

// 带占空比的传感器读取
float sensor_read_with_duty_cycle(sensor_power_t *sensor) {
    // 开启电源
    gpio_set_level(sensor->power_pin, 1);
    sensor->is_powered = true;

    // 等待稳定
    delay_ms(sensor->wakeup_time_ms);

    // 读取数据
    float value = read_sensor_data();

    // 关闭电源
    gpio_set_level(sensor->power_pin, 0);
    sensor->is_powered = false;

    return value;
}

// 传感器平均功耗计算
float sensor_average_power(sensor_power_t *sensor,
                            uint32_t sample_interval_ms) {
    float active_time_s = (sensor->wakeup_time_ms + sensor->sample_time_ms) / 1000.0f;
    float sleep_time_s = sample_interval_ms / 1000.0f - active_time_s;

    float active_energy = sensor->active_current_ma * 3.3f * active_time_s;
    float sleep_energy = sensor->sleep_current_ua * 3.3f * sleep_time_s / 1000;

    return (active_energy + sleep_energy) / (active_time_s + sleep_time_s);
}
```

### 4.2 自适应采样

```c
// 自适应采样
typedef struct {
    float last_value;
    float threshold;
    uint32_t min_interval_ms;
    uint32_t max_interval_ms;
    uint32_t current_interval_ms;
    uint32_t stable_count;
} adaptive_sampler_t;

uint32_t adaptive_sample_interval(adaptive_sampler_t *sampler, float current_value) {
    float delta = fabs(current_value - sampler->last_value);

    if (delta > sampler->threshold) {
        // 变化大，增加采样率
        sampler->current_interval_ms = sampler->min_interval_ms;
        sampler->stable_count = 0;
    } else {
        sampler->stable_count++;
        if (sampler->stable_count > 10) {
            // 稳定后降低采样率
            sampler->current_interval_ms *= 2;
            if (sampler->current_interval_ms > sampler->max_interval_ms) {
                sampler->current_interval_ms = sampler->max_interval_ms;
            }
        }
    }

    sampler->last_value = current_value;
    return sampler->current_interval_ms;
}
```

```c
typedef struct {
    bool is_connected;
    uint32_t last_activity;
    uint32_t idle_timeout_ms;
    wifi_power_save_t ps_config;
} wifi_manager_t;

void wifi_power_manage(wifi_manager_t *mgr) {
    uint32_t now = get_tick_count();

    if (mgr->is_connected) {
        if (now - mgr->last_activity > mgr->idle_timeout_ms) {
            // 空闲超时，断开WiFi
            esp_wifi_disconnect();
            esp_wifi_stop();
            mgr->is_connected = false;
        }
    }
}

void wifi_wake_up(wifi_manager_t *mgr) {
    if (!mgr->is_connected) {
        esp_wifi_start();
        esp_wifi_connect();
        mgr->is_connected = true;
    }
    mgr->last_activity = get_tick_count();
}
```

## 六、系统级优化

### 6.1 任务调度优化

```c
// 低功耗任务调度
typedef struct {
    task_t *tasks;
    int task_count;
    uint32_t idle_threshold_ms;
} lp_scheduler_t;

void lp_scheduler_run(lp_scheduler_t *sched) {
    while (1) {
        bool task_executed = false;

        // 执行就绪任务
        for (int i = 0; i < sched->task_count; i++) {
            if (is_task_ready(&sched->tasks[i])) {
                execute_task(&sched->tasks[i]);
                task_executed = true;
            }
        }

        // 没有任务就进入低功耗
        if (!task_executed) {
            uint32_t next_wakeup = get_next_wakeup_time();
            uint32_t sleep_ms = next_wakeup - get_tick_count();

            if (sleep_ms > sched->idle_threshold_ms) {
                enter_low_power(&(lp_config_t){
                    .mode = LP_MODE_STOP0,
                    .wakeup_timer_ms = sleep_ms,
                });
            } else {
                __WFI();  // 短暂等待
            }
        }
    }
}
```

### 6.2 功耗监控

```c
// 功耗统计
typedef struct {
    uint32_t active_time_ms;
    uint32_t sleep_time_ms;
    float total_charge_uah;
    float avg_current_ma;
    uint32_t wakeup_count;
} power_stats_t;

static power_stats_t stats = {0};
static uint32_t last_state_change = 0;
static bool is_active = true;

void power_stats_start(void) {
    last_state_change = get_tick_count();
    is_active = true;
}

void power_stats_enter_sleep(void) {
    uint32_t now = get_tick_count();
    stats.active_time_ms += now - last_state_change;
    last_state_change = now;
    is_active = false;
    stats.wakeup_count++;
}

void power_stats_exit_sleep(void) {
    uint32_t now = get_tick_count();
    stats.sleep_time_ms += now - last_state_change;
    last_state_change = now;
    is_active = true;
}

power_stats_t power_stats_get(void) {
    // 更新当前状态时间
    uint32_t now = get_tick_count();
    if (is_active) {
        stats.active_time_ms += now - last_state_change;
    } else {
        stats.sleep_time_ms += now - last_state_change;
    }
    last_state_change = now;

    // 计算平均电流
    uint32_t total_time = stats.active_time_ms + stats.sleep_time_ms;
    if (total_time > 0) {
        stats.avg_current_ma = stats.total_charge_uah * 3600 / total_time;
    }

    return stats;
}
```
