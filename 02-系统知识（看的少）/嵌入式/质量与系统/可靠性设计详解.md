# 可靠性设计详解

## 核心概念

- **MTBF** - 平均无故障时间
- **FMEA** - 失效模式与影响分析
- **冗余设计** - 提高系统可靠性
- **降额设计** - 降低应力水平

---

## 一、可靠性指标

### 1.1 基本指标

```c
// 可靠性指标
typedef struct {
    float mtbf;           // 平均无故障时间(小时)
    float mttr;           // 平均修复时间(小时)
    float availability;   // 可用性
    float failure_rate;   // 失效率(1/小时)
} reliability_metrics_t;

// 计算可用性
float calc_availability(float mtbf, float mttr) {
    return mtbf / (mtbf + mttr);
}

// 失效率计算(串联系统)
float series_failure_rate(float *rates, int count) {
    float total = 0;
    for (int i = 0; i < count; i++) {
        total += rates[i];
    }
    return total;
}

// 失效率计算(并联系统)
float parallel_failure_rate(float *rates, int count) {
    float product = 1;
    for (int i = 0; i < count; i++) {
        product *= rates[i];
    }
    return product;
}
```

```c
float predict_system_reliability(void) {

        float rate = components[i].base_rate *
```

## 二、FMEA分析

### 2.1 FMEA表格

```c
// FMEA条目
typedef struct {
    char item[32];           // 分析项目
    char function[64];       // 功能
    char failure_mode[64];   // 失效模式
    char effect[128];        // 失效影响
    int severity;            // 严重度(S: 1-10)
    char cause[128];         // 失效原因
    int occurrence;          // 发生度(O: 1-10)
    char detection[64];      // 检测方法
    int detection_score;     // 检测度(D: 1-10)
    int rpn;                 // 风险优先数(S×O×D)
    char action[128];        // 改进措施
} fmea_item_t;

// RPN计算
int calc_rpn(int severity, int occurrence, int detection) {
    return severity * occurrence * detection;
}

// FMEA示例
fmea_item_t fmea_example = {
    .item = "电源模块",
    .function = "提供3.3V电源",
    .failure_mode = "输出电压偏高",
    .effect = "MCU损坏，系统失效",
    .severity = 8,
    .cause = "反馈电阻开路",
    .occurrence = 2,
    .detection = "电压监测电路",
    .detection_score = 3,
    .rpn = 48,
    .action = "增加过压保护，冗余监测"
};
```

## 三、降额设计

### 3.1 降额准则

```c
// 降额等级
typedef enum {
    DERATE_LEVEL_1,  // 一级降额(最严格)
    DERATE_LEVEL_2,  // 二级降额
    DERATE_LEVEL_3,  // 三级降额(基本)
} derate_level_t;

// 降额参数
typedef struct {
    const char *component;
    float voltage_derate;   // 电压降额系数
    float current_derate;   // 电流降额系数
    float power_derate;     // 功率降额系数
    float temperature_derate;// 温度降额
} derating_t;

// 降额标准
derating_t derating_standards[] = {
    // 一级降额
    {"电容(电解)",  0.6,  0.7,  0.5,  85},
    {"电容(陶瓷)",  0.5,  0.7,  0.5,  85},
    {"电阻",        0.7,  0.7,  0.5,  85},
    {"二极管",      0.7,  0.7,  0.5,  85},
    {"MOSFET",      0.7,  0.7,  0.5,  85},
    {"IC",          0.8,  0.8,  0.7,  85},
};

// 降额检查
bool check_derating(const char *component, float actual_voltage,
                    float rated_voltage, float actual_current,
                    float rated_current) {
    // 查找降额标准
    for (int i = 0; i < sizeof(derating_standards)/sizeof(derating_standards[0]); i++) {
        if (strcmp(derating_standards[i].component, component) == 0) {
            float max_voltage = rated_voltage * derating_standards[i].voltage_derate;
            float max_current = rated_current * derating_standards[i].current_derate;

            if (actual_voltage > max_voltage || actual_current > max_current) {
                printf("Derating violation: %s\n", component);
                return false;
            }
            return true;
        }
    }
    return true;
}
```

## 四、冗余设计

### 4.1 冗余架构

```c
// 冗余类型
typedef enum {
    REDUNDANCY_NONE,        // 无冗余
    REDUNDANCY_ACTIVE,      // 主动冗余(热备份)
    REDUNDANCY_STANDBY,     // 待机冗余(冷备份)
    REDUNDANCY_VOTING,      // 表决冗余(三模冗余)
} redundancy_type_t;

// 双机热备
typedef struct {
    uint32_t primary_data;
    uint32_t backup_data;
    bool primary_healthy;
    bool backup_healthy;
    redundancy_type_t type;
} redundant_system_t;

uint32_t get_redundant_data(redundant_system_t *sys) {
    if (sys->primary_healthy) {
        return sys->primary_data;
    } else if (sys->backup_healthy) {
        // 切换到备份
        printf("Switching to backup\n");
        return sys->backup_data;
    } else {
        // 两个都故障
        printf("Both units failed\n");
        return 0;
    }
}

// 三模表决
uint32_t triple_voting(uint32_t a, uint32_t b, uint32_t c) {
    if (a == b) return a;
    if (b == c) return b;
    if (a == c) return a;
    return a;  // 无法表决，返回第一个
}
```

### 4.2 看门狗冗余

```c
// 外部看门狗
typedef struct {
    gpio_num_t wdi_pin;    // 喂狗引脚
    gpio_num_t reset_pin;  // 复位输出
    uint32_t timeout_ms;
    uint32_t last_feed;
} external_wdg_t;

void external_wdg_init(external_wdg_t *wdg) {
    gpio_set_direction(wdg->wdi_pin, GPIO_MODE_OUTPUT);
    gpio_set_direction(wdg->reset_pin, GPIO_MODE_INPUT);
}

void external_wdg_feed(external_wdg_t *wdg) {
    gpio_set_level(wdg->wdi_pin, 1);
    delay_us(1);
    gpio_set_level(wdg->wdi_pin, 0);
    wdg->last_feed = get_tick();
}

bool external_wdg_check(external_wdg_t *wdg) {
    // 检查是否超时
    if (get_tick() - wdg->last_feed > wdg->timeout_ms) {
        // 看门狗超时，外部复位
        return false;
    }
    return true;
}
```

```c
float temperature_derating(float current_temp, float max_temp) {
    float derating_factor = 1.0f - (current_temp - 25.0f) / (max_temp - 25.0f) * 0.5f;
    return (derating_factor > 0.5f) ? derating_factor : 0.5f;
}
```

## 六、寿命设计

### 6.1 电解电容寿命

```c
// 电解电容寿命计算
typedef struct {
    float rated_life_hours;   // 额定寿命
    float rated_temperature;  // 额定温度
    float actual_temperature; // 实际温度
    float ripple_current_ratio; // 纹波电流比
} capacitor_life_t;

float calc_capacitor_life(capacitor_life_t *cap) {
    // Arrhenius模型
    float dt = cap->actual_temperature - cap->rated_temperature;
    float life = cap->rated_life_hours * powf(2, -dt / 10);

    // 纹波电流修正
    life *= powf(0.5, cap->ripple_current_ratio);

    return life;
}

// 示例: 105°C/2000小时电容在65°C工作
capacitor_life_t cap = {
    .rated_life_hours = 2000,
    .rated_temperature = 105,
    .actual_temperature = 65,
    .ripple_current_ratio = 0.5,
};
float life = calc_capacitor_life(&cap);
printf("Expected life: %.0f hours (%.1f years)\n", life, life / 8760);
```
