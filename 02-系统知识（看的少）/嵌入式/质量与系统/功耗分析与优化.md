```c
// 电流测量方法
/*
 * 1. 串联电阻法:
 *    - 串联精密电阻(0.1-1Ω)
 *    - 测量电阻压降
 *    - I = V/R
 *
 * 2. 霍尔传感器:
 *    - 非接触测量
 *    - 适合大电流
 *
 * 3. 专用芯片:
 *    - INA219 (I2C接口)
 *    - INA226 (高精度)
 *    - 功率分析仪
 */

// INA219驱动
#define INA219_ADDR 0x40
#define INA219_REG_CONFIG 0x00
#define INA219_REG_SHUNT 0x01
#define INA219_REG_BUS 0x02
#define INA219_REG_POWER 0x03
#define INA219_REG_CURRENT 0x04
#define INA219_REG_CALIBRATION 0x05

void ina219_init(void) {
    // 配置: 32V范围, 320mV分流, 连续测量
    i2c_write_reg16(INA219_ADDR, INA219_REG_CONFIG, 0x399F);

    // 校准: 电流分辨率 = 0.1mA
    i2c_write_reg16(INA219_ADDR, INA219_REG_CALIBRATION, 4096);
}

float ina219_read_current_mA(void) {
    int16_t raw = (int16_t)i2c_read_reg16(INA219_ADDR, INA219_REG_CURRENT);
    return raw * 0.1f;  // mA
}

float ina219_read_power_mW(void) {
    uint16_t raw = i2c_read_reg16(INA219_ADDR, INA219_REG_POWER);
    return raw * 2.0f;  // mW
}
```

### 1.2 功耗分析

```c
// 功耗分解
typedef struct {
    const char *name;
    float current_mA;
    float duration_ms;
    float energy_mJ;
} power_component_t;

void analyze_power(power_component_t *components, int count) {
    float total_energy = 0;

    printf("Power Analysis:\n");
    printf("%-20s %10s %10s %10s\n", "Component", "mA", "ms", "mJ");
    printf("----------------------------------------------------\n");

    for (int i = 0; i < count; i++) {
        components[i].energy_mJ = components[i].current_mA *
                                   components[i].duration_ms / 1000.0f;
        total_energy += components[i].energy_mJ;

        printf("%-20s %10.2f %10.1f %10.3f\n",
               components[i].name,
               components[i].current_mA,
               components[i].duration_ms,
               components[i].energy_mJ);
    }

    printf("----------------------------------------------------\n");
    printf("%-20s %10s %10s %10.3f\n", "Total", "", "", total_energy);
}
```

```c
// 事件驱动替代轮询
typedef enum {
    EVT_TIMER,
    EVT_SENSOR,
    EVT_UART,
    EVT_BUTTON,
    EVT_COUNT
} event_type_t;

typedef struct {
    event_type_t type;
    uint32_t data;
} event_t;

// 事件队列
QueueHandle_t event_queue;

void event_driven_task(void *param) {
    event_t event;

    while (1) {
        // 阻塞等待事件(不消耗CPU)
        if (xQueueReceive(event_queue, &event, portMAX_DELAY)) {
            switch (event.type) {
                case EVT_TIMER:
                    handle_timer();
                    break;
                case EVT_SENSOR:
                    handle_sensor(event.data);
                    break;
                case EVT_UART:
                    handle_uart(event.data);
                    break;
                case EVT_BUTTON:
                    handle_button();
                    break;
            }
        }
    }
}

// 中断触发事件
void sensor_isr(void) {
    event_t event = {EVT_SENSOR, read_sensor()};
    xQueueSendFromISR(event_queue, &event, NULL);
}
```

### 2.2 间歇性工作

```c
// 间歇性工作模式
typedef struct {
    uint32_t work_duration_ms;
    uint32_t sleep_duration_ms;
    float avg_current_mA;
} duty_cycle_t;

void intermittent_work(duty_cycle_t *config) {
    while (1) {
        // 工作阶段
        uint32_t start = get_tick();
        while (get_tick() - start < config->work_duration_ms) {
            do_work();
        }

        // 睡眠阶段
        enter_deep_sleep(config->sleep_duration_ms);

        // 唤醒后继续
    }
}

// 计算平均功耗
float calc_avg_power(duty_cycle_t *config, float active_mA, float sleep_uA) {
    float duty = (float)config->work_duration_ms /
                 (config->work_duration_ms + config->sleep_duration_ms);
    return active_mA * duty + sleep_uA / 1000.0f * (1 - duty);
}
```

## 三、能量收集

### 3.1 太阳能收集

```c
// 太阳能MPPT(最大功率点跟踪)
typedef struct {
    float voltage;
    float current;
    float power;
    float max_power;
    float optimal_voltage;
} mppt_t;

void mppt_update(mppt_t *mppt, float v, float i) {
    mppt->voltage = v;
    mppt->current = i;
    mppt->power = v * i;

    // P&O算法(Perturb and Observe)
    static float prev_power = 0;
    static float prev_voltage = 0;
    static float step = 0.01f;

    if (mppt->power > prev_power) {
        if (v > prev_voltage) {
            mppt->optimal_voltage = v + step;
        } else {
            mppt->optimal_voltage = v - step;
        }
    } else {
        if (v > prev_voltage) {
            mppt->optimal_voltage = v - step;
        } else {
            mppt->optimal_voltage = v + step;
        }
    }

    if (mppt->power > mppt->max_power) {
        mppt->max_power = mppt->power;
    }

    prev_power = mppt->power;
    prev_voltage = v;
}
```

### 3.2 能量存储

```c
// 超级电容管理
typedef struct {
    float capacitance;     // F
    float voltage;         // V
    float max_voltage;     // V
    float energy;          // J
} supercap_t;

void supercap_init(supercap_t *cap, float capacitance, float max_voltage) {
    cap->capacitance = capacitance;
    cap->max_voltage = max_voltage;
    cap->voltage = 0;
    cap->energy = 0;
}

void supercap_charge(supercap_t *cap, float current, float dt) {
    cap->voltage += current * dt / cap->capacitance;
    if (cap->voltage > cap->max_voltage) {
        cap->voltage = cap->max_voltage;
    }
    cap->energy = 0.5f * cap->capacitance * cap->voltage * cap->voltage;
}

float supercap_discharge(supercap_t *cap, float power, float dt) {
    float energy_needed = power * dt;
    if (energy_needed > cap->energy) {
        return -1;  // 能量不足
    }

    cap->energy -= energy_needed;
    cap->voltage = sqrtf(2 * cap->energy / cap->capacitance);
    return cap->voltage;
}
```

## 四、电池管理

### 4.1 电量计

```c
// 库仑计数法
typedef struct {
    float capacity_mah;      // 电池容量
    float remaining_mah;     // 剩余容量
    float current_ma;        // 当前电流
    uint32_t last_update;    // 上次更新时间
    float soc;               // 电量百分比
} coulomb_counter_t;

void coulomb_update(coulomb_counter_t *cc, float current_ma) {
    uint32_t now = get_tick();
    float dt_hours = (now - cc->last_update) / 3600000.0f;

    cc->remaining_mah -= current_ma * dt_hours;
    if (cc->remaining_mah < 0) cc->remaining_mah = 0;
    if (cc->remaining_mah > cc->capacity_mah) cc->remaining_mah = cc->capacity_mah;

    cc->soc = cc->remaining_mah / cc->capacity_mah * 100;
    cc->current_ma = current_ma;
    cc->last_update = now;
}

// 电压法估算(辅助)
float estimate_soc_by_voltage(float voltage) {
    // 锂电池放电曲线
    if (voltage >= 4.2f) return 100;
    if (voltage >= 4.03f) return 90;
    if (voltage >= 3.86f) return 80;
    if (voltage >= 3.83f) return 70;
    if (voltage >= 3.79f) return 60;
    if (voltage >= 3.75f) return 50;
    if (voltage >= 3.73f) return 40;
    if (voltage >= 3.71f) return 30;
    if (voltage >= 3.69f) return 20;
    if (voltage >= 3.61f) return 10;
    return 0;
}
```

### 4.2 电池寿命预测

```c
// 电池寿命预测
typedef struct {
    float battery_mah;
    float avg_current_ma;
    float self_discharge_per_day;  // 自放电率(%/天)
} battery_life_t;

float predict_battery_life(battery_life_t *bat) {
    // 工作电流消耗
    float working_hours = bat->battery_mah / bat->avg_current_ma;

    // 自放电消耗
    float self_discharge_hours = 100.0f / bat->self_discharge_per_day * 24;

    // 取较小值
    return fminf(working_hours, self_discharge_hours);
}

// 电池寿命优化建议
void suggest_optimizations(battery_life_t *bat) {
    float current = bat->avg_current_ma;

    printf("Current consumption: %.2f mA\n", current);
    printf("Estimated life: %.1f hours\n", predict_battery_life(bat));

    if (current > 10) {
        printf("Suggestions:\n");
        printf("- Enable deep sleep mode\n");
        printf("- Reduce sensor sampling rate\n");
        printf("- Use power-efficient communication\n");
    }
}
```

### 5.1 功耗日志

```c
// 功耗日志记录
typedef struct {
    uint32_t timestamp;
    float current_ma;
    float voltage_v;
    uint8_t state;
    char activity[32];
} power_log_t;

#define LOG_SIZE 1000
static power_log_t power_log[LOG_SIZE];
static int log_index = 0;

void log_power(float current, float voltage, uint8_t state, const char *activity) {
    if (log_index >= LOG_SIZE) return;

    power_log[log_index].timestamp = get_tick();
    power_log[log_index].current_ma = current;
    power_log[log_index].voltage_v = voltage;
    power_log[log_index].state = state;
    strncpy(power_log[log_index].activity, activity, 31);
    log_index++;
}

void dump_power_log(void) {
    printf("Timestamp,Current(mA),Voltage(V),State,Activity\n");
    for (int i = 0; i < log_index; i++) {
        printf("%u,%.2f,%.2f,%d,%s\n",
               power_log[i].timestamp,
               power_log[i].current_ma,
               power_log[i].voltage_v,
               power_log[i].state,
               power_log[i].activity);
    }
}
```
