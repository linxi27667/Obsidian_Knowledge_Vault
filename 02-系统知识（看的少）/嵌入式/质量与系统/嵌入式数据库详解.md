# 嵌入式数据库详解

## 核心概念

- **SQLite** - 轻量级关系数据库
- **KV存储** - 键值存储
- **时序数据库** - 时间序列数据
- **环形缓冲** - 循环存储

## 一、SQLite

### 1.1 SQLite基础

```c
#include <sqlite3.h>

// 打开数据库
sqlite3 *db;
int rc = sqlite3_open("sensor.db", &db);
if (rc != SQLITE_OK) {
    fprintf(stderr, "Cannot open database: %s\n", sqlite3_errmsg(db));
    return -1;
}

// 执行SQL
char *err_msg = NULL;
rc = sqlite3_exec(db, "CREATE TABLE IF NOT EXISTS sensors ("
                  "id INTEGER PRIMARY KEY AUTOINCREMENT,"
                  "name TEXT NOT NULL,"
                  "value REAL,"
                  "timestamp DATETIME DEFAULT CURRENT_TIMESTAMP)",
                  NULL, NULL, &err_msg);
if (rc != SQLITE_OK) {
    fprintf(stderr, "SQL error: %s\n", err_msg);
    sqlite3_free(err_msg);
}

// 插入数据
const char *sql = "INSERT INTO sensors (name, value) VALUES (?, ?)";
sqlite3_stmt *stmt;
sqlite3_prepare_v2(db, sql, -1, &stmt, NULL);
sqlite3_bind_text(stmt, 1, "temperature", -1, SQLITE_STATIC);
sqlite3_bind_double(stmt, 2, 25.5);
sqlite3_step(stmt);
sqlite3_finalize(stmt);

// 查询数据
const char *query = "SELECT * FROM sensors WHERE name = ?";
sqlite3_prepare_v2(db, query, -1, &stmt, NULL);
sqlite3_bind_text(stmt, 1, "temperature", -1, SQLITE_STATIC);

while (sqlite3_step(stmt) == SQLITE_ROW) {
    int id = sqlite3_column_int(stmt, 0);
    const char *name = sqlite3_column_text(stmt, 1);
    double value = sqlite3_column_double(stmt, 2);
    const char *timestamp = sqlite3_column_text(stmt, 3);

    printf("ID: %d, Name: %s, Value: %.2f, Time: %s\n",
           id, name, value, timestamp);
}

sqlite3_finalize(stmt);

// 关闭数据库
sqlite3_close(db);
```

### 1.2 事务处理

```c
// 开始事务
sqlite3_exec(db, "BEGIN TRANSACTION", NULL, NULL, NULL);

// 批量插入
const char *sql = "INSERT INTO sensors (name, value) VALUES (?, ?)";
sqlite3_stmt *stmt;
sqlite3_prepare_v2(db, sql, -1, &stmt, NULL);

for (int i = 0; i < 1000; i++) {
    sqlite3_bind_text(stmt, 1, "temperature", -1, SQLITE_STATIC);
    sqlite3_bind_double(stmt, 2, 20.0 + rand() % 10);
    sqlite3_step(stmt);
    sqlite3_reset(stmt);
}

sqlite3_finalize(stmt);

// 提交事务
sqlite3_exec(db, "COMMIT", NULL, NULL, NULL);

// 回滚事务
// sqlite3_exec(db, "ROLLBACK", NULL, NULL, NULL);

// WAL模式(写前日志，提高并发性能)
sqlite3_exec(db, "PRAGMA journal_mode=WAL", NULL, NULL, NULL);

// 同步模式
sqlite3_exec(db, "PRAGMA synchronous=NORMAL", NULL, NULL, NULL);
```

### 1.3 高级特性

```c
// 索引
sqlite3_exec(db, "CREATE INDEX IF NOT EXISTS idx_name ON sensors(name)",
             NULL, NULL, NULL);
sqlite3_exec(db, "CREATE INDEX IF NOT EXISTS idx_time ON sensors(timestamp)",
             NULL, NULL, NULL);

// 视图
sqlite3_exec(db, "CREATE VIEW IF NOT EXISTS recent_sensors AS "
             "SELECT * FROM sensors ORDER BY timestamp DESC LIMIT 100",
             NULL, NULL, NULL);

// 触发器
sqlite3_exec(db, "CREATE TRIGGER IF NOT EXISTS log_insert "
             "AFTER INSERT ON sensors "
             "BEGIN "
             "  INSERT INTO audit_log (action, table_name, record_id) "
             "  VALUES ('INSERT', 'sensors', NEW.id); "
             "END",
             NULL, NULL, NULL);

// 聚合查询
const char *agg_sql = "SELECT name, AVG(value), MIN(value), MAX(value), COUNT(*) "
                      "FROM sensors GROUP BY name";
sqlite3_prepare_v2(db, agg_sql, -1, &stmt, NULL);

while (sqlite3_step(stmt) == SQLITE_ROW) {
    const char *name = sqlite3_column_text(stmt, 0);
    double avg = sqlite3_column_double(stmt, 1);
    double min = sqlite3_column_double(stmt, 2);
    double max = sqlite3_column_double(stmt, 3);
    int count = sqlite3_column_int(stmt, 4);

    printf("%s: avg=%.2f, min=%.2f, max=%.2f, count=%d\n",
           name, avg, min, max, count);
}

sqlite3_finalize(stmt);

// 分页查询
int page = 1;
int page_size = 20;
char page_sql[256];
snprintf(page_sql, sizeof(page_sql),
         "SELECT * FROM sensors ORDER BY id LIMIT %d OFFSET %d",
         page_size, (page - 1) * page_size);
```

## 二、KV存储

### 2.1 简单KV实现

```c
// KV存储结构
typedef struct {
    char key[64];
    uint8_t value[256];
    size_t value_len;
    uint32_t crc;
    uint32_t timestamp;
} kv_entry_t;

// KV存储管理
typedef struct {
    uint8_t *flash_base;
    size_t flash_size;
    size_t entry_size;
    int max_entries;
    int current_entry;
} kv_store_t;

// 初始化KV存储
int kv_init(kv_store_t *store, uint8_t *flash_base, size_t flash_size) {
    store->flash_base = flash_base;
    store->flash_size = flash_size;
    store->entry_size = sizeof(kv_entry_t);
    store->max_entries = flash_size / store->entry_size;
    store->current_entry = 0;

    // 扫描现有条目
    for (int i = 0; i < store->max_entries; i++) {
        kv_entry_t *entry = (kv_entry_t *)(flash_base + i * store->entry_size);
        if (entry->crc != 0 && entry->crc == crc32(entry, sizeof(kv_entry_t) - 4)) {
            store->current_entry = i + 1;
        }
    }

    return 0;
}

// 写入KV
int kv_set(kv_store_t *store, const char *key, const void *value, size_t len) {
    if (store->current_entry >= store->max_entries) {
        // 需要垃圾回收
        kv_gc(store);
    }

    kv_entry_t entry;
    memset(&entry, 0, sizeof(entry));
    strncpy(entry.key, key, sizeof(entry.key) - 1);
    memcpy(entry.value, value, len);
    entry.value_len = len;
    entry.timestamp = get_timestamp();
    entry.crc = crc32(&entry, sizeof(entry) - 4);

    // 写入Flash
    int offset = store->current_entry * store->entry_size;
    flash_write(store->flash_base + offset, &entry, sizeof(entry));

    store->current_entry++;

    return 0;
}

// 读取KV
int kv_get(kv_store_t *store, const char *key, void *value, size_t *len) {
    // 从后向前搜索(最新值)
    for (int i = store->current_entry - 1; i >= 0; i--) {
        kv_entry_t *entry = (kv_entry_t *)(store->flash_base + i * store->entry_size);

        if (entry->crc != 0 &&
            entry->crc == crc32(entry, sizeof(entry) - 4) &&
            strcmp(entry.key, key) == 0) {
            memcpy(value, entry->value, entry->value_len);
            *len = entry->value_len;
            return 0;
        }
    }

    return -1;  // 未找到
}

// 垃圾回收
int kv_gc(kv_store_t *store) {
    // 收集有效条目
    kv_entry_t *valid_entries = malloc(store->max_entries * sizeof(kv_entry_t));
    int valid_count = 0;

    // 标记有效条目(保留每个key的最新值)
    for (int i = store->current_entry - 1; i >= 0; i--) {
        kv_entry_t *entry = (kv_entry_t *)(store->flash_base + i * store->entry_size);

        if (entry->crc != 0 && entry->crc == crc32(entry, sizeof(entry) - 4)) {
            // 检查是否已存在
            bool exists = false;
            for (int j = 0; j < valid_count; j++) {
                if (strcmp(valid_entries[j].key, entry->key) == 0) {
                    exists = true;
                    break;
                }
            }

            if (!exists) {
                valid_entries[valid_count++] = *entry;
            }
        }
    }

    // 擦除Flash
    flash_erase(store->flash_base, store->flash_size);

    // 重写有效条目
    store->current_entry = 0;
    for (int i = valid_count - 1; i >= 0; i--) {
        int offset = store->current_entry * store->entry_size;
        flash_write(store->flash_base + offset, &valid_entries[i], sizeof(kv_entry_t));
        store->current_entry++;
    }

    free(valid_entries);
    return 0;
}
```

## 三、时序数据

### 3.1 时序存储

```c
// 时序数据点
typedef struct {
    uint32_t timestamp;
    float value;
    uint8_t quality;  // 数据质量
} ts_data_point_t;

// 时序缓冲区
typedef struct {
    ts_data_point_t *buffer;
    size_t capacity;
    size_t head;
    size_t tail;
    size_t count;
} ts_buffer_t;

// 初始化
ts_buffer_t *ts_buffer_create(size_t capacity) {
    ts_buffer_t *buf = malloc(sizeof(ts_buffer_t));
    buf->buffer = malloc(capacity * sizeof(ts_data_point_t));
    buf->capacity = capacity;
    buf->head = 0;
    buf->tail = 0;
    buf->count = 0;
    return buf;
}

// 写入数据点
int ts_write(ts_buffer_t *buf, uint32_t timestamp, float value, uint8_t quality) {
    buf->buffer[buf->head].timestamp = timestamp;
    buf->buffer[buf->head].value = value;
    buf->buffer[buf->head].quality = quality;

    buf->head = (buf->head + 1) % buf->capacity;

    if (buf->count < buf->capacity) {
        buf->count++;
    } else {
        buf->tail = (buf->tail + 1) % buf->capacity;  // 覆盖旧数据
    }

    return 0;
}

// 查询时间范围
int ts_query_range(ts_buffer_t *buf, uint32_t start, uint32_t end,
                   ts_data_point_t *result, size_t max_results) {
    int count = 0;

    for (size_t i = 0; i < buf->count; i++) {
        size_t idx = (buf->tail + i) % buf->capacity;
        ts_data_point_t *dp = &buf->buffer[idx];

        if (dp->timestamp >= start && dp->timestamp <= end) {
            if (count < max_results) {
                result[count++] = *dp;
            }
        }
    }

    return count;
}

// 聚合查询
typedef struct {
    float min;
    float max;
    float avg;
    float sum;
    int count;
} ts_aggregate_t;

ts_aggregate_t ts_aggregate(ts_buffer_t *buf, uint32_t start, uint32_t end) {
    ts_aggregate_t agg = {0};
    agg.min = FLT_MAX;
    agg.max = FLT_MIN;

    for (size_t i = 0; i < buf->count; i++) {
        size_t idx = (buf->tail + i) % buf->capacity;
        ts_data_point_t *dp = &buf->buffer[idx];

        if (dp->timestamp >= start && dp->timestamp <= end) {
            agg.sum += dp->value;
            agg.count++;

            if (dp->value < agg.min) agg.min = dp->value;
            if (dp->value > agg.max) agg.max = dp->value;
        }
    }

    if (agg.count > 0) {
        agg.avg = agg.sum / agg.count;
    }

    return agg;
}

// 降采样
int ts_downsample(ts_buffer_t *buf, uint32_t start, uint32_t end,
                  uint32_t interval, ts_data_point_t *result, size_t max_results) {
    int count = 0;
    uint32_t current_bucket = start;

    float sum = 0;
    int bucket_count = 0;

    for (size_t i = 0; i < buf->count; i++) {
        size_t idx = (buf->tail + i) % buf->capacity;
        ts_data_point_t *dp = &buf->buffer[idx];

        if (dp->timestamp >= start && dp->timestamp <= end) {
            if (dp->timestamp < current_bucket + interval) {
                sum += dp->value;
                bucket_count++;
            } else {
                // 输出当前桶的平均值
                if (bucket_count > 0 && count < max_results) {
                    result[count].timestamp = current_bucket;
                    result[count].value = sum / bucket_count;
                    count++;
                }

                // 移动到下一个桶
                current_bucket += interval;
                sum = dp->value;
                bucket_count = 1;
            }
        }
    }

    // 输出最后一个桶
    if (bucket_count > 0 && count < max_results) {
        result[count].timestamp = current_bucket;
        result[count].value = sum / bucket_count;
        count++;
    }

    return count;
}
```

## 四、环形缓冲

### 4.1 环形缓冲实现

```c
// 环形缓冲区
typedef struct {
    uint8_t *buffer;
    size_t size;
    size_t head;
    size_t tail;
    size_t count;
} ring_buffer_t;

// 创建环形缓冲
ring_buffer_t *ring_buffer_create(size_t size) {
    ring_buffer_t *rb = malloc(sizeof(ring_buffer_t));
    rb->buffer = malloc(size);
    rb->size = size;
    rb->head = 0;
    rb->tail = 0;
    rb->count = 0;
    return rb;
}

// 写入数据
size_t ring_buffer_write(ring_buffer_t *rb, const uint8_t *data, size_t len) {
    size_t written = 0;

    while (written < len && rb->count < rb->size) {
        rb->buffer[rb->head] = data[written];
        rb->head = (rb->head + 1) % rb->size;
        rb->count++;
        written++;
    }

    return written;
}

// 读取数据
size_t ring_buffer_read(ring_buffer_t *rb, uint8_t *data, size_t len) {
    size_t read = 0;

    while (read < len && rb->count > 0) {
        data[read] = rb->buffer[rb->tail];
        rb->tail = (rb->tail + 1) % rb->size;
        rb->count--;
        read++;
    }

    return read;
}

// 查看数据(不移除)
size_t ring_buffer_peek(ring_buffer_t *rb, uint8_t *data, size_t len) {
    size_t read = 0;
    size_t pos = rb->tail;

    while (read < len && read < rb->count) {
        data[read] = rb->buffer[pos];
        pos = (pos + 1) % rb->size;
        read++;
    }

    return read;
}

// 可用空间
size_t ring_buffer_available(ring_buffer_t *rb) {
    return rb->size - rb->count;
}

// 已用空间
size_t ring_buffer_used(ring_buffer_t *rb) {
    return rb->count;
}

// 清空
void ring_buffer_flush(ring_buffer_t *rb) {
    rb->head = 0;
    rb->tail = 0;
    rb->count = 0;
}

// 查找模式
int ring_buffer_find(ring_buffer_t *rb, const uint8_t *pattern, size_t pattern_len) {
    if (pattern_len > rb->count) return -1;

    for (size_t i = 0; i <= rb->count - pattern_len; i++) {
        bool match = true;
        for (size_t j = 0; j < pattern_len; j++) {
            size_t idx = (rb->tail + i + j) % rb->size;
            if (rb->buffer[idx] != pattern[j]) {
                match = false;
                break;
            }
        }
        if (match) return i;
    }

    return -1;
}
```

## 五、Flash存储优化

### 5.1 磨损均衡

```c
// 磨损均衡器
typedef struct {
    uint32_t *block_counters;   // 块擦除计数
    int total_blocks;
    int current_block;
    uint32_t max_erase_count;
} wear_leveler_t;

// 初始化
int wl_init(wear_leveler_t *wl, int total_blocks) {
    wl->total_blocks = total_blocks;
    wl->block_counters = calloc(total_blocks, sizeof(uint32_t));
    wl->current_block = 0;
    wl->max_erase_count = 0;

    // 读取现有计数(从Flash元数据)
    load_wear_counters(wl);

    return 0;
}

// 选择下一个块(均衡策略)
int wl_select_block(wear_leveler_t *wl) {
    // 选择擦除次数最少的块
    int min_block = 0;
    uint32_t min_count = UINT32_MAX;

    for (int i = 0; i < wl->total_blocks; i++) {
        if (wl->block_counters[i] < min_count) {
            min_count = wl->block_counters[i];
            min_block = i;
        }
    }

    return min_block;
}

// 擦除并更新计数
int wl_erase(wear_leveler_t *wl, int block) {
    // 擦除Flash块
    flash_erase_block(block);

    // 更新计数
    wl->block_counters[block]++;
    if (wl->block_counters[block] > wl->max_erase_count) {
        wl->max_erase_count = wl->block_counters[block];
    }

    // 保存计数
    save_wear_counters(wl);

    return 0;
}

// 获取磨损状态
float wl_get_wear_level(wear_leveler_t *wl) {
    uint32_t min_count = UINT32_MAX;
    uint32_t max_count = 0;

    for (int i = 0; i < wl->total_blocks; i++) {
        if (wl->block_counters[i] < min_count) min_count = wl->block_counters[i];
        if (wl->block_counters[i] > max_count) max_count = wl->block_counters[i];
    }

    if (max_count == 0) return 1.0f;

    return (float)min_count / max_count;
}
```

## 附录：数据库对比

| 数据库 | 类型 | 特点 | 适用场景 |
|--------|------|------|----------|
| SQLite | 关系型 | SQL支持、事务 | 结构化数据 |
| LevelDB | KV | LSM树、高性能 | 键值存储 |
| RocksDB | KV | 优化、可配置 | 大数据量 |
| InfluxDB | 时序 | 时间索引、聚合 | 传感器数据 |
| Redis | 内存 | 高速、丰富结构 | 缓存 |
