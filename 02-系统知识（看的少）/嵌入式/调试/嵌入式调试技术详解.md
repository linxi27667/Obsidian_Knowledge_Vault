### 2.2 GDB脚本

```python
# GDB Python脚本
import gdb

class DumpRegisters(gdb.Command):
    """打印所有寄存器"""
    def __init__(self):
        super().__init__("dump-regs", gdb.COMMAND_USER)

    def invoke(self, arg, from_tty):
        regs = ["r0", "r1", "r2", "r3", "r4", "r5", "r6", "r7",
                "r8", "r9", "r10", "r11", "r12", "sp", "lr", "pc"]
        for reg in regs:
            val = gdb.parse_and_eval(f"${reg}")
            print(f"{reg:>4} = {int(val):#010x}")

DumpRegisters()

class DumpMemory(gdb.Command):
    """转储内存区域"""
    def __init__(self):
        super().__init__("dump-mem", gdb.COMMAND_USER)

    def invoke(self, arg, from_tty):
        args = gdb.string_to_argv(arg)
        if len(args) < 2:
            print("Usage: dump-mem <addr> <size>")
            return

        addr = int(args[0], 0)
        size = int(args[1], 0)

        inferior = gdb.selected_inferior()
        mem = inferior.read_memory(addr, size)

        for i in range(0, size, 16):
            hex_str = " ".join(f"{b:02x}" for b in mem[i:i+16])
            ascii_str = "".join(chr(b) if 32 <= b < 127 else "." for b in mem[i:i+16])
            print(f"{addr+i:#010x}: {hex_str:<48} {ascii_str}")

DumpMemory()

class RTTViewer(gdb.Command):
    """RTT查看器"""
    def __init__(self):
        super().__init__("rtt-view", gdb.COMMAND_USER)
        self.running = False

    def invoke(self, arg, from_tty):
        self.running = not self.running
        if self.running:
            print("RTT viewer started")
            gdb.events.stop.connect(self.on_stop)
        else:
            print("RTT viewer stopped")
            gdb.events.stop.disconnect(self.on_stop)

    def on_stop(self, event):
        # 读取RTT缓冲区
        try:
            rtt_addr = gdb.parse_and_eval("_SEGGER_RTT")
            up_buffer = rtt_addr["aUp"][0]
            buffer_addr = up_buffer["pBuffer"]
            wr_ofs = up_buffer["WrOfs"]
            rd_ofs = up_buffer["RdOfs"]

            if wr_ofs != rd_ofs:
                size = wr_ofs - rd_ofs
                mem = gdb.selected_inferior().read_memory(buffer_addr + rd_ofs, size)
                print(mem.tobytes().decode('utf-8', errors='replace'), end='')
        except:
            pass

RTTViewer()
```

```c
    SEGGER_RTT_ConfigUpBuffer(0, "Terminal", buffer_up, sizeof(buffer_up),
                               SEGGER_RTT_MODE_NO_BLOCK_SKIP);

    SEGGER_RTT_ConfigDownBuffer(0, "Terminal", buffer_down, sizeof(buffer_down),
                                 SEGGER_RTT_MODE_NO_BLOCK_SKIP);
}

// RTT输出
void rtt_printf(const char *fmt, ...) {
    va_list args;
    va_start(args, fmt);
    SEGGER_RTT_vprintf(0, fmt, &args);
    va_end(args);
}

// RTT输入
int rtt_read(void *buf, size_t len) {
    return SEGGER_RTT_Read(0, buf, len);
}
```

### 3.2 RTT日志系统

```c
// 日志级别
typedef enum {
    LOG_LEVEL_ERROR = 0,
    LOG_LEVEL_WARN,
    LOG_LEVEL_INFO,
    LOG_LEVEL_DEBUG,
    LOG_LEVEL_VERBOSE,
} log_level_t;

static log_level_t current_level = LOG_LEVEL_INFO;

// 日志宏
#define LOGE(tag, fmt, ...) do { \
    if (current_level >= LOG_LEVEL_ERROR) \
        rtt_printf("[E][%s] " fmt "\n", tag, ##__VA_ARGS__); \
} while(0)

#define LOGW(tag, fmt, ...) do { \
    if (current_level >= LOG_LEVEL_WARN) \
        rtt_printf("[W][%s] " fmt "\n", tag, ##__VA_ARGS__); \
} while(0)

#define LOGI(tag, fmt, ...) do { \
    if (current_level >= LOG_LEVEL_INFO) \
        rtt_printf("[I][%s] " fmt "\n", tag, ##__VA_ARGS__); \
} while(0)

#define LOGD(tag, fmt, ...) do { \
    if (current_level >= LOG_LEVEL_DEBUG) \
        rtt_printf("[D][%s] " fmt "\n", tag, ##__VA_ARGS__); \
} while(0)

// 使用示例
void sensor_task(void) {
    LOGI("SENSOR", "Starting sensor task");

    while (1) {
        float temp = read_temperature();
        LOGD("SENSOR", "Temperature: %.2f", temp);

        if (temp > 50) {
            LOGW("SENSOR", "High temperature: %.2f", temp);
        }

        vTaskDelay(pdMS_TO_TICKS(1000));
    }
}
```

## 四、CoreDump

### 4.1 崩溃转储

```c
// 寄存器上下文
typedef struct {
    uint32_t r0;
    uint32_t r1;
    uint32_t r2;
    uint32_t r3;
    uint32_t r4;
    uint32_t r5;
    uint32_t r6;
    uint32_t r7;
    uint32_t r8;
    uint32_t r9;
    uint32_t r10;
    uint32_t r11;
    uint32_t r12;
    uint32_t sp;
    uint32_t lr;
    uint32_t pc;
    uint32_t xpsr;
} core_regs_t;

// HardFault处理
void HardFault_Handler(void) {
    core_regs_t regs;

    // 保存寄存器
    __asm volatile (
        "TST LR, #4\n"
        "ITE EQ\n"
        "MRSEQ R0, MSP\n"
        "MRSNE R0, PSP\n"
        "MOV %0, R0\n"
        : "=r" (regs)
    );

    // 保存到Flash
    save_coredump(&regs);

    // 复位
    NVIC_SystemReset();
}

// 保存CoreDump
void save_coredump(core_regs_t *regs) {
    // 写入Flash
    uint32_t addr = COREDUMP_FLASH_ADDR;

    // 写入魔数
    flash_write(addr, "CORE", 4);
    addr += 4;

    // 写入寄存器
    flash_write(addr, regs, sizeof(core_regs_t));
    addr += sizeof(core_regs_t);

    // 写入栈内容
    uint32_t stack_size = 256;
    flash_write(addr, (void *)regs->sp, stack_size);
}
```

### 4.2 CoreDump分析

```python
# CoreDump解析工具
import struct
import sys

def parse_coredump(filename):
    with open(filename, 'rb') as f:
        data = f.read()

    # 检查魔数
    if data[:4] != b'CORE':
        print("Invalid coredump file")
        return

    # 解析寄存器
    offset = 4
    regs = struct.unpack_from('<17I', data, offset)
    offset += 17 * 4

    print("=== Register Dump ===")
    names = ['R0', 'R1', 'R2', 'R3', 'R4', 'R5', 'R6', 'R7',
             'R8', 'R9', 'R10', 'R11', 'R12', 'SP', 'LR', 'PC', 'XPSR']

    for i, name in enumerate(names):
        print(f"{name:>4} = {regs[i]:#010x}")

    # 解析栈
    stack = data[offset:offset+256]
    print("\n=== Stack Dump ===")
    for i in range(0, len(stack), 4):
        val = struct.unpack_from('<I', stack, i)[0]
        print(f"SP+{i:03x}: {val:#010x}")

    # 分析崩溃原因
    pc = regs[15]
    lr = regs[14]
    print(f"\n=== Crash Analysis ===")
    print(f"PC: {pc:#010x}")
    print(f"LR: {lr:#010x}")

    # 检查XPSR
    xpsr = regs[16]
    exception_number = xpsr & 0x1F
    print(f"Exception: {exception_number}")

if __name__ == '__main__':
    parse_coredump(sys.argv[1])
```

```c
    CoreDebug->DEMCR |= CoreDebug_DEMCR_TRCENA_Msk;

uint32_t dwt_get_cycles(void) {
    return DWT->CYCCNT;
}

// 性能测量
typedef struct {
    uint32_t start_cycles;
    uint32_t end_cycles;
    const char *name;
} perf_measure_t;

void perf_start(perf_measure_t *perf, const char *name) {
    perf->name = name;
    perf->start_cycles = dwt_get_cycles();
}

void perf_end(perf_measure_t *perf) {
    perf->end_cycles = dwt_get_cycles();
    uint32_t cycles = perf->end_cycles - perf->start_cycles;
    printf("%s: %lu cycles\n", perf->name, cycles);
```

### 5.2 能耗分析

```c
// 能耗测量
typedef struct {
    float voltage_v;
    float current_ma;
    uint32_t duration_ms;
    float energy_mj;
} energy_measure_t;

energy_measure_t measure_energy(float voltage, float current, uint32_t duration) {
    energy_measure_t m;
    m.voltage_v = voltage;
    m.current_ma = current;
    m.duration_ms = duration;
    m.energy_mj = voltage * current * duration / 1000;
    return m;
}

// 能耗报告
void print_energy_report(energy_measure_t *measures, int count) {
    float total_energy = 0;
    printf("\n=== Energy Report ===\n");
    printf("%-30s %10s %10s %10s\n", "Component", "Current", "Duration", "Energy");
    printf("--------------------------------------------------------------\n");

    for (int i = 0; i < count; i++) {
        printf("%-30s %7.1f mA %7lu ms %7.2f mJ\n",
               "Component",
               measures[i].current_ma,
               measures[i].duration_ms,
               measures[i].energy_mj);
        total_energy += measures[i].energy_mj;
    }

    printf("\nTotal energy: %.2f mJ\n", total_energy);
}
```

```c
// 内存泄漏检测
typedef struct {
    void *ptr;
    size_t size;
    const char *file;
    int line;
} alloc_record_t;

#define MAX_ALLOC_RECORDS 100
static alloc_record_t alloc_records[MAX_ALLOC_RECORDS];
static int alloc_count = 0;

void *debug_malloc(size_t size, const char *file, int line) {
    void *ptr = malloc(size);
    if (ptr && alloc_count < MAX_ALLOC_RECORDS) {
        alloc_records[alloc_count].ptr = ptr;
        alloc_records[alloc_count].size = size;
        alloc_records[alloc_count].file = file;
        alloc_records[alloc_count].line = line;
        alloc_count++;
    }
    return ptr;
}

void debug_free(void *ptr, const char *file, int line) {
    for (int i = 0; i < alloc_count; i++) {
        if (alloc_records[i].ptr == ptr) {
            alloc_records[i] = alloc_records[--alloc_count];
            break;
        }
    }
    free(ptr);
}

#define malloc(size) debug_malloc(size, __FILE__, __LINE__)
#define free(ptr) debug_free(ptr, __FILE__, __LINE__)
```
