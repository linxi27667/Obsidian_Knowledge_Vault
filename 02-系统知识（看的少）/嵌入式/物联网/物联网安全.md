# 物联网安全

## 核心概念

- **安全启动** - 验证固件完整性
- **安全通信** - 加密传输
- **安全存储** - 保护密钥和数据
- **安全更新** - 防篡改OTA

## 一、安全启动

### 1.1 安全启动流程

```
ROM Bootloader
    │
    ├── 验证Bootloader签名
    │       │
    │       ├── 成功 → 继续
    │       └── 失败 → 停止/恢复
    │
    ├── Bootloader验证APP签名
    │       │
    │       ├── 成功 → 跳转APP
    │       └── 失败 → 停止/恢复
    │
    └── APP运行
```

### 1.2 RSA签名验证

```c
// RSA-2048签名验证
typedef struct {
    uint8_t modulus[256];      // N
    uint8_t exponent[4];       // e (通常65537)
    uint8_t signature[256];    // 签名
} rsa_key_t;

bool verify_signature(const uint8_t *data, size_t data_len,
                      rsa_key_t *key) {
    // 计算数据哈希(SHA-256)
    uint8_t hash[32];
    sha256(data, data_len, hash);

    // RSA解密签名
    uint8_t decrypted[256];
    rsa_decrypt(key->signature, decrypted, key->modulus, key->exponent, 256);

    // PKCS#1 v1.5填充验证
    // 00 01 FF FF ... FF 00 ASN.1 HASH
    if (decrypted[0] != 0x00 || decrypted[1] != 0x01) return false;

    int i = 2;
    while (i < 256 && decrypted[i] == 0xFF) i++;
    if (decrypted[i] != 0x00) return false;
    i++;

    // 跳过ASN.1头
    // 验证哈希
    return memcmp(&decrypted[i + 19], hash, 32) == 0;  // 19 = ASN.1头长度
}

// 安全启动验证
bool secure_boot_verify(uint32_t app_addr, uint32_t app_size, rsa_key_t *key) {
    // 读取APP区域
    uint8_t *app_data = (uint8_t *)app_addr;

    // 签名存储在APP末尾
    uint8_t *signature = app_data + app_size;
    uint32_t data_len = app_size;

    return verify_signature(app_data, data_len, key);
}
```

## 二、安全通信

### 2.1 TLS/DTLS

```c
// mbedTLS配置
#include "mbedtls/ssl.h"
#include "mbedtls/net_sockets.h"
#include "mbedtls/entropy.h"
#include "mbedtls/ctr_drbg.h"

typedef struct {
    mbedtls_ssl_context ssl;
    mbedtls_ssl_config conf;
    mbedtls_net_context server_fd;
    mbedtls_entropy_context entropy;
    mbedtls_ctr_drbg_context ctr_drbg;
    mbedtls_x509_crt ca_crt;
} tls_client_t;

int tls_connect(tls_client_t *ctx, const char *host, const char *port) {
    int ret;

    // 初始化
    mbedtls_net_init(&ctx->server_fd);
    mbedtls_ssl_init(&ctx->ssl);
    mbedtls_ssl_config_init(&ctx->conf);
    mbedtls_x509_crt_init(&ctx->ca_crt);
    mbedtls_entropy_init(&ctx->entropy);
    mbedtls_ctr_drbg_init(&ctx->ctr_drbg);

    // 随机数生成器
    ret = mbedtls_ctr_drbg_seed(&ctx->ctr_drbg, mbedtls_entropy_func,
                                 &ctx->entropy, (const unsigned char *)"client", 6);
    if (ret != 0) return ret;

    // 建立TCP连接
    ret = mbedtls_net_connect(&ctx->server_fd, host, port, MBEDTLS_NET_PROTO_TCP);
    if (ret != 0) return ret;

    // 配置TLS
    ret = mbedtls_ssl_config_defaults(&ctx->conf,
                                       MBEDTLS_SSL_IS_CLIENT,
                                       MBEDTLS_SSL_TRANSPORT_STREAM,
                                       MBEDTLS_SSL_PRESET_DEFAULT);
    if (ret != 0) return ret;

    // 设置CA证书
    mbedtls_ssl_conf_authmode(&ctx->conf, MBEDTLS_SSL_VERIFY_REQUIRED);
    mbedtls_ssl_conf_ca_chain(&ctx->conf, &ctx->ca_crt, NULL);

    // 配置随机数
    mbedtls_ssl_conf_rng(&ctx->conf, mbedtls_ctr_drbg_random, &ctx->ctr_drbg);

    // 设置主机名
    mbedtls_ssl_set_hostname(&ctx->ssl, host);

    // 设置IO回调
    mbedtls_ssl_set_bio(&ctx->ssl, &ctx->server_fd,
                         mbedtls_net_send, mbedtls_net_recv, NULL);

    // TLS握手
    ret = mbedtls_ssl_setup(&ctx->ssl, &ctx->conf);
    if (ret != 0) return ret;

    while ((ret = mbedtls_ssl_handshake(&ctx->ssl)) != 0) {
        if (ret != MBEDTLS_ERR_SSL_WANT_READ && ret != MBEDTLS_ERR_SSL_WANT_WRITE) {
            return ret;
        }
    }

    return 0;
}

int tls_send(tls_client_t *ctx, const uint8_t *data, size_t len) {
    return mbedtls_ssl_write(&ctx->ssl, data, len);
}

int tls_receive(tls_client_t *ctx, uint8_t *buf, size_t buf_len) {
    return mbedtls_ssl_read(&ctx->ssl, buf, buf_len);
}
```

### 2.2 MQTT over TLS

```c
// 安全MQTT连接
void secure_mqtt_connect(const char *host, const char *ca_cert,
                          const char *client_cert, const char *client_key) {
    // TLS配置
    tls_client_t tls;
    tls_connect(&tls, host, "8883");  // MQTT over TLS端口

    // MQTT CONNECT包
    uint8_t connect[] = {
        0x10,  // CONNECT
        0x16,  // 剩余长度
        0x00, 0x04, 'M', 'Q', 'T', 'T',  // 协议名
        0x04,  // 协议级别
        0xC2,  // 连接标志(用户名+密码+清理会话)
        0x00, 0x3C,  // Keep Alive: 60s
        0x00, 0x04, 't', 'e', 's', 't',  // Client ID
        0x00, 0x04, 'u', 's', 'e', 'r',  // 用户名
        0x00, 0x04, 'p', 'a', 's', 's',  // 密码
    };

    tls_send(&tls, connect, sizeof(connect));
}
```

## 三、安全存储

### 3.1 密钥管理

```c
// 安全密钥存储
typedef struct {
    uint8_t device_key[32];      // 设备密钥
    uint8_t session_key[32];     // 会话密钥
    uint8_t ota_key[32];         // OTA密钥
    bool initialized;
} key_store_t;

// 使用eFuse存储根密钥
void store_root_key(const uint8_t *key) {
    // 写入eFuse(一次性可编程)
    efuse_write(EFUSE_BLK_KEY0, key, 32);
}

// 派生密钥
void derive_key(const uint8_t *root_key, const char *context,
                uint8_t *derived_key) {
    // HKDF密钥派生
    uint8_t prk[32];
    hkdf_extract(root_key, 32, (const uint8_t *)context, strlen(context), prk);
    hkdf_expand(prk, 32, derived_key, 32);
}

// 密钥轮换
void rotate_key(key_store_t *ks) {
    // 生成新会话密钥
    uint8_t new_key[32];
    random_bytes(new_key, 32);

    // 安全擦除旧密钥
    secure_memzero(ks->session_key, 32);

    // 存储新密钥
    memcpy(ks->session_key, new_key, 32);
}
```

### 3.2 安全存储API

```c
// NVS加密存储
void secure_nvs_write(const char *key, const uint8_t *data, size_t len) {
    // 使用设备密钥加密
    uint8_t encrypted[len + 16];  // +16 for IV and tag
    uint8_t iv[12];
    uint8_t tag[16];

    random_bytes(iv, 12);
    aes_gcm_encrypt(data, len, device_key, iv, encrypted, tag);

    // 存储: IV + 密文 + Tag
    uint8_t stored[12 + len + 16];
    memcpy(stored, iv, 12);
    memcpy(stored + 12, encrypted, len);
    memcpy(stored + 12 + len, tag, 16);

    nvs_write(key, stored, sizeof(stored));
}

int secure_nvs_read(const char *key, uint8_t *data, size_t max_len) {
    uint8_t stored[256];
    int stored_len = nvs_read(key, stored, sizeof(stored));
    if (stored_len < 28) return -1;

    uint8_t *iv = stored;
    uint8_t *encrypted = stored + 12;
    uint8_t *tag = stored + stored_len - 16;
    size_t len = stored_len - 28;

    return aes_gcm_decrypt(encrypted, len, device_key, iv, tag, data) ? len : -1;
}
```

## 四、安全OTA

### 4.1 安全OTA流程

```c
// 安全OTA更新
typedef struct {
    uint32_t version;
    uint32_t size;
    uint8_t hash[32];        // SHA-256
    uint8_t signature[256];  // RSA签名
    char url[128];
} ota_manifest_t;

bool secure_ota_update(ota_manifest_t *manifest) {
    // 1. 下载固件
    uint8_t *firmware = download_file(manifest->url, manifest->size);
    if (!firmware) return false;

    // 2. 验证哈希
    uint8_t hash[32];
    sha256(firmware, manifest->size, hash);
    if (memcmp(hash, manifest->hash, 32) != 0) {
        printf("Hash verification failed\n");
        free(firmware);
        return false;
    }

    // 3. 验证签名
    if (!verify_signature(firmware, manifest->size, &ota_public_key)) {
        printf("Signature verification failed\n");
        free(firmware);
        return false;
    }

    // 4. 写入Flash(双bank)
    write_to_inactive_bank(firmware, manifest->size);

    // 5. 设置更新标志
    set_update_flag();

    // 6. 重启
    esp_restart();

    return true;
}
```

### 4.2 回滚机制

```c
// OTA回滚保护
typedef struct {
    uint32_t magic;
    uint32_t version;
    uint32_t state;      // 0: 正常, 1: 待确认, 2: 已确认
    uint32_t boot_count;
    uint32_t max_boot_attempts;
} ota_state_t;

#define OTA_MAGIC 0x544F4132  // "OTA2"

bool ota_confirm(void) {
    ota_state_t state;
    read_ota_state(&state);

    if (state.magic != OTA_MAGIC) return false;

    // 标记为已确认
    state.state = 2;
    write_ota_state(&state);

    return true;
}

void ota_check_on_boot(void) {
    ota_state_t state;
    read_ota_state(&state);

    if (state.magic != OTA_MAGIC) return;

    if (state.state == 1) {
        // 待确认状态
        state.boot_count++;

        if (state.boot_count > state.max_boot_attempts) {
            // 启动失败，回滚
            printf("OTA rollback triggered\n");
            rollback_to_previous();
            esp_restart();
        }

        write_ota_state(&state);
    }
}
```

## 五、安全审计

### 5.1 安全日志

```c
// 安全事件日志
typedef enum {
    SEC_EVENT_BOOT,
    SEC_EVENT_AUTH_FAIL,
    SEC_EVENT_KEY_ROTATION,
    SEC_EVENT_OTA_START,
    SEC_EVENT_OTA_COMPLETE,
    SEC_EVENT_TAMPER,
    SEC_EVENT_COUNT
} security_event_t;

typedef struct {
    uint32_t timestamp;
    security_event_t event;
    uint32_t data;
    uint8_t mac[16];  // 消息认证码
} security_log_t;

void log_security_event(security_event_t event, uint32_t data) {
    security_log_t log;
    log.timestamp = get_timestamp();
    log.event = event;
    log.data = data;

    // 计算MAC(防篡改)
    hmac_sha256(log_key, sizeof(log_key),
                (uint8_t *)&log, sizeof(log) - 16,
                log.mac);

    // 写入安全存储
    append_security_log(&log);
}

// 验证日志完整性
bool verify_security_log(security_log_t *log) {
    uint8_t computed_mac[16];
    hmac_sha256(log_key, sizeof(log_key),
                (uint8_t *)log, sizeof(*log) - 16,
                computed_mac);
    return memcmp(log->mac, computed_mac, 16) == 0;
}
```
