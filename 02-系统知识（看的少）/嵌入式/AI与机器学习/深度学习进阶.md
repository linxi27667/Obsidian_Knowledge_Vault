# 深度学习进阶

## 核心概念

- **深度学习** - 多层神经网络学习

- **循环神经网络(RNN)** - 序列数据处理

### 1.2 残差连接

```c
// ResNet残差块
void residual_block(float *input, float *output, int channels) {
    float conv1_out[channels];
    float conv2_out[channels];

    // 卷积层1
    conv2d(input, conv1_out, channels, 3, 1);
    batch_norm(conv1_out, channels);
    relu(conv1_out, channels);

    // 卷积层2
    conv2d(conv1_out, conv2_out, channels, 3, 1);
    batch_norm(conv2_out, channels);

    // 残差连接
    for (int i = 0; i < channels; i++) {
        output[i] = conv2_out[i] + input[i];
    }
    relu(output, channels);
}
```

### 1.3 深度可分离卷积

```c
// MobileNet核心：深度可分离卷积
// 标准卷积: 3×3×C×C = 9C² 参数
// 深度可分离: 3×3×C + 1×1×C×C' = 9C + CC' 参数

void depthwise_conv(float *input, float *output, int channels, int stride) {
    float kernel[3][3];

    for (int c = 0; c < channels; c++) {
        for (int y = 0; y < out_h; y++) {
            for (int x = 0; x < out_w; x++) {
                float sum = 0;
                for (int ky = 0; ky < 3; ky++) {
                    for (int kx = 0; kx < 3; kx++) {
                        int iy = y * stride + ky - 1;
                        int ix = x * stride + kx - 1;
                        if (iy >= 0 && iy < in_h && ix >= 0 && ix < in_w) {
                            sum += input[(c * in_h + iy) * in_w + ix] * kernel[ky][kx];
                        }
                    }
                }
                output[(c * out_h + y) * out_w + x] = sum;
            }
        }
    }
}

void pointwise_conv(float *input, float *output, int in_ch, int out_ch) {
    float kernel[out_ch][in_ch];

    for (int oc = 0; oc < out_ch; oc++) {
        for (int y = 0; y < h; y++) {
            for (int x = 0; x < w; x++) {
                float sum = 0;
                for (int ic = 0; ic < in_ch; ic++) {
                    sum += input[(ic * h + y) * w + x] * kernel[oc][ic];
                }
                output[(oc * h + y) * w + x] = sum;
            }
        }
    }
}
```

## 二、RNN/LSTM

### 2.1 RNN结构

```
输入: x₁, x₂, ..., xₜ
隐藏状态: hₜ = f(Wₓh·xₜ + Wₕh·hₜ₋₁ + bₕ)
输出: yₜ = Wₕy·hₜ + bᵧ
```

---

### 2.2 LSTM单元

```c
typedef struct {
    float Wi[4];  // 输入门权重
    float Wf[4];  // 遗忘门权重
    float Wc[4];  // 候选细胞权重
    float Wo[4];  // 输出门权重
    float Ui[4];  // 输入门隐藏权重
    float Uf[4];  // 遗忘门隐藏权重
    float Uc[4];  // 候选细胞隐藏权重
    float Uo[4];  // 输出门隐藏权重
    float c;      // 细胞状态
    float h;      // 隐藏状态
} lstm_cell_t;

float sigmoid(float x) {
    return 1.0f / (1.0f + expf(-x));
}

float tanh_f(float x) {
    return tanhf(x);
}

void lstm_step(lstm_cell_t *cell, float x) {
    // 输入门
    float i = sigmoid(cell->Wi[0] * x + cell->Ui[0] * cell->h);

    // 遗忘门
    float f = sigmoid(cell->Wf[0] * x + cell->Uf[0] * cell->h);

    // 候选细胞
    float c_tilde = tanh_f(cell->Wc[0] * x + cell->Uc[0] * cell->h);

    // 输出门
    float o = sigmoid(cell->Wo[0] * x + cell->Uo[0] * cell->h);

    // 更新细胞状态
    cell->c = f * cell->c + i * c_tilde;

    // 更新隐藏状态
    cell->h = o * tanh_f(cell->c);
}
```

## 三、Transformer

### 3.1 自注意力机制

```c
// 自注意力: Q, K, V
void self_attention(float *Q, float *K, float *V, float *output,
                    int seq_len, int d_model) {
    float scores[seq_len][seq_len];
    float d_k = sqrtf(d_model);

    // 计算注意力分数: scores = Q * K^T / sqrt(d_k)
    for (int i = 0; i < seq_len; i++) {
        for (int j = 0; j < seq_len; j++) {
            float dot = 0;
            for (int k = 0; k < d_model; k++) {
                dot += Q[i * d_model + k] * K[j * d_model + k];
            }
            scores[i][j] = dot / d_k;
        }
    }

    // Softmax
    for (int i = 0; i < seq_len; i++) {
        float max_val = scores[i][0];
        for (int j = 1; j < seq_len; j++) {
            if (scores[i][j] > max_val) max_val = scores[i][j];
        }
        float sum = 0;
        for (int j = 0; j < seq_len; j++) {
            scores[i][j] = expf(scores[i][j] - max_val);
            sum += scores[i][j];
        }
        for (int j = 0; j < seq_len; j++) {
            scores[i][j] /= sum;
        }
    }

    // 输出 = scores * V
    for (int i = 0; i < seq_len; i++) {
        for (int k = 0; k < d_model; k++) {
            float sum = 0;
            for (int j = 0; j < seq_len; j++) {
                sum += scores[i][j] * V[j * d_model + k];
            }
            output[i * d_model + k] = sum;
        }
    }
}
```

### 3.2 多头注意力

```c
void multi_head_attention(float *Q, float *K, float *V, float *output,
                          int seq_len, int d_model, int num_heads) {
    int d_k = d_model / num_heads;

    float head_outputs[num_heads][seq_len * d_k];

    for (int h = 0; h < num_heads; h++) {
        // 每个头的Q, K, V切片
        float *Qh = Q + h * d_k;
        float *Kh = K + h * d_k;
        float *Vh = V + h * d_k;

        self_attention(Qh, Kh, Vh, head_outputs[h], seq_len, d_k);
    }

    // 拼接所有头
    for (int i = 0; i < seq_len; i++) {
        for (int h = 0; h < num_heads; h++) {
            memcpy(output + i * d_model + h * d_k,
                   head_outputs[h] + i * d_k,
                   d_k * sizeof(float));
        }
    }
}
```

### 3.3 前馈网络

```c
void feed_forward(float *input, float *output, int d_model, int d_ff) {
    float hidden[d_ff];

    // 第一层: ReLU(W1 * x + b1)
    for (int i = 0; i < d_ff; i++) {
        float sum = 0;
        for (int j = 0; j < d_model; j++) {
            sum += W1[i * d_model + j] * input[j];
        }
        hidden[i] = fmaxf(0, sum + b1[i]);  // ReLU
    }

    // 第二层: W2 * hidden + b2
    for (int i = 0; i < d_model; i++) {
        float sum = 0;
        for (int j = 0; j < d_ff; j++) {
            sum += W2[i * d_ff + j] * hidden[j];
        }
        output[i] = sum + b2[i];
    }
}
```

## 四、模型优化

### 4.1 量化

```c
// INT8量化
typedef struct {
    float scale;
    int8_t zero_point;
} quant_params_t;

int8_t quantize(float value, quant_params_t *params) {
    int32_t q = (int32_t)(value / params->scale) + params->zero_point;
    if (q > 127) q = 127;
    if (q < -128) q = -128;
    return (int8_t)q;
}

float dequantize(int8_t value, quant_params_t *params) {
    return (value - params->zero_point) * params->scale;
}

// INT8矩阵乘法
void matmul_int8(int8_t *A, int8_t *B, int32_t *C,
                 int M, int N, int K,
                 quant_params_t *params_a, quant_params_t *params_b) {
    for (int i = 0; i < M; i++) {
        for (int j = 0; j < N; j++) {
            int32_t sum = 0;
            for (int k = 0; k < K; k++) {
                sum += (int32_t)A[i * K + k] * (int32_t)B[k * N + j];
            }
            C[i * N + j] = sum;
        }
    }
}
```

### 4.2 剪枝

```c
// 权重剪枝
void prune_weights(float *weights, int size, float threshold) {
    for (int i = 0; i < size; i++) {
        if (fabsf(weights[i]) < threshold) {
            weights[i] = 0.0f;
        }
    }
}

// 稀疏矩阵乘法
void sparse_matmul(float *values, int *indices, int nnz,
                   float *x, float *y, int n) {
    memset(y, 0, n * sizeof(float));
    for (int i = 0; i < nnz; i++) {
        y[indices[i]] += values[i] * x[indices[i]];
    }
}
```

## 五、边缘部署

### 5.1 TFLite Micro

```c
#include "tensorflow/lite/micro/micro_interpreter.h"
#include "tensorflow/lite/micro/micro_mutable_op_resolver.h"
#include "tensorflow/lite/schema/schema_generated.h"

// 模型数据(编译时嵌入)
#include "model_data.h"

// 内存分配
constexpr int kTensorArenaSize = 32 * 1024;
uint8_t tensor_arena[kTensorArenaSize];

void inference(float *input, float *output) {
    // 加载模型
    const tflite::Model *model = tflite::GetModel(g_model_data);

    // 注册算子
    static tflite::MicroMutableOpResolver<10> resolver;
    resolver.AddConv2D();
    resolver.AddMaxPool2D();
    resolver.AddFullyConnected();
    resolver.AddSoftmax();

    // 创建解释器
    static tflite::MicroInterpreter interpreter(
        model, resolver, tensor_arena, kTensorArenaSize);

    // 分配内存
    interpreter.AllocateTensors();

    // 获取输入张量
    TfLiteTensor *input_tensor = interpreter.input(0);
    memcpy(input_tensor->data.f, input, input_tensor->bytes);

    // 执行推理
    interpreter.Invoke();

    // 获取输出
    TfLiteTensor *output_tensor = interpreter.output(0);
    memcpy(output, output_tensor->data.f, output_tensor->bytes);
}
```
