# 传感器驱动详解

## 核心概念

- **传感器驱动** - 传感器通信与数据获取
- **校准** - 消除系统误差
- **补偿** - 修正环境影响
- **融合** - 多传感器组合

## 一、温度传感器

### 1.1 DS18B20 (1-Wire)

```c
// DS18B20驱动
#define DS18B20_CMD_CONVERT     0x44
#define DS18B20_CMD_READ_SCRATCHPAD  0xBE
#define DS18B20_CMD_SKIP_ROM    0xCC

// 1-Wire时序(基于GPIO)
void ow_write_bit(int bit) {
    gpio_set_direction(OW_PIN, GPIO_MODE_OUTPUT);
    gpio_set_level(OW_PIN, 0);
    if (bit) {
        delay_us(6);
        gpio_set_level(OW_PIN, 1);
        delay_us(64);
    } else {
        delay_us(60);
        gpio_set_level(OW_PIN, 1);
        delay_us(10);
    }
}

int ow_read_bit(void) {
    gpio_set_direction(OW_PIN, GPIO_MODE_OUTPUT);
    gpio_set_level(OW_PIN, 0);
    delay_us(6);
    gpio_set_direction(OW_PIN, GPIO_MODE_INPUT);
    delay_us(9);
    int bit = gpio_get_level(OW_PIN);
    delay_us(55);
    return bit;
}

// 读取温度
float ds18b20_read_temp(void) {
    // 复位
    ow_reset();

    // Skip ROM
    ow_write_byte(DS18B20_CMD_SKIP_ROM);

    // 启动转换
    ow_write_byte(DS18B20_CMD_CONVERT);
    vTaskDelay(pdMS_TO_TICKS(750));

    // 再次复位
    ow_reset();
    ow_write_byte(DS18B20_CMD_SKIP_ROM);

    // 读取暂存器
    ow_write_byte(DS18B20_CMD_READ_SCRATCHPAD);
    uint8_t lsb = ow_read_byte();
    uint8_t msb = ow_read_byte();

    int16_t raw = (msb << 8) | lsb;
    return raw / 16.0f;
}
```

### 1.2 DHT11/DHT22 (单总线)

```c
// DHT22驱动
typedef struct {
    float temperature;
    float humidity;
    bool valid;
} dht22_data_t;

dht22_data_t dht22_read(void) {
    dht22_data_t data = {0};

    // 主机发送起始信号
    gpio_set_direction(DHT_PIN, GPIO_MODE_OUTPUT);
    gpio_set_level(DHT_PIN, 0);
    vTaskDelay(pdMS_TO_TICKS(2));
    gpio_set_level(DHT_PIN, 1);
    delay_us(30);

    // 切换到输入
    gpio_set_direction(DHT_PIN, GPIO_MODE_INPUT);

    // 等待DHT响应(低电平80μs)
    if (!wait_level(0, 100)) return data;
    // 等待DHT响应(高电平80μs)
    if (!wait_level(1, 100)) return data;

    // 读取40位数据
    uint8_t bytes[5] = {0};
    for (int i = 0; i < 40; i++) {
        // 低电平50μs
        if (!wait_level(0, 70)) return data;

        // 高电平: 26-28μs=0, 70μs=1
        int duration = wait_level(1, 100);
        if (duration < 0) return data;

        bytes[i / 8] <<= 1;
        if (duration > 40) bytes[i / 8] |= 1;
    }

    // 校验
    uint8_t checksum = bytes[0] + bytes[1] + bytes[2] + bytes[3];
    if (checksum == bytes[4]) {
        data.humidity = ((bytes[0] << 8) | bytes[1]) / 10.0f;
        data.temperature = ((bytes[2] << 8) | bytes[3]) / 10.0f;
        data.valid = true;
    }

    return data;
}
```

## 二、IMU传感器

### 2.1 MPU6050 (I2C)

```c
// MPU6050寄存器
#define MPU6050_ADDR         0x68
#define MPU6050_REG_PWR_MGMT 0x6B
#define MPU6050_REG_ACCEL_X  0x3B
#define MPU6050_REG_GYRO_X   0x43

typedef struct {
    int16_t ax, ay, az;
    int16_t gx, gy, gz;
} mpu6050_raw_t;

void mpu6050_init(void) {
    // 唤醒
    i2c_write_reg(MPU6050_ADDR, MPU6050_REG_PWR_MGMT, 0x00);

    // 配置采样率: 1kHz / (1+9) = 100Hz
    i2c_write_reg(MPU6050_ADDR, 0x19, 9);

    // 配置低通滤波: 44Hz
    i2c_write_reg(MPU6050_ADDR, 0x1A, 0x03);

    // 配置陀螺仪: ±500°/s
    i2c_write_reg(MPU6050_ADDR, 0x1B, 0x08);

    // 配置加速度计: ±4g
    i2c_write_reg(MPU6050_ADDR, 0x1C, 0x08);
}

mpu6050_raw_t mpu6050_read(void) {
    mpu6050_raw_t raw;
    uint8_t buf[14];

    i2c_read_regs(MPU6050_ADDR, MPU6050_REG_ACCEL_X, buf, 14);

    raw.ax = (buf[0] << 8) | buf[1];
    raw.ay = (buf[2] << 8) | buf[3];
    raw.az = (buf[4] << 8) | buf[5];
    raw.gx = (buf[8] << 8) | buf[9];
    raw.gy = (buf[10] << 8) | buf[11];
    raw.gz = (buf[12] << 8) | buf[13];

    return raw;
}

// 转换为物理单位
void mpu6050_convert(mpu6050_raw_t *raw, float *ax, float *ay, float *az,
                     float *gx, float *gy, float *gz) {
    // ±4g → 8192 LSB/g
    *ax = raw->ax / 8192.0f;
    *ay = raw->ay / 8192.0f;
    *az = raw->az / 8192.0f;

    // ±500°/s → 65.5 LSB/(°/s)
    *gx = raw->gx / 65.5f;
    *gy = raw->gy / 65.5f;
    *gz = raw->gz / 65.5f;
}
```

## 三、气压传感器

### 3.1 BMP280

```c
// BMP280驱动
typedef struct {
    uint16_t dig_T1;
    int16_t  dig_T2, dig_T3;
    uint16_t dig_P1;
    int16_t  dig_P2, dig_P3, dig_P4, dig_P5;
    int16_t  dig_P6, dig_P7, dig_P8, dig_P9;
} bmp280_calib_t;

static bmp280_calib_t calib;
static int32_t t_fine;

void bmp280_init(void) {
    // 读取校准数据
    uint8_t cal[26];
    i2c_read_regs(BMP280_ADDR, 0x88, cal, 26);

    calib.dig_T1 = (cal[1] << 8) | cal[0];
    calib.dig_T2 = (cal[3] << 8) | cal[2];
    calib.dig_T3 = (cal[5] << 8) | cal[4];
    // ... 解析其他校准参数

    // 配置: 温度过采样16x, 气压过采样16x, 正常模式
    i2c_write_reg(BMP280_ADDR, 0xF4, 0x57);
    i2c_write_reg(BMP280_ADDR, 0xF5, 0x90);  // 500ms间隔
}

float bmp280_read_temperature(void) {
    uint8_t buf[3];
    i2c_read_regs(BMP280_ADDR, 0xFA, buf, 3);
    int32_t adc_T = (buf[0] << 12) | (buf[1] << 4) | (buf[2] >> 4);

    int32_t var1 = ((((adc_T >> 3) - ((int32_t)calib.dig_T1 << 1))) *
                    ((int32_t)calib.dig_T2)) >> 11;
    int32_t var2 = (((((adc_T >> 4) - ((int32_t)calib.dig_T1)) *
                      ((adc_T >> 4) - ((int32_t)calib.dig_T1))) >> 12) *
                    ((int32_t)calib.dig_T3)) >> 14;

    t_fine = var1 + var2;
    return (t_fine * 5 + 128) >> 8 / 100.0f;
}

float bmp280_read_pressure(void) {
    uint8_t buf[3];
    i2c_read_regs(BMP280_ADDR, 0xF7, buf, 3);
    int32_t adc_P = (buf[0] << 12) | (buf[1] << 4) | (buf[2] >> 4);

    int64_t var1 = ((int64_t)t_fine) - 128000;
    int64_t var2 = var1 * var1 * (int64_t)calib.dig_P6;
    var2 = var2 + ((var1 * (int64_t)calib.dig_P5) << 17);
    var2 = var2 + (((int64_t)calib.dig_P4) << 35);
    var1 = ((var1 * var1 * (int64_t)calib.dig_P3) >> 8) +
           ((var1 * (int64_t)calib.dig_P2) << 12);
    var1 = (((((int64_t)1) << 47) + var1)) * ((int64_t)calib.dig_P1) >> 33;

    if (var1 == 0) return 0;

    int64_t p = 1048576 - adc_P;
    p = (((p << 31) - var2) * 3125) / var1;
    var1 = (((int64_t)calib.dig_P9) * (p >> 13) * (p >> 13)) >> 25;
    var2 = (((int64_t)calib.dig_P8) * p) >> 19;
    p = ((p + var1 + var2) >> 8) + (((int64_t)calib.dig_P7) << 4);

    return (uint32_t)p / 256.0f / 100.0f;  // hPa
}
```

### 4.1 BH1750 光照传感器

```c
// BH1750驱动
#define BH1750_ADDR 0x23
#define BH1750_CMD_POWER_ON   0x01
#define BH1750_CMD_CONT_HRES  0x10  // 连续高分辨率

void bh1750_init(void) {
    i2c_write_byte(BH1750_ADDR, BH1750_CMD_POWER_ON);

float bh1750_read_lux(void) {
    uint8_t buf[2];
    i2c_read(BH1750_ADDR, buf, 2);

    uint16_t raw = (buf[0] << 8) | buf[1];
    return raw / 1.2f;  // lx
}
```

### 4.2 MAX30102 心率血氧

```c
// MAX30102驱动
void max30102_init(void) {
    // 复位
    i2c_write_reg(MAX30102_ADDR, 0x09, 0x40);
    vTaskDelay(pdMS_TO_TICKS(100));

    // 配置FIFO
    i2c_write_reg(MAX30102_ADDR, 0x08, 0x0F);  // 平均4次, 满中断

    // SpO2配置: 100Hz, 18bit, 1600μs
    i2c_write_reg(MAX30102_ADDR, 0x0A, 0x27);

    // LED电流: IR=12.5mA, Red=12.5mA
    i2c_write_reg(MAX30102_ADDR, 0x0C, 0x24);
    i2c_write_reg(MAX30102_ADDR, 0x0D, 0x24);
}

typedef struct {
    uint32_t red;
    uint32_t ir;
} max30102_sample_t;

max30102_sample_t max30102_read(void) {
    max30102_sample_t sample;
    uint8_t buf[6];

    i2c_read_regs(MAX30102_ADDR, 0x07, buf, 6);

    sample.ir  = ((uint32_t)buf[0] << 16) | ((uint32_t)buf[1] << 8) | buf[2];
    sample.red = ((uint32_t)buf[3] << 16) | ((uint32_t)buf[4] << 8) | buf[5];

    sample.ir &= 0x03FFFF;
    sample.red &= 0x03FFFF;

    return sample;
}

// 简单心率计算(峰值检测)
int calc_heart_rate(uint32_t *ir_buffer, int count, int sample_rate) {
    int peaks = 0;
    int last_peak = 0;

    for (int i = 2; i < count - 2; i++) {
        if (ir_buffer[i] > ir_buffer[i-1] && ir_buffer[i] > ir_buffer[i+1] &&
            ir_buffer[i] > ir_buffer[i-2] && ir_buffer[i] > ir_buffer[i+2]) {
            if (i - last_peak > sample_rate / 3) {  // 最小间隔200ms
                peaks++;
                last_peak = i;
            }
        }
    }

    float duration = (float)count / sample_rate;
    return (int)(peaks / duration * 60);  // BPM
}
```

### 5.1 SGP30 空气质量

```c
// SGP30驱动
void sgp30_init(void) {
    uint8_t cmd[2] = {0x20, 0x03};  // IAQ Init
    i2c_write(SGP30_ADDR, cmd, 2);
    vTaskDelay(pdMS_TO_TICKS(10));
}

typedef struct {
    uint16_t tvoc;  // ppb
    uint16_t co2;   // ppm
} sgp30_data_t;

sgp30_data_t sgp30_read(void) {
    sgp30_data_t data;
    uint8_t cmd[2] = {0x20, 0x08};  // Measure IAQ
    uint8_t buf[6];

    i2c_write(SGP30_ADDR, cmd, 2);
    vTaskDelay(pdMS_TO_TICKS(12));

    i2c_read(SGP30_ADDR, buf, 6);

    data.co2 = (buf[0] << 8) | buf[1];
    data.tvoc = (buf[3] << 8) | buf[4];

    return data;
}
```

## 六、传感器校准

### 6.1 多点校准

```c
// 多点线性校准
typedef struct {
    int num_points;
    float raw[8];
    float calibrated[8];
} calibration_table_t;

float apply_calibration(calibration_table_t *cal, float raw) {
    // 找到所在区间
    for (int i = 0; i < cal->num_points - 1; i++) {
        if (raw >= cal->raw[i] && raw <= cal->raw[i+1]) {
            float ratio = (raw - cal->raw[i]) / (cal->raw[i+1] - cal->raw[i]);
            return cal->calibrated[i] +
                   ratio * (cal->calibrated[i+1] - cal->calibrated[i]);
        }
    }
    return raw;
}
```

### 6.2 温度补偿

```c
// 温度补偿公式
typedef struct {
    float offset;
    float gain;
    float temp_coeff;
    float ref_temp;
} compensation_params_t;

float apply_compensation(float raw, float temperature,
                         compensation_params_t *params) {
    float temp_diff = temperature - params->ref_temp;
    return (raw + params->offset) * params->gain *
           (1.0f + params->temp_coeff * temp_diff);
}
```

## 七、传感器滤波

### 7.1 滑动平均

```c
// 滑动平均滤波器
typedef struct {
    float buffer[32];
    int size;
    int index;
    float sum;
} moving_avg_t;

float moving_avg_update(moving_avg_t *f, float new_val) {
    f->sum -= f->buffer[f->index];
    f->buffer[f->index] = new_val;
    f->sum += new_val;
    f->index = (f->index + 1) % f->size;
    return f->sum / f->size;
}
```

### 7.2 中值滤波

```c
// 中值滤波器
float median_filter(float *buffer, int size) {
    // 排序
    for (int i = 0; i < size - 1; i++) {
        for (int j = i + 1; j < size; j++) {
            if (buffer[j] < buffer[i]) {
                float temp = buffer[i];
                buffer[i] = buffer[j];
                buffer[j] = temp;
            }
        }
    }
    return buffer[size / 2];
}
```
