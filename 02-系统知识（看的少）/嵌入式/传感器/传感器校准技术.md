# 传感器校准技术

## 核心概念

- **校准** - 消除系统误差
- **补偿** - 温度等环境因素修正
- **滤波** - 消除随机噪声
- **融合** - 多传感器数据结合

---

## 一、校准方法

### 1.1 线性校准

```c
// 线性校准: y = ax + b
typedef struct {
    float gain;      // 斜率a
    float offset;    // 偏移b
} linear_cal_t;

// 两点校准
void calibrate_two_point(float raw1, float ref1,
                         float raw2, float ref2,
                         linear_cal_t *cal) {
    cal->gain = (ref2 - ref1) / (raw2 - raw1);
    cal->offset = ref1 - cal->gain * raw1;
}

float apply_calibration(float raw, linear_cal_t *cal) {
    return cal->gain * raw + cal->offset;
}
```

### 1.2 多点校准

```c
// 多点分段线性校准
#define CAL_POINTS 5

typedef struct {
    float raw[CAL_POINTS];
    float ref[CAL_POINTS];
    int count;
} multi_point_cal_t;

// 多点校准(分段插值)
float calibrate_multi_point(float raw, multi_point_cal_t *cal) {
    // 查找所在区间
    for (int i = 0; i < cal->count - 1; i++) {
        if (raw >= cal->raw[i] && raw <= cal->raw[i + 1]) {
            // 线性插值
            float t = (raw - cal->raw[i]) / (cal->raw[i + 1] - cal->raw[i]);
            return cal->ref[i] + t * (cal->ref[i + 1] - cal->ref[i]);
        }
    }

    // 超出范围，外推
    if (raw < cal->raw[0]) {
        float t = (raw - cal->raw[0]) / (cal->raw[1] - cal->raw[0]);
        return cal->ref[0] + t * (cal->ref[1] - cal->ref[0]);
    } else {
        int last = cal->count - 1;
        float t = (raw - cal->raw[last - 1]) / (cal->raw[last] - cal->raw[last - 1]);
        return cal->ref[last - 1] + t * (cal->ref[last] - cal->ref[last - 1]);
    }
}
```

### 1.3 多项式校准

```c
// 多项式校准: y = a0 + a1*x + a2*x^2 + ...
typedef struct {
    float coeffs[4];  // 最多3次多项式
    int order;
} poly_cal_t;

float calibrate_polynomial(float x, poly_cal_t *cal) {
    float result = 0;
    float x_pow = 1.0f;

    for (int i = 0; i <= cal->order; i++) {
        result += cal->coeffs[i] * x_pow;
        x_pow *= x;
    }

    return result;
}

// 最小二乘法拟合
void least_squares_fit(float *x, float *y, int n,
                       int order, poly_cal_t *cal) {
    // 构建正规方程组
    float A[4][4] = {0};
    float b[4] = {0};

    for (int i = 0; i <= order; i++) {
        for (int j = 0; j <= order; j++) {
            for (int k = 0; k < n; k++) {
                A[i][j] += powf(x[k], i + j);
            }
        }
        for (int k = 0; k < n; k++) {
            b[i] += y[k] * powf(x[k], i);
        }
    }

    // 高斯消元求解
    gaussian_elimination(A, b, order + 1, cal->coeffs);
    cal->order = order;
}
```

## 二、温度补偿

### 2.1 温度漂移补偿

```c
// 温度漂移模型
typedef struct {
    float t0;           // 参考温度
    float sensitivity;  // 灵敏度温度系数(ppm/°C)
    float offset_tc;    // 偏移温度系数
} temp_comp_t;

float compensate_temperature(float raw, float temp, temp_comp_t *tc) {
    float dt = temp - tc->t0;

    // 灵敏度补偿
    float sensitivity_factor = 1.0f + tc->sensitivity * dt / 1e6f;

    // 偏移补偿
    float offset = tc->offset_tc * dt;

    return (raw - offset) / sensitivity_factor;
}

// 多项式温度补偿
typedef struct {
    float ref_temp;
    float temp_coeff[3];  // 温度系数
} poly_temp_comp_t;

float poly_temp_compensate(float raw, float temp, poly_temp_comp_t *ptc) {
    float dt = temp - ptc->ref_temp;
    float correction = ptc->temp_coeff[0] * dt +
                       ptc->temp_coeff[1] * dt * dt +
                       ptc->temp_coeff[2] * dt * dt * dt;
    return raw - correction;
}
```

### 2.2 传感器自校准

```c
// 自动校准流程
typedef enum {
    CAL_STATE_IDLE,
    CAL_STATE_ZERO,
    CAL_STATE_SPAN,
    CAL_STATE_DONE,
} cal_state_t;

typedef struct {
    cal_state_t state;
    float zero_raw;
    float span_raw;
    float zero_ref;
    float span_ref;
    linear_cal_t result;
} auto_cal_t;

void auto_calibrate_step(auto_cal_t *cal, float raw, float ref,
                         bool button_pressed) {
    switch (cal->state) {
        case CAL_STATE_IDLE:
            if (button_pressed) {
                printf("Apply zero reference and press button\n");
                cal->state = CAL_STATE_ZERO;
            }
            break;

        case CAL_STATE_ZERO:
            if (button_pressed) {
                cal->zero_raw = raw;
                cal->zero_ref = ref;
                printf("Zero point recorded: raw=%.2f ref=%.2f\n", raw, ref);
                printf("Apply span reference and press button\n");
                cal->state = CAL_STATE_SPAN;
            }
            break;

        case CAL_STATE_SPAN:
            if (button_pressed) {
                cal->span_raw = raw;
                cal->span_ref = ref;
                printf("Span point recorded: raw=%.2f ref=%.2f\n", raw, ref);

                // 计算校准参数
                calibrate_two_point(cal->zero_raw, cal->zero_ref,
                                    cal->span_raw, cal->span_ref,
                                    &cal->result);
                printf("Calibration: gain=%.6f offset=%.4f\n",
                       cal->result.gain, cal->result.offset);
                cal->state = CAL_STATE_DONE;
            }
            break;

        case CAL_STATE_DONE:
            break;
    }
}
```

## 三、滤波算法

### 3.1 滑动平均

```c
// 滑动平均滤波器
typedef struct {
    float buffer[32];
    int size;
    int index;
    float sum;
} moving_avg_t;

float moving_average(moving_avg_t *ma, float new_value) {
    // 移除旧值
    ma->sum -= ma->buffer[ma->index];

    // 添加新值
    ma->buffer[ma->index] = new_value;
    ma->sum += new_value;

    // 更新索引
    ma->index = (ma->index + 1) % ma->size;

    return ma->sum / ma->size;
}

// 指数移动平均(EMA)
typedef struct {
    float alpha;  // 平滑系数(0-1)
    float value;
} ema_t;

float exponential_moving_average(ema_t *ema, float new_value) {
    ema->value = ema->alpha * new_value + (1 - ema->alpha) * ema->value;
    return ema->value;
}
```

### 3.2 中值滤波

```c
// 中值滤波
float median_filter(float *buffer, int size) {
    // 复制并排序
    float sorted[32];
    memcpy(sorted, buffer, size * sizeof(float));

    for (int i = 0; i < size - 1; i++) {
        for (int j = 0; j < size - i - 1; j++) {
            if (sorted[j] > sorted[j + 1]) {
                float temp = sorted[j];
                sorted[j] = sorted[j + 1];
                sorted[j + 1] = temp;
            }
        }
    }

    return sorted[size / 2];
}

// 滑动中值滤波
typedef struct {
    float buffer[32];
    int size;
    int index;
} sliding_median_t;

float sliding_median(sliding_median_t *sm, float new_value) {
    sm->buffer[sm->index] = new_value;
    sm->index = (sm->index + 1) % sm->size;

    return median_filter(sm->buffer, sm->size);
}
```

### 3.3 卡尔曼滤波

```c
// 一维卡尔曼滤波
typedef struct {
    float x;      // 状态估计
    float P;      // 估计误差协方差
    float Q;      // 过程噪声
    float R;      // 测量噪声
    float K;      // 卡尔曼增益
} kalman1d_t;

float kalman_filter(kalman1d_t *k, float measurement) {
    // 预测
    k->P = k->P + k->Q;

    // 更新
    k->K = k->P / (k->P + k->R);
    k->x = k->x + k->K * (measurement - k->x);
    k->P = (1 - k->K) * k->P;

    return k->x;
}

// 二维卡尔曼(位置+速度)
typedef struct {
    float x[2];   // [位置, 速度]
    float P[2][2];
    float Q[2][2];
    float R;
    float dt;
} kalman2d_t;

void kalman2d_predict(kalman2d_t *k) {
    // 状态预测: x = F * x
    float x_pred = k->x[0] + k->dt * k->x[1];
    float v_pred = k->x[1];

    // 协方差预测: P = F * P * F^T + Q
    float P00 = k->P[0][0] + k->dt * (k->P[1][0] + k->P[0][1] + k->dt * k->P[1][1]) + k->Q[0][0];
    float P01 = k->P[0][1] + k->dt * k->P[1][1];
    float P10 = k->P[1][0] + k->dt * k->P[1][1];
    float P11 = k->P[1][1] + k->Q[1][1];

    k->x[0] = x_pred;
    k->x[1] = v_pred;
    k->P[0][0] = P00;
    k->P[0][1] = P01;
    k->P[1][0] = P10;
    k->P[1][1] = P11;
}

void kalman2d_update(kalman2d_t *k, float measurement) {
    float y = measurement - k->x[0];
    float S = k->P[0][0] + k->R;
    float K0 = k->P[0][0] / S;
    float K1 = k->P[1][0] / S;

    k->x[0] += K0 * y;
    k->x[1] += K1 * y;

    k->P[0][0] *= (1 - K0);
    k->P[0][1] -= K0 * k->P[0][1];
    k->P[1][0] -= K1 * k->P[0][0];
    k->P[1][1] -= K1 * k->P[0][1];
}
```

## 四、多传感器融合

### 4.1 加权融合

```c
// 加权平均融合
typedef struct {
    float *weights;
    float *values;
    int count;
} weighted_fusion_t;

float weighted_fusion(weighted_fusion_t *wf) {
    float sum = 0;
    float weight_sum = 0;

    for (int i = 0; i < wf->count; i++) {
        sum += wf->weights[i] * wf->values[i];
        weight_sum += wf->weights[i];
    }

    return sum / weight_sum;
}

// 基于方差的加权
float variance_weighted_fusion(float *values, float *variances, int count) {
    float weight_sum = 0;
    float weighted_sum = 0;

    for (int i = 0; i < count; i++) {
        float weight = 1.0f / variances[i];
        weighted_sum += weight * values[i];
        weight_sum += weight;
    }

    return weighted_sum / weight_sum;
}
```

### 4.2 IMU融合

```c
// 互补滤波(加速度计+陀螺仪)
typedef struct {
    float angle;      // 融合角度
    float alpha;      // 滤波系数
    float dt;         // 采样周期
} complementary_filter_t;

float complementary_filter_update(complementary_filter_t *cf,
                                  float gyro_rate, float accel_angle) {
    // 陀螺仪积分
    float gyro_angle = cf->angle + gyro_rate * cf->dt;

    // 互补滤波
    cf->angle = cf->alpha * gyro_angle + (1 - cf->alpha) * accel_angle;

    return cf->angle;
}

// Madgwick滤波器(四元数)
typedef struct {
    float q[4];       // 四元数 [w, x, y, z]
    float beta;       // 收敛速度
    float dt;
} madgwick_t;

void madgwick_update(madgwick_t *mw,
                     float gx, float gy, float gz,
                     float ax, float ay, float az,
                     float mx, float my, float mz) {
    float q0 = mw->q[0], q1 = mw->q[1], q2 = mw->q[2], q3 = mw->q[3];

    // 归一化加速度
    float norm = sqrtf(ax*ax + ay*ay + az*az);
    ax /= norm; ay /= norm; az /= norm;

    // 梯度下降步
    float f1 = 2*(q1*q3 - q0*q2) - ax;
    float f2 = 2*(q0*q1 + q2*q3) - ay;
    float f3 = 2*(0.5f - q1*q1 - q2*q2) - az;

    // 雅可比矩阵
    float J[3][4] = {
        {-2*q2, 2*q3, -2*q0, 2*q1},
        {2*q1, 2*q0, 2*q3, 2*q2},
        {0, -4*q1, -4*q2, 0}
    };

    // 梯度
    float g[4] = {0};
    for (int i = 0; i < 4; i++) {
        g[i] = J[0][i]*f1 + J[1][i]*f2 + J[2][i]*f3;
    }

    // 归一化
    norm = sqrtf(g[0]*g[0] + g[1]*g[1] + g[2]*g[2] + g[3]*g[3]);
    for (int i = 0; i < 4; i++) g[i] /= norm;

    // 陀螺仪积分 + 梯度下降修正
    float qDot0 = 0.5f * (-q1*gx - q2*gy - q3*gz) - mw->beta * g[0];
    float qDot1 = 0.5f * (q0*gx + q2*gz - q3*gy) - mw->beta * g[1];
    float qDot2 = 0.5f * (q0*gy - q1*gz + q3*gx) - mw->beta * g[2];
    float qDot3 = 0.5f * (q0*gz + q1*gy - q2*gx) - mw->beta * g[3];

    // 积分
    mw->q[0] += qDot0 * mw->dt;
    mw->q[1] += qDot1 * mw->dt;
    mw->q[2] += qDot2 * mw->dt;
    mw->q[3] += qDot3 * mw->dt;

    // 归一化四元数
    norm = sqrtf(mw->q[0]*mw->q[0] + mw->q[1]*mw->q[1] +
                 mw->q[2]*mw->q[2] + mw->q[3]*mw->q[3]);
    for (int i = 0; i < 4; i++) mw->q[i] /= norm;
}
```

## 五、异常检测

### 5.1 范围检测

```c
// 数据有效性检查
typedef struct {
    float min_value;
    float max_value;
    float max_rate_of_change;
    float last_value;
    uint32_t last_time;
} validity_check_t;

bool check_validity(validity_check_t *vc, float value, uint32_t timestamp) {
    // 范围检查
    if (value < vc->min_value || value > vc->max_value) {
        return false;
    }

    // 变化率检查
    float dt = (timestamp - vc->last_time) / 1000.0f;
    if (dt > 0) {
        float rate = fabsf(value - vc->last_value) / dt;
        if (rate > vc->max_rate_of_change) {
            return false;
        }
    }

    vc->last_value = value;
    vc->last_time = timestamp;
    return true;
}
```

### 5.2 统计异常检测

```c
// Z-score异常检测
typedef struct {
    float mean;
    float variance;
    int count;
    float threshold;  // 通常3.0
} zscore_detector_t;

bool detect_anomaly_zscore(zscore_detector_t *zd, float value) {
    // 在线更新均值和方差
    zd->count++;
    float delta = value - zd->mean;
    zd->mean += delta / zd->count;
    zd->variance += delta * (value - zd->mean);

    if (zd->count < 10) return false;  // 样本不足

    float stddev = sqrtf(zd->variance / zd->count);
    if (stddev < 1e-6f) return false;

    float z = fabsf((value - zd->mean) / stddev);
    return z > zd->threshold;
}
```
