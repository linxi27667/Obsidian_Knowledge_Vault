# 传感器融合

## 核心概念

- **传感器融合** - 组合多个传感器数据
- **互补滤波** - 结合高低频特性
- **卡尔曼滤波** - 最优状态估计
- **姿态估计** - 计算设备方向

---

## 一、传感器融合概述

### 1.1 融合目的

| 目的 | 说明 |
|------|------|
| 提高精度 | 减少单一传感器误差 |
| 提高可靠性 | 冗余互补 |
| 扩展范围 | 覆盖单一传感器局限 |
| 降低成本 | 用多个低成本传感器替代高成本传感器 |

---

### 1.2 IMU传感器

| 传感器 | 测量量 | 特点 |
|--------|--------|------|
| 加速度计 | 加速度(含重力) | 长期稳定，短期噪声大 |
| 陀螺仪 | 角速度 | 短期精确，长期漂移 |
| 磁力计 | 磁场方向 | 绝对方向，易受干扰 |

---

## 二、互补滤波

### 2.1 原理

```
加速度计: 低频准确 → 姿态
陀螺仪:   高频准确 → 角速度

互补滤波: 姿态 = α × (姿态 + 陀螺仪积分) + (1-α) × 加速度计姿态
```

---

### 2.2 实现

```c
typedef struct {
    float angle;     // 融合角度
    float alpha;     // 滤波系数(0.96-0.99)
    float dt;        // 采样周期
} complementary_filter_t;

void complementary_init(complementary_filter_t *f, float alpha, float dt) {
    f->angle = 0;
    f->alpha = alpha;
    f->dt = dt;
}

float complementary_update(complementary_filter_t *f,
                           float gyro_rate,    // 陀螺仪角速度(deg/s)
                           float accel_angle)  // 加速度计角度(deg)
{
    // 陀螺仪积分 + 加速度计修正
    f->angle = f->alpha * (f->angle + gyro_rate * f->dt) +
               (1.0f - f->alpha) * accel_angle;

    return f->angle;
}

// 计算Roll角度(加速度计)
float accel_roll(float ax, float ay, float az) {
    return atan2f(ay, az) * 180.0f / M_PI;
}

// 计算Pitch角度(加速度计)
float accel_pitch(float ax, float ay, float az) {
    return atan2f(-ax, sqrtf(ay*ay + az*az)) * 180.0f / M_PI;
}
```

### 2.3 完整IMU融合

```c
typedef struct {
    complementary_filter_t roll;
    complementary_filter_t pitch;
    float yaw;
    float mag_declination;  // 磁偏角
} imu_fusion_t;

void imu_fusion_init(imu_fusion_t *imu, float dt) {
    complementary_init(&imu->roll, 0.98f, dt);
    complementary_init(&imu->pitch, 0.98f, dt);
    imu->yaw = 0;
    imu->mag_declination = 0;
}

void imu_fusion_update(imu_fusion_t *imu,
                       float gx, float gy, float gz,  // 陀螺仪
                       float ax, float ay, float az,  // 加速度计
                       float mx, float my, float mz,  // 磁力计
                       float dt)
{
    // 加速度计角度
    float accel_roll_angle = accel_roll(ax, ay, az);
    float accel_pitch_angle = accel_pitch(ax, ay, az);

    // 互补滤波
    float roll = complementary_update(&imu->roll, gx, accel_roll_angle);
    float pitch = complementary_update(&imu->pitch, gy, accel_pitch_angle);

    // 磁力计计算Yaw
    float roll_rad = roll * M_PI / 180.0f;
    float pitch_rad = pitch * M_PI / 180.0f;

    float mx_comp = mx * cosf(pitch_rad) + mz * sinf(pitch_rad);
    float my_comp = mx * sinf(roll_rad) * sinf(pitch_rad) +
                    my * cosf(roll_rad) -
                    mz * sinf(roll_rad) * cosf(pitch_rad);

    float yaw = atan2f(-my_comp, mx_comp) * 180.0f / M_PI + imu->mag_declination;

    // 陀螺仪积分(短期)
    imu->yaw += gz * dt;

    // 融合(简单平均)
    imu->yaw = 0.98f * imu->yaw + 0.02f * yaw;

    // 规范化到0-360
    if (imu->yaw < 0) imu->yaw += 360.0f;
    if (imu->yaw >= 360) imu->yaw -= 360.0f;
}
```

## 三、卡尔曼滤波

### 3.1 一维卡尔曼滤波

```c
typedef struct {
    float x;      // 状态估计
    float P;      // 估计误差协方差
    float Q;      // 过程噪声
    float R;      // 测量噪声
    float K;      // 卡尔曼增益
} kalman1d_t;

void kalman1d_init(kalman1d_t *k, float Q, float R, float P0) {
    k->x = 0;
    k->P = P0;
    k->Q = Q;
    k->R = R;
}

float kalman1d_update(kalman1d_t *k, float measurement) {
    // 预测
    k->P = k->P + k->Q;

    // 更新
    k->K = k->P / (k->P + k->R);
    k->x = k->x + k->K * (measurement - k->x);
    k->P = (1 - k->K) * k->P;

    return k->x;
}
```

### 3.2 二维卡尔曼滤波(位置+速度)

```c
typedef struct {
    float x[2];     // 状态: [位置, 速度]
    float P[2][2];  // 协方差矩阵
    float Q[2][2];  // 过程噪声
    float R;        // 测量噪声
    float dt;       // 采样周期
} kalman2d_t;

void kalman2d_init(kalman2d_t *k, float dt) {
    k->dt = dt;
    k->x[0] = 0; k->x[1] = 0;

    // 初始协方差
    k->P[0][0] = 1; k->P[0][1] = 0;
    k->P[1][0] = 0; k->P[1][1] = 1;

    // 过程噪声
    k->Q[0][0] = 0.01; k->Q[0][1] = 0;
    k->Q[1][0] = 0; k->Q[1][1] = 0.01;

    k->R = 0.1;
}

void kalman2d_predict(kalman2d_t *k) {
    // 状态预测: x = F * x
    float x0 = k->x[0] + k->dt * k->x[1];
    float x1 = k->x[1];
    k->x[0] = x0;
    k->x[1] = x1;

    // 协方差预测: P = F * P * F' + Q
    float P00 = k->P[0][0] + k->dt * (k->P[1][0] + k->P[0][1] + k->dt * k->P[1][1]) + k->Q[0][0];
    float P01 = k->P[0][1] + k->dt * k->P[1][1];
    float P10 = k->P[1][0] + k->dt * k->P[1][1];
    float P11 = k->P[1][1] + k->Q[1][1];

    k->P[0][0] = P00; k->P[0][1] = P01;
    k->P[1][0] = P10; k->P[1][1] = P11;
}

void kalman2d_update(kalman2d_t *k, float z) {
    // 卡尔曼增益: K = P * H' * (H * P * H' + R)^-1
    float S = k->P[0][0] + k->R;
    float K0 = k->P[0][0] / S;
    float K1 = k->P[1][0] / S;

    // 状态更新: x = x + K * (z - H * x)
    float y = z - k->x[0];
    k->x[0] += K0 * y;
    k->x[1] += K1 * y;

    // 协方差更新: P = (I - K * H) * P
    float P00 = (1 - K0) * k->P[0][0];
    float P01 = (1 - K0) * k->P[0][1];
    float P10 = k->P[1][0] - K1 * k->P[0][0];
    float P11 = k->P[1][1] - K1 * k->P[0][1];

    k->P[0][0] = P00; k->P[0][1] = P01;
    k->P[1][0] = P10; k->P[1][1] = P11;
}
```

## 四、四元数

### 4.1 四元数基础

```c
typedef struct {
    float w, x, y, z;
} quaternion_t;

// 单位四元数
quaternion_t quat_identity(void) {
    return (quaternion_t){1, 0, 0, 0};
}

// 四元数乘法
quaternion_t quat_mul(quaternion_t a, quaternion_t b) {
    return (quaternion_t){
        .w = a.w*b.w - a.x*b.x - a.y*b.y - a.z*b.z,
        .x = a.w*b.x + a.x*b.w + a.y*b.z - a.z*b.y,
        .y = a.w*b.y - a.x*b.z + a.y*b.w + a.z*b.x,
        .z = a.w*b.z + a.x*b.y - a.y*b.x + a.z*b.w
    };
}

// 四元数归一化
quaternion_t quat_normalize(quaternion_t q) {
    float norm = sqrtf(q.w*q.w + q.x*q.x + q.y*q.y + q.z*q.z);
    return (quaternion_t){
        q.w/norm, q.x/norm, q.y/norm, q.z/norm
    };
}

// 旋转四元数
quaternion_t quat_from_axis_angle(float ax, float ay, float az, float angle) {
    float half = angle / 2.0f;
    float s = sinf(half);
    return (quaternion_t){
        .w = cosf(half),
        .x = ax * s,
        .y = ay * s,
        .z = az * s
    };
}

// 四元数转欧拉角
void quat_to_euler(quaternion_t q, float *roll, float *pitch, float *yaw) {
    *roll = atan2f(2*(q.w*q.x + q.y*q.z), 1 - 2*(q.x*q.x + q.y*q.y));
    *pitch = asinf(2*(q.w*q.y - q.z*q.x));
    *yaw = atan2f(2*(q.w*q.z + q.x*q.y), 1 - 2*(q.y*q.y + q.z*q.z));
}
```

### 4.2 Madgwick滤波器

```c
typedef struct {
    quaternion_t q;
    float beta;  // 收敛速度
} madgwick_t;

void madgwick_init(madgwick_t *m, float beta) {
    m->q = quat_identity();
    m->beta = beta;
}

void madgwick_update(madgwick_t *m,
                     float gx, float gy, float gz,
                     float ax, float ay, float az,
                     float mx, float my, float mz,
                     float dt)
{
    float q0 = m->q.w, q1 = m->q.x, q2 = m->q.y, q3 = m->q.z;

    // 归一化加速度计
    float norm = sqrtf(ax*ax + ay*ay + az*az);
    ax /= norm; ay /= norm; az /= norm;

    // 归一化磁力计
    norm = sqrtf(mx*mx + my*my + mz*mz);
    mx /= norm; my /= norm; mz /= norm;

    // 梯度下降步
    float s0, s1, s2, s3;

    // 加速度计梯度
    float f1 = 2*(q1*q3 - q0*q2) - ax;
    float f2 = 2*(q0*q1 + q2*q3) - ay;
    float f3 = 2*(0.5f - q1*q1 - q2*q2) - az;

    s0 = -2*q2*f1 + 2*q1*f2;
    s1 = 2*q3*f1 + 2*q0*f2 - 4*q1*f3;
    s2 = -2*q0*f1 + 2*q3*f2 - 4*q2*f3;
    s3 = 2*q1*f1 + 2*q2*f2;

    // 归一化步长
    norm = sqrtf(s0*s0 + s1*s1 + s2*s2 + s3*s3);
    s0 /= norm; s1 /= norm; s2 /= norm; s3 /= norm;

    // 四元数导数
    float qDot0 = 0.5f*(-q1*gx - q2*gy - q3*gz) - m->beta*s0;
    float qDot1 = 0.5f*(q0*gx + q2*gz - q3*gy) - m->beta*s1;
    float qDot2 = 0.5f*(q0*gy - q1*gz + q3*gx) - m->beta*s2;
    float qDot3 = 0.5f*(q0*gz + q1*gy - q2*gx) - m->beta*s3;

    // 积分
    m->q.w += qDot0 * dt;
    m->q.x += qDot1 * dt;
    m->q.y += qDot2 * dt;
    m->q.z += qDot3 * dt;

    // 归一化
    m->q = quat_normalize(m->q);
}
```
